Source code for tngxmm.photometry.measure_lx

"""Campaign aperture Lx of a merged count image (the ``lx_r500/lx_1mpc/lx_1p5mpc`` columns).

Holds :func:`measure_lx` (plain annulus-mean aperture Lx), :func:`t_vir_keV`,
:func:`emsce_mean_exposure` and ``ARF_BAND_MEAN_CM2`` (the spec field
``photometry.arf_band_mean_cm2``).  The cosmology comes from
``tngxmm.core.cosmology``; centring from :func:`tngxmm.photometry.find_centre.find_centre`.

Convention of :func:`measure_lx`::

    Lx(rest 0.5-2 keV, unabsorbed) = N_net / t_eff / ECF(T, z, N_H) * 4 pi d_L^2

* background: mean counts per pixel in the 1.1-1.6 aperture-radius annulus,
  restricted to the coverage map when one is given (pixels with
  cov <= 1 % of max are dropped); the source aperture must be >= 98 % covered;
* no compensation factors: mocks carry (1+z) arrival and Galactic absorption
  like real images, so both are read with the same call;
* t_eff = ARF-weighted per-camera live time (:func:`emsce_mean_exposure`).

The campaign tables (``photometry.lx_bkg = 'masked-clipped'``, campaign check C6) use
the same geometry, coverage rules and conversion with point sources masked in
the aperture and the annulus, a sector-clipped background and no clipping of the
net counts: :func:`campaign_lx` / :func:`measure_lx_flagged` write NaN plus the
flags ``lx_<ap>_flag`` (ok | nondetection | masked | uncovered) and the signed
``lx_<ap>_signed`` (:mod:`tngxmm.photometry.estimate_lx_background`).

The ML-facing photometry uses
:mod:`tngxmm.photometry.measure_fixed_apertures` instead (one angular
background, fixed apertures, coverage flags).

References
----------
xmm_ccf_sas : band-mean effective areas and the ECF.
cosmology_planck2015 : distances (``tngxmm.core.cosmology``).
astropy : FITS I/O and cosmology.
scipy : ``ndimage.zoom``.
"""

from __future__ import annotations

from dataclasses import dataclass
from functools import lru_cache
from pathlib import Path
from typing import Any

from tngxmm.cites import cites, collect_cites


def __getattr__(name: str) -> Any:  # PEP 562: module constant, value from the spec
    if name == "ARF_BAND_MEAN_CM2":
        from tngxmm.photometry._settings import default_settings

        return dict(default_settings().arf_band_mean_cm2)
    raise AttributeError(name)


@lru_cache(maxsize=1)
def _default_mu() -> float:
    """The ``mu_tvir`` of the production preset (``PRODUCTION_PRESET``), whatever preset the view was made with."""
    from tngxmm.config import PRODUCTION_PRESET, get_preset

    return float(get_preset(PRODUCTION_PRESET).constants.mu_tvir)

@cites("cosmology_planck2015")
def t_vir_keV(M500c_msun: float, R500c_kpc: float, mu: float | None = None) -> float:
    """Virial temperature [keV] from M500c and R500c (mu = 0.588, spec ``constants.mu_tvir``)."""
    if mu is None:
        mu = _default_mu()
    m_p, G, k_B = 1.6726e-24, 6.674e-8, 1.381e-16
    return (mu * m_p * G * (M500c_msun * 1.989e33)
            / (2.0 * k_B * (R500c_kpc * 3.0856776e21)) * 8.617e-8)

@cites("xmm_ccf_sas")
def emsce_mean_exposure(fits_path: str | Path | None = None, *, header: Any = None, settings: Any = None) -> float:
    """ARF-weighted effective exposure [s] from the EMSCE/EMSCI header keywords.

    t_eff = sum_i A_i t_i / sum_j A_j, j over ALL three ECF cameras (MOS1,
    MOS2, PN), i over the cameras present (a missing camera enters with t = 0).
    Camera identity from EMSCI001..N; a header with exactly three EMSCE values
    and no EMSCI falls back to the (MOS1, MOS2, PN) write order; otherwise NaN.
    The band-mean ARFs A come from ``settings`` (the caller's preset,
    :func:`tngxmm.photometry._settings.photometry_settings`), else from the production preset.
    """
    from tngxmm.photometry._settings import default_settings

    if header is None:
        from astropy.io import fits

        with fits.open(str(fits_path)) as hdul:
            header = hdul[0].header
    hdr = header
    t_vals = [float(hdr[k]) for k in sorted(hdr.keys()) if str(k).startswith("EMSCE")
              and not str(k).startswith("EMSCI")]
    names = [str(hdr[k]).strip().upper() for k in sorted(hdr.keys())
             if str(k).startswith("EMSCI")]
    if not t_vals:
        return float("nan")
    if len(names) != len(t_vals):
        if len(t_vals) == 3:
            names = ["MOS1", "MOS2", "PN"]
        else:
            return float("nan")   # cannot identify cameras: refuse to guess
    arf = (settings or default_settings()).arf_band_mean_cm2
    A_tot = sum(arf.values())
    num = sum(arf.get(n, 0.0) * t for n, t in zip(names, t_vals))
    return num / A_tot

@cites("xmm_ccf_sas", "cosmology_planck2015", "astropy", "scipy")
def measure_lx(fits_path: str | Path, apt_kpc: float, z_obs: float, exposure_s: float,
               nH_1e22: float, T_keV: float = 5.0, expmap: Any = None,
               pix_arcsec: float | None = None, center: Any = None, *, settings: Any = None) -> float:
    """Background-subtracted rest-frame 0.5-2 keV Lx [erg/s] in a circle of ``apt_kpc``.

    ``exposure_s``: effective exposure (:func:`emsce_mean_exposure`);
    ``nH_1e22``: Galactic column of the field (donor N_H for mocks);
    ``expmap``: coverage map on the image grid (strongly recommended);
    ``center``: see :func:`tngxmm.photometry.find_centre.find_centre`;
    ``settings``: the photometry numbers of the caller's preset
    (:func:`tngxmm.photometry._settings.photometry_settings`; default the production preset's).
    Net counts are clipped at zero (a non-detection gives 0.0).  Returns NaN when unmeasurable (aperture < 98 % covered, < 200 annulus
    pixels, or, without coverage, an annulus beyond 0.95 of the half frame).
    """
    import numpy as np
    from astropy.io import fits
    from scipy.ndimage import zoom as _zoom

    from tngxmm.core.cosmology import get_cosmology
    from tngxmm.photometry._ecf_adapter import ecf_lookup
    from tngxmm.photometry._settings import default_settings
    from tngxmm.photometry.find_centre import find_centre

    st = settings or default_settings()
    if pix_arcsec is None:
        pix_arcsec = st.pix_arcsec
    cosmo = get_cosmology()
    z_obs = max(z_obs, st.z_floor)

    with fits.open(str(fits_path)) as hdul:
        data = hdul[0].data.astype(np.float32)
        hdr = hdul[0].header

    file_pix = abs(hdr.get("CDELT2", pix_arcsec / 3600.0)) * 3600.0
    if abs(file_pix - pix_arcsec) > 0.05:
        fac = file_pix / pix_arcsec
        data = _zoom(data, fac, order=1) * (1.0 / fac**2)
        if expmap is not None:
            expmap = _zoom(expmap, fac, order=1)

    d_A = cosmo.angular_diameter_distance(z_obs).to("kpc").value
    r_px = apt_kpc * 206265.0 / d_A / pix_arcsec

    h, w = data.shape
    cx, cy = find_centre(data, hdr, center, pix_arcsec, settings=st)

    yy, xx = np.ogrid[:h, :w]
    dist = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
    a_in, a_out = st.bkg_annulus_aperture_frac
    src = dist <= r_px
    ann = (dist > a_in * r_px) & (dist < a_out * r_px)

    if expmap is not None and expmap.shape == data.shape:
        # coverage-aware geometry: the SOURCE aperture must be fully on-detector,
        # the background annulus only needs enough covered pixels
        cov = expmap > st.coverage_threshold_frac * np.nanmax(expmap)
        if src.sum() == 0 or cov[src].mean() < st.aperture_min_coverage_frac:
            return np.nan
        ann &= cov
        if ann.sum() < st.bkg_min_pixels:
            return np.nan
    else:
        # no coverage information: conservative frame guard
        if a_out * r_px > 0.95 * min(h, w) / 2:
            return np.nan
        if ann.sum() < st.bkg_min_pixels or src.sum() == 0:
            return np.nan

    n_net = max(float(data[src].sum())
                - float(data[ann].mean()) * float(src.sum()), 0.0)

    ecf = ecf_lookup(T_keV, z_obs, nH_1e22)
    d_L = cosmo.luminosity_distance(z_obs).to("cm").value
    return n_net / exposure_s / ecf * 4.0 * np.pi * d_L ** 2

def sources_to_pixels(sources: Any, header: Any) -> Any:
    """(N, 2) 0-indexed pixel positions of ``[(ra, dec, ...), ...]`` in the WCS of
    ``header`` (the sky lists of :mod:`tngxmm.massfit.mask_point_sources`)."""
    import numpy as np
    from astropy.wcs import WCS

    src = list(sources or [])
    if not src:
        return np.zeros((0, 2))
    ra = np.array([float(s[0]) for s in src])
    dec = np.array([float(s[1]) for s in src])
    x, y = WCS(header, naxis=2).all_world2pix(ra, dec, 0)
    xy = np.column_stack([np.asarray(x, float), np.asarray(y, float)])
    return xy[np.all(np.isfinite(xy), axis=1)]

    """Read ``fits_path`` (resampled to ``pix_arcsec`` as :func:`measure_lx`), its
    coverage, the centre (:func:`~tngxmm.photometry.find_centre.find_centre`) and the
    mask radius of every source.  ``center`` and the header's ``CLUSTX/CLUSTY`` are pixels of the file grid and are carried
    to the resampled grid when the image is resampled.  Sources: ``sources`` = sky list ``[(ra, dec, w)]``
    (converted with the image WCS) and/or ``sources_xy`` = 0-indexed pixels of the
    file grid.  ``settings``: the photometry numbers of the caller's preset (pixel size, coverage threshold;
    :func:`tngxmm.photometry._settings.photometry_settings`; default the production preset's)."""
    import numpy as np
    from astropy.io import fits
    from scipy.ndimage import zoom as _zoom

    from tngxmm.photometry._settings import default_settings
    from tngxmm.photometry.estimate_lx_background import LxBackgroundOptions, source_radii_px
    from tngxmm.photometry.find_centre import find_centre

    opts = options or LxBackgroundOptions(method="masked-clipped")
    st = settings or default_settings()
    if pix_arcsec is None:
        pix_arcsec = st.pix_arcsec
    with fits.open(str(fits_path)) as hdul:
        data = hdul[0].data.astype(np.float32)
        hdr = hdul[0].header
    xy = sources_to_pixels(sources, hdr)
    if sources_xy is not None and len(sources_xy):
        xy = np.vstack([xy, np.asarray(sources_xy, float).reshape(-1, 2)])
    file_pix = abs(hdr.get("CDELT2", pix_arcsec / 3600.0)) * 3600.0
    if abs(file_pix - pix_arcsec) > 0.05:
        fac = file_pix / pix_arcsec
        data = _zoom(data, fac, order=1) * (1.0 / fac**2)
        if expmap is not None:
            expmap = _zoom(expmap, fac, order=1)
        xy = (xy + 0.5) * fac - 0.5
        # an explicit centre, or the CLUSTX/CLUSTY of the header, is in pixels of the FILE grid: carry it to the resampled grid
        # by the rule of the source positions ('auto' is searched on the resampled image)
        if center is None and "CLUSTX" in hdr and "CLUSTY" in hdr:
            center = (float(hdr["CLUSTX"]), float(hdr["CLUSTY"]))
        if isinstance(center, (tuple, list, np.ndarray)):
            center = ((float(center[0]) + 0.5) * fac - 0.5, (float(center[1]) + 0.5) * fac - 0.5)
    cx, cy = find_centre(data, hdr, center, pix_arcsec, settings=st)
    covered = None
    if expmap is not None and np.shape(expmap) == data.shape:
        e = np.nan_to_num(np.asarray(expmap, np.float64))
        covered = e > st.coverage_threshold_frac * np.nanmax(e)
    cov_or_all = covered if covered is not None else np.ones(data.shape, bool)
    radii = (source_radii_px(data, cov_or_all, xy, pix_arcsec, opts) if len(xy)
             else np.zeros(0))
    return LxImage(data=data, covered=covered, cx=float(cx), cy=float(cy),
                   pix_arcsec=float(pix_arcsec), sources_xy=xy, radii_px=radii)

@cites("xmm_ccf_sas", "cosmology_planck2015")
def measure_lx_flagged(image: LxImage, apt_kpc: float, z_obs: float, exposure_s: float,
                       nH_1e22: float, T_keV: float = 5.0, *, options: Any = None,
                       settings: Any = None, inner_kpc: float = 0.0) -> dict[str, Any]:
    """Masked-clipped aperture Lx [erg/s] of a :class:`LxImage`, with its flag.

    ``inner_kpc`` > 0 measures the annulus ``inner_kpc < r <= apt_kpc`` instead of the disc (a core-excised
    luminosity, :func:`core_excised_lx`): the source pixels, the coverage rules, the ring fill and the masked fraction
    are those of the annulus, and the background is the one of the disc of radius ``apt_kpc`` (its 1.1-1.6 annulus),
    so the whole annulus has one background from outside it.

    Same geometry, coverage rules and conversion as :func:`measure_lx` (``settings``: the photometry numbers of the
    caller's preset, :func:`tngxmm.photometry._settings.photometry_settings`; default the production preset's); the
    background and the aperture sum follow
    :mod:`tngxmm.photometry.estimate_lx_background`.  Returns ``lx`` (finite only
    when ``flag == 'ok'``), ``lx_signed`` (the unclipped value; NaN for ``uncovered`` and for ``masked`` caused by a failed background), ``flag`` (ok | nondetection | masked | uncovered),
    ``n_net`` (signed net counts), ``bkg_per_pix``, ``n_src_pix``,
    ``n_src_masked_pix``, ``n_bkg_pix`` (annulus pixels used) and
    ``n_sources_masked`` (listed sources whose mask reaches the aperture or
    the annulus), ``mask_frac`` (masked fraction of the aperture pixels) and
    ``area_corr`` (masked-area correction factor, see below).

    Masked-area correction (:func:`~tngxmm.photometry.estimate_lx_background.ring_filled_sum`)::

        N_net = sum_unmasked C + sum_rings n_mask(k) <C>_k  -  B N_ap
        area_corr = N_net / (sum_unmasked C - B N_unmasked)

    ``<C>_k`` the mean counts of the unmasked covered aperture pixels in ring k
    (width ``lx_fill_ring_arcsec`` about the centre), ``n_mask(k)`` its masked
    covered pixels, ``B`` the background per pixel, ``N_ap`` all aperture pixels.
    ``area_corr`` is the factor the correction applied to the net counts of the
    unmasked pixels (1 with no mask; NaN when those are <= 0).

    ``nondetection``: net counts <= 0; ``masked``: more than ``max_masked_frac``
    of the aperture lies under source masks, or fewer than ``bkg_min_pixels``
    annulus pixels survive masking and clipping; ``uncovered``: the
    coverage rules (aperture < 98 % covered, < 200 covered annulus pixels, or
    without a coverage map an annulus beyond 0.95 of the half frame).

    ``options.fill_uncovered`` (the seed photometry): the uncovered aperture pixels are
    ring-filled like the masked ones, the 98 % rule is replaced by the masked-area
    limit on masked plus uncovered pixels (``mask_frac`` counts both), and an
    aperture more than ``max_masked_frac`` uncovered is ``uncovered``."""
    import numpy as np

    from tngxmm.core.cosmology import get_cosmology
    from tngxmm.photometry._ecf_adapter import ecf_lookup
    from tngxmm.photometry._settings import default_settings
    from tngxmm.photometry.estimate_lx_background import (LxBackgroundOptions, ring_filled_sum,
                                                          sector_clipped_mean, source_mask)

    opts = options or LxBackgroundOptions(method="masked-clipped")
    st = settings or default_settings()
    cosmo = get_cosmology()
    z_obs = max(z_obs, st.z_floor)
    nan = float("nan")
    out: dict[str, Any] = dict(lx=nan, lx_signed=nan, flag="uncovered", n_net=nan,
                               mask_frac=nan, area_corr=nan,
                               bkg_per_pix=nan, n_src_pix=0, n_src_masked_pix=0, n_bkg_pix=0,
                               n_sources_masked=0)
    data, pix = image.data, image.pix_arcsec
    d_A = cosmo.angular_diameter_distance(z_obs).to("kpc").value
    r_px = apt_kpc * 206265.0 / d_A / pix
    h, w = data.shape
    yy, xx = np.ogrid[:h, :w]
    dist = np.sqrt((xx - image.cx) ** 2 + (yy - image.cy) ** 2)
    a_in, a_out = st.bkg_annulus_aperture_frac
    src = dist <= r_px
    if inner_kpc > 0:
        src &= dist > inner_kpc * 206265.0 / d_A / pix
    ann = (dist > a_in * r_px) & (dist < a_out * r_px)
    out["n_src_pix"] = int(src.sum())

    # geometric coverage rules (those of measure_lx)
    fill_gap = bool(getattr(opts, "fill_uncovered", False)) and image.covered is not None
    if image.covered is not None:
        cov = image.covered
        if src.sum() == 0:
            return out
        if fill_gap:
            if 1.0 - cov[src].mean() > opts.max_masked_frac:
                return out
        elif cov[src].mean() < st.aperture_min_coverage_frac:
            return out
        ann &= cov
        if ann.sum() < st.bkg_min_pixels:
            return out
    else:
        cov = np.ones(data.shape, bool)
        if a_out * r_px > 0.95 * min(h, w) / 2:
            return out
        if ann.sum() < st.bkg_min_pixels or src.sum() == 0:
            return out

    # point-source masks (sources whose disc reaches the aperture or the annulus)
    xy, rad = np.asarray(image.sources_xy).reshape(-1, 2), np.asarray(image.radii_px)
    protect = min(opts.protect_core_arcsec / pix, opts.protect_core_frac * r_px)
    if len(xy):
        dc = np.hypot(xy[:, 0] - image.cx, xy[:, 1] - image.cy)
        near = (dc - rad < a_out * r_px) & (dc >= protect)
        xy, rad = xy[near], rad[near]
    out["n_sources_masked"] = int(len(xy))
    masked = source_mask(data.shape, xy, rad, image.cx, image.cy, protect)
    if not opts.mask_aperture:
        masked &= ~src
    n_src_masked = int((src & masked).sum())
    out["n_src_masked_pix"] = n_src_masked

    bkg, n_bkg = sector_clipped_mean(data, ann & ~masked, image.cx, image.cy, opts.sectors,
                                     opts.clip_sigma)
    out["n_bkg_pix"] = int(n_bkg)
    if n_bkg < st.bkg_min_pixels or not np.isfinite(bkg):
        out["flag"] = "masked"
        return out
    if fill_gap:
        # uncovered aperture pixels take the ring mean of the covered unmasked ones, as masked pixels do
        masked = masked | (src & ~cov)
        n_src_masked = int((src & masked).sum())
        cov = np.ones(data.shape, bool)
    total, _ = ring_filled_sum(data, src, masked, cov, dist, opts.fill_ring_arcsec / pix)
    n_net = total - bkg * float(src.sum())
    # what the masked-area correction did: the masked fraction of the aperture and the
    # factor between the corrected net counts and the net counts of the unmasked pixels
    keep = src & ~masked
    n_net_unmasked = float(data[keep].sum()) - bkg * float(keep.sum())
    out["mask_frac"] = n_src_masked / float(src.sum())
    out["area_corr"] = (1.0 if n_src_masked == 0
                        else float(n_net / n_net_unmasked) if n_net_unmasked > 0 else nan)
    ecf = ecf_lookup(T_keV, z_obs, nH_1e22)
    d_L = cosmo.luminosity_distance(z_obs).to("cm").value
    lx_signed = n_net / exposure_s / ecf * 4.0 * np.pi * d_L ** 2
    out.update(n_net=float(n_net), bkg_per_pix=float(bkg), lx_signed=float(lx_signed))
    if n_src_masked > opts.max_masked_frac * float(src.sum()):
        out["flag"] = "masked"
    elif not (n_net > 0):
        out["flag"] = "nondetection"
    else:
        out.update(flag="ok", lx=float(lx_signed))
    return out

def _masked_lx_columns(merged: str | Path, aps: list[tuple[str, float]], z_obs: float,
                       exposure_s: float, nh_1e22: float, T_keV: float, expmap: Any = None, *,
                       spec: Any, sources: Any = None, sources_xy: Any = None, center: Any = None,
                       kind: str = "mock", obsid: str | None = None,
                       pps_cache: str | None = None, strict: bool = False,
                       fill_uncovered: bool = False, source_list: str | None = None) -> dict[str, Any]:
    """Masked-clipped Lx columns (``<name>``, ``_flag``, ``_signed``, ``_mask_frac``,
    ``_area_corr`` per aperture, ``lx_n_sources_masked``, ``centre_x``, ``centre_y``) of
    the apertures ``aps`` = [(column, radius kpc)] on one image: the body of
    :func:`campaign_lx` and :func:`seed_lx` (one source list, one prepared image).  The sources are the SAS lists
    under ``massfit.source_list = 'sas'`` (both sides).  ``strict``: a missing list raises instead of falling back to
    the clipped background alone (the seeds: a fit target is not seeded with a mask the other side would have).
    ``fill_uncovered``: the seed's ring fill of an aperture that leaves the field (:func:`measure_lx_flagged`).
    Every number comes from ``spec``: the estimator options and the photometry settings of that preset
    (:func:`tngxmm.photometry._settings.photometry_settings`), never another preset's."""
    import dataclasses

    from tngxmm.photometry._settings import photometry_settings
    from tngxmm.photometry.estimate_lx_background import LxBackgroundOptions

    st = photometry_settings(spec)
    opts = dataclasses.replace(LxBackgroundOptions.from_spec(spec), fill_uncovered=bool(fill_uncovered))
    sas = str(source_list or getattr(spec.massfit, "source_list", "default")) == "sas"
    no_list = False
    if sources is None and sources_xy is None:
        try:
            sources, _ = view_lx_sources(merged, opts, kind=kind, obsid=obsid,
                                         pps_cache=pps_cache, sas=sas)
        except (FileNotFoundError, KeyError, ValueError):
            if strict:
                raise
            # no source table, or no usable threshold for the donor (PPS list absent):
            # the sector-clipped background alone; lx_n_sources_masked = -1 records it
            sources, no_list = [], True
    img = prepare_lx_image(merged, expmap, center=center, sources=sources, sources_xy=sources_xy,
                           options=opts, settings=st)
    out: dict[str, Any] = {}
    n_masked = -1 if no_list else 0
    for ap in aps:
        name, r_ap = ap[0], ap[1]
        r_in = float(ap[2]) if len(ap) > 2 else 0.0            # (name, radius[, inner radius]) [kpc]
        res = measure_lx_flagged(img, r_ap, z_obs, exposure_s, nh_1e22, T_keV, options=opts, settings=st,
                                 inner_kpc=r_in)
        out[name] = res["lx"]
        out[f"{name}_flag"] = res["flag"]
        out[f"{name}_signed"] = res["lx_signed"]
        out[f"{name}_mask_frac"] = res["mask_frac"]
        out[f"{name}_area_corr"] = res["area_corr"]
        if not no_list:
            n_masked = max(n_masked, int(res["n_sources_masked"]))
    out["lx_n_sources_masked"] = n_masked
    out["centre_x"], out["centre_y"] = img.cx, img.cy
    return out

def core_excised_lx(merged: str | Path, r500_kpc: float, z_obs: float, exposure_s: float,
                    nh_1e22: float, T_keV: float, expmap: Any = None, *, spec: Any,
