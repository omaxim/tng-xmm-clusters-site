Source code for tngxmm.ingest.simba_remote

"""Ingest of SIMBA from its original GIZMO snapshots, read over HTTP range requests.

The SIMBA team serves every snapshot of m100n1024 (run s50) as one HDF5 file of about 263 GB
(``http://simba.roe.ac.uk/simdata/m100n1024/s50/snapshots/``).  Its datasets are contiguous and
uncompressed, so a row range of any field is one byte range of the file, and the server answers
range requests.  The rows are not sorted in space as a whole, but consecutive rows are close (a run
of 8192 gas rows spans about 3 Mpc/h at z = 0.19), so the particles of a cluster cube come in long
runs of rows.  The reader therefore works in three steps:

* :func:`layout`: the HDF5 metadata, read once through h5py over range requests: byte offset,
  shape and dtype of every dataset;
* :func:`build_block_index`: one pass over the Coordinates of each particle type, keeping the
  bounding box of every block of :data:`BLOCK` rows (``index_<snap>.npz``, a few MB, reused by
  every later cutout of the snapshot);
* :func:`select_cubes` and :func:`read_rows`: the rows inside each cube (coordinates of the
  overlapping blocks only), then each field at those rows as merged byte ranges.

:func:`canonical_gas_arrays` converts the SIMBA gas fields to the canonical ones (descriptor
``simba``) and :func:`ingest_snapshot` writes the canonical cubes of a list of halos.

Halos: centres, R500c / R200c and masses come from the Subfind catalogue of the TNG-format
conversion on Vera (:func:`halo_rows`).  That catalogue was computed on the same particles in
the same comoving frame (ckpc/h); the snapshot numbers are the same.  SIMBA's own Caesar
catalogues use a different FoF and centre definition, so the Subfind rows also keep the halo
definition of TNG and EAGLE.

Rules (descriptor ``simba``): T from the internal energy and the electron abundance, as GIZMO
computes it; star forming ``StarFormationRate > 0``; wind ``DelayTime > 0`` (a particle
hydrodynamically decoupled after a kinetic wind launch, Davé et al. 2019 section 2.3); no
net-heating flag; the eleven tracked elements (``Metallicity`` columns Z, He, C, N, O, Ne, Mg,
Si, S, Ca, Fe), X_H = 1 - Z - He, the untracked metals as ``other``.  Dust-phase metals
(``Dust_Metallicity``) are not added to the gas.

Politeness: one request at a time, no parallel connections.

References
----------
simba (simulation), simba_data (the snapshot service), gizmo (code, units, wind decoupling),
tng_format_conversions (the Subfind catalogue).
"""

from __future__ import annotations

#: snapshot URL of m100n1024 s50
URL = "http://simba.roe.ac.uk/simdata/m100n1024/s50/snapshots/snap_m100n1024_{snap:03d}.hdf5"
#: rows per block of the bounding-box index
BLOCK = 4096
#: rows per request of the coordinate pass (25 MB of float32 triplets)
SCAN_ROWS = 512 * BLOCK
#: two runs of wanted rows closer than this many rows are fetched as one range
GAP_ROWS = 4096
#: upper bound of one range request [bytes]
MAX_REQUEST = 256 << 20
#: GIZMO code units of SIMBA (cm, g, cm/s; the same as the TNG-format conversion's header)
UNIT_LENGTH_CM = 3.085678e21
UNIT_MASS_G = 1.989e43
UNIT_VELOCITY_CMS = 1.0e5
#: proton mass [g] and Boltzmann constant [erg/K] (CODATA 2018, the values astropy uses)
M_P = 1.67262192369e-24
K_B = 1.380649e-16
#: columns of the SIMBA ``Metallicity`` field (Davé et al. 2019 section 2.1)
METALLICITY_COLUMNS = ("Z", "He", "C", "N", "O", "Ne", "Mg", "Si", "S", "Ca", "Fe")
#: canonical element columns of the SIMBA cubes
ELEMENTS = ("H", "He", "C", "N", "O", "Ne", "Mg", "Si", "S", "Ca", "Fe", "other")
GAS = ("Coordinates", "Velocities", "Masses", "Density", "InternalEnergy", "ElectronAbundance",
       "StarFormationRate", "Metallicity", "DelayTime", "SmoothingLength", "ParticleIDs")
KIND_PTYPE = {"gas": 0, "dm": 1, "stars": 4, "bh": 5}


def _retry(fn: Any, tries: int = 6, wait: float = 15.0) -> Any:
    """``fn()`` with retries on any error (network), waiting ``wait`` x attempt seconds."""
    for k in range(tries):
        try:
            return fn()
        except Exception:  # noqa: BLE001
            if k == tries - 1:
                raise
            time.sleep(wait * (k + 1))
    return None

class HttpFile:
    """Byte ranges of one remote file (``requests``, one connection, retries); counts bytes and requests."""

    def __init__(self, url: str) -> None:
        import requests

        self.url = url
        self.s = requests.Session()
        head = _retry(lambda: self.s.head(url, timeout=60))
        head.raise_for_status()
        self.size = int(head.headers["Content-Length"])
        self.etag = head.headers.get("ETag", "")
        self.last_modified = head.headers.get("Last-Modified", "")
        self.bytes_read = 0
        self.requests = 0

    def read(self, start: int, nbytes: int) -> bytes:
        """``nbytes`` bytes from ``start``."""
        def get():
            r = self.s.get(self.url, headers={"Range": f"bytes={start}-{start + nbytes - 1}"}, timeout=600)
            if r.status_code != 206 or len(r.content) != nbytes:
                raise OSError(f"range {start}+{nbytes}: HTTP {r.status_code}, {len(r.content)} bytes")
            return r.content

        b = _retry(get)
        self.bytes_read += nbytes
        self.requests += 1
        return b






class _RangeIO(io.RawIOBase):
    """File object over :class:`HttpFile` for h5py's metadata reads (1 MB blocks, cached)."""

    def __init__(self, hf: HttpFile, block: int = 1 << 20) -> None:
        self.hf, self.pos, self.block, self.cache = hf, 0, block, {}

    def seekable(self) -> bool:
        return True

    def readable(self) -> bool:
        return True

    def tell(self) -> int:
        return self.pos

    def seek(self, off: int, whence: int = 0) -> int:
        self.pos = off if whence == 0 else self.pos + off if whence == 1 else self.hf.size + off
        return self.pos

    def _blk(self, i: int) -> bytes:
        if i not in self.cache:
            a = i * self.block
            self.cache[i] = self.hf.read(a, min(self.block, self.hf.size - a))
        return self.cache[i]

    def readinto(self, buf: Any) -> int:
        n = min(len(buf), self.hf.size - self.pos)
        out = bytearray()
        while len(out) < n:
            i, o = divmod(self.pos + len(out), self.block)
            out += self._blk(i)[o:o + n - len(out)]
        buf[:n] = out
        self.pos += n
        return n

def layout(hf: HttpFile) -> dict:
    """``{'header': {...}, 'datasets': {'PartType0/Coordinates': {'offset', 'shape', 'dtype'}, ...}}`` of a
    snapshot; every dataset must be contiguous and uncompressed."""
    import h5py
    import numpy as np

    out: dict = {"datasets": {}}
    with h5py.File(_RangeIO(hf), "r") as f:
        out["header"] = {k: (v.tolist() if isinstance(v, np.ndarray) else v.item() if hasattr(v, "item") else v)
                         for k, v in f["Header"].attrs.items()}
        for pt in ("PartType0", "PartType1", "PartType4", "PartType5"):
            if pt not in f:
                continue
            for k, d in f[pt].items():
                if d.chunks is not None or d.compression is not None:
                    raise ValueError(f"{pt}/{k}: chunked or compressed; the range reader needs contiguous data")
                out["datasets"][f"{pt}/{k}"] = {"offset": int(d.id.get_offset()), "shape": list(d.shape),
                                                "dtype": d.dtype.str}
    return out




def _row_bytes(ds: dict) -> int:
    import numpy as np

    return int(np.dtype(ds["dtype"]).itemsize * int(np.prod(ds["shape"][1:], dtype=np.int64)))

def read_slice(hf: HttpFile, ds: dict, lo: int, hi: int) -> Any:
    """Rows ``lo:hi`` of a dataset of :func:`layout`."""
    import numpy as np

    rb = _row_bytes(ds)
    raw = hf.read(int(ds["offset"]) + lo * rb, (hi - lo) * rb)
    return np.frombuffer(raw, ds["dtype"]).reshape((hi - lo,) + tuple(ds["shape"][1:]))

def read_rows(hf: HttpFile, ds: dict, rows: Any, gap: int = GAP_ROWS) -> Any:
    """A dataset at the sorted, unique ``rows``: runs closer than ``gap`` rows are one range request
    (split at :data:`MAX_REQUEST`)."""
    import numpy as np

    rows = np.asarray(rows, np.int64)
    shape = tuple(ds["shape"][1:])
    if len(rows) == 0:
        return np.zeros((0,) + shape, ds["dtype"])
    rb = _row_bytes(ds)
    cap = max(1, MAX_REQUEST // rb)
    cut = np.flatnonzero(np.diff(rows) > gap) + 1
    out = []
    for run in np.split(rows, cut):
        lo = int(run[0])
        while lo <= int(run[-1]):
            hi = min(int(run[-1]) + 1, lo + cap)
            block = read_slice(hf, ds, lo, hi)
            sel = run[(run >= lo) & (run < hi)]
            out.append(block[sel - lo])
            lo = hi
    return np.concatenate(out)

def build_block_index(hf: HttpFile, lay: dict, out: str | Path, *, ptypes: tuple = (0, 1, 4, 5),
                      echo: Any = print) -> Path:
    """``out`` (npz): per particle type the per-block minimum and maximum of the Coordinates (``lo<pt>``,
    ``hi<pt>``, float32, comoving code units) over blocks of :data:`BLOCK` rows, from one pass over the
    Coordinates; with the URL, ETag and the block size."""
    import numpy as np

    out = Path(out)
    arrays: dict = {}
    t0 = time.time()
    for pt in ptypes:
        ds = lay["datasets"].get(f"PartType{pt}/Coordinates")
        if ds is None:
            continue
        n = int(ds["shape"][0])
        nb = (n + BLOCK - 1) // BLOCK
        lo, hi = np.empty((nb, 3), np.float32), np.empty((nb, 3), np.float32)
        for r0 in range(0, n, SCAN_ROWS):
            r1 = min(n, r0 + SCAN_ROWS)
            x = read_slice(hf, ds, r0, r1)
            b0 = r0 // BLOCK
            for j, s in enumerate(range(0, r1 - r0, BLOCK)):
                blk = x[s:s + BLOCK]
                lo[b0 + j], hi[b0 + j] = blk.min(axis=0), blk.max(axis=0)
            if echo and (r0 // SCAN_ROWS) % 50 == 0:
                echo(f"  index PartType{pt}: {r1}/{n} rows, {hf.bytes_read / 1e9:.1f} GB, "
                     f"{(time.time() - t0) / 60:.1f} min")
        arrays[f"lo{pt}"], arrays[f"hi{pt}"] = lo, hi
    tmp = out.with_name(out.name + ".tmp.npz")
    np.savez(tmp, block=BLOCK, url=hf.url, etag=hf.etag, **arrays)
    tmp.rename(out)
    return out




def _overlapping_blocks(lo: Any, hi: Any, centre: Any, half: float, box: float) -> Any:
    """Blocks whose bounding box can hold a row of the cube (per-axis minimum image; a block that spans
    more than half the box on an axis, i.e. wraps, always counts on that axis)."""
    import numpy as np

    mid = 0.5 * (lo.astype(np.float64) + hi.astype(np.float64))
    ext = 0.5 * (hi.astype(np.float64) - lo.astype(np.float64))
    d = np.asarray(centre, np.float64) - mid
    d -= box * np.round(d / box)
    ok = (np.abs(d) <= half + ext) | (2 * ext > 0.5 * box)
    return np.flatnonzero(ok.all(axis=1))

def select_cubes(hf: HttpFile, lay: dict, index: Any, pt: int, cubes: list[tuple], box: float) -> list:
    """Rows of particle type ``pt`` inside each cube ``(centre, half)`` (comoving code units, per-axis
    minimum image): the coordinates of the union of overlapping blocks are read once."""
    import numpy as np

    ds = lay["datasets"][f"PartType{pt}/Coordinates"]
    n = int(ds["shape"][0])
    lo, hi = index[f"lo{pt}"], index[f"hi{pt}"]
    per = [_overlapping_blocks(lo, hi, c, h, box) for c, h in cubes]
    blocks = np.unique(np.concatenate(per)) if per else np.zeros(0, np.int64)
    out: list = [[] for _ in cubes]
    if not len(blocks):
        return [np.zeros(0, np.int64) for _ in cubes]
    cut = np.flatnonzero(np.diff(blocks) > 1) + 1
    for run in np.split(blocks, cut):
        r0, r1 = int(run[0]) * BLOCK, min(n, (int(run[-1]) + 1) * BLOCK)
        step = max(BLOCK, (MAX_REQUEST // _row_bytes(ds)) // BLOCK * BLOCK)
        for s0 in range(r0, r1, step):
            s1 = min(r1, s0 + step)
            x = read_slice(hf, ds, s0, s1).astype(np.float64)
            for k, (c, h) in enumerate(cubes):
                d = x - np.asarray(c, np.float64)
                d -= box * np.round(d / box)
                m = np.flatnonzero(np.all(np.abs(d) <= h, axis=1))
                if len(m):
                    out[k].append(s0 + m)
    return [np.concatenate(o) if o else np.zeros(0, np.int64) for o in out]

@cites("simba", "gizmo")
def canonical_gas_arrays(raw: dict, centre: Any, box: float, h: float, a: float) -> dict:
    """Canonical gas arrays (schema dtypes) from the native SIMBA gas fields at the cube's rows.

    Positions ``Coordinates (a / h)`` at the minimum image about ``centre`` [ckpc/h]; velocity
    ``Velocities sqrt(a)``; mass ``Masses 1e10 / h``; density ``Density (h^2 / a^3) UM / UL^3``;
    T = (2/3) u UV^2 mu m_p / k_B, mu = 4 / (1 + 3 X_H + 4 X_H x_e), the relation of the TNG gas
    (and of GIZMO for an H-He gas with x_e = n_e / n_H); X_H = 1 - Z - He."""
    import numpy as np

    from tngxmm.ingest import schema
    from tngxmm.sim.read_halo_source import minimum_image

    x = minimum_image(raw["Coordinates"], centre, box)
    met = np.asarray(raw["Metallicity"], np.float32)
    Z, He = met[:, 0], met[:, 1]
    X_H = (np.float32(1.0) - Z - He).astype(np.float32)
    tracked = met[:, 2:].astype(np.float64).sum(axis=1)
    other = np.maximum(Z.astype(np.float64) - tracked, 0.0).astype(np.float32)
    x_e = np.asarray(raw["ElectronAbundance"], np.float32)
    u = np.asarray(raw["InternalEnergy"], np.float32)
    mu = 4.0 / (1.0 + 3.0 * X_H + 4.0 * X_H * x_e)
    T = (2.0 / 3.0) * u * UNIT_VELOCITY_CMS ** 2 * mu * M_P / K_B
    sfr = np.asarray(raw["StarFormationRate"], np.float32)
    flags = (sfr > 0).astype(np.uint8) * np.uint8(schema.FLAG_STAR_FORMING)
    flags |= (np.asarray(raw["DelayTime"]) > 0).astype(np.uint8) * np.uint8(schema.FLAG_WIND)
    X = np.concatenate([X_H[:, None], met[:, 1:], other[:, None]], axis=1)
    return {"pos_kpc": np.asarray(x, np.float64) * (a / h),
            "vel_kms": (np.asarray(raw["Velocities"], np.float32) * np.float32(np.sqrt(a))).astype(np.float32),
            "mass_msun": np.asarray(raw["Masses"], np.float64) * 1e10 / h,
            "rho_gcm3": np.asarray(raw["Density"], np.float64) * (h ** 2 / a ** 3) * UNIT_MASS_G / UNIT_LENGTH_CM ** 3,
            "u_kms2": u, "T_K": np.asarray(T, np.float32), "x_e": x_e, "Z": Z.astype(np.float32),
            "X_el": X.astype(np.float32), "sfr_msun_yr": sfr, "flags": flags,
            "ids": np.asarray(raw["ParticleIDs"]).astype(np.uint64),
            "hsml_kpc": (np.asarray(raw["SmoothingLength"], np.float64) * a / h).astype(np.float32)}

def halo_rows(base_path: str, snap: int, halo_ids: Any) -> list[dict]:
    """The Subfind rows of ``halo_ids`` from the TNG-format conversion's group catalogue (``base_path``):
    ``{'halo_id', 'group', 'subhalo'}`` with the native rows (code units)."""
    from tngxmm.sim.read_halo_source import import_illustris_python

    il = import_illustris_python()
    rows = []
    for hid in halo_ids:
        g = il.groupcat.loadSingle(base_path, int(snap), haloID=int(hid))
        s = (il.groupcat.loadSingle(base_path, int(snap), subhaloID=int(g["GroupFirstSub"]))
             if int(g["GroupFirstSub"]) >= 0 else {})
        rows.append({"halo_id": int(hid), "group": g, "subhalo": s})
    return rows

@cites("simba", "simba_data", "gizmo", "tng_format_conversions")
def ingest_snapshot(snap: int, rows: list[dict], out_path: str | Path, index_dir: str | Path, *,
                    half_side_r500: float = 5.66, r200_cover: float = 3.0, kinds: tuple = ("dm", "stars", "bh"),
                    url: str | None = None, sim: str = "simba", echo: Any = print) -> list[dict]:
    """The canonical cubes of the halos ``rows`` (:func:`halo_rows`) of snapshot ``snap``, written to
    ``out_path`` (:mod:`tngxmm.ingest.cube_io`; halos already there are skipped).  The half-side is
    max(``half_side_r500`` R500c, 1.001 ``r200_cover`` R200c), so that the truth sphere lies inside every
    cube.  The layout and the block index of the snapshot are kept in ``index_dir`` and reused."""
    import numpy as np

    from tngxmm.ingest import cube_io

    out_path = Path(out_path)
    done = set(cube_io.halos_in(out_path)) if out_path.exists() else set()
    rows = [r for r in rows if int(r["halo_id"]) not in done]
    if not rows:
        return []
    url = url or URL.format(snap=int(snap))
    hf = HttpFile(url)
    index_dir = Path(index_dir)
    index_dir.mkdir(parents=True, exist_ok=True)
    lay_p = index_dir / f"layout_{int(snap):03d}.json"
    if lay_p.exists():
        lay = json.loads(lay_p.read_text())
    else:
        lay = layout(hf)
        lay.update({"url": url, "etag": hf.etag, "last_modified": hf.last_modified, "size": hf.size})
        lay_p.write_text(json.dumps(lay, indent=1) + "\n")
    if lay.get("etag") != hf.etag:
        raise ValueError(f"{url}: ETag {hf.etag} differs from the stored layout's {lay.get('etag')}")
    idx_p = index_dir / f"index_{int(snap):03d}.npz"
    if not idx_p.exists():
        echo(f"snap {snap}: block index (one pass over the coordinates)")
        build_block_index(hf, lay, idx_p, echo=echo)
    index = np.load(idx_p)
    hdr = lay["header"]
    box, h, a = float(hdr["BoxSize"]), float(hdr["HubbleParam"]), float(hdr["Time"])
    z = float(hdr["Redshift"])
    cubes, halves = [], []
    for r in rows:
        g = r["group"]
        half = max(float(half_side_r500) * float(g["Group_R_Crit500"]),
                   1.001 * float(r200_cover) * float(g["Group_R_Crit200"]))
        cubes.append((np.asarray(g["GroupPos"], np.float64), half))
        halves.append(half)
    t0, b0 = time.time(), hf.bytes_read
    sel = {kind: select_cubes(hf, lay, index, KIND_PTYPE[kind], cubes, box) for kind in ("gas",) + tuple(kinds)}
    echo(f"snap {snap}: rows selected for {len(rows)} halos, {(hf.bytes_read - b0) / 1e9:.2f} GB, "
         f"{(time.time() - t0) / 60:.1f} min")

    def fields_at(pt: int, names: tuple, per_halo: list) -> list[dict]:
        union = np.unique(np.concatenate(per_halo)) if per_halo else np.zeros(0, np.int64)
        res: list[dict] = [{} for _ in per_halo]
        for name in names:
            v = read_rows(hf, lay["datasets"][f"PartType{pt}/{name}"], union)
            for k, rr in enumerate(per_halo):
                res[k][name] = v[np.searchsorted(union, rr)]
        return res

    gas = fields_at(0, GAS, sel["gas"])
    other = {kind: fields_at(KIND_PTYPE[kind], ("Coordinates", "Masses"), sel[kind]) for kind in kinds}
    recs = []
    for k, r in enumerate(rows):
        g, s = r["group"], r["subhalo"]
        centre = cubes[k][0]
        out = {"gas": canonical_gas_arrays(gas[k], centre, box, h, a)}
        from tngxmm.sim.read_halo_source import minimum_image

        for kind in kinds:
            p = other[kind][k]
            out[kind] = {"pos_kpc": np.asarray(minimum_image(p["Coordinates"], centre, box), np.float64) * (a / h),
                         "mass_msun": np.asarray(p["Masses"], np.float64) * 1e10 / h}
        r500 = float(g["Group_R_Crit500"])
        half = halves[k]
        report = {"shape": "box", "half_side_kpc": half * a / h, "half_side_r500c": half / r500,
                  "r500c_kpc": r500 * a / h, "n_cube": int(len(out["gas"]["mass_msun"])), "n_lowres_gas_dropped": 0,
                  "coarse_gas_kept": True, "n_coarse": 0, "box_source": "simba-http-ranges"}
        attrs = {"centre_kpc": centre * (a / h), "centre_native": centre, "r500c_kpc": r500 * a / h,
                 "m500c_msun": float(g["Group_M_Crit500"]) * 1e10 / h,
                 "r200c_kpc": float(g["Group_R_Crit200"]) * a / h,
                 "m200c_msun": float(g["Group_M_Crit200"]) * 1e10 / h,
                 "half_side_r500c": half / r500, "half_side_kpc": half * a / h, "complete": True,
                 "cutout_report_json": json.dumps(report),
                 "bulk_vel_kms": np.asarray(g["GroupVel"], np.float64) / a}
        file_attrs = {"sim": sim, "snap": int(snap), "z": z, "a": a, "h": h, "msun_in_g": UNIT_MASS_G / 1e10,
                      "family": "gizmo", "box_size_ckpc_h": box, "remote_snapshot": url, "remote_etag": hf.etag,
                      "halo_catalogue": "Subfind of the TNG-format conversion (tng_format_conversions)"}
        e = cube_io.write_halo(out_path, int(r["halo_id"]), out, attrs, file_attrs=file_attrs, elements=ELEMENTS,
                               catalog={"group": g, "subhalo": s})
        recs.append({"halo_id": int(r["halo_id"]), "m500c_msun": attrs["m500c_msun"], "counts": e["counts"]})
        echo(f"simba snap {snap} halo {r['halo_id']} M500c {attrs['m500c_msun']:.2e}: {e['counts']}")
    echo(f"snap {snap}: {len(recs)} cubes, {hf.bytes_read / 1e9:.2f} GB in {hf.requests} requests, "
         f"{(time.time() - t0) / 60:.1f} min")
    return recs

def main(argv: list[str] | None = None) -> int:
    """``python -m tngxmm.ingest.simba_remote <snap> <cube_root> <index_dir> --halo ID [ID ...] [--base PATH]``."""
    import argparse

    from tngxmm.ingest import cube_io, descriptor

    p = argparse.ArgumentParser(prog="python -m tngxmm.ingest.simba_remote")
    p.add_argument("snap", type=int)
    p.add_argument("cube_root")
    p.add_argument("index_dir")
    p.add_argument("--halo", type=int, nargs="+", required=True, help="Subfind FoF indices of the conversion")
    p.add_argument("--base", default=None, help="the conversion's output directory (default: descriptor simba_tngformat)")
    p.add_argument("--half", type=float, default=5.66)
    p.add_argument("--r200-cover", type=float, default=3.0)
    p.add_argument("--sim", default="simba")
    a = p.parse_args(argv)
    base = a.base or descriptor.load("simba_tngformat")["base_path"]
    rows = halo_rows(base, a.snap, a.halo)
    ingest_snapshot(a.snap, rows, cube_io.cube_path(a.cube_root, a.sim, a.snap), a.index_dir, half_side_r500=a.half,
                    r200_cover=a.r200_cover, sim=a.sim, echo=lambda m: print(m, flush=True))
    return 0
