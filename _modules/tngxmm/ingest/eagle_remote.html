Source code for tngxmm.ingest.eagle_remote

"""Ingest of EAGLE Ref-L0100N1504 from its original snapshots, streamed from the hdfstream service.

The EAGLE particle data (The EAGLE team 2017) are served by the same hdfstream service as the
FLAMINGO data (``cosma:/EAGLE/Fiducial_models/RefL0100N1504``): 256 files per snapshot, the
particles of each type sorted by the Peano-Hilbert key of their cell on a 2^HashBits grid (HashBits
= 6, cells of 1.06 cMpc/h), with a hash table per file (``HashTable/PartType<n>``:
``FirstKeyInFile``, ``LastKeyInFile`` and, per key of the file, ``NumParticleInCell``).

* :class:`HashIndex`: the rows of every key, per file and particle type, from the hash tables;
* :func:`key_cells`: the grid cell of every key, measured from the data (the cell of the first,
  middle and last particle of each key; all three must agree), so that no Peano-Hilbert code is
  needed; stored once per HashBits;
* :func:`read_cube`: the particles of one type inside a cube: the keys of the cells that overlap
  the cube padded by one cell, their coordinates, the exact per-axis minimum-image cut, then the
  other fields at the selected rows (index-list requests per file);
* :func:`canonical_gas_arrays`: EAGLE gas fields -> canonical gas arrays (descriptor ``eagle``);
* :func:`ingest_snapshot`: the canonical cubes of a list of halos of one snapshot.

Units come from each dataset's own attributes (``CGSConversionFactor``, ``aexp-scale-exponent``,
``h-scale-exponent``).  Halos: centres, radii and masses from the Subfind catalogue of the
TNG-format conversion on Vera (same particles; ckpc/h there, cMpc/h here), as for SIMBA
(:func:`tngxmm.ingest.simba_remote.halo_rows`).

Rules (descriptor ``eagle``): T as stored (EAGLE's own, from the internal energy with the
ionisation state of its cooling tables); u as stored; the electron abundance is not stored and is
taken as full ionisation of every element of the particle's smoothed composition
(:func:`electron_abundance_full`).  That holds for the gas that emits in the mock: above 3e5 K H
and He are fully ionised in collisional equilibrium (Bryans et al. 2009), and the metals carry
under one per cent of the electrons, so their partial ionisation moves n_e by less than that.
Gas below the emission threshold gets the same rule but emits nothing.  Star forming ``StarFormationRate > 0``; no wind particles; no net-heating
flag; the smoothed element abundances and metallicity (EAGLE's cooling uses them), the particle
metallicity beside them (``Z_particle``).

Politeness: one request at a time.

References
----------
eagle, eagle_particles (simulation and particle data release), wiersma2009a and wiersma2009b
(the stored temperatures and the smoothed abundances), bryans2009 and iupac2013 (the electron
abundance), flamingo_dr and hdfstream (the service and its client), tng_format_conversions (the
Subfind catalogue).
"""

def _retry(fn: Any, tries: int = 6, wait: float = 15.0) -> Any:
    for k in range(tries):
        try:
            return fn()
        except Exception:  # noqa: BLE001
            if k == tries - 1:
                raise
            time.sleep(wait * (k + 1))
    return None

def snapshot_dir(snap: int, server: str = SERVER, run: str = RUN) -> Any:
    """The remote directory of snapshot ``snap``."""
    import hdfstream

    return hdfstream.open(server, f"/{run}/snapshot_{int(snap):03d}_{SNAP_TAGS[int(snap)]}")




def _phys(ds: Any, a: float, h: float) -> float:
    """Factor from the stored values of ``ds`` to physical CGS (the dataset's own attributes)."""
    at = ds.attrs
    return (float(at["CGSConversionFactor"]) * a ** float(at["aexp-scale-exponent"])
            * h ** float(at["h-scale-exponent"]))

class HashIndex:
    """Rows of every Peano-Hilbert key, per file, of the particle types of one snapshot."""

    def __init__(self, d: Any, snap: int, ptypes: tuple = (0, 1, 4, 5), cache: str | Path | None = None) -> None:
        import numpy as np

        self.d, self.snap = d, int(snap)
        self.files = [f"snap_{int(snap):03d}_{SNAP_TAGS[int(snap)]}.{i}.hdf5" for i in range(256)]
        f0 = d[self.files[0]]
        hdr = f0["Header"].attrs
        self.nfiles = int(hdr["NumFilesPerSnapshot"])
        self.files = self.files[:self.nfiles]
        self.box = float(hdr["BoxSize"])
        self.h, self.a, self.z = float(hdr["HubbleParam"]), float(hdr["ExpansionFactor"]), float(hdr["Redshift"])
        self.bits = int(f0["HashTable"].attrs["HashBits"])
        self.ncell = 1 << self.bits
        self.mass_table = np.asarray(hdr["MassTable"], np.float64)
        self._handles: dict = {0: f0}
        self.requests = 0
        cp = Path(cache) if cache else None
        if cp is not None and cp.exists():
            z = np.load(cp)
            self.tab = {pt: (z[f"file{pt}"], z[f"key{pt}"], z[f"lo{pt}"], z[f"n{pt}"]) for pt in ptypes}
            return
        self.tab = {}
        for pt in ptypes:
            first = np.asarray(_retry(lambda pt=pt: f0[f"HashTable/PartType{pt}/FirstKeyInFile"][:]), np.int64)
            last = np.asarray(_retry(lambda pt=pt: f0[f"HashTable/PartType{pt}/LastKeyInFile"][:]), np.int64)
            fi, ke, lo, nn = [], [], [], []
            for i in range(self.nfiles):
                if last[i] < first[i]:
                    continue
                cnt = np.asarray(_retry(lambda i=i, pt=pt: self.handle(i)[f"HashTable/PartType{pt}/NumParticleInCell"][:]),
                                 np.int64)
                keys = np.arange(first[i], last[i] + 1)
                off = np.concatenate([[0], np.cumsum(cnt)[:-1]])
                m = cnt > 0
                fi.append(np.full(m.sum(), i, np.int32))
                ke.append(keys[m])
                lo.append(off[m])
                nn.append(cnt[m])
            self.tab[pt] = (np.concatenate(fi), np.concatenate(ke), np.concatenate(lo), np.concatenate(nn))
        if cp is not None:
            cp.parent.mkdir(parents=True, exist_ok=True)
            arrays = {}
            for pt, (fi, ke, lo, nn) in self.tab.items():
                arrays.update({f"file{pt}": fi, f"key{pt}": ke, f"lo{pt}": lo, f"n{pt}": nn})
            tmp = cp.with_name(cp.name + ".tmp.npz")
            np.savez(tmp, **arrays)
            tmp.rename(cp)

    def handle(self, i: int) -> Any:
        """The remote file ``i`` (cached)."""
        if i not in self._handles:
            self._handles[i] = _retry(lambda: self.d[self.files[i]])
        return self._handles[i]

    def read(self, i: int, path: str, rows: Any) -> Any:
        """``path`` of file ``i`` at the sorted ``rows`` (index lists, split at :data:`MAX_INDEX`)."""
        import numpy as np

        ds = self.handle(i)[path]
        parts = []
        for s in range(0, len(rows), MAX_INDEX):
            r = rows[s:s + MAX_INDEX]
            if len(r) and int(r[-1]) - int(r[0]) + 1 == len(r):
                v = _retry(lambda r=r: ds[int(r[0]):int(r[-1]) + 1])
            else:
                v = _retry(lambda r=r: ds[r])
            self.requests += 1
            parts.append(np.asarray(v))
        return np.concatenate(parts) if parts else np.zeros(0)

@cites("eagle_particles")
def key_cells(idx: HashIndex, out: str | Path, pt: int = 1) -> Any:
    """``out`` (npy, int16 (n_keys, 3)): the grid cell of every key, from the first, middle and last particle
    of the key in the snapshot (particle type ``pt``, then the others for keys without such particles);
    raises when the three disagree.  Keys with no particle at all get -1."""
    import numpy as np

    out = Path(out)
    if out.exists():
        return np.load(out)
    nkeys = idx.ncell ** 3
    cells = np.full((nkeys, 3), -1, np.int16)
    size = idx.box / idx.ncell
    for p in (pt,) + tuple(q for q in idx.tab if q != pt):
        fi, ke, lo, nn = idx.tab[p]
        todo = cells[ke, 0] < 0
        for i in np.unique(fi[todo]):
            m = todo & (fi == i)
            rows = np.stack([lo[m], lo[m] + nn[m] // 2, lo[m] + nn[m] - 1], axis=1)
            u, inv = np.unique(rows.ravel(), return_inverse=True)
            x = idx.read(int(i), f"PartType{p}/Coordinates", u)[inv].reshape(-1, 3, 3)
            c = np.floor(x / size).astype(np.int64) % idx.ncell
            if not (np.all(c[:, 0] == c[:, 1]) and np.all(c[:, 0] == c[:, 2])):
                bad = np.flatnonzero(~np.all((c[:, 0] == c[:, 1]) & (c[:, 0] == c[:, 2]), axis=1))
                raise ValueError(f"file {i} type {p}: {len(bad)} keys whose particles lie in different cells")
            cells[ke[m]] = c[:, 0]
    np.save(out, cells)
    return cells




def _keys_of_cube(cells: Any, ncell: int, size: float, centre: Any, half: float) -> Any:
    """Keys whose cell overlaps the cube padded by one cell (periodic)."""
    import numpy as np

    lo = np.floor((np.asarray(centre) - half) / size).astype(np.int64) - 1
    hi = np.floor((np.asarray(centre) + half) / size).astype(np.int64) + 1
    want = np.zeros((ncell, ncell, ncell), bool)
    ix, iy, iz = (np.arange(lo[k], hi[k] + 1) % ncell for k in range(3))
    want[np.ix_(ix, iy, iz)] = True
    ok = cells[:, 0] >= 0
    keys = np.flatnonzero(ok)
    c = cells[ok]
    return keys[want[c[:, 0], c[:, 1], c[:, 2]]]

def read_cube(idx: HashIndex, cells: Any, centre: Any, half: float, kind: str, fields: tuple) -> dict:
    """Every particle of ``kind`` inside the cube of half-side ``half`` about ``centre`` (cMpc/h, per-axis
    minimum image): ``{field: stored values}``, ``Coordinates`` as the offsets from the centre."""
    import numpy as np

    pt = KIND_PTYPE[kind]
    fi, ke, lo, nn = idx.tab[pt]
    keys = _keys_of_cube(cells, idx.ncell, idx.box / idx.ncell, centre, half)
    m = np.isin(ke, keys)
    parts: dict = {k: [] for k in fields}
    for i in np.unique(fi[m]):
        s = m & (fi == i)
        rows = np.concatenate([np.arange(a, a + b) for a, b in zip(lo[s], nn[s])])
        rows.sort()
        x = idx.read(int(i), f"PartType{pt}/Coordinates", rows).astype(np.float64) - np.asarray(centre, np.float64)
        x -= idx.box * np.round(x / idx.box)
        inside = np.all(np.abs(x) <= half, axis=1)
        if not inside.any():
            continue
        sel = rows[inside]
        for k in fields:
            if k == "Coordinates":
                parts[k].append(x[inside])
            else:
                parts[k].append(idx.read(int(i), f"PartType{pt}/{k}", sel))
    return {k: (np.concatenate(v) if v else np.zeros((0, 3) if k == "Coordinates" else 0)) for k, v in parts.items()}

@cites("bryans2009", "iupac2013")
def electron_abundance_full(X: dict, other: Any) -> Any:
    """n_e / n_H of a fully ionised gas of mass fractions ``X`` (element -> array) and untracked metal mass
    fraction ``other``: sum over elements of (X_i / A_i) Z_i, over X_H / A_H."""
    import numpy as np

    ne = sum(np.asarray(X[e], np.float64) / ATOMIC[e][1] * ATOMIC[e][0] for e in X)
    ne = ne + np.asarray(other, np.float64) * OTHER_Z_OVER_A
    xh = np.asarray(X["H"], np.float64) / ATOMIC["H"][1]
    return np.where(xh > 0, ne / np.where(xh > 0, xh, 1.0), 0.0)

@cites("eagle", "eagle_particles", "wiersma2009a", "wiersma2009b")
def canonical_gas_arrays(raw: dict, centre_cmpc_h: Any, idx: HashIndex, conv: dict) -> dict:
    """Canonical gas arrays (schema dtypes) from the stored EAGLE gas fields of :func:`read_cube`; ``conv``:
    the physical CGS factor of each field (:func:`_phys`)."""
    import numpy as np

    from tngxmm.ingest import schema

    pos = (np.asarray(centre_cmpc_h, np.float64) + raw["Coordinates"]) * conv["Coordinates"] / 3.085678e21
    X = {e: np.asarray(raw[f"SmoothedElementAbundance/{n}"], np.float32) for e, n in ELEMENT_NAMES}
    Z = np.asarray(raw["SmoothedMetallicity"], np.float32)
    tracked = sum(X[e].astype(np.float64) for e in ("C", "N", "O", "Ne", "Mg", "Si", "Fe"))
    other = np.maximum(Z.astype(np.float64) - tracked, 0.0)
    sfr = np.asarray(raw["StarFormationRate"], np.float32)
    flags = (sfr > 0).astype(np.uint8) * np.uint8(schema.FLAG_STAR_FORMING)
    return {"pos_kpc": pos,
            "vel_kms": (np.asarray(raw["Velocity"], np.float64) * conv["Velocity"] / 1e5).astype(np.float32),
            "mass_msun": np.asarray(raw["Mass"], np.float64) * conv["Mass"] / MSUN_G,
            "rho_gcm3": np.asarray(raw["Density"], np.float64) * conv["Density"],
            "u_kms2": (np.asarray(raw["InternalEnergy"], np.float64) * conv["InternalEnergy"] / 1e10).astype(np.float32),
            "T_K": np.asarray(raw["Temperature"], np.float32),
            "x_e": electron_abundance_full(X, other).astype(np.float32),
            "Z": Z, "Z_particle": np.asarray(raw["Metallicity"], np.float32),
            "X_el": np.stack([X[e] for e, _ in ELEMENT_NAMES] + [other.astype(np.float32)], axis=1),
            "sfr_msun_yr": sfr, "flags": flags, "ids": np.asarray(raw["ParticleIDs"]).astype(np.uint64),
            "hsml_kpc": (np.asarray(raw["SmoothingLength"], np.float64) * conv["SmoothingLength"]
                         / 3.085678e21).astype(np.float32)}

@cites("eagle", "eagle_particles", "flamingo_dr", "hdfstream", "tng_format_conversions")
def ingest_snapshot(snap: int, rows: list[dict], out_path: str | Path, index_dir: str | Path, *,
                    half_side_r500: float = 5.66, r200_cover: float = 3.0, kinds: tuple = ("dm", "stars", "bh"),
                    sim: str = "eagle", echo: Any = print) -> list[dict]:
    """The canonical cubes of the halos ``rows`` (:func:`tngxmm.ingest.simba_remote.halo_rows` on the conversion,
    code units ckpc/h) of snapshot ``snap``, written to ``out_path``; halos already there are skipped.  Half-side
    max(``half_side_r500`` R500c, 1.001 ``r200_cover`` R200c).  The hash index and the key cells are kept in
    ``index_dir``."""
    import numpy as np

    from tngxmm.ingest import cube_io

    out_path = Path(out_path)
    done = set(cube_io.halos_in(out_path)) if out_path.exists() else set()
    rows = [r for r in rows if int(r["halo_id"]) not in done]
    if not rows:
        return []
    index_dir = Path(index_dir)
    d = snapshot_dir(snap)
    t0 = time.time()
    idx = HashIndex(d, snap, cache=index_dir / f"hash_{int(snap):03d}.npz")
    cells = key_cells(idx, index_dir / f"key_cells_{int(snap):03d}.npy")
    echo(f"snap {snap}: hash index and key cells in {(time.time() - t0) / 60:.1f} min, {idx.requests} requests")
    f0 = idx.handle(0)
    a, h = idx.a, idx.h
    conv = {k: _phys(f0[f"PartType0/{k}"], a, h) for k in GAS}
    gas_fields = GAS + tuple(f"SmoothedElementAbundance/{n}" for _, n in ELEMENT_NAMES)
    kind_fields = {"dm": ("Coordinates",), "stars": ("Coordinates", "Mass"), "bh": ("Coordinates", "Mass")}
    recs = []
    for r in rows:
        t1 = time.time()
        g, s = r["group"], r["subhalo"]
        centre = np.asarray(g["GroupPos"], np.float64) / 1e3                    # ckpc/h -> cMpc/h
        half = max(float(half_side_r500) * float(g["Group_R_Crit500"]),
                   1.001 * float(r200_cover) * float(g["Group_R_Crit200"])) / 1e3
        raw = read_cube(idx, cells, centre, half, "gas", gas_fields)
        out = {"gas": canonical_gas_arrays(raw, centre, idx, conv)}
        attrs: dict = {}
        for kind in kinds:
            pt = KIND_PTYPE[kind]
            p = read_cube(idx, cells, centre, half, kind, kind_fields[kind])
            fpos = _phys(f0[f"PartType{pt}/Coordinates"], a, h) if f"PartType{pt}" in f0 else conv["Coordinates"]
            out[kind] = {"pos_kpc": (centre + p["Coordinates"]) * fpos / 3.085678e21}
            if "Mass" in p:
                fm = _phys(f0[f"PartType{pt}/Mass"], a, h) if f"PartType{pt}" in f0 else conv["Mass"]
                out[kind]["mass_msun"] = np.asarray(p["Mass"], np.float64) * fm / MSUN_G
            else:
                attrs[f"{kind}_mass_msun_const"] = float(idx.mass_table[pt]) * 1e10 / h
        ckpc = 1e3 * a / h
        r500 = float(g["Group_R_Crit500"]) / 1e3
        report = {"shape": "box", "half_side_kpc": half * ckpc, "half_side_r500c": half / r500,
                  "r500c_kpc": r500 * ckpc, "n_cube": int(len(out["gas"]["mass_msun"])), "n_lowres_gas_dropped": 0,
                  "coarse_gas_kept": True, "n_coarse": 0, "box_source": "eagle-hash-hdfstream"}
        attrs.update({"centre_kpc": centre * ckpc, "centre_native": centre, "r500c_kpc": r500 * ckpc,
                      "m500c_msun": float(g["Group_M_Crit500"]) * 1e10 / h,
                      "r200c_kpc": float(g["Group_R_Crit200"]) / 1e3 * ckpc,
                      "m200c_msun": float(g["Group_M_Crit200"]) * 1e10 / h,
                      "half_side_r500c": half / r500, "half_side_kpc": half * ckpc, "complete": True,
                      "cutout_report_json": json.dumps(report),
                      "bulk_vel_kms": np.asarray(g["GroupVel"], np.float64) / a})
        file_attrs = {"sim": sim, "snap": int(snap), "z": idx.z, "a": a, "h": h, "msun_in_g": MSUN_G,
                      "family": "eagle", "box_size_ckpc_h": idx.box * 1e3,
                      "remote_snapshot": f"{SERVER}:/{RUN}/snapshot_{int(snap):03d}_{SNAP_TAGS[int(snap)]}",
                      "halo_catalogue": "Subfind of the TNG-format conversion (tng_format_conversions)"}
        e = cube_io.write_halo(out_path, int(r["halo_id"]), out, attrs, file_attrs=file_attrs, elements=ELEMENTS,
                               catalog={"group": g, "subhalo": s})
        recs.append({"halo_id": int(r["halo_id"]), "m500c_msun": attrs["m500c_msun"], "counts": e["counts"],
                     "seconds": round(time.time() - t1, 1)})
        echo(f"eagle snap {snap} halo {r['halo_id']} M500c {attrs['m500c_msun']:.2e}: {e['counts']} in "
             f"{recs[-1]['seconds']} s, {idx.requests} requests so far")
    return recs

def main(argv: list[str] | None = None) -> int:
    """``python -m tngxmm.ingest.eagle_remote <snap> <cube_root> <index_dir> --rows rows.json``: the halo rows
    are written beforehand by :func:`tngxmm.ingest.simba_remote.halo_rows` (an environment with illustris_python),
    since the streaming environment has only numpy, h5py and hdfstream."""
    import argparse

    import numpy as np

    from tngxmm.ingest import cube_io

    p = argparse.ArgumentParser(prog="python -m tngxmm.ingest.eagle_remote")
    p.add_argument("snap", type=int)
    p.add_argument("cube_root")
    p.add_argument("index_dir")
    p.add_argument("--rows", required=True, help="JSON list of {halo_id, group, subhalo} (native catalogue rows)")
    p.add_argument("--half", type=float, default=5.66)
    p.add_argument("--r200-cover", type=float, default=3.0)
    p.add_argument("--sim", default="eagle")
    a = p.parse_args(argv)
    rows = json.loads(Path(a.rows).read_text())
    for r in rows:
        for part in ("group", "subhalo"):
            r[part] = {k: np.asarray(v) for k, v in r[part].items()}
    ingest_snapshot(a.snap, rows, cube_io.cube_path(a.cube_root, a.sim, a.snap), a.index_dir, half_side_r500=a.half,
                    r200_cover=a.r200_cover, sim=a.sim, echo=lambda m: print(m, flush=True))
    return 0
