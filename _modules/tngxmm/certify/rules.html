Source code for tngxmm.certify.rules

"""Pre-registered certification rules, applied to the analysis tables.

No threshold lives in this module. A rule is a ``params:certify.rules.rules`` entry:

    id:        {design: "DESIGN §4 R3'", table: mock_fits, filter: {tag: v4s1, qc_excluded: false},
                column: ln_br_1, test: precision, max_halfwidth95: 0.07, ...}

The tables are those of the ``analysis`` pipeline: ``mock_fits``, ``real_fits``, ``closure_fits``. Every statistic
is the median of a column over the filtered rows, with a bootstrap over rows, or over ``group_by`` (one value per
group, e.g. per cluster) when given, or, with ``resample_by`` (a column such as ``halo_group``), the median of all
rows with a bootstrap that resamples whole groups (every view of a drawn halo is kept;
:func:`tngxmm.donors.halo_groups.halo_bootstrap`). ``group_by`` and ``resample_by`` exclude each other;
``resample_by`` applies to the median-based tests and to ``consistency`` only, and a ``fraction`` or ``flag_ratio``
rule that names it raises. With ``resample_by`` a row without a group label raises, and fewer than two groups give
no interval: the rule is INCOMPLETE, whatever ``min_units``. ``n_boot`` (default 10000), ``seed`` and the
catastrophic band come from the params' top level.

Tests (each returns PASS, FAIL, INCONCLUSIVE or REPORT; any test returns INCOMPLETE when its rows are empty, so a
missing input never passes or fails a rule):
- ``interval3``: PASS if the 95% interval of the median lies inside ``pass_inside`` (ln units, or ``ratio: true``
  to give ratios); FAIL if it lies entirely outside ``fail_outside``; else INCONCLUSIVE. Optional ``max_fail_frac``
  (failed fits of the filter's rows) and ``max_catastrophic``.
- ``window``: PASS if the median itself (not its interval) lies inside ``pass_inside``, else FAIL: the form of
  DESIGN v2's closure rule R4 (|median - 1| <= 0.08, every redshift median in [0.85, 1.15]).  Optional
  ``max_fail_frac`` and ``max_catastrophic`` as for ``interval3``.
- ``precision``: PASS if the 95% half-width is at most ``max_halfwidth95`` and, if set, the catastrophic fraction is at
  most ``max_catastrophic``; else FAIL.  With ``max_blocks`` (DESIGN v4), a half-width above the limit while fewer
  than ``max_blocks`` blocks of views have been added (``blocks``, raised by one with each block) is INCOMPLETE:
  the sample grows, the rule is read again, and only a miss after the last block FAILs.
- ``equivalence``: like ``interval3`` on a ``scale`` x the median of a PAIRED difference (``pair: {key: [...],
  minus: {filter}}``: the column of the rule's rows minus the same column of the ``minus`` rows on the same key).
- ``fraction``: PASS if the fraction of rows where ``flag`` is true is at most ``max_frac``.
- ``consistency``: two medians (``a`` and ``b``, each {table, filter, column, group_by?}) agree within the quadrature
  sum of their 95% half-widths: CONSISTENT / INCONSISTENT (a REPORT).
- ``report``: the median and interval only.
- ``tail``: PASS if the catastrophic fraction (rows outside the catastrophic band) is at most ``max_catastrophic``;
  else FAIL.
- ``flag_ratio``: the ``flags`` rates of two filters agree within ``max_factor`` (PASS / FAIL), or differ by at most
  ``max_abs_diff`` (a flag rare on both sides cannot fail on the factor); a rate is the Jeffreys estimate
  (k + 1/2) / (n + 1), so a flag never seen on one side gives a finite ratio; ``unique`` de-duplicates the rows of a
  side on those columns, and a side with fewer than ``min_n`` fits makes the rule INCOMPLETE.

The median-based tests (``interval3``, ``window``, ``precision``, ``equivalence``, ``report``, ``tail``) with ``min_units`` are
INCOMPLETE while their statistic rests on fewer units (rows, or groups with ``group_by`` or ``resample_by``): the
sample size the rule is designed for, so a partial or toy run never passes or fails a gate.  ``consistency``,
``fraction`` and ``flag_ratio`` do not read ``min_units``.  A ``min_units`` counts the units of the rule as written:
R3p's 111 (``conf/base/parameters/certify.yml``) counts views, one per cluster in its sample, and must be derived again
in halos before that rule takes ``resample_by``.
"""
from __future__ import annotations

import math
from typing import Any

from tngxmm.cites import collect_cites


def _filter(df, flt: dict | None):
    for k, v in (flt or {}).items():
        if k not in df.columns:
            raise KeyError(f"filter column {k!r} not in the table (columns: {sorted(df.columns)[:20]}...)")
        df = df[df[k].isin(v)] if isinstance(v, list) else df[df[k] == v]
    return df

def boot_median(x, groups=None, n: int = 10000, seed: int = 20261015, *, units=None) -> tuple:
    """(median, lo68, hi68, lo95, hi95, n_units) of the median, bootstrap over rows or over groups (group medians);
    with ``units`` (one label per row, e.g. the halo group) the median of all rows, resampling whole units
    (:func:`tngxmm.donors.halo_groups.halo_bootstrap`; ``n_units`` = units with a finite value; a NaN interval below
    two units; a row without a label raises ``ValueError``)."""
    import numpy as np
    import pandas as pd

    if units is not None:
        if groups is not None:
            raise ValueError("group_by and resample_by exclude each other")
        from tngxmm.donors.halo_groups import halo_bootstrap

        r = halo_bootstrap(x, units, n_boot=n, seed=seed)
        return (r["value"], r["lo68"], r["hi68"], r["lo95"], r["hi95"], r["n_groups"])
    x = np.asarray(x, float)
    ok = np.isfinite(x)
    if groups is not None:
        v = pd.Series(x[ok]).groupby(np.asarray(groups)[ok]).median().to_numpy()
    else:
        v = x[ok]
    if len(v) == 0:
        return (float("nan"),) * 5 + (0,)
    rng = np.random.default_rng(seed)
    b = np.median(v[rng.integers(0, len(v), (n, len(v)))], axis=1)
    return (float(np.median(v)), *[float(q) for q in np.percentile(b, [16, 84, 2.5, 97.5])], int(len(v)))




def _band(rule, key, ratio):
    lo, hi = rule[key]
    return (math.log(lo), math.log(hi)) if ratio else (float(lo), float(hi))


def _catastrophic(x, band):
    import numpy as np

    x = np.asarray(x, float)
    x = x[np.isfinite(x)]
    lo, hi = math.log(band[0]), math.log(band[1])
    return float(((x < lo) | (x > hi)).mean()) if len(x) else float("nan")

def evaluate(tables: dict[str, Any], params: dict[str, Any]) -> dict[str, Any]:
    """Every rule of ``params['rules']`` on ``tables`` -> {id: {verdict, numbers..., design}}."""
    n, seed = int(params.get("n_boot", 10000)), int(params.get("seed", 20261015))
    cat_band = params.get("catastrophic", [0.67, 1.5])
    out = {}
    for rid, r in params["rules"].items():
        t = tables[r["table"]] if "table" in r else None
        rows = _filter(t, r.get("filter")) if t is not None else None
        res: dict[str, Any] = dict(design=r.get("design", ""), test=r["test"])
        test = r["test"]
        if test == "consistency":
            stats = {}
            for side in ("a", "b"):
                s = r[side]
                d = _filter(tables[s["table"]], s.get("filter"))
                g = d[s["group_by"]].to_numpy() if s.get("group_by") else None
                u = d[s["resample_by"]].to_numpy() if s.get("resample_by") else None
                stats[side] = boot_median(d[s["column"]], g, n, seed, units=u)
            a, b = stats["a"], stats["b"]
            half = math.hypot((a[4] - a[3]) / 2, (b[4] - b[3]) / 2)
            v = ("INCOMPLETE" if not (a[5] and b[5]) or half != half else
                 "CONSISTENT" if abs(a[0] - b[0]) < half else "INCONSISTENT")
            res.update(a=a, b=b, verdict=v)
            out[rid] = res
            continue
        if test in ("fraction", "flag_ratio") and (r.get("resample_by") or any(
                isinstance(r.get(s), dict) and r[s].get("resample_by") for s in ("a", "b"))):
            raise ValueError(f"rule {rid}: resample_by does not apply to the test {test!r} (it reads no interval)")
        if test == "fraction":
            f = float(rows[r["flag"]].astype(bool).mean()) if len(rows) else float("nan")
            res.update(fraction=f, n=int(len(rows)),
                       verdict="INCOMPLETE" if not len(rows) else "PASS" if f <= float(r["max_frac"]) else "FAIL")
            out[rid] = res
            continue
        if test == "flag_ratio":
            # one row per unit (``unique``: e.g. one per fit when a table has a row per fit and reference);
            # Jeffreys rates (k + 1/2) / (n + 1); a flag passes when its rates agree within ``max_factor``
            # OR differ by at most ``max_abs_diff`` (a flag rare on both sides cannot fail on the factor of
            # two tiny rates, whose ratio would only reflect the two sample sizes)
            rates, sizes, failing = {}, {}, []
            for side in ("a", "b"):
                d = _filter(tables[r[side]["table"]], r[side].get("filter"))
                if r[side].get("unique"):
                    d = d.drop_duplicates(subset=r[side]["unique"])
                sizes[side] = len(d)
                rates[side] = {fl: (float(d[fl].astype(bool).sum()) + 0.5) / (len(d) + 1.0) for fl in r["flags"]}
            if min(sizes.values()) < max(1, int(r.get("min_n", 1))):
                res.update(rates=rates, sizes=sizes, worst_factor=float("nan"), verdict="INCOMPLETE")
                out[rid] = res
                continue
            tol = float(r.get("max_abs_diff", 0.0))
            for fl in r["flags"]:
                ra, rb = rates["a"][fl], rates["b"][fl]
                if max(ra, rb) / min(ra, rb) > float(r["max_factor"]) and abs(ra - rb) > tol:
                    failing.append(fl)
            worst = max(max(rates["a"][fl], rates["b"][fl]) / min(rates["a"][fl], rates["b"][fl]) for fl in r["flags"])
            res.update(rates=rates, sizes=sizes, worst_factor=worst, failing=failing,
                       verdict="FAIL" if failing else "PASS")
            out[rid] = res
            continue
        col = rows[r["column"]]
        rb = r.get("resample_by")
        if test == "equivalence":
            key = r["pair"]["key"]
            other = _filter(t, r["pair"]["minus"])
            extra = [c for c in (r.get("group_by"), rb) if c and c not in key and c != r["column"]]
            m = rows[key + [r["column"]] + extra].merge(other[key + [r["column"]]], on=key, suffixes=("", "_minus"))
            col = (m[r["column"]] - m[r["column"] + "_minus"]) * float(r.get("scale", 1.0))
            rows = m
        g = rows[r["group_by"]].to_numpy() if r.get("group_by") else None
        u = rows[rb].to_numpy() if rb else None
        st = boot_median(col, g, n, seed, units=u)
        res.update(median=st[0], ci68=st[1:3], ci95=st[3:5], n_units=st[5])
        if not st[5] or st[5] < int(r.get("min_units", 1)) or st[3] != st[3] or st[4] != st[4]:
            res["verdict"] = "INCOMPLETE"            # no units, too few, or no interval (one halo)
            out[rid] = res
            continue
        if "max_catastrophic" in r or test in ("precision", "interval3", "tail", "window"):
            res["catastrophic"] = _catastrophic(col, cat_band)
        if t is not None and "max_fail_frac" in r:
            allrows = _filter(t, {k: v for k, v in (r.get("filter") or {}).items() if k != "qc_excluded"})
            res["fail_frac"] = float((allrows.status.astype(str) != "ok").mean()) if len(allrows) else float("nan")
        ratio = bool(r.get("ratio", False))
        lo95, hi95 = st[3], st[4]
        if test in ("interval3", "equivalence"):
            plo, phi = _band(r, "pass_inside", ratio)
            if "fail_outside" in r:
                flo, fhi = _band(r, "fail_outside", ratio)
                failed = hi95 < flo or lo95 > fhi
            else:
                failed = False
            v = "PASS" if (plo <= lo95 and hi95 <= phi) else ("FAIL" if failed else "INCONCLUSIVE")
        elif test == "window":
            plo, phi = _band(r, "pass_inside", ratio)
            v = "PASS" if plo <= st[0] <= phi else "FAIL"
        elif test == "precision":
            v = "PASS" if (hi95 - lo95) / 2 <= float(r["max_halfwidth95"]) else "FAIL"
            if v == "FAIL" and int(r.get("blocks", 0)) < int(r.get("max_blocks", 0)):
                # DESIGN v4: a width not met while blocks of views may still be added is not judged yet
                v = "INCOMPLETE"
                res["reason"] = (f"half-width above {r['max_halfwidth95']}: add a block of views "
                                 f"({int(r.get('blocks', 0))} of {int(r['max_blocks'])} added)")
        elif test == "tail":
            v = "PASS" if res["catastrophic"] <= float(r["max_catastrophic"]) else "FAIL"
        elif test == "report":
            v = "REPORT"
        else:
            raise ValueError(f"rule {rid}: unknown test {test!r}")
        if "max_catastrophic" in r and res["catastrophic"] > float(r["max_catastrophic"]) and v == "PASS":
            v = "FAIL"
        if "max_fail_frac" in r and res.get("fail_frac", 0.0) > float(r["max_fail_frac"]):
            v = "FAIL"
        res["verdict"] = v
        out[rid] = res
    return out

def gate(verdicts: dict[str, Any], params: dict[str, Any]) -> dict[str, Any]:
    """The launch gate: OPEN when every rule listed in ``params['gate']`` has verdict PASS.  With ``params['hse_claim']``
    (DESIGN v4) also the verdict of the hydrostatic-bias claim, which is not part of the gate: PASS when every rule it
    lists passes, FAIL when one fails, else INCOMPLETE."""
    need = list(params.get("gate", []))
    missing = [k for k in need if k not in verdicts]
    closed = [k for k in need if verdicts.get(k, {}).get("verdict") != "PASS"]
    out = dict(gate="OPEN" if not closed and not missing else "CLOSED", closed_by=closed, missing=missing,
               design=params.get("design", ""))
    if params.get("hse_claim"):
        words = [verdicts.get(k, {}).get("verdict", "INCOMPLETE") for k in params["hse_claim"]]
        out["hse_claim"] = ("FAIL" if "FAIL" in words else "PASS" if all(w == "PASS" for w in words)
                            else "INCOMPLETE")
    return out

def report_markdown(verdicts: dict[str, Any], g: dict[str, Any]) -> str:
    """The certification report: one line per rule with its numbers, its DESIGN section and its verdict."""
    lines = [f"# Certification report ({g.get('design', '')})", "", f"**Launch gate: {g['gate']}**"
             + (f" (closed by {', '.join(g['closed_by'])})" if g["closed_by"] else "")
             + (f"; hydrostatic-bias claim: **{g['hse_claim']}**" if "hse_claim" in g else ""), ""]
    for rid, v in verdicts.items():
        if "median" in v:
            nums = (f"median {math.exp(v['median']):.3f} [95%: {math.exp(v['ci95'][0]):.3f}, {math.exp(v['ci95'][1]):.3f}]"
                    f" (n={v['n_units']})" if v["median"] == v["median"] else "no data")
            if "catastrophic" in v:
                nums += f", catastrophic {v['catastrophic']:.0%}"
        elif "fraction" in v:
            nums = f"fraction {v['fraction']:.1%} (n={v['n']})"
        elif "worst_factor" in v:
            nums = (f"worst flag-rate factor {v['worst_factor']:.2f}"
                    + (f", failing {', '.join(v['failing'])}" if v.get("failing") else "")
                    + (f" (n = {v['sizes']['a']} / {v['sizes']['b']})" if v.get("sizes") else ""))
        else:
            nums = f"a {math.exp(v['a'][0]):.3f} vs b {math.exp(v['b'][0]):.3f}"
        lines.append(f"- **{rid}** ({v['design']}, {v['test']}): {nums} -> **{v['verdict']}**")
    return "\n".join(lines) + "\n"
