Source code for tngxmm.donor_tables.pipeline

"""Kedro pipeline ``donor_tables``: the per-donor tables of the data package (not a stage of ``__default__``).

Every mock view is observed under the conditions of one real XMM-Newton observation, its donor.  The
instrument stage reads what it needs about the donor from tables of the data package
(:mod:`tngxmm.instrument.donor_tables`); this pipeline BUILDS them, from the public archive products of the
donors (PPS images and exposure maps, event lists, flare time series) and the calibration files, one node per
table, into ``$TNGXMM_DATA/tables/`` (sha256 in its ``SHA256SUMS``)::

    obs.observations + donor_ebv_sfd ─build_donor_nh_total─> donor_nh_total   (Willingale et al. 2013)
    PPS images + exposure maps ─build_donor_background─> donor_background     (in-FOV flat level, live times)
    PPS MOS images + CCD maps ─build_donor_soft_ccd─> donor_soft_ccd           (anomalous-state MOS CCDs)
    donors.targets + donors.max_targets + donor_background + core offsets + active CCDs + obs_seed_lx ─build_donor_conditions─> donor_conditions

    obs_centres + obs_seed_lx ─fetch_pps_archive─> pps_archive_report      (event lists + flare series cache)
    CCF FWC files ─build_fwc_reference─> donor_fwc_reference               (work area, pickle; one per FWC epoch)
    catalogue + archive + FWC + soft CCDs ─measure_mos_background─> mos_background_status  (one JSON per donor)
    status + catalogue + FWC + soft CCDs ─build_mos_background─> donor_mos_background  (corner QPB, IN/OUT protons)
    catalogue + archive + FWC + soft CCDs ─build_qpb_corners─> donor_qpb_corners  (corner QPB per band, CCD, quadrant)

Parameters: ``params:donor_tables.*`` (``conf/base/parameters/pipeline.yml``).  Not part of ``__default__``: the
nodes read the archive products of every donor and some take hours.
"""

from __future__ import annotations

from typing import Any

from tngxmm.cites import adapts, collect_cites
from tngxmm.donor_tables import archive as archive_mod
from tngxmm.donor_tables import fwc as fwc_mod

def extra_donor_rows(observations: Any) -> Any:
    """Target-like rows (the columns :func:`tngxmm.instrument.conditions_donor.build_condition_table` reads) of the
    donors of ``observations`` that are not in the bank: such an observation is in no replica or maximal list, and
    its conditions are those of its row of the observation table."""
    import pandas as pd

    if "in_bank" not in observations:                  # a table of the bank alone
        return pd.DataFrame({"donor_id": pd.Series(dtype=str)})
    o = observations[~observations.in_bank.astype(bool) & observations.is_donor.astype(bool)]
    if o.empty:
        return pd.DataFrame({"donor_id": pd.Series(dtype=str)})
    return pd.DataFrame({
        "donor_id": o.obsid.astype(str).str.zfill(10), "ref_name": o.cluster_name, "ref_sample": o["sample"],
        "ref_z": o.z.astype(float), "exposure_pn_donor_s": o.exposure_pn_s, "exposure_mos1_donor_s": o.exposure_mos1_s,
        "exposure_mos2_donor_s": o.exposure_mos2_s, "nh_donor_1e22cm2": o.nh_1e22cm2, "filter_pn": o.filter_pn,
        "filter_mos1": o.filter_mos1, "filter_mos2": o.filter_mos2, "pn_mode": o.pn_mode,
        "submode_mos1": o.submode_mos1, "submode_mos2": o.submode_mos2}).reset_index(drop=True)

@adapts(build_condition_table)
def build_donor_conditions(targets: Any, max_targets: Any, background: Any, core_offsets: Any, active_ccds: Any,
                           seed_lx: Any, observations: Any = None) -> Any:
    """The condition table of every donor of EITHER target list (the replica list first, so a donor of both keeps its
    replica row): a physics donor of the maximal list needs its own row under the conditions rule ``nn``.  With
    ``observations`` the donors added to the bank later follow (:func:`extra_donor_rows`); the rows of the lists'
    donors do not depend on them.  The table then has the column ``lends``: the flag ``pool`` as the condition table
    computes it (the donor can lend its conditions: imaging modes, live time, a core offset).  A donor outside the
    bank (``in_bank`` False) lends its conditions only to the views paired with it (rule ``donor``, read through
    ``lends``): its ``pool`` flag is False, so the pool of the rule ``nn``, the spreads that standardise its
    distances and the donors a bank preset is held to are those of the bank.

    The table copies the core offsets (where the cluster sits on the detector) and the background levels, so its
    provenance records the core-offset and background tables it was built from and the centre table the core offsets
    were computed at (from the core-offset table's own provenance); the conditions donor reads the package copy only
    when all three are the current ones (:func:`tngxmm.instrument.donor_tables.check_table`).  A core-offset table read
    from a file must itself be current (:func:`tngxmm.donors.build_core_offsets.check_core_offsets_table`)."""
    import pandas as pd

    from tngxmm.core.datasets import PROVENANCE_ATTR, table_path
    from tngxmm.donor_tables.provenance import attach, recorded_input
    from tngxmm.donors.build_core_offsets import check_core_offsets_table
    from tngxmm.instrument.conditions_donor import MEASURE_VERSION

    core_file = getattr(core_offsets, "attrs", {}).get("filepath")
    if core_file:
        check_core_offsets_table(table_path(core_offsets))
    lists = [targets, max_targets] + ([extra_donor_rows(observations)] if observations is not None else [])
    t = build_condition_table(pd.concat(lists, ignore_index=True), background, core_offsets, active_ccds, seed_lx)
    inputs = {"targets": targets, "max_targets": max_targets, "background": background, "core_offsets": core_offsets,
              "active_ccds": active_ccds, "seed_lx": seed_lx}
    if observations is not None:
        t["lends"] = t["pool"].astype(bool)
        if "in_bank" in observations:
            outside = set(observations.obsid[~observations.in_bank.astype(bool)].astype(str).str.zfill(10))
            t.loc[t["obsid"].astype(str).str.zfill(10).isin(outside), "pool"] = False
        inputs["observations"] = observations
    attach(t, "tngxmm.instrument.conditions_donor.build_condition_table", inputs,
           n_rows=len(t), n_pool=int(t["pool"].astype(bool).sum()), measure_version=MEASURE_VERSION)
    t.attrs[PROVENANCE_ATTR]["inputs"]["centres"] = recorded_input(core_file, "centres")
    return t

@adapts(nh_total_table)
def build_donor_nh_total(observations: Any, ebv: Any) -> Any:
    """The total Galactic column of every donor's field: N_HI of the observation table (HI4PI) with the SFD
    E(B-V) at the donor's position (``donor_ebv_sfd``; for the observations added to the bank the column of
    ``obs/catalogue/inputs_extra/sky_columns.csv``, from the same service) through Willingale et al. (2013).  The table carries its
    provenance (the sha256 of the observation and reddening tables, the commit), which the instrument stage checks
    (:func:`tngxmm.instrument.donor_tables.check_table`)."""
    import pandas as pd

    from tngxmm.donor_tables.provenance import attach
    from tngxmm.obs.extra_observations import load_extra_inputs

    t = observations[["obsid", "nh_1e22cm2"]].rename(columns={"nh_1e22cm2": "nh_hi_1e22"})
    e = ebv[["obsid", "ebv_sfd"]].assign(obsid=ebv.obsid.astype(str).str.zfill(10))
    extra = load_extra_inputs().get("extra_sky")       # the observations added to the bank: E(B-V) kept in the package
    inputs = {"observations": observations, "ebv_sfd": ebv}
    if extra is not None:
        from tngxmm.obs.extra_observations import EXTRA_FILES, INPUTS_EXTRA_DIR

        inputs["ebv_sfd_extra"] = INPUTS_EXTRA_DIR / EXTRA_FILES["extra_sky"]
        x = extra[["obsid", "ebv_sfd"]].assign(obsid=extra.obsid.astype(str).str.zfill(10))
        e = pd.concat([e, x[~x.obsid.isin(set(e.obsid))]], ignore_index=True)
    t = t.assign(obsid=t.obsid.astype(str).str.zfill(10)).merge(e, on="obsid", how="left")
    if t.ebv_sfd.isna().any():
        raise ValueError(f"no E(B-V) for {int(t.ebv_sfd.isna().sum())} donors of the observation table")
    return attach(nh_total_table(t), "tngxmm.instrument.galactic_column.build_table", inputs, n_rows=len(t))

def fetch_pps_archive(centres: Any, seed_lx: Any, params: dict[str, Any]) -> Any:
    """Event lists and flare series of every donor of the catalogue that the PPS archive cache lacks, fetched from
    the public archive; the report (per donor: listed, cached, fetched, failed products, and whether the listing
    was skipped on a recorded negative result).  The report carries the run's provenance (stamps, the archive
    settings, the counts of :func:`~tngxmm.donor_tables.archive.coverage` and the skipped donors), which the
    dataset writes next to it (``pps_archive_report.provenance.json``).  ``params['refetch_none']`` (default false)
    asks the archive again about the recorded negative results."""
    from tngxmm.core.datasets import PROVENANCE_ATTR
    from tngxmm.core.provenance import provenance_stamp

    ids = donor_catalogue(centres, seed_lx).obsid
    kinds, refetch = list(params["kinds"]), params.get("refetch_none", False)
    report = archive_mod.fetch_archive(ids, params["root"], params["base_url"], kinds, params["streams"],
                                       refetch_none=refetch)
    report.attrs[PROVENANCE_ATTR] = provenance_stamp(
        root=str(params["root"]), base_url=params["base_url"], kinds=kinds, refetch_none=refetch, coverage=archive_mod.coverage(report, kinds),
        skipped={k: archive_mod.skipped_donors(report, k) for k in kinds})
    return report

def measure_mos_background(centres: Any, seed_lx: Any, fwc_reference: dict[str, Any], archive_report: Any,

    """The MOS corner / IN-OUT measurements of the donors of the run (``shard`` of the parameters) that have no

    from tngxmm.instrument.donor_inout_sp import cxb_band_rates

    shard = params["shard"]
    cat = donor_catalogue(centres, seed_lx)
    status = mos_mod.measure_all(cat, fwc_reference, cxb_band_rates(params["ccf_dir"]), params["archive_root"],
                                 params["image_root"], params["work_dir"], params["cameras"], params["workers"], shard,
                                 left_out=anomalous_ccds_by_observation(soft_ccd))
    work.finish_shard(dict.fromkeys(status.obsid), list(cat.obsid), shard, "mos_background")

def build_mos_background(status: Any, centres: Any, seed_lx: Any, fwc_reference: dict[str, Any], soft_ccd: Any,
                         params: dict[str, Any]) -> Any:
    """The MOS background table of all donors from the stored results; refuses when a donor has none
    (``status`` orders the node after the measurements).  The table carries its provenance (the sha256 of the centre
    and seed tables the IN regions were drawn about, the cameras, the calibration the values are referred to, the
    commit), which the instrument stage checks against the current centre table."""
    from tngxmm.donor_tables.provenance import attach
    from tngxmm.instrument.donor_inout_sp import cxb_band_rates
    from tngxmm.instrument.donor_soft_ccd import anomalous_ccds_by_observation

    cxb = cxb_band_rates(params["ccf_dir"])
    cams = list(params["cameras"])
    left_out = anomalous_ccds_by_observation(soft_ccd)
    t = mos_mod.assemble_table(donor_catalogue(centres, seed_lx), fwc_reference, cxb, cams, params["archive_root"],
                               params["image_root"], params["work_dir"], left_out=left_out)
    # the FWC reference of each camera: its rule and, per epoch of working CCDs, the band conversions
    fwc = {c: mos_mod._reference_signature(fwc_reference[c]) for c in cams}
    return attach(t, "tngxmm.donor_tables.mos_background.measure_all and assemble_table",
                  {"centres": centres, "seed_lx": seed_lx, "soft_ccd": soft_ccd}, cameras=cams,
                  measure_version=mos_mod.MEASURE_VERSION,
                  archive_root=str(params["archive_root"]), image_root=str(params["image_root"]),
                  work=str(params["work_dir"]), ccf_dir=str(params["ccf_dir"]),
                  cxb_10_12kev_counts_s_arcmin2={c: float(cxb[c]) for c in cams}, fwc_reference=fwc,
                  fwc_rule=fwc_mod.reference_rule(fwc_reference, cams), n_donors=len(t))

def build_qpb_corners(centres: Any, seed_lx: Any, fwc_reference: dict[str, Any], archive_report: Any, soft_ccd: Any,
                      params: dict[str, Any]) -> Any:
    """The corner QPB table of every EPIC camera of the donors (:mod:`tngxmm.donor_tables.qpb_corners`): the donors
    without a current stored result in the work area are measured (resumable), then the table is assembled from the
    stored results of the catalogue's donors.  With ``shard`` of the parameters only that share of the donors is
    measured and the run ends with :class:`~tngxmm.donor_tables.work.ShardDone`, as the other measuring nodes; an
    unsharded run assembles the table.  ``archive_report`` orders the node after the archive fetch;
    ``soft_ccd`` gives the MOS CCDs in the anomalous state, which the scales leave out as the fit leaves them out of
    its images.  The table carries its provenance (the sha256 of the centre, seed and anomalous-CCD tables, the
    measurement version and the rule of the FWC reference the scales refer to, the commit), which the instrument
    stage checks (:func:`tngxmm.instrument.donor_tables.check_table`)."""
    from tngxmm.donor_tables.provenance import attach
    from tngxmm.instrument.donor_soft_ccd import anomalous_ccds_by_observation

    cams = list(params["cameras"])
    cat = donor_catalogue(centres, seed_lx)
    shard = params.get("shard")
    share = cat[cat.obsid.isin(mos_mod.select_shard(list(cat.obsid), work.check_shard(shard)))]
    qpb_mod.measure_all(share, fwc_reference, params["archive_root"], params["image_root"], params["work_dir"], cams,
                        int(params["workers"]), left_out=anomalous_ccds_by_observation(soft_ccd))
    work.finish_shard(dict.fromkeys(share.obsid), list(cat.obsid), shard, "qpb_corners")
    t = qpb_mod.assemble_table(params["work_dir"], cams)
    t = t[t.obsid.astype(str).isin(set(cat.obsid.astype(str)))].reset_index(drop=True)
    return attach(t, "tngxmm.donor_tables.qpb_corners.measure_all and assemble_table",
                  {"centres": centres, "seed_lx": seed_lx, "soft_ccd": soft_ccd}, cameras=cams,
                  measure_version=qpb_mod.MEASURE_VERSION, fwc_rule=fwc_mod.reference_rule(fwc_reference, cams),
                  archive_root=str(params["archive_root"]), image_root=str(params["image_root"]),
                  work=str(params["work_dir"]), n_donors=len(t))

def create_pipeline(**kwargs: Any) -> Any:
    """observations, archive products, calibration files -> the donor tables of the data package."""
    from tngxmm.cites import stage_node, stage_pipeline

    return stage_pipeline([
        stage_node(build_donor_nh_total, ["obs.observations", "donor_ebv_sfd"], "donor_nh_total",
                   name="build_donor_nh_total"),
        stage_node(fetch_pps_archive, ["obs_centres", "obs_seed_lx",
                                       "params:donor_tables.archive"], "pps_archive_report",
                   name="fetch_pps_archive"),
        stage_node(build_fwc_reference, "params:donor_tables.fwc_reference", "donor_fwc_reference",
                   name="build_fwc_reference"),
        stage_node(measure_mos_background, ["obs_centres", "obs_seed_lx",
                                            "donor_fwc_reference", "pps_archive_report", "donor_soft_ccd",
                                            "params:donor_tables.mos_background"], "mos_background_status",
                   name="measure_mos_background"),
        stage_node(build_mos_background, ["mos_background_status", "obs_centres",
                                          "obs_seed_lx", "donor_fwc_reference", "donor_soft_ccd",
                                          "params:donor_tables.mos_background"], "donor_mos_background",
                   name="build_mos_background"),
        stage_node(build_qpb_corners, ["obs_centres", "obs_seed_lx", "donor_fwc_reference", "pps_archive_report",
                                       "donor_soft_ccd", "params:donor_tables.qpb_corners"], "donor_qpb_corners",
                   name="build_qpb_corners"),
        stage_node(build_background, ["obs_centres", "obs_seed_lx", "params:donor_tables"],
                   "donor_background", name="build_donor_background"),
        stage_node(build_soft_ccd, ["obs_centres", "obs_seed_lx", "donors.core_offsets",
                                    "params:donor_tables"], "donor_soft_ccd", name="build_donor_soft_ccd"),
        stage_node(build_donor_conditions, ["donors.targets", "donors.max_targets", "donor_background", "donors.core_offsets",
                                           "donor_active_ccds", "obs_seed_lx", "obs.observations"],
                   "donor_conditions", name="build_donor_conditions"),
    ], NAMESPACE)
