Source code for tngxmm.core.diskguard

"""Writes that survive a full disk: atomic files, completeness tests, the disk-full stop.

A production run can hit the quota in the middle of any write.  Three rules keep such a run resumable,
and every stage driver (generation pool, SAS queue, fit batch) follows them through this module:

* a product is written under a temporary name in its own directory (``.<name>.<pid>.tmp``), synced and
  renamed (:func:`atomic_path`, :func:`atomic_open`, :func:`write_text`, :func:`write_json`), so a file with
  its final name is complete.  An item with several files gets its done-marker last, after the files
  were synced (:func:`sync_tree`), and loses the marker first when it is made again;
* a failure caused by the disk (``ENOSPC`` / ``EDQUOT``, or the message an external tool leaves:
  :func:`is_disk_error`, and :func:`failure_status`, which also asks the disk itself with a probe write) is
  not a failure of the item: its status is ``interrupted: disk`` (:data:`INTERRUPTED`), it is recorded
  nowhere and the next run makes it again;
* after the first such item the driver takes no new item (:func:`request_stop`, seen by the forked workers
  through :func:`stop_requested`), lets the running ones end and exits with :data:`EXIT_DISK` after one line
  starting with ``DISK FULL:`` (:func:`disk_message`).

The guard (:func:`check_space`, :class:`SpaceGuard`) runs before the first item and every few minutes.
``os.statvfs`` gives the free space of the file system, not of a user's quota, so the guard also writes
and removes a few MB in the output root (:func:`probe_write`): that write fails when the quota's hard limit
is reached.  Neither sees how much room is left under the quota.

Settings (environment; the defaults need no tuning): ``TNGXMM_DISK_PROBE_MB`` (8; 0 = no probe write),
``TNGXMM_DISK_MIN_FREE_GB`` (1, file-system free space) and ``TNGXMM_DISK_CHECK_S`` (120, the least time
between two checks of a running driver).
"""

from __future__ import annotations

import contextlib
import errno
import json
import os
import socket
import time
from collections.abc import Iterator
from pathlib import Path
from typing import Any

#: exit code of a driver stopped by a full disk (the value of ENOSPC); the job scripts pass it on
EXIT_DISK = 28
#: status prefix of an item stopped by a full disk: never written to a results table, made again on resume
INTERRUPTED = "interrupted: disk"
#: first words of the one log line a stopped driver prints
LOG_PREFIX = "DISK FULL:"

ENV_PROBE_MB = "TNGXMM_DISK_PROBE_MB"
ENV_MIN_FREE_GB = "TNGXMM_DISK_MIN_FREE_GB"
ENV_CHECK_S = "TNGXMM_DISK_CHECK_S"
DEFAULT_PROBE_MB = 8.0
DEFAULT_MIN_FREE_GB = 1.0
DEFAULT_CHECK_S = 120.0

_DISK_ERRNOS = (errno.ENOSPC, errno.EDQUOT)
#: what tools that do not pass the errno on write into their message or log (lower case)
_DISK_TEXT = ("no space left on device", "disk quota exceeded", "quota exceeded", "enospc", "edquot")

class DiskFull(OSError):
    """A write failed because the disk or the quota is full (errno ENOSPC)."""

    def __init__(self, message: str) -> None:
        """``message``: where and what failed."""
        super().__init__(errno.ENOSPC, message)
        self.message = message

    def __str__(self) -> str:
        return self.message

def text_says_disk(text: str | bytes | None) -> bool:
    """True when ``text`` (a message, the stderr or the log of a tool) holds a disk-full message."""
    if not text:
        return False
    if isinstance(text, bytes):
        text = text.decode(errors="replace")
    low = text.lower()
    return any(t in low for t in _DISK_TEXT)

def is_disk_error(exc: BaseException | None) -> bool:
    """True when ``exc``, or an exception it was raised from or while handling, is a disk-full error: an
    ``OSError`` with errno ENOSPC or EDQUOT, or any exception whose message says so (h5py and cfitsio raise
    without the errno)."""
    seen = 0
    while exc is not None and seen < 8:
        if isinstance(exc, OSError) and exc.errno in _DISK_ERRNOS:
            return True
        if text_says_disk(str(exc)):
            return True
        exc = exc.__cause__ or exc.__context__
        seen += 1
    return False

def interrupted_status(detail: Any = "") -> str:
    """``interrupted: disk: <detail>`` (one line, at most 150 characters of detail)."""
    d = " ".join(str(detail).split())[:150]
    return f"{INTERRUPTED}: {d}" if d else INTERRUPTED

def failure_status(exc: BaseException, root: str | Path | None = None) -> str | None:
    """The status of an item that failed with ``exc``: ``interrupted: disk: ...`` when the failure is the
    disk's, else None (the caller records its own error).  The disk is blamed when the exception says so
    (:func:`is_disk_error`) or, with ``root``, when a probe write there fails: an external tool that ran
    out of space often reports something else."""
    if is_disk_error(exc):
        return interrupted_status(f"{type(exc).__name__}: {exc}")
    if root is not None:
        try:
            probe_write(root)
        except DiskFull as e:
            return interrupted_status(f"{type(exc).__name__}: {exc} [{e}]")
        except OSError:
            pass
    return None

def tmp_name(path: str | Path) -> Path:
    """The temporary name of ``path``: ``.<name>.<pid>.tmp`` in the same directory."""
    p = Path(path)
    return p.with_name(f".{p.name}.{os.getpid()}.tmp")

def fsync_file(path: str | Path) -> None:
    """Flush ``path`` to the disk; a write the kernel had only buffered fails here, not later."""
    fd = os.open(str(path), os.O_RDONLY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)




def _fsync_dir(path: Path) -> None:
    try:
        fsync_file(path)
    except OSError:          # not every file system syncs a directory; the rename is done either way
        pass

@contextlib.contextmanager
def atomic_path(path: str | Path, fsync: bool = True) -> Iterator[Path]:
    """Yield a temporary path next to ``path`` for a writer that takes a file name; on a clean exit the
    file is synced and renamed to ``path``, on an exception it is removed and ``path`` is left as it was."""
    path = Path(path)
    tmp = tmp_name(path)
    try:
        yield tmp
        if fsync:
            fsync_file(tmp)
        os.replace(tmp, path)
    except BaseException:
        with contextlib.suppress(OSError):
            tmp.unlink()
        raise
    if fsync:
        _fsync_dir(path.parent)

@contextlib.contextmanager
def atomic_open(path: str | Path, mode: str = "w", fsync: bool = True, **kwargs: Any) -> Iterator[Any]:
    """``open(path, mode)`` through :func:`atomic_path`: the file appears under its name only when the block
    ended without an exception."""
    if "w" not in mode:
        raise ValueError("atomic_open writes a new file: mode 'w' or 'wb'")
    with atomic_path(path, fsync) as tmp, open(tmp, mode, **kwargs) as fh:
        yield fh
        fh.flush()

def write_text(path: str | Path, text: str, fsync: bool = True) -> Path:
    """Write ``text`` to ``path`` atomically."""
    with atomic_open(path, "w", fsync) as fh:

def copy_file(src: str | Path, dst: str | Path, fsync: bool = True) -> Path:
    """Copy ``src`` to ``dst`` atomically."""
    import shutil

    with atomic_path(dst, fsync) as tmp:
        shutil.copyfile(src, tmp)
    return Path(dst)

def sync_tree(root: str | Path) -> dict[str, int]:
    """Flush every regular file below ``root`` to the disk (links are not followed) and return
    ``{relative path: size}``.  Called before an item's done-marker is written, so that a write error the
    kernel deferred stops the item instead of leaving a marker next to a short file."""
    root = Path(root)
    sizes: dict[str, int] = {}
    for d, _dirs, files in os.walk(root):
        for f in sorted(files):
            p = Path(d) / f
            if p.is_symlink() or not p.is_file():
                continue
            fsync_file(p)
            sizes[str(p.relative_to(root))] = p.stat().st_size
    return sizes

def remove_stale_tmp(directory: str | Path, older_than_s: float = 0.0) -> list[str]:
    """Remove the temporary files (``.<name>.<pid>.tmp``) left in ``directory`` (not below it) by a writer
    that was killed, and return their names.  ``older_than_s``: leave younger files alone, for a directory
    other processes may be writing to; an item's own directory is cleaned with 0 when the item starts."""
    d = Path(directory)
    now = time.time()
    gone = []
    for p in d.glob(".*.tmp"):
        try:
            if p.is_file() and not p.is_symlink() and now - p.stat().st_mtime >= older_than_s:
                p.unlink()
                gone.append(p.name)
        except OSError:
            continue
    return gone

def fits_complete(path: str | Path, hdus: tuple[str | int, ...] = ()) -> bool:
    """True when the FITS file (plain or gzipped) opens, holds the HDUs named in ``hdus`` and every HDU's
    data can be read to its end.  A file cut by a full disk fails one of these."""
    import warnings

    from astropy.io import fits

    p = Path(path)
    try:
        if not p.is_file() or p.stat().st_size == 0:
            return False
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")           # astropy's own "may have been truncated": the answer is False
            with fits.open(str(p), memmap=False) as h:
                for k in hdus:
                    h[k]                              # noqa: B018 - KeyError / IndexError when absent
                for hdu in h:
                    info = hdu.fileinfo()
                    if info and not str(p).lower().endswith((".gz", ".ftz")):
                        if info["datLoc"] + info["datSpan"] > p.stat().st_size:
                            return False
                    hdu.data                          # noqa: B018 - reads (and unzips) the data
        return True
    except Exception:  # noqa: BLE001 - any failure to read is an incomplete file
        return False

def hdf5_complete(path: str | Path, names: tuple[str, ...] = ()) -> bool:
    """True when the HDF5 file opens (a truncated one does not) and holds the objects in ``names``."""
    import h5py

    try:
        with h5py.File(str(path), "r") as f:
            return all(n in f for n in names)
    except Exception:  # noqa: BLE001

def npz_complete(path: str | Path, names: tuple[str, ...] = ()) -> bool:
    """True when the ``.npz`` archive opens, passes its checksums and holds the arrays in ``names``."""
    import zipfile

    try:
        with zipfile.ZipFile(str(path)) as z:
            have = {n[:-4] if n.endswith(".npy") else n for n in z.namelist()}
            return z.testzip() is None and all(n in have for n in names)
    except Exception:  # noqa: BLE001
        return False

def csv_tail_complete(path: str | Path) -> bool:
    """True when the text file is empty or ends with a newline (its last line was written whole)."""
    p = Path(path)
    size = p.stat().st_size
    if size == 0:
        return True
    with open(p, "rb") as fh:
        fh.seek(size - 1)
        return fh.read(1) == b"\n"

def repair_csv_tail(path: str | Path) -> int:
    """Cut a last line without its newline off an append-only table (a row a full disk or a kill left
    half written) and return the number of bytes removed.  A missing file is left alone."""
    p = Path(path)
    if not p.is_file() or csv_tail_complete(p):
        return 0
    data = p.read_bytes()
    keep = data.rfind(b"\n") + 1
    with open(p, "r+b") as fh:
        fh.truncate(keep)
    return len(data) - keep

def existing_dir(path: str | Path) -> Path:
    """``path``, or its nearest parent that exists (the directory a probe of a root not made yet uses)."""
    p = Path(path).absolute()
    while not p.is_dir() and p != p.parent:
        p = p.parent
    return p

def free_gb(root: str | Path) -> float:
    """Free space [GB] of the file system that holds ``root`` (``os.statvfs``; not the quota)."""
    st = os.statvfs(existing_dir(root))
    return st.f_bavail * st.f_frsize / 1e9

def probe_write(root: str | Path, mb: float | None = None) -> None:
    """Write ``mb`` MB (default ``$TNGXMM_DISK_PROBE_MB``, 8) to a file in ``root``, sync it and remove it.
    :class:`DiskFull` when the write fails for lack of space or quota; other errors pass through.  This is
    the only test that sees an exhausted quota; it does not tell how much room is left."""
    mb = _env_float(ENV_PROBE_MB, DEFAULT_PROBE_MB) if mb is None else float(mb)
    if mb <= 0:
        return
    d = existing_dir(root)
    p = d / f".diskprobe.{socket.gethostname()}.{os.getpid()}.tmp"
    block = b"\xa5" * (1 << 20)
    left = int(mb * (1 << 20))
    try:
        with open(p, "wb") as fh:
            while left > 0:
                fh.write(block[:left])
                left -= len(block)
            fh.flush()
            os.fsync(fh.fileno())
    except OSError as e:
        if is_disk_error(e):
            raise DiskFull(f"a probe write of {mb:g} MB in {d} failed ({e.strerror or e})") from e
        raise
    finally:
        with contextlib.suppress(OSError):
            p.unlink()

def check_space(root: str | Path, *, min_free_gb: float | None = None, probe_mb: float | None = None) -> None:
    """:class:`DiskFull` when the file system of ``root`` has less than ``min_free_gb`` free
    (``$TNGXMM_DISK_MIN_FREE_GB``, 1) or a probe write there fails (:func:`probe_write`)."""
    need = _env_float(ENV_MIN_FREE_GB, DEFAULT_MIN_FREE_GB) if min_free_gb is None else float(min_free_gb)
    have = free_gb(root)
    if have < need:
        raise DiskFull(f"{have:.2f} GB free on the file system of {root}, {need:g} GB needed")
    probe_write(root, probe_mb)

class SpaceGuard:
    """:func:`check_space` on some roots, at most once per ``interval_s`` (``$TNGXMM_DISK_CHECK_S``, 120)."""

    def __init__(self, *roots: str | Path, interval_s: float | None = None) -> None:
        """``roots``: the directories the run writes to (one probe per root)."""
        self.roots = [Path(r) for r in roots if r]
        self.interval_s = _env_float(ENV_CHECK_S, DEFAULT_CHECK_S) if interval_s is None else float(interval_s)
        self._last = 0.0



[docs]
    def check(self, force: bool = False) -> None:
        """Check the roots if the interval has passed (or ``force``); :class:`DiskFull` when one is full.
        A root that cannot be probed for another reason (no permission, gone) is left to the write that
        will name it."""
        now = time.monotonic()
        if not force and now - self._last < self.interval_s:
            return
        self._last = now
        for r in self.roots:
            try:
                check_space(r)
            except DiskFull:
                raise
            except OSError:
                continue

def completed_or_tick(futures: Any, interval_s: float) -> Iterator[Any]:
    """The futures as they end (as ``concurrent.futures.as_completed``), and ``None`` whenever ``interval_s``
    passes without one ending: the driver checks the disk on a tick, so a disk that fills while every worker
    is busy for hours is noticed without waiting for the next item to end."""
    from concurrent.futures import FIRST_COMPLETED, wait

    pending = set(futures)
    while pending:
        done, pending = wait(pending, timeout=max(float(interval_s), 1.0), return_when=FIRST_COMPLETED)
        if not done:
            yield None
        yield from done
