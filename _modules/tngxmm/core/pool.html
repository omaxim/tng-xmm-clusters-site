Source code for tngxmm.core.pool

"""Campaign process pool: one task per halo, the gas-load semaphore, the resumable results CSV.

The pool holds the orchestration of a campaign: the task list, the gas-load
semaphore, the resume logic and the results-CSV writer.  The per-view generation and
its seeding belong to ``tngxmm.sim.generate_view``; the pool only schedules it.

Contract with the generation code:

* one task per HALO, submitted in descending-mass order (longest processing
  time first); the task function loops over its views in
  order and returns one row per view;
* processes are forked (``fork`` context) so the warmed APEC tables
  and responses are shared copy-on-write, and the gas-load semaphore created
  here is inherited by every worker;
* the pool never seeds and never draws: every random number of a view comes
  from the per-view streams created inside the task function, so results do
  not depend on the number of workers or on scheduling (tested);
* rows are written in completion order; compare result tables sorted by
  ``(halo_id, view)``, never bytewise.

The gas-load throttle: the raw FoF load of a massive TNG-Cluster halo peaks at
8-19 GB, and with one task per halo every giant would load at t = 0 and
exhaust the node (2026-07-08: ``BrokenProcessPool`` at 40 workers / 406 GB).
:func:`gas_load_slot` caps the number of CONCURRENT raw loads; the generation
code wraps its raw load in it::

    from tngxmm.core.pool import gas_load_slot
    with gas_load_slot():
        gas = load_halo_gas(...)

References
----------
pandas : McKinney 2010; The pandas development team (results CSV, resume).
numpy  : Harris et al. 2020 (float conversion of rows, change counting).
tngcluster : Nelson et al. 2024 (the halos the campaign iterates over).
"""

def gas_load_concurrency(spec: Any | None = None, environ: dict | None = None) -> int:
    """Number of concurrent raw gas loads: ``$GAS_LOAD_CONCURRENCY`` if set (the job
    scripts export it), else ``spec.runtime.gas_load_concurrency``, else 8."""
    env = os.environ if environ is None else environ
    if env.get(GAS_LOAD_ENV):
        n = int(env[GAS_LOAD_ENV])
    elif spec is not None:
        n = int(spec.runtime.gas_load_concurrency)
    else:
        n = DEFAULT_GAS_LOAD_CONCURRENCY
    if n < 1:
        raise ValueError(f"gas-load concurrency must be >= 1, got {n}")
    return n

def install_gas_semaphore(n: int, context: Any | None = None) -> Any:
    """Create the bounded gas-load semaphore in the PARENT, before the workers fork."""
    import multiprocessing

    global _GAS_SEM
    ctx = context or multiprocessing.get_context("fork")
    _GAS_SEM = ctx.BoundedSemaphore(int(n))
    return _GAS_SEM

def clear_gas_semaphore() -> None:
    """Forget the semaphore (after the pool has shut down)."""
    global _GAS_SEM
    _GAS_SEM = None

@contextlib.contextmanager
def gas_load_slot() -> Iterator[None]:
    """Hold one gas-load slot for the duration of a raw load (no-op without a pool)."""
    sem = _GAS_SEM
    if sem is None:
        yield
        return
    with sem:
        yield

def resolve_callable(ref: str | Callable) -> Callable:
    """``"package.module:function"`` -> the function (callables pass through)."""
    if callable(ref):
        return ref
    mod, _, name = str(ref).partition(":")
    if not name:
        raise ValueError(f"expected 'module:function', got {ref!r}")
    try:
        return getattr(importlib.import_module(mod), name)
    except (ImportError, AttributeError) as e:
        raise click.ClickException(
            f"{ref} is not available yet ({e})") from e

def _of_sim(tg: Any, sim: str | None) -> Any:
    """Rows of one simulation when the targets table names several (``sim`` column;
    the maximal list holds TNG-Cluster and TNG300 views)."""
    if sim is None or "sim" not in tg.columns:
        return tg
    return tg[tg["sim"].astype(str) == str(sim)]

def excluded_donors(spec: Any) -> dict[str, str]:
    """``{obsid (10 digits): reason}`` of the donors that cannot supply a generated view
    (``spec.runtime.exclude_donors``): the one rule of the generation, applied by ``campaign run``,
    the Kedro plan, the donor-table check and the realism selection and split."""
    from tngxmm.schemas.ids import normalise_obsid

    return {normalise_obsid(o): str(r) for o, r in getattr(spec.runtime, "exclude_donors", ())}

def donor_column(table: Any) -> str | None:
    """The column of a targets table that names the donor of a view (``donor_id``, or the legacy
    ``donor_obsid``), None when it has none."""
    return next((c for c in ("donor_id", "donor_obsid") if c in table.columns), None)

def split_excluded(table: Any, exclude: dict[str, str] | None) -> tuple[Any, Any]:
    """``(kept, excluded)`` rows of a targets table: ``excluded`` are the views of the donors in
    ``exclude`` (:func:`excluded_donors`).  Both keep the table's index; nothing is excluded
    when ``exclude`` is empty or the table names no donor."""
    col = donor_column(table)
    if not exclude or col is None:
        return table, table.iloc[0:0]
    from tngxmm.schemas.ids import normalise_obsid

    hit = table[col].map(lambda v: normalise_obsid(v) in exclude if v == v and v is not None else False)
    return table[~hit], table[hit]

@cites("pandas")
def read_excluded_rows(targets_csv: str | Path, snap: int, sim: str | None,
                       exclude: dict[str, str] | None) -> list[dict[str, Any]]:
    """The results rows of the views of one snapshot that are planned but not generated: one per
    excluded view, with the view-schema columns of its targets row, ``sim``, ``snap``, ``halo_id``,
    ``view`` and ``status`` = :func:`excluded_status` (the reason of the donor)."""
    import pandas as pd

    from tngxmm.schemas.ids import normalise_obsid

    tg = pd.read_csv(targets_csv, dtype={"donor_id": str, "ref_obsid": str, "mock_obsid": str, "view_id": str},
                     low_memory=False)
    tg = _of_sim(tg[tg["snap"] == snap], sim)
    _, drop = split_excluded(tg, exclude)
    if not len(drop):
        return []
    cols = read_replica_view_columns(drop, snap, sim)
    col = donor_column(drop)
    rows = []
    for _, r in drop.iterrows():
        key = (int(r["halo_id"]), int(r["view"]))
        rows.append({**cols.get(key, {}), "halo_id": key[0], "view": key[1], "snap": int(r["snap"]),
                     "sim": str(r["sim"]) if "sim" in drop.columns else sim,
                     "status": excluded_status(exclude[normalise_obsid(r[col])])})
    return rows

                 exclude: dict[str, str] | None = None) -> tuple[set, dict, dict]:
    """``(halo ids, {halo: {view: z_obs}}, {halo: {view: donor obsid}})`` (the third is filled when the CSV has a donor column, ``donor_id`` of the replica tables or the legacy ``donor_obsid``: :func:`donor_column`) of one snapshot
    of a targets CSV (``--targets-csv``; the donor obsid is zero-padded to 10).  ``exclude``
    (:func:`excluded_donors`): the views of these donors are left out; a halo all of whose views are
    left out is not in the halo ids."""
    import pandas as pd

    from tngxmm.schemas.ids import normalise_obsid

    tg = pd.read_csv(targets_csv)
    tg = tg[tg["snap"] == snap]
    if tg.empty:                                   # no view of this snapshot in the list
        return set(), {}, {}
    tg = _of_sim(tg, sim)
    tg, _ = split_excluded(tg, exclude)
    want = {int(h) for h in tg["halo_id"]}
    zfix: dict[int, dict[int, float]] = {}
    zdon: dict[int, dict[int, str]] = {}
    dcol = donor_column(tg)
    for _, r in tg.iterrows():
        zfix.setdefault(int(r["halo_id"]), {})[int(r["view"])] = float(r["z_obs"])
        if dcol is not None and r[dcol] == r[dcol]:  # the real cluster that lent its redshift (not NaN)
            # normalise_obsid = str(...).zfill(10), also when iterrows made it a float
            zdon.setdefault(int(r["halo_id"]), {})[int(r["view"])] = normalise_obsid(r[dcol])
    return want, zfix, zdon

def replica_package(row: Any, replica_index: bool = False) -> dict[str, Any]:
    """The donor package of one replica targets row (``targets_v4.csv``): the donor
    the replica assigned to this view, with its (jittered) per-camera live times,
    N_H, filters, PN mode and out-of-time fraction, in the key set of
    :func:`tngxmm.donors.draw_donor.draw_donor` plus ``filters`` and the MOS submodes.
    ``replica_index``: add ``replica_k``, the row's replica index; only the observing-conditions rule
    'nn' reads it (it picks the conditions donor of the view), and the package is written to obs_meta.json,
    so the other rules leave it out (their products are those of release 1.1).

    A row whose halo-snapshot has no physics donor (``paired`` False, release 1.2,
    :func:`tngxmm.donors.pair_projections.paired_value`) is refused: generating unpaired views is not supported yet."""
    import math

    from tngxmm.schemas.ids import normalise_obsid

    get = row.get if hasattr(row, "get") else (lambda k, d=None: getattr(row, k, d))
    flag = get("paired")
    if flag is not None:
        from tngxmm.donors.pair_projections import paired_value

        if not paired_value(flag):
            raise ValueError(f"replica row {get('view_id')}: its halo-snapshot has no physics donor (paired False); "
                             "generating unpaired views is not supported yet (release 1.2, E1)")

    def num(k):
        try:
            v = float(get(k))
        except (TypeError, ValueError):
            return None
        return v if math.isfinite(v) else None

    def txt(k):
        v = get(k)
        return v if isinstance(v, str) and v else None

    cams = {c: num(e) for c, e, _ in REPLICA_CAMERAS if num(e) is not None and num(e) > 0}
    if not cams:
        raise ValueError(f"replica row {get('view_id')}: no camera with a live time")
    m_ref = num("ref_m500c_cat_msun")
    real = {c: num(e) for c, e in REPLICA_REAL_EXPOSURE if num(e) is not None and num(e) > 0}
    return {"donor": txt("ref_name"), "sample": txt("ref_sample"),
            "obsid": normalise_obsid(get("donor_id")), "z_donor": num("ref_z"),
            "cameras_s": cams, **({"cameras_real_s": real} if real else {}), "nh_1e22": num("nh_1e22cm2"),
            "jittered": bool(get("exposure_jittered")),
            "m500_donor_1e14": (m_ref / 1e14 if m_ref is not None else None),
            "match_windows": None,
            "pn_submode": txt("pn_mode"), "pn_oot_frac": num("pn_oot_frac"),
            "filters": {c: txt(f) for c, _, f in REPLICA_CAMERAS if c in cams and txt(f)},
            "submode_mos1": txt("submode_mos1"), "submode_mos2": txt("submode_mos2"),
            "view_id": txt("view_id"),
            **({"replica_k": (int(num("replica_k")) if num("replica_k") else None)} if replica_index else {}),
            "source": "replica"}

def halo_projection_anchors(tg: Any, n_projections: int) -> dict[int, dict[str, str]]:
    """``{halo: {view (a string, a JSON key): anchor obsid}}`` of the halo-snapshots of ``tg`` (the rows of one
    simulation and snapshot of a targets table, read BEFORE any exclusion): the per-projection conditions draw of
    release 1.2 needs the anchors of every projection of a halo-snapshot, whichever of its views a run generates.

    With a ``conditions_anchors`` column (written by :func:`tngxmm.donors.pair_projections.pair_projections`) every row
    carries the whole map: the rows of one halo-snapshot must agree on it and with their own anchor
    (``conditions_anchor_id``, else ``donor_id``), so a list cut to some views still gives the full map.  Without it the
    map is built from the rows, which must then hold every view.  The map must cover exactly the views 0 ..
    ``n_projections`` - 1 of the projection scheme; ``ValueError`` otherwise, or for a row without an anchor."""
    from tngxmm.donors.pair_projections import ANCHORS_COLUMN, parse_anchor_map
    from tngxmm.schemas.ids import normalise_obsid

    want = set(range(int(n_projections)))
    col = "conditions_anchor_id" if "conditions_anchor_id" in tg.columns else "donor_id"
    stored = ANCHORS_COLUMN in tg.columns
    out: dict[int, dict[int, str]] = {}
    for _, r in tg.iterrows():
        h, v, a = int(r["halo_id"]), int(r["view"]), r[col]
        if a is None or (isinstance(a, float) and a != a) or not str(a).strip():
            raise ValueError(f"halo {h} view {v}: no {col} for the per-projection conditions draw")
        own = normalise_obsid(a)
        if stored:
            m = parse_anchor_map(r[ANCHORS_COLUMN])
            if h in out and out[h] != m:
                raise ValueError(f"halo {h}: its rows disagree on {ANCHORS_COLUMN}")
            if m.get(v) != own:
                raise ValueError(f"halo {h} view {v}: {ANCHORS_COLUMN} names {m.get(v)}, the row's {col} is {own}")
            out[h] = m
        else:
            out.setdefault(h, {})[v] = own
    for h, m in out.items():
        if set(m) != want:
            raise ValueError(f"halo {h}: the anchors cover views {sorted(m)}, the projection scheme has views "
                             f"0-{int(n_projections) - 1}; a target list that holds only some views needs the "
                             f"{ANCHORS_COLUMN} column (donors.pair_projections)")
    return {h: {str(v): m[v] for v in sorted(m)} for h, m in out.items()}

@cites("pandas")
def read_replica_packages(targets: str | Path | Any, snap: int,
                          sim: str | None = None, exclude: dict[str, str] | None = None,
                          replica_index: bool = False, projection_anchors: bool = False,
                          n_projections: int | None = None) -> dict[int, dict[int, dict]]:
    """``{halo: {view: donor package}}`` of one snapshot of a replica targets table
    (a path or a DataFrame); see :func:`replica_package` (``replica_index``).  ``exclude``: as in
    :func:`read_targets`.  ``projection_anchors`` (the per-projection conditions draw of release 1.2, which alone
    reads it; needs ``n_projections``, the views of the projection scheme): every package of a halo-snapshot also
    carries ``conditions_anchors``, the anchors of ALL projections of that halo-snapshot (:func:`halo_projection_anchors`,
    read before ``exclude``), so neither an excluded donor nor a list cut to some views changes the conditions donor
    of another view (:func:`tngxmm.instrument.conditions_donor.draw_projection_conditions`)."""
    import pandas as pd

    tg = (targets if isinstance(targets, pd.DataFrame)
          else pd.read_csv(targets, dtype={"donor_id": str, "ref_obsid": str, "conditions_anchor_id": str}))
    tg = tg[tg["snap"] == snap]
    if tg.empty:                                   # no view of this snapshot in the list
        return set(), {}, {}
    tg = _of_sim(tg, sim)
    anchors = None
    if projection_anchors:
        if n_projections is None:
            raise ValueError("projection_anchors needs n_projections (the views of the projection scheme)")
        anchors = halo_projection_anchors(tg, n_projections)         # every row of the halo-snapshot, before exclude
    tg, _ = split_excluded(tg, exclude)
    out: dict[int, dict[int, dict]] = {}
    for _, r in tg.iterrows():
        out.setdefault(int(r["halo_id"]), {})[int(r["view"])] = replica_package(r, replica_index)
    if anchors is not None:
        for h, views in out.items():
            for p in views.values():
                p["conditions_anchors"] = dict(anchors[h])
    return out

def read_replica_view_columns(targets: str | Path | Any, snap: int,
                              sim: str | None = None,
                              exclude: dict[str, str] | None = None) -> dict[tuple[int, int], dict[str, Any]]:
    """``{(halo_id, view): {column: value}}`` of :data:`REPLICA_VIEW_COLUMNS` for one
    snapshot of a replica targets table; columns absent from the table are left out.
    ``exclude``: as in :func:`read_targets`."""
    import math

    import pandas as pd

    tg = (targets if isinstance(targets, pd.DataFrame)
          else pd.read_csv(targets, dtype={"donor_id": str, "ref_obsid": str,
                                           "mock_obsid": str, "view_id": str}))
    tg = tg[tg["snap"] == snap]
    if tg.empty:                                   # no view of this snapshot in the list
        return set(), {}, {}
    tg = _of_sim(tg, sim)
    tg, _ = split_excluded(tg, exclude)
    cols = [c for c in REPLICA_VIEW_COLUMNS if c in tg.columns]
    # optional columns only where the table fills them (a table mixing simulations
    # leaves held_out_resolution empty on the TNG-Cluster and TNG300 rows)
    opt = [c for c in OPTIONAL_REPLICA_VIEW_COLUMNS if c in tg.columns and tg[c].notna().any()]
    cols += opt
    out: dict[tuple[int, int], dict[str, Any]] = {}
    for _, r in tg.iterrows():
        d: dict[str, Any] = {}
        for c in cols:
            v = r[c]
            if c == "donor_id" and isinstance(v, str):
                from tngxmm.schemas.ids import normalise_obsid

                v = normalise_obsid(v)
            elif c in ("rep", "zoom_target_id") and not (isinstance(v, float) and math.isnan(v)):
                v = int(v)
            elif c in ("fov_fit", "held_out_resolution") and isinstance(v, str):
                v = v.strip().lower() == "true"
            elif c in ("fov_fit", "held_out_resolution") and not (isinstance(v, float)
                                                                  and math.isnan(v)):
                v = bool(v)
            if c in opt and isinstance(v, float) and math.isnan(v):
                continue
            d[c] = v
        out[(int(r["halo_id"]), int(r["view"]))] = d
    return out

def safe_call(task_fn: Callable, task: dict) -> list[dict]:
    """Run one task; a crash becomes one ``fatal:`` row.  A crash caused by a full disk, and a task that
    starts after the run was stopped for it, give one ``interrupted: disk`` row instead
    (:mod:`tngxmm.core.diskguard`): the pool does not record it and the next run makes the halo."""
    if diskguard.stop_requested():
        return [{**task, "status": diskguard.interrupted_status("not started")}]
    try:
        return task_fn(task)
    except Exception as e:
        rec2 = dict(task)
        rec2["status"] = (diskguard.failure_status(e, task.get("out_root"))
                          or f"fatal: {traceback.format_exc().splitlines()[-1][:150]}")
        return [rec2]

class ResultsWriter:
    """Append-only results CSV (``csv.DictWriter`` with a flush per row).

    A row whose ``(halo_id, view)`` is already ``ok`` in the file is not written
    again.  The table is a log, not the record of what is done: a view is finished when its directory holds
    its done-marker (``obs_meta.json``), and a view whose row was lost is written again (``ok (cached)``) by
    the next run.  A last line a kill or a full disk left without its end is cut off when the file is
    opened; a row that cannot be written for lack of space raises :class:`tngxmm.core.diskguard.DiskFull`.
    ``stamp`` (e.g. spec hash, package version, git hash) is merged into
    every row and its keys are appended to the columns.  ``view_columns``
    (``{(halo_id, view): {column: value}}``, the replica's view-schema columns)
    fills those columns of a view's row; a key the row itself carries wins.
    """

    def __init__(self, csv_path: str | Path, columns: Iterable[str] = RESULT_COLUMNS,
                 stamp: dict | None = None,
                 view_columns: dict[tuple[int, int], dict[str, Any]] | None = None) -> None:
        """Open ``csv_path`` for appending (the header is written only for a new file).  ``columns``: the column order
        (default :data:`RESULT_COLUMNS`, plus the keys of ``stamp``); ``stamp``: values merged into every row (spec
        hash, version, commit); ``view_columns``: per-``(halo_id, view)`` extra columns.  Reads the views already
        ``ok`` or excluded in an existing file so that a restart does not repeat them."""
        self.path = Path(csv_path)
        self.stamp = dict(stamp or {})
        self.view_columns = view_columns or {}
        cols = list(columns)
        cols += [k for k in self.stamp if k not in cols]
        self.columns = cols
        diskguard.repair_csv_tail(self.path)        # a half-written last row would swallow the next one
        self.seen = read_done_keys(self.path)
        self.excluded = read_excluded_keys(self.path)
        new_file = not self.path.exists() or self.path.stat().st_size == 0
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._fh = open(self.path, "a", newline="")
        self._w = csv.DictWriter(self._fh, fieldnames=self.columns, extrasaction="ignore")
        if new_file:
            self._put(None)

    def _put(self, row: dict | None) -> None:
        """Write the header (``row`` None) or one row and flush; a full disk raises ``DiskFull``."""
        try:
            if row is None:
                self._w.writeheader()
            else:
                self._w.writerow(row)
            self._fh.flush()
        except OSError as e:
            if diskguard.is_disk_error(e):
                raise diskguard.DiskFull(f"results table {self.path}: {e}") from e
            raise

    def write(self, row: dict) -> bool:
        """Write ``row`` unless its view is already recorded; True if written."""
        key = (row.get("halo_id"), row.get("view"))
        if key in self.seen:
            return False
        vc = self.view_columns.get((_as_int(key[0]), _as_int(key[1])), {})
        full = {**vc, **row, **self.stamp}
        self._put({k: _plain(v) for k, v in full.items() if k in self.columns})
        self.seen.add(key)
        return True

    def write_excluded(self, row: dict) -> bool:
        """Record an excluded view (``status`` = :func:`excluded_status`) unless it is ``ok`` in the file or
        recorded with the same status; True if written.  The row is not added to the done keys: the view is
        generated when the exclusion is lifted."""
        key = (_as_int(row.get("halo_id")), _as_int(row.get("view")))
        if key in self.seen or self.excluded.get(key) == row["status"]:
            return False
        self._put({k: _plain(v) for k, v in {**row, **self.stamp}.items() if k in self.columns})
        self.excluded[key] = row["status"]
        return True

@cites("pandas")
def run_pool(tasks: list[dict], task_fn: str | Callable, *, workers: int,
             results_csv: str | Path, gas_load_concurrency: int = DEFAULT_GAS_LOAD_CONCURRENCY,
             columns: Iterable[str] = RESULT_COLUMNS, stamp: dict | None = None,
             n_views_per_task: int = len(VIEWS),
             echo: Callable[[str], Any] = print,
             view_columns: dict[tuple[int, int], dict[str, Any]] | None = None,
             excluded_rows: Iterable[dict[str, Any]] = ()
             ) -> dict[str, Any]:
    """Run ``task_fn(task) -> [row, ...]`` over ``tasks`` on ``workers`` forked processes.

    Rows go to ``results_csv`` as they complete (resumable: views already ``ok`` in
    the file are not written twice).  ``view_columns``: see :class:`ResultsWriter`.
    ``excluded_rows`` (:func:`read_excluded_rows`) are recorded first, once, with their status.
    Returns ``{n_views, n_done, n_ok, n_excluded, csv}``.

    A full disk stops the run (:mod:`tngxmm.core.diskguard`): the directory of ``results_csv`` is probed
    before the first task and every few minutes; at the first row with status ``interrupted: disk``, or
    when a probe or a write of the table fails, the tasks not yet started are cancelled, the running ones
    end, the interrupted rows are not written and ``DiskFull`` is raised.
    """
    import multiprocessing
    from concurrent.futures import ProcessPoolExecutor

    fn = resolve_callable(task_fn)
    ctx = multiprocessing.get_context("fork")
    install_gas_semaphore(gas_load_concurrency, ctx)
    # the views actually generated: a task with ``z_obs_fixed`` runs only the listed views (the views of excluded donors
    # are recorded as rows, not run), so the progress denominator leaves them out
    n_views = sum(len(t["z_obs_fixed"]) if t.get("z_obs_fixed") is not None else n_views_per_task for t in tasks)
    t0, done, n_ok, n_excl = time.time(), 0, 0, 0
    out_dir = Path(results_csv).parent
    out_dir.mkdir(parents=True, exist_ok=True)
    diskguard.remove_stale_tmp(out_dir, older_than_s=3600.0)     # temporary and probe files of a killed run
    guard = diskguard.SpaceGuard(out_dir)
    try:
        guard.check(force=True)              # before anything is computed
    except diskguard.DiskFull as e:
        raise diskguard.DiskFull(diskguard.disk_message("campaign run", 0, out_dir, e)) from e
    diskguard.install_stop(ctx)
    n_disk, why = 0, ""
    try:
        with ResultsWriter(results_csv, columns, stamp, view_columns) as w, \
                ProcessPoolExecutor(max_workers=workers, mp_context=ctx) as pool:
            n_excl = sum(w.write_excluded(r) for r in excluded_rows)
            futs = {pool.submit(safe_call, fn, t): t for t in tasks}

            def stop(reason: Any) -> None:       # no new task; the running ones end by themselves
                nonlocal why
                why = why or str(reason)
                diskguard.request_stop()
                for f in futs:
                    f.cancel()

            def check_disk() -> None:
                if not why:
                    try:
                        guard.check()
                    except diskguard.DiskFull as e:
                        stop(e)

            for fut in diskguard.completed_or_tick(futs, guard.interval_s):
                if fut is None:                  # nothing ended for a while: look at the disk all the same
                    check_disk()
                    continue
                if fut.cancelled():
                    z = futs[fut].get("z_obs_fixed")
                    n_disk += len(z) if z is not None else n_views_per_task
                    continue
                try:
                    rows = fut.result()
                except Exception as e:           # a worker that died (the pool is broken): the disk's doing?
                    disk = diskguard.failure_status(e, out_dir)
                    if disk is None:
                        raise
                    rows = [{**futs[fut], "status": disk}]
                for r in rows:  # one row per view
                    done += 1
                    if str(r.get("status", "")).startswith("ok"):
                        n_ok += 1
                    echo(progress_line(done, n_views, t0, r))
                    if isinstance(r.get("mem"), dict):   # core.membudget report of the task
                        echo(memory_line(r))
                    if diskguard.is_interrupted(r.get("status")):
                        n_disk += 1
                        stop(r["status"])
                        continue
                    try:
                        w.write(r)
                    except diskguard.DiskFull as e:      # the view is on disk; its row comes with the next run
                        n_disk += 1
                        stop(e)
                check_disk()
    finally:
        clear_gas_semaphore()
        diskguard.clear_stop()
    if why:
        raise diskguard.DiskFull(diskguard.disk_message("campaign run", n_disk, out_dir, why))
    echo(f"done: {n_ok}/{n_views} ok  -> {results_csv}")
    return {"n_views": n_views, "n_done": done, "n_ok": n_ok, "n_excluded": n_excl, "csv": str(results_csv)}

@cites("tngcluster", "pandas")
def run_campaign(spec: Any, snap: int, *, out_root: str | Path, targets_csv: str | None = None,
                 max_halos: int = 0, workers: int | None = None, dry_run: bool = False,
                 source: str | Path | None = None, dependencies: str = "package",
                 select_fn: str | Callable = DEFAULT_SELECT_FN,
                 task_fn: str | Callable = DEFAULT_TASK_FN,
                 warm_fn: str | Callable | None = DEFAULT_WARM_FN,
                 stamp: dict | None = None, echo: Callable[[str], Any] = print) -> dict[str, Any]:
    """One snapshot of a campaign (the per-view code lives in the task function).

    ``spec`` is a :class:`tngxmm.config.CampaignSpec`.  Halos come from ``select_fn``
    (signature ``(sim, snap, mass_cut, primary_only=..., source=...)``); a
    targets CSV restricts them and fixes ``z_obs`` per view; the warm-up runs once
    in the parent before the fork.  ``source`` (default ``spec.sim.base_path``) is
    the simulation output, or a fixture cutout; ``dependencies`` names the donors /
    instrument / photometry binding of the task function (``package``, or a factory
    ``module:function``).
    """
    sim = spec.sim.sim_name
    workers = int(workers or spec.runtime.workers)
    n_gas = gas_load_concurrency(spec)
    seed_offset = int(spec.seed_offset)
    no_dither_cap = not bool(spec.instrument.dither_cap)
    mass_cut = float(spec.sim.mass_cut_m500c_msun)
    echo(f"=== campaign {spec.name}  sim={sim} snap={snap} mass_cut={mass_cut:.1e} Msun "
         f"(physical) tvir_frac={spec.sim.tvir_frac} workers={workers} "
         f"gas_load_concurrency={n_gas} ===")
    if dependencies != "package" and ":" not in dependencies:
        raise click.ClickException("dependencies must be 'package' or a factory 'module:function'")
    source = str(source or spec.sim.base_path)
    from tngxmm.sim.generate_view import SimOptions

    _o = SimOptions.from_spec(spec)
    if _o.cube_store == "canonical" and spec.sim.source is not None:
        # a preset that names a descriptor: the halos of its canonical cubes (no native catalogue read)
        from tngxmm.ingest.canonical_source import select_cube_halos

        halos = select_cube_halos(_o.cube_root, sim, snap, mass_cut)
    else:
        halos = resolve_callable(select_fn)(sim, snap, mass_cut, primary_only=spec.sim.primary_only,
                                            source=source)
    if max_halos > 0:
        halos = halos[:max_halos]
    zfix = zdon = None
    exclude = excluded_donors(spec)
    excluded_rows: list[dict[str, Any]] = []
    if targets_csv:
        if not out_root:
            raise click.ClickException("--targets-csv requires --out-root")
        want, zfix, zdon = read_targets(targets_csv, snap, sim=sim, exclude=exclude)
        excluded_rows = read_excluded_rows(targets_csv, snap, sim, exclude)
        if excluded_rows:
            echo(f"  {len(excluded_rows)} views of excluded donors are not generated "
                 f"({sorted({r['status'] for r in excluded_rows})})")
        halos = [h for h in halos if h["halo_id"] in want]
        missing = want - {h["halo_id"] for h in halos}
        if missing:
            raise click.ClickException(
                f"--targets-csv: {len(missing)} halo_ids not in the halo selection for snap "
                f"{snap} (e.g. {sorted(missing)[:5]}); check the mass cut / primary-only")
    tasks = make_halo_tasks(halos, tvir_frac=spec.sim.tvir_frac, seed_offset=seed_offset,
                            no_dither_cap=no_dither_cap, drop_events=spec.runtime.drop_events,
                            zfix_by_halo=zfix, zdon_by_halo=zdon,
                            out_root=str(out_root), preset=spec.name, source=source,
                            dependencies=dependencies)
    if getattr(spec.sim, "z_obs_source", None) == "replica":
        # the replica assigns every view its donor: no random donor draw
        if not targets_csv:
            raise click.ClickException(f"preset {spec.name}: z_obs_source 'replica' needs "
                                       "--targets-csv (campaigns/<preset>/targets_<preset>.csv)")
        from tngxmm.instrument.conditions_donor import conditions_of
        from tngxmm.instrument.donor_tables import check_donor_tables

        # the replica index of a view is the package's only under the rule 'nn' (it picks the conditions donor); the
        # anchors of all projections of a halo-snapshot only under the per-projection draw (release 1.2), which reads them
        from tngxmm.sim.project_luminosity import n_views

        cs = conditions_of(spec)
        pk = read_replica_packages(targets_csv, snap, sim=sim, exclude=exclude,
                                   replica_index=cs.get("rule") == "nn",
                                   projection_anchors=bool(cs.get("per_projection")),
                                   n_projections=n_views(getattr(spec.sim, "projection", None)))

        check_donor_tables({str(p["obsid"]) for v in pk.values() for p in v.values()}, spec)
        for t in tasks:
            t["donor_packages"] = pk.get(t["halo_id"])
        view_columns = read_replica_view_columns(targets_csv, snap, sim=sim, exclude=exclude)
    else:
        view_columns = None
    n_per = len(spec.sim.views)
    echo(f"  {len(tasks)} halo tasks x {n_per} views")
    if dry_run:
        return {"n_views": len(tasks) * n_per, "n_done": 0, "n_ok": 0, "csv": None,
                "tasks": tasks}
    if warm_fn is not None:
        echo("Pre-warming APEC tables + camera responses (shared copy-on-write via fork)")
        resolve_callable(warm_fn)(spec)
    from tngxmm.sim.generate_view import result_columns   # tngref / truth columns of the spec

    columns = result_columns(spec)
    if view_columns is not None:
        # replica: the view-schema identity and donor columns (INTERFACES 4.1)
        optional = tuple(c for c in OPTIONAL_REPLICA_VIEW_COLUMNS
                         if any(c in d for d in view_columns.values()))
        columns = (*REPLICA_VIEW_COLUMNS, *optional, *columns[:-1], *REPLICA_ROW_COLUMNS,
                   columns[-1])
    return run_pool(tasks, task_fn, workers=workers, gas_load_concurrency=n_gas,
                    results_csv=results_csv_path(out_root, sim, snap), stamp=stamp,
                    columns=columns, n_views_per_task=n_per, echo=echo,
                    view_columns=view_columns, excluded_rows=excluded_rows)

def _stamp_for(spec: Any) -> dict[str, Any]:
    from tngxmm import __version__
    from tngxmm.core.provenance import git_hash

    return {"spec_hash": spec.spec_hash(), "package_version": __version__,
            "git_hash": git_hash() or ""}

@click.option("--workers", type=int, default=0, help="0 = spec.runtime.workers.")

@click.option("--cube-root", default=None, type=click.Path(file_okay=False),

                   "'canonical', runtime.cube_root; environment TNGXMM_CUBE_STORE / TNGXMM_CUBE_ROOT).")
@cites("click")
def run_command(preset: str, snap: tuple[int, ...], out_root: str, targets_csv: str | None,
                workers: int, max_halos: int, task_fn: str, source: str | None,
                dependencies: str, no_warm: bool, dry_run: bool, cube_root: str | None = None) -> None:
    """Generate the views of one or more snapshots with the process pool."""
    from tngxmm.config import get_preset

    if cube_root:                       # the workers inherit the environment (sim.generate_view)
        os.environ["TNGXMM_CUBE_STORE"] = "canonical"
        os.environ["TNGXMM_CUBE_ROOT"] = str(Path(cube_root).resolve())

    spec = get_preset(preset)
    stamp = _stamp_for(spec)
    for s in snap or tuple(spec.sim.snaps):
        click.echo(f"=== {spec.name} snap {s} === {time.strftime('%c')}")
        try:
            if not dry_run:                  # before the halo selection and the warm-up, which take minutes
                try:
                    diskguard.SpaceGuard(out_root).check(force=True)
                except diskguard.DiskFull as e:
                    raise diskguard.DiskFull(diskguard.disk_message("campaign run", 0, out_root, e)) from e
            run_campaign(spec, s, out_root=out_root, targets_csv=targets_csv,
                         max_halos=max_halos, workers=workers or None, dry_run=dry_run,
                         source=source, dependencies=dependencies, task_fn=task_fn,
                         warm_fn=None if no_warm else DEFAULT_WARM_FN, stamp=stamp,
                         echo=click.echo)
        except diskguard.DiskFull as e:      # a full disk: the later snapshots are not started either
            click.echo(str(e))
            raise SystemExit(diskguard.EXIT_DISK) from e
