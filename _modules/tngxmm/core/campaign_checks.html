"""Campaign acceptance checks C1-C17: ``tngxmm campaign check <tree>``.

The checks run on any campaign tree through the CampaignSpec preset (one try-block
per check; ``pts_sources.fits`` is read as the ASCII table it is, C6 is centred on
the dithered cluster).  Read-only; seconds to minutes on the login node.

Tree layout: ``<root>/results_<sim>_snap_<snap:03d>.csv`` and
``<root>/<sim>/<snap:03d>/<halo_id:07d>/view<v>/`` holding ``obs_meta.json``,
``image_merged.fits`` (header ``CLUSTX/CLUSTY``), ``image_merged_cov.fits``,
``pts_sources.fits``, optionally ``events_v<v>_abs.h5`` and ``percam/`` with
8 (``product_bands = 'pps-2-5'``) or 10 (the default ``'pps-1-5'``) IMAGE/EXPMAP files per simulated camera
(INTERFACES section 3) and, under ``instrument.qpb_model_images``, the model images of the fit's ESAS background
beside them (``P<obsid><code>{QPBEXP,SPMAP,CCDMAP[,QPBALT]}_<band>.FTZ``,
:func:`tngxmm.instrument.simulate_instrument.camera_model_images`).

=====  ================================================================  ======
check  rule                                                              scope
=====  ================================================================  ======
C1     every target view (targets CSV, else every row and view directory)  all
       has an ``ok`` row; no error left; no unexpected extra view
C2     products: kinds x bands per-camera files per simulated camera (the     sample
       conditions donor's cameras under the rule 'nn'; the kinds are
       IMAGE_ and EXPMAP, plus the background model images QPBEXP_,
       SPMAP_, CCDMAP_ under ``instrument.qpb_model_images`` and QPBALT_
       under ``'on+alt'``), required header
       keys (INTERFACES 3), ``image_merged.fits`` with CLUSTX/CLUSTY,
       coverage map, AGN list, ``obs_meta.json``; with ``sim.tngref_map``
       ``tngref_sb_map.fits``: every band, npix, header keys,
       ``PROVNCE = 'TEMETEXACT'`` and the row's ``VIEW_ID``; with
       ``sim.truth_profiles`` the halo's ``truth_profile.h5``: format,
       every dataset and attribute, shapes, snap/halo, M_tot(<r) monotonic,
       M_tot(<R500c) within 5 % of M500c
C3     provenance in ``obs_meta``: z_obs source, seed offset, dither cap,  sample
       z_eff_min as the preset; z_obs equal to the target's
C4     donor core offset applied: the applied offset equals the donor's   sample
       core offset (donor table; under the rule 'nn' the conditions
       donor's plus the pointing jitter, from conditions.json) and
       CLUSTX/Y is the optical axis plus that offset rotated by POSANGLE
C5     AGN sky differs from the reference campaign's same view             sample
C6     per aperture (R500, 1, 1.5 Mpc, centred on CLUSTX/CLUSTY):         sample
       ``lx_<ap>_flag == 'uncovered'`` exactly where the aperture is
       < 98 % on the detector (coverage map) or the covered annulus
       < 200 px, the Lx NaN (never 0) unless the flag is ``ok``, the
       signed value consistent with the flag; a tree without the flag
       columns fails
C7     INFO: wall clock per view per worker from the job log
C8     INFO: disk per view
C9     ``lx_intr_*`` equal the reference campaign's (same gas, same axis)   sample
C10    event-file ``z_eff`` equals ``z_obs`` (no Z_EFF_MIN clamp)           sample + z<0.05
C11    duplicate views: at most one ``ok`` row per (snap, halo, view)       all
C12    no NaN/inf in the required columns of ``ok`` rows; the nullable    all
       ``m500c_hse_msun`` is NaN exactly when ``hse_crossing`` is
       ``none`` or ``fit-failed``
C13    seeds: ``obs_meta`` seed = the preset's derivation; collisions fail  all
C14    EXPMAP==0: counts on zero-exposure pixels per camera < 0.3 % for    sample
       views with >= 2000 counts, and stacked over the fainter ones
C15    obsid: one per-camera obsid per view in the tree's convention        all
       (``{snap:03d}{halo:08d}v{view}``), mock obsids unique, donor
       obsids 10-digit strings
C16    provenance columns of the results table: the view schema + spec      all
       hash (the stamp may be an earlier fit configuration of the same
       generation hash)
C17    (``sim.truth_selection_hse = 'sphere'``) spherical truth: every      all
       halo-snapshot of an ``ok`` row has a ``truth_profile.h5`` of the
       spherical stage (``truth_selection = 'sphere'``), complete (every
       chunk scanned); its non-SF shell filling factor at R500c is in
       [0.9, 1.1] where the cells tile space (moving mesh; sanity: a
       sphere fills its shells), reported but not judged for SPH and
       meshless runs, whose kernel volumes m / rho need not add up to
       the shell volume, in either direction (Dehnen & Aly 2012,
       Sect. 2.5); the row's truth
       columns equal those recomputed from the file with the row's z_obs
       (the apply step ran); TNG-Cluster spheres reaching low-resolution
       DM within r_max are counted (INFO in the message)
=====  ================================================================  ======

Exit status 0 when no check FAILs, 1 otherwise.  Reports:
``<report-dir>/campaign_check.{md,csv}``, ``campaign_check_issues.csv`` (one
row per offending view) and ``campaign_check_expmap.csv`` (C14 per camera).

References
----------
astropy : Astropy Collaboration 2013, 2018, 2022 (FITS products).
h5py    : Collette 2013 (event files, C10).
numpy, pandas : array and table handling.
cosmology_planck2015 : the TNG cosmology of the C6 aperture radius.
xmm_ccf_sas : the PPS-like per-camera IMAGE/EXPMAP products (C2, C14, C15).
"""

def _product_kinds(spec: Any) -> tuple[str, ...]:
    """Per-camera product kinds of preset ``spec``: :data:`PPS_KINDS` and the model images of
    ``instrument.qpb_model_images`` (:data:`MODEL_KINDS`)."""
    mode = str(getattr(spec.instrument, "qpb_model_images", None) or "off")
    if mode not in MODEL_KINDS:
        raise ValueError(f"unknown instrument.qpb_model_images {mode!r}; known: {sorted(MODEL_KINDS)}")
    return PPS_KINDS + MODEL_KINDS[mode]


def _product_bands(spec: Any) -> tuple[str, ...]:
    """Band codes of the per-camera products of preset ``spec``."""
    five = str(spec.instrument.product_bands) == "pps-1-5"
    return ((BAND_1,) if five else ()) + BANDS

def _d_a_kpc(z: float) -> float:
    """Angular-diameter distance [kpc] of the pipeline (1001-point trapezoid)."""
    import numpy as np

    from tngxmm.core.cosmology import C_KM_S, H0_KM_S_MPC, OM0

    zz = np.linspace(0, z, 1001)
    ez = np.sqrt(OM0 * (1 + zz) ** 3 + (1 - OM0))
    trapz = getattr(np, "trapezoid", None) or np.trapz
    return (C_KM_S / H0_KM_S_MPC) * 1e3 * float(trapz(1 / ez, zz)) / (1 + z)

def load_results(root: str | Path) -> tuple[Any, Any]:
    """All rows of ``results_*_snap_*.csv`` and the de-duplicated table (a resubmitted
    job appends a fresh row for a view whose earlier row was an error: keep the ok row)."""
    import pandas as pd

    fs = sorted(glob.glob(str(Path(root) / "results_*_snap_*.csv")))
    if not fs:
        empty = pd.DataFrame(columns=["sim", *KEY, "status"])
        return empty, empty.copy()

    def read(f: str) -> Any:
        head = set(pd.read_csv(f, nrows=0).columns)
        return pd.read_csv(f, dtype={k: v for k, v in RESULT_DTYPES.items() if k in head})

    raw = pd.concat([read(f) for f in fs], ignore_index=True)
    d = raw.copy()
    d["_ok"] = d["status"].astype(str).str.startswith("ok")
    d = d.sort_values("_ok", kind="stable").drop_duplicates(KEY, keep="last").drop(columns="_ok")
    return raw, d.reset_index(drop=True)

def _expected_cameras(meta: dict | None, conditions: dict | None = None, table: Any = None) -> list[str] | None:
    """The cameras whose products a view must have: those of the donor package of ``obs_meta.json`` (the physics donor's),
    or, under the observing-conditions rule ``nn`` (``conditions``: the view's ``conditions.json``), those the conditions
    donor ``chosen`` has data for in the condition ``table`` (a finite live time): the instrument ran with its
    cameras, which need not be the physics donor's.  None when they cannot be told (no package, or the conditions
    donor is not in the table)."""
    if conditions is not None:
        if table is None:
            return None
        hit = table[table["obsid"].astype(str).str.zfill(10) == str(conditions.get("chosen") or "").zfill(10)]
        if not len(hit):
            return None
        row = hit.iloc[0]
        return [c for c, s in (("PN", "pn"), ("MOS1", "mos1"), ("MOS2", "mos2"))
                if math.isfinite(float(row[f"exposure_{s}_s"]))]
    try:
        cams = list((meta or {})["donor_package"]["cameras_s"])
    except (KeyError, TypeError):
        return None
    return [c for c in ("PN", "MOS1", "MOS2") if c in cams]

class _Ctx:
    def __init__(self, root: Path, spec: Any, raw: Any, df: Any, targets: Any,
                 reference_root: Path | None, sample: int, expmap_dir: str,
                 expmap_threshold: float, expmap_min_counts: int,
                 core_offsets: Any = None) -> None:
        self.root, self.spec, self.raw, self.df, self.targets = root, spec, raw, df, targets
        self.core_offsets = core_offsets      # donor core-offset table path (None: default)
        self.reference_root = reference_root
        self.sample = sample
        self.expmap_dir = expmap_dir
        self.expmap_threshold = expmap_threshold
        self.expmap_min_counts = expmap_min_counts
        self.issues: list[dict[str, Any]] = []
        self.expmap_rows: list[dict[str, Any]] = []
        #: target views of the donors the preset excludes (``runtime.exclude_donors``): planned, not generated
        self.excluded: Any = None
        self.exclude_reasons: dict[str, str] = {}

    def bad(self, check: str, key: tuple | None, detail: str) -> None:
        self.issues.append({"check": check, "view": _tag(key) if key else "",
                            "detail": str(detail)[:300]})

    def n_bad(self, check: str) -> int:
        return sum(1 for i in self.issues if i["check"] == check)

    def examples(self, check: str, k: int = 3) -> list[str]:
        return [f"{i['view']}: {i['detail']}" if i["view"] else i["detail"]
                for i in self.issues if i["check"] == check][:k]

    def result(self, check: str, title: str, n: int, msg: str, *, info: bool = False,
               ok: bool | None = None) -> CheckResult:
        nb = self.n_bad(check)
        good = (nb == 0) if ok is None else ok
        st = INFO if info else (PASS if good else FAIL)
        return CheckResult(check, title, st, n, nb, msg, self.examples(check))


def _ok_rows(df: Any) -> Any:
    return df[df["status"].astype(str).str.startswith("ok")]


def _c1(cx: _Ctx) -> CheckResult:
    import pandas as pd

    df = cx.df
    if cx.targets is not None:
        tg = cx.targets
        j = tg.merge(df, on=KEY, how="left", suffixes=("_t", ""))
        ok = j["status"].astype(str).str.startswith("ok")
        for _, r in j[~ok].iterrows():
            cx.bad("C1", (int(r.snap), int(r.halo_id), int(r["view"])),
                   f"status {str(r['status'])[:80]}")
        known = tg[KEY] if cx.excluded is None else pd.concat([tg[KEY], cx.excluded[KEY]])
        extra = df.merge(known, on=KEY, how="left", indicator=True)
        for _, r in extra[extra["_merge"] == "left_only"].iterrows():
            cx.bad("C1", (int(r.snap), int(r.halo_id), int(r["view"])), "row not in the targets")
        n = len(tg)
        msg = f"{int(ok.sum())}/{n} target views ok; rows in CSVs {len(df)}"
        if cx.excluded is not None and len(cx.excluded):
            rec = int(df["status"].astype(str).str.startswith("excluded").sum())
            msg += (f"; {len(cx.excluded)} further target views excluded (donor not usable: "
                    f"{cx.exclude_reasons}), {rec} recorded in the CSVs")
    else:
        for _, r in df[~df["status"].astype(str).str.startswith("ok")].iterrows():
            cx.bad("C1", (int(r.snap), int(r.halo_id), int(r["view"])),
                   f"status {str(r['status'])[:80]}")
        n = len(df)
        msg = f"{n - cx.n_bad('C1')}/{n} rows ok"
    on_disk = set()
    for sim in sorted(set(df["sim"].astype(str))) or ["tng-cluster"]:
        for p in (cx.root / sim).glob("[0-9][0-9][0-9]/[0-9]*/view[0-9]"):
            on_disk.add((int(p.parent.parent.name), int(p.parent.name), int(p.name[4:])))
    in_csv = {(int(a), int(b), int(c)) for a, b, c in df[KEY].itertuples(index=False)}
    for k in sorted(on_disk - in_csv):
        cx.bad("C1", k, "view directory without a results row")
    msg += f"; {len(on_disk)} view directories, {len(on_disk - in_csv)} without a row"
    return cx.result("C1", "results complete", n, msg)


def _c11(cx: _Ctx) -> CheckResult:

    dup = ok[ok.duplicated(KEY, keep=False)]
    for k, _ in dup.groupby(KEY):
        cx.bad("C11", tuple(int(x) for x in k), "more than one ok row")
    superseded = len(raw) - len(cx.df) - (len(dup) - dup.drop_duplicates(KEY).shape[0])
    return cx.result("C11", "duplicate views", len(raw),
                     f"{len(raw)} rows, {len(cx.df)} distinct views; {superseded} error rows "
                     "superseded by a later ok row (resubmission)")

def _c17(cx: _Ctx) -> CheckResult:
    """Spherical truth present, complete, sane and applied to the rows (module table)."""
    import numpy as np

    from tngxmm.sim.compute_hse_mass import TRUTH_COLUMNS
    from tngxmm.sim.generate_view import SimOptions, view_truth_columns
    from tngxmm.sim.truth_profile import hse_from_truth_profile, path_for
    from tngxmm.ingest.descriptor import for_sim_name, volumes_tile
    from tngxmm.sim.truth_sphere import FILL_SANITY, SELECTION

    opts = SimOptions.from_spec(cx.spec)
    ok = _ok_rows(cx.df)
    seen: dict[tuple, Any] = {}
    fills, n_contam, n_out = [], 0, 0
    judge_fill = volumes_tile(for_sim_name(cx.spec.sim.sim_name, cx.spec.sim.source))
    for _, r in ok.iterrows():
        key = (int(r.snap), int(r.halo_id), int(r["view"]))
        hk = (str(r.sim), key[0], key[1])
        if hk not in seen:
            seen[hk] = None
            p = path_for(view_dir(cx.root, str(r.sim), *key).parent)
            try:
                hse = hse_from_truth_profile(p)
            except Exception as e:
                cx.bad("C17", key, f"no spherical truth file: {e!r}"[:200])
                continue
            a = hse["attrs"]
            if a.get("truth_selection") != SELECTION:
                cx.bad("C17", key, f"truth_selection {a.get('truth_selection')!r} (the stage "
                                   "`campaign truth-sphere` has not run on this halo)")
                continue
            if not bool(a.get("sphere_complete", False)):
                cx.bad("C17", key, "sphere from an incomplete snapshot pass")
            f = float(a.get("sphere_fill_nonsf_r500c", float("nan")))
            fills.append(f)
            if not (FILL_SANITY[0] <= f <= FILL_SANITY[1]):
                n_out += 1
                if judge_fill:
                    cx.bad("C17", key, f"non-SF filling factor at R500c {f:.3f}")
            n_contam += int(bool(a.get("sphere_contaminated", False)))
            seen[hk] = hse
        hse = seen[hk]
        if hse is None:
            continue
        tc = view_truth_columns(hse, float(r["z_obs"]), opts)
        diff = []
        for c in TRUTH_COLUMNS:
            if c not in r.index:
                continue
            want, got = tc[c], r[c]
            if isinstance(want, float):
                g = float(got)
                same = (math.isnan(want) and math.isnan(g)) or bool(
                    np.isclose(g, want, rtol=1e-12, atol=0.0))
            else:
                same = str(got) == str(want)
            if not same:
                diff.append(c)
        if diff:
            cx.bad("C17", key, f"truth columns differ from the spherical file (apply not run?): "
                               f"{diff[:4]}")
    fl = np.asarray([f for f in fills if np.isfinite(f)])
    med = f"{np.median(fl):.3f} [{fl.min():.3f}, {fl.max():.3f}]" if len(fl) else "n/a"
    return cx.result("C17", "spherical hydrostatic truth", len(seen),
                     f"{len(seen)} halo-snapshots; non-SF fill(R500c) median [min, max] {med}; "
                     f"{n_out} outside [{FILL_SANITY[0]}, {FILL_SANITY[1]}]"
                     f"{'' if judge_fill else ' (kernel volumes: reported, not judged)'}; "
                     f"{n_contam} spheres reach low-resolution DM within r_max")


def _c12(cx: _Ctx) -> CheckResult:
    import numpy as np
    import pandas as pd

    from tngxmm.schemas.tables import VIEW_TABLE

    ok = VIEW_TABLE.rename_legacy(_ok_rows(cx.df))
    required = [c.name for c in VIEW_TABLE.columns if not c.nullable and c.name in ok.columns]
    for c in required:
        s = ok[c]
        if pd.api.types.is_numeric_dtype(s):
            badm = ~np.isfinite(s.astype(float))
        else:
            badm = s.isna()
        for _, r in ok[badm].iterrows():
            cx.bad("C12", (int(r.snap), int(r.halo_id), int(r["view"])), f"{c} = {r[c]}")
    note = ""
    if "m500c_hse_msun" in ok.columns:
        # the headline HSE mass is nullable: NaN exactly when hse_crossing is 'none' or
        # 'fit-failed' (a fitted headline whose Vikhlinin fits did not converge)
        nan_m = ~np.isfinite(ok["m500c_hse_msun"].astype(float))
        if "hse_crossing" in ok.columns:
            flag = ok["hse_crossing"].astype(str)
            null = flag.isin(["none", "fit-failed"])
            badm = (nan_m & ~null) | (~nan_m & null)
            why = "m500c_hse_msun = {m} with hse_crossing = {f}"
            note = f"; hse_crossing {flag.value_counts().to_dict()}"
        else:
            badm = nan_m
            why = "m500c_hse_msun = {m} and no hse_crossing column"
        for _, r in ok[badm].iterrows():
            cx.bad("C12", (int(r.snap), int(r.halo_id), int(r["view"])),
                   why.format(m=r["m500c_hse_msun"], f=r.get("hse_crossing")))
    return cx.result("C12", "no NaN in required columns", len(ok),
                     f"{len(required)} required columns checked on {len(ok)} ok rows" + note)


def _int_columns(schema: Any, df: Any) -> Any:
    """Integer columns of ``schema`` that pandas read as float (the empty cells of
    the error rows in the same CSV) back to nullable integers when every value is
    integral; other columns untouched."""
    import numpy as np

    df = df.copy()
    for c in schema.columns:
        if c.dtype != "int" or c.name not in df.columns or df[c.name].dtype.kind != "f":
            continue
        s = df[c.name]
        v = s.dropna().to_numpy()
        if np.all(np.isfinite(v)) and np.all(v == np.round(v)) and np.all(np.abs(v) < 2**53):
            df[c.name] = s.astype("Int64")
    return df


def _stamp_generation_hashes(root: Path) -> dict[str, str]:
    """``{full spec hash: generation hash}`` recorded by the job start files of the tree
    (``provenance/job_*_start.json``, here and in the parent campaign directory)."""
    import json

    out: dict[str, str] = {}
    for d in (root / "provenance", root.parent / "provenance"):
        for f in sorted(d.glob("job_*_start.json")) if d.is_dir() else ():
            try:
                rec = json.loads(f.read_text())
            except (OSError, ValueError):
                continue
            if rec.get("spec_hash") and rec.get("generation_hash"):
                out[str(rec["spec_hash"])] = str(rec["generation_hash"])
    return out


def _c16_spec_hash(cx: _Ctx) -> str:
    """Row stamps against the current spec.  A row is a view product: its stamp is
    accepted when it equals the full ``spec_hash`` or when the *generation* hash of the
    stamped spec equals the current one (``massfit`` settings changed afterwards, which
    cannot change a generated view; the fit columns of such a row are then those of the
    earlier fit configuration, and the note says so).  The generation hash of a stamp is
    read from the job start files of the tree.  Fit products (``result.json``) are
    compared with the full hash by their own readers."""
    full, gen = cx.spec.spec_hash(), cx.spec.generation_hash()
    known = _stamp_generation_hashes(cx.root)
    stamps = cx.df["spec_hash"].astype(str)
    older = sorted({s for s in set(stamps) - {full} if known.get(s) == gen})
    bad = cx.df[~stamps.isin({full, *older})]
    if len(bad):
        cx.bad("C16", None, f"{len(bad)} rows with spec_hash != {full} and no matching generation "
                            f"hash {gen} ({sorted(set(bad['spec_hash'].astype(str)))[:3]})")
    return (f"{int(stamps.isin(older).sum())} rows stamped by an earlier fit configuration "
            f"(same generation hash {gen}: {', '.join(older)})") if older else ""

def _c13(cx: _Ctx, metas: dict[tuple, dict | None]) -> CheckResult:
    seeds: dict[int, list[tuple]] = {}
    n = 0
    for key, meta in metas.items():
        if meta is None or "seed" not in meta:
            cx.bad("C13", key, "no seed in obs_meta")
            continue
        n += 1
        s = int(meta["seed"])
        seeds.setdefault(s, []).append(key)
        want = view_seed(cx.spec, key, meta)
        if s != want:
            cx.bad("C13", key, f"seed {s} != seed_for {want}")
    coll = {s: ks for s, ks in seeds.items() if len(ks) > 1}
    msg = f"{n} seeds; {len(coll)} collisions"
    if coll:
        ex = "; ".join(f"{s}: {', '.join(_tag(k) for k in ks)}" for s, ks in list(coll.items())[:3])
        msg += f" ({ex})"
        for s, ks in coll.items():
            cx.bad("C13", ks[0], f"seed {s} shared by {len(ks)} views")
    return cx.result("C13", "per-view seeds", n, msg)


def _c15(cx: _Ctx, metas: dict[tuple, dict | None], files: dict[tuple, list[str]]) -> CheckResult:

        mock_ids[mo] = key

            m = _PERCAM_RE.match(nm)
            if not m:
                cx.bad("C15", key, f"not a per-camera product name: {nm}")
                continue
            obsids.add(m["obsid"])
        if len(obsids) > 1:
            cx.bad("C15", key, f"several obsids in one view: {sorted(obsids)}")
        for ob in obsids:
            if ob == mo or re.fullmatch(re.escape(mo) + r"r[1-9]\d*", ob):
                conv["mock obsid"] = conv.get("mock obsid", 0) + 1
            else:
                cx.bad("C15", key, f"per-camera obsid {ob!r} is not the mock obsid {mo!r}")
        meta = metas.get(key)
        donor = ((meta or {}).get("donor_package") or {}).get("obsid")

    if len(conv) > 1:
        cx.bad("C15", None, f"mixed per-camera naming conventions in one tree: {conv}")
    return cx.result("C15", "obsid uniqueness", len(files),
                     f"conventions {conv or 'none'}; {len(mock_ids)} distinct mock obsids")

def _donor_offset_check(cx: _Ctx, key: tuple, h: Any, meta: dict | None,
                        table: Any, conditions: dict | None = None) -> float | None:
    """C4: the applied offset (header
    DITHERX/Y, mask frame [arcmin]) must equal the donor's core offset in the donor
    table (header CORDXREQ/Y too), and the cluster pixel (CLUSTX/Y) must be the
    optical axis plus that offset rotated by POSANGLE.  Returns the offset [arcmin].

    Under the observing-conditions rule ``nn`` (``conditions``: the view's ``conditions.json``) the view is observed
    under the conditions donor ``chosen``: it must be usable in the table, and the offset is the one the record
    states (its core offset plus the pointing jitter), not the physics donor's."""
    import numpy as np

    obsid = str(((meta or {}).get("donor_package") or {}).get("obsid") or "").zfill(10)
    if conditions is not None:
        obsid = str(conditions.get("chosen") or "").zfill(10)
    hit = table[(table.obsid == obsid) & (table.usable.astype(str).str.lower() == "true")]
    if not len(hit):
        cx.bad("C4", key, f"donor {obsid} not usable in the donor core-offset table"
                          f"{' (conditions donor)' if conditions is not None else ''}")
        return None
    want = (float(hit.core_dx_mask_arcmin.iloc[0]), float(hit.core_dy_mask_arcmin.iloc[0]))
    if conditions is not None:
        want = tuple(float(x) for x in conditions["core_offset_arcmin"])
    if str(h.get("PLACEMNT", "")) != "donor-offset":
        cx.bad("C4", key, f"PLACEMNT {h.get('PLACEMNT')!r} != 'donor-offset'")
        return None
    req = (float(h["CORDXREQ"]), float(h["CORDYREQ"]))
    app = (float(h["DITHERX"]), float(h["DITHERY"]))
    tol = C4_DONOR_OFFSET_TOL_ARCMIN
    if max(abs(req[0] - want[0]), abs(req[1] - want[1])) > tol:
        cx.bad("C4", key, f"requested offset {req} != "
                          f"{'conditions record' if conditions is not None else 'donor table'} {want} ({obsid})")
    capped = bool(h.get("CORCAPPD", False))
    if capped and bool(cx.spec.instrument.dither_cap):
        if math.hypot(*app) > math.hypot(*want) + tol:
            cx.bad("C4", key, f"capped offset {app} longer than the donor's {want}")
    elif max(abs(app[0] - want[0]), abs(app[1] - want[1])) > tol:
        cx.bad("C4", key, f"applied offset {app} != donor offset {want} ({obsid}"
                          f"{', capped' if capped else ''})")
    # cluster pixel = optical axis + rotated applied offset (4"/px product grid)
    n4 = int(h["NAXIS1"])
    c0 = (n4 - 1) / 2 if bool(cx.spec.instrument.half_pixel_fix) else n4 / 2
    pa = np.deg2rad(float(h.get("POSANGLE", 0.0)))
    ca, sa = np.cos(pa), np.sin(pa)
    scale = 60.0 / float(cx.spec.photometry.pix_arcsec)
    ex = c0 + (app[0] * ca - app[1] * sa) * scale
    ey = c0 + (app[0] * sa + app[1] * ca) * scale
    dpx = math.hypot(float(h["CLUSTX"]) - ex, float(h["CLUSTY"]) - ey)
    if dpx > C4_CLUSTER_PIX_TOL:
        cx.bad("C4", key, f"CLUSTX/Y ({float(h['CLUSTX']):.3f}, {float(h['CLUSTY']):.3f}) "
                          f"{dpx:.3f} px from the rotated donor offset ({ex:.3f}, {ey:.3f})")
    return math.hypot(*app)


def _c6_lx_flags(cx: _Ctx, key: tuple, r: Any, h: Any, covmask: Any, pix_arcmin: float) -> None:
    """C6, per aperture (R500, 1, 1.5 Mpc, centred on CLUSTX/CLUSTY): the flag is one of ok | nondetection | masked |
    uncovered; ``uncovered`` exactly where the aperture is < 98 % covered or the covered
    1.1-1.6 r annulus has < 200 pixels (with tolerance bands); the Lx is
    finite and > 0 exactly when the flag is ``ok`` and NaN otherwise (never 0: no silent
    clipping); the signed value equals it when ``ok``, is <= 0 for ``nondetection`` and
    NaN for ``uncovered``."""
    import numpy as np

    from tngxmm.schemas.tables import LX_FLAGS

    yy, xx = np.ogrid[:covmask.shape[0], :covmask.shape[1]]
    d = np.hypot(xx - float(h["CLUSTX"]), yy - float(h["CLUSTY"]))
    d_a = _d_a_kpc(float(r.z_obs))
    for col, kpc in C6_APERTURES:
        ap = float(r.R500c_kpc) if kpc is None else kpc
        r_px = ap / d_a * 206265.0 / (pix_arcmin * 60.0)
        src = d <= r_px
        frac = float(covmask[src].mean()) if src.sum() else 0.0
        n_ann = int((covmask & (d > 1.1 * r_px) & (d < 1.6 * r_px)).sum())
        flag, lx = r[f"{col}_flag"], float(r[col])
        sgn = float(r[f"{col}_signed"]) if f"{col}_signed" in r else float("nan")
        tag = f"{col}: flag {flag!r}, Lx {lx:.4g}"
        if flag not in LX_FLAGS:
            cx.bad("C6", key, f"{tag}: unknown flag")
            continue
        if lx == 0.0:
            cx.bad("C6", key, f"{tag}: 0 stored (unmeasurable must be NaN with its flag)")
        if (flag == "ok") != (np.isfinite(lx) and lx > 0):
            cx.bad("C6", key, f"{tag}: Lx finite and > 0 exactly when the flag is 'ok'")
        covered = frac >= C6_MEASURED_IF_ABOVE and n_ann >= 200 + C6_ANNULUS_PIX_TOL
        uncovered = frac < C6_ZERO_IF_BELOW or n_ann < 200 - C6_ANNULUS_PIX_TOL
        if covered and flag == "uncovered":
            cx.bad("C6", key, f"{tag} but aperture coverage {frac:.3f}, {n_ann} annulus px")
        if uncovered and flag != "uncovered":
            cx.bad("C6", key, f"{tag} but aperture coverage {frac:.3f}, {n_ann} annulus px")
        if flag == "ok" and not np.isclose(sgn, lx, rtol=1e-12, atol=0.0):
            cx.bad("C6", key, f"{tag}: signed {sgn:.4g} differs")
        if flag == "nondetection" and not (sgn <= 0):
            cx.bad("C6", key, f"{tag}: signed {sgn:.4g} not <= 0")
        if flag == "uncovered" and np.isfinite(sgn):
            cx.bad("C6", key, f"{tag}: signed {sgn:.4g} finite")

def _c7_log(log: str) -> str | None:
    """Wall clock per view per worker from the progress lines of a job log."""
    if not log or not os.path.exists(log):
        return None
    txt = open(log).read()
    nw = re.findall(r"workers=(\d+)", txt)
    segs = re.split(r"=== .*snap \d+ ===", txt)
    wall, nviews = 0.0, 0
    for seg in segs:
        t = [float(m.group(1)) for m in re.finditer(r"\[\s*\d+/\s*\d+ \|\s*([\d.]+)m\]", seg)]
        if t:
            wall += max(t)
            nviews += len(t)
    if not (nviews and nw):
        return "no progress lines found"
    return (f"{nviews} view completions, {wall:.1f} min wall over the snapshots on {nw[-1]} "
            f"workers; ~{wall * 60 * int(nw[-1]) / nviews:.0f} s/view/worker")

    run("C4", c4)
    if ref is None:
        run("C5", lambda: CheckResult("C5", "AGN sky independent of the reference", SKIP, 0, 0,
                                      "no --reference-root"))
        run("C9", lambda: CheckResult("C9", "lx_intr equal the reference", SKIP, 0, 0,
                                      "no --reference-root"))
    else:
        run("C5", lambda: cx.result("C5", "AGN sky independent of the reference", n,
                                    f"AGN skies compared with {ref}"))
        run("C9", lambda: cx.result("C9", "lx_intr equal the reference", n,
                                    f"lx_intr_* compared with {ref}"))
    def c6() -> CheckResult:
        ok = _ok_rows(cx.df)
        counts = {c: ok[c].astype(str).value_counts().to_dict() for c in C6_FLAG_COLUMNS
                  if c in ok.columns}
        return cx.result("C6", "Lx flags vs aperture coverage", n,
                         "lx_r500/1mpc/1p5mpc flags consistent with the 98 % aperture and "
                         f"200-pixel annulus coverage rules, NaN (never 0) unless ok; {counts}")

    run("C6", c6)

    def c10() -> CheckResult:
        n_low, zmins = _c10_low_z(cx, j, metas)
        n_ev = vr["n_zeff"] + n_low
        if n_ev:
            return cx.result("C10", "no Z_EFF_MIN clamp", n_ev,
                             f"event-file z_eff == z_obs for {vr['n_zeff']} sampled + {n_low} "
                             "low-z views with events")
        if zmins:
            zcol = "z_obs_t" if "z_obs_t" in j.columns else "z_obs"
            zmin = float(ok_j[zcol].min())
            return cx.result("C10", "no Z_EFF_MIN clamp", len(zmins),
                             f"no event files; obs_meta z_eff_min {sorted(set(zmins))} vs min "
                             f"z_obs {zmin:.4f}", ok=max(zmins) <= zmin + 1e-9 and not cx.n_bad("C10"))
        return CheckResult("C10", "no Z_EFF_MIN clamp", SKIP, 0, 0,
                           "no event files and no view below z = 0.05")
    run("C10", c10)

    def c14() -> CheckResult:
        er = cx.expmap_rows
        if not er:
            return CheckResult("C14", "EXPMAP==0 fraction per camera", SKIP, 0, 0,
                               "no IMAGE/EXPMAP pairs found")
        e = pd.DataFrame(er)
        parts = []
        for cam, g in e.groupby("camera"):
            parts.append(f"{cam}: median {g.fraction.median():.4f}, max {g.fraction.max():.4f} "
                         f"over {len(g)} views")
        return cx.result("C14", "EXPMAP==0 fraction per camera", len(e),
                         f"threshold {cx.expmap_threshold:.4f} (views >= {cx.expmap_min_counts} "
                         f"counts, stacked below; EXPMAP from {cx.expmap_dir}/); "
                         + "; ".join(parts))
    run("C14", c14)
    msg7 = _c7_log(log or "")
    if msg7 is not None:
        run("C7", lambda: CheckResult("C7", "wall clock (log)", INFO, 0, 0, msg7))
    if vr["sizes"]:
        s = vr["sizes"]
        run("C8", lambda: CheckResult(
            "C8", "disk per view", INFO, len(s), 0,
            f"median {np.median(s):.0f} MB, max {max(s):.0f} MB -> "
            f"{np.median(s) * len(j) / 1e3:.1f} GB for {len(j)} views"))
    order = {f"C{i}": i for i in range(1, 17)}
    res.sort(key=lambda r: order.get(r.check, 99))
    return Report(str(root), preset, res, cx.issues, cx.expmap_rows, len(df), n)




def _infer_preset(root: Path, df: Any) -> str:
    """The preset a tree was generated with, from its ``campaign`` column: the preset of that
    name, else the preset whose stream key (``sim.rng_campaign``) it is.  A tree with results
    but without the column is an error."""
    if "campaign" in df.columns and len(df):
        c = str(df["campaign"].iloc[0])
        if c:
            from tngxmm.config import PRESETS, get_preset

            if c in PRESETS:
                return c
            keyed = [n for n in PRESETS if get_preset(n).sim.rng_campaign == c]
            if len(keyed) == 1:
                return keyed[0]
            raise click.ClickException(f"{root}: campaign tag {c!r} names no preset; use --preset")
    if not len(df):
        from tngxmm.config import PRODUCTION_PRESET

        return PRODUCTION_PRESET
    raise click.ClickException(f"{root}: the results tables have no `campaign` column; "
                               "name the preset with --preset")

def write_report(rep: Report, out_dir: str | Path) -> dict[str, Path]:
    """Markdown + CSV report, the issue list and the C14 table into ``out_dir``."""
    import pandas as pd

    d = Path(out_dir)
    d.mkdir(parents=True, exist_ok=True)
    rows = [{"check": r.check, "title": r.title, "status": r.status, "n_checked": r.n_checked,
             "n_bad": r.n_bad, "message": r.message} for r in rep.results]
    paths = {"csv": d / "campaign_check.csv", "md": d / "campaign_check.md",
             "issues": d / "campaign_check_issues.csv", "expmap": d / "campaign_check_expmap.csv"}
    pd.DataFrame(rows).to_csv(paths["csv"], index=False)
    pd.DataFrame(rep.issues, columns=["check", "view", "detail"]).to_csv(paths["issues"],
                                                                         index=False)
    pd.DataFrame(rep.expmap, columns=["view", "camera", "counts", "counts_on_zero_expmap",
                                      "fraction"]).to_csv(paths["expmap"], index=False)
    lines = [f"# Campaign check: `{rep.root}`", "",
             f"Preset `{rep.preset}`; {rep.n_rows} views in the results tables, "
             f"{rep.n_sampled} checked per view.  RESULT: **{'PASS' if rep.passed else 'FAIL'}**",
             "", "| check | title | status | checked | bad | message |",
             "|---|---|---|---|---|---|"]
    for r in rep.results:
        msg = r.message.replace("|", "/")
        lines.append(f"| {r.check} | {r.title} | {r.status} | {r.n_checked} | {r.n_bad} | {msg} |")
    ex = [r for r in rep.results if r.examples]
    if ex:
        lines += ["", "## Examples", ""]
        for r in ex:
            lines += [f"- **{r.check}**: " + "; ".join(e.replace("|", "/") for e in r.examples)]
    paths["md"].write_text("\n".join(lines) + "\n")
    return paths

def format_report(rep: Report) -> str:
    """Console form (one line per check)."""
    out = []
    for r in rep.results:
        ex = f"; e.g. {r.examples[:2]}" if r.examples and r.status == FAIL else ""
        out.append(f"  {r.check} {r.status}: {r.message}{ex}")
    out.append("RESULT: " + ("PASS" if rep.passed else
                             "FAIL (" + ", ".join(r.check for r in rep.results
                                                  if r.status == FAIL) + ")"))
    return "\n".join(out)

                  report_dir: str | None, expmap_dir: str, expmap_threshold: float,
