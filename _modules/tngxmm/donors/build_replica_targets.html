Source code for tngxmm.donors.build_replica_targets

"""Replica sampler: K mass- and epoch-matched mock views per reference observation.

Every reference observation (the 638 bank observations, ``observations.csv``)
receives K views of DIFFERENT TNG-Cluster halo-snapshots whose catalogue M500c
lies within a two-sided window around the observation's matching mass, taken
from the snapshot nearest to the observation's redshift.  Each view copies the
observation's observing conditions (per-camera live times and camera set,
Galactic N_H, FILTER per camera, PN submode) and is observed at the
observation's redshift with a 3 % relative jitter.  There are no importance
weights anywhere: the population is matched by construction and checked by
gate P (:mod:`tngxmm.donors.validate_population`).

Algorithm (every number is a :class:`ReplicaSpec` field):

0. inputs: the observation table, a halo table (one row per halo-snapshot:
   ``snap, z_snap, halo_id, m500c_msun, r500c_kpc`` [+ ``zoom_target_id``]),
   the FROZEN ``split_by_zoom_target.csv`` (an input, never re-derived) and the
   ``zoom_target_ids.csv`` lookup;
1. epoch slice: the snapshot nearest in z (midpoint edges), only if
   |z_ref - z_snap| <= ``epoch_max_dz``; the second-nearest is the fallback;
2. matching mass (``matching_mass``): the default is the literature
   hydrostatic-bias-corrected mass M_cat/(1-b)_lit (``literature_one_minus_b``,
   :mod:`tngxmm.obs.hydrostatic_bias`: PSZ2 MSZ / 0.78, ACT DR6 M500c and the
   lensing masses unchanged); also catalogue M500 (convention A).
   Every row stores the (1-b) applied, its source and, as a recorded result
   that is never applied, the project's empirical G3 value;
3. window: |log M - log M_target| <= 0.10 dex, widened to 0.15 then 0.20 only
   if fewer than the missing number of unused candidates exist;
4. order: hardest observation first (fewest candidates at 0.10 dex), random
   within ties;
5. split: per draw, the split with the largest deficit against 72/13/15 among
   those with candidates, then the halo nearest in mass inside it;
6. reuse cap: at most ``halo_reuse_cap`` views per halo-snapshot, one per
   projection axis, never two from one halo-snapshot for one observation;
7. z_obs = z_ref * exp(N(0, 0.03)), no floor, no clamp;
8. conditions copied from the same donor; the exposure / N_H jitter
   (0.15 common + 0.05 per camera, 0.10 on N_H; lognormal) is applied and the
   donor values are stored;
9. ``fov_fit`` = theta500 <= 13.5 arcmin, recorded, NEVER used for selection;
10. observations with no candidate get no replica (``covered = False``).

Seeds: per view and purpose from ``tngxmm.core.rng.seed_for`` (INTERFACES
section 6); where that call cannot be made the sha256-v1 derivation is reproduced
locally (:func:`seed_for_view`) and every row records which one was used.  The campaign-level draws of the sampler (order
ties, split and pick tie-breaks) use one Generator seeded with
``ReplicaSpec.sampler_seed``.

References
----------
tngcluster  : Nelson et al. 2024 (halo catalogue, zoom targets).
psz2, act_dr6 : catalogue masses and redshifts of the reference observations.
xmm_mission : XMM-Newton observing conditions copied from the donors.
hi4pi       : Galactic N_H of the donor fields.
"""

@dataclass(frozen=True)
class ReplicaSpec:
    """Every tunable of the replica sampler (defaults = the recommended values).

    :meth:`from_campaign` takes the preset's value (``CampaignSpec.donors`` / ``.sim``)
    wherever it is decided and keeps the default otherwise.
    """

    campaign: str = "v4"
    sim: str = "tng-cluster"
    reference_sample: str = "bank638"            # 'bank638' | 'sz_defined'
    halos_per_observation: int = 3               # K
    #: 'literature_one_minus_b' = M_cat / (1-b)_literature (default);
    #: 'catalogue' = M_cat (convention A)
    matching_mass: str = "literature_one_minus_b"
    #: literature (1-b) per mass class (tngxmm.obs.hydrostatic_bias: Planck 2015 XXIV CCCP baseline
    #: for the PSZ2 MSZ; ACT DR6 M500c is already lensing-calibrated; lensing masses uncorrected)
    one_minus_b_literature: tuple = (("planck_psz2", 0.78), ("act_dr6", 1.0), ("wl", 1.0))
    one_minus_b_literature_source: tuple = (("planck_psz2", "planck2015_xxiv:tab2:cccp"),
                                            ("act_dr6", "act_dr6:sec2.5:M500c"),
                                            ("wl", "hoekstra2015,applegate2014:lensing"))
    #: EMPIRICAL (1-b) = M_ref/M_fit offsets of this project, a recorded result (column only, never applied):
    #: G3 Q4b (MASSFIT_FINDINGS §3b, controlled regime theta500 < 5') re-derived on the corrected
    #: Planck15 matching masses by tngxmm.massfit.one_minus_b
    one_minus_b_empirical_g3: tuple = (("planck_psz2", 1.11), ("act_dr6", 1.32), ("wl", 0.89))
    one_minus_b_empirical_g3_source: str = ("massfit.one_minus_b: G3 Q4b, theta500 < 5 arcmin, corrected "
                                            "Planck15 matching masses (2026-10-08)")
    mass_window_ladder_dex: tuple = (0.10, 0.15, 0.20)
    epoch_max_dz: float = 0.15                   # |z_ref - z_snap| cap
    epoch_candidates: int = 2                    # nearest snapshot, then the second-nearest
    snaps: tuple = CAMPAIGN_SNAPS + (50,)
    include_hiz_snaps: bool = False              # optional: add snapshots 45 and 40
    hiz_snaps: tuple = (45, 40)
    halo_reuse_cap: int = 3                      # views per halo-snapshot, one per axis
    #: reference observations left out (their real clusters get no mocks): donors whose
    #: measured core offset is implausible (donor core-offset table, > 6 arcmin)
    exclude_obsids: tuple = ()
    axes: tuple = (0, 1, 2)
    order: str = "hardest_first"
    split_fractions: tuple = (("train", 0.72), ("val", 0.13), ("test", 0.15))
    z_jitter_frac: float = 0.03                  # sigma of ln(z_obs / z_ref)
    z_floor: Any = None                          # none
    z_eff_min: Any = None                        # no clamp
    exposure_jitter: bool = True
    jitter_scale_sigma: float = 0.15
    jitter_cam_sigma: float = 0.05
    jitter_nh_sigma: float = 0.10
    fov_fit_arcmin: float = 13.5                 # flag only
    aperture_max_kpc: float = 1500.0
    photon_flag_threshold: float = 4e7           # flag only, no cap
    sampler_seed: int = 20261007
    seed_scheme: str = "sha256-v1"               # INTERFACES section 6
    n_folds: int = 5                             # donor-grouped folds
    fold_seed: int = 20261007                    # = donors.assign_donor_folds.DONOR_FOLD_SEED

    def spec_hash(self, length: int = 16) -> str:
        """First ``length`` hex digits of the sha256 of the settings' sorted JSON; recorded in the provenance of the
        replica target list."""
        js = json.dumps(self.to_dict(), sort_keys=True, separators=(",", ":"), default=str)
        return hashlib.sha256(js.encode()).hexdigest()[:length]

    @classmethod
    def from_campaign(cls, spec: Any, **overrides: Any) -> ReplicaSpec:
        """The replica settings of a CampaignSpec (a ``None`` field keeps the default)."""
        kw: dict[str, Any] = {"campaign": getattr(spec.sim, "rng_campaign", None) or spec.name}
        d = spec.donors
        for name in ("reference_sample", "halos_per_observation", "halo_reuse_cap", "matching_mass",
                     "exclude_obsids"):
            v = getattr(d, name, None)
            if v is not None:
                kw[name] = v
        snaps = getattr(spec.sim, "snaps", None)
        if snaps is not None:
            kw["snaps"] = tuple(snaps)
        for name in ("jitter_scale_sigma", "jitter_cam_sigma", "jitter_nh_sigma", "fov_fit_arcmin",
                     "aperture_max_kpc"):
            if hasattr(d, name):
                kw[name] = getattr(d, name)
        kw.update(overrides)
        return cls(**kw)






# --------------------------------------------------------------------------- seeds --

_MASK63 = (1 << 63) - 1


def _seed_sha256_v1(sim: str, snap: int, halo_id: int, view: int, purpose: str, campaign: str,
                    realisation: int = 0, camera: str | None = None,
                    namespace: str = "tngxmm") -> int:
    """The sha256-v1 seed derivation, exact."""
    msg = (f"{namespace}|rng|sha256-v1|{campaign}|{sim}|snap={snap:03d}|halo={halo_id:d}"
           f"|view={view:d}|real={realisation:d}|{purpose}|{camera or '-'}")
    return int.from_bytes(hashlib.sha256(msg.encode("ascii")).digest()[:8], "big") & _MASK63

def seed_for_view(sim: str, snap: int, halo_id: int, view: int, purpose: str,
                  campaign: str = "v4", realisation: int = 0) -> tuple[int, str]:
    """Seed of one per-view stream and the name of the scheme that produced it.

    Calls ``tngxmm.core.rng.seed_for``; where that raises, the local sha256-v1

@cites("tngcluster")
def prepare_halo_units(halos: Any, splits: Any, zoom_target_ids: Any | None, spec: ReplicaSpec) -> tuple[Any, dict]:
    """Expand the halo table to drawable units (one per halo-snapshot and projection axis).

    ``zoom_target_id`` comes from the halo table or, if absent there, from the
    ``zoom_target_ids`` lookup (snap, halo_id); the split comes ONLY from the
    frozen ``splits`` table.  Halos of other snapshots, or without a zoom
    target or split, are dropped and counted in the returned report.
    """
    import numpy as np

    h = halos.copy()
    rep: dict[str, Any] = {"halo_rows_in": int(len(h))}
    h = h[h.snap.isin(spec.snapshots())].copy()
    rep["halo_rows_in_snaps"] = int(len(h))
    if zoom_target_ids is not None:
        lut = zoom_target_ids[["snap", "halo_id", "zoom_target_id"]].rename(
            columns={"zoom_target_id": "_zt_lut"})
        h = h.merge(lut, on=["snap", "halo_id"], how="left")
        if "zoom_target_id" in h:
            both = h.zoom_target_id.notna() & h._zt_lut.notna()
            rep["zoom_target_conflicts"] = int((h.zoom_target_id[both] != h._zt_lut[both]).sum())
            h["zoom_target_id"] = h.zoom_target_id.fillna(h._zt_lut)
        else:
            h["zoom_target_id"] = h._zt_lut
        h = h.drop(columns="_zt_lut")
    no_zt = h.zoom_target_id.isna()
    rep["dropped_no_zoom_target"] = int(no_zt.sum())
    h = h[~no_zt].copy()
    h["zoom_target_id"] = h.zoom_target_id.astype(int)
    h = h.merge(splits[["zoom_target_id", "split"]], on="zoom_target_id", how="left")
    no_sp = h.split.isna()
    rep["dropped_no_split"] = int(no_sp.sum())
    h = h[~no_sp].copy()
    h = h.sort_values(["snap", "halo_id"]).reset_index(drop=True)
    h["hs_index"] = np.arange(len(h))
    u = h.loc[h.index.repeat(len(spec.axes))].copy()
    u["view"] = np.tile(np.asarray(spec.axes, int), len(h))
    u["log_m500c"] = np.log10(u.m500c_msun.astype(float))
    rep["halo_snaps"] = int(len(h))
    rep["units"] = int(len(u))
    rep["halo_snaps_per_snap"] = {int(k): int(v) for k, v in h.groupby("snap").size().items()}
    return u.reset_index(drop=True), rep

@cites("psz2", "act_dr6")
def select_reference(observations: Any, spec: ReplicaSpec) -> Any:
    """Reference observations of the sampler with their matching-mass targets (A and B)."""
    import numpy as np

    o = observations
    extra = spec.reference_sample == "bank638+extra"      # the bank and the observations added to it
    if spec.reference_sample == "bank638":
        o = o[o.in_bank.astype(bool)]
    elif extra:
        pass
    elif spec.reference_sample == "sz_defined":
        o = o[o.in_sz_defined_set.astype(bool)]
    else:
        raise ValueError(f"unknown reference_sample {spec.reference_sample!r}")
    if spec.exclude_obsids:
        o = o[~o.obsid.astype(str).str.zfill(10).isin({str(x).zfill(10) for x in spec.exclude_obsids})]
    o = o.copy().reset_index(drop=True)
    lit = dict(spec.one_minus_b_literature)
    lit_src = dict(spec.one_minus_b_literature_source)
    emp = dict(spec.one_minus_b_empirical_g3)
    # the extra observations have no bias factor: their targets by catalogue mass are empty (they are paired on
    # the fitted mass, tngxmm.obs.extra_observations)
    missing = sorted(set(o.mass_class.astype(str)) - set(lit) - ({"extra"} if extra else set()))
    if missing:
        raise ValueError(f"no literature (1-b) for mass class(es) {missing}")
    added = (o.mass_class.astype(str) == "extra").to_numpy()
    o["log_m500c_cat"] = np.log10(o.m500_catalogue_msun.astype(float))
    o.loc[added, "log_m500c_cat"] = np.nan              # never a matching mass (a survey mass in its own convention)
    o["one_minus_b_literature"] = o.mass_class.map(lit).astype(float)
    o["one_minus_b_literature_source"] = o.mass_class.map(lit_src).fillna("").astype(str)
    o["one_minus_b_empirical_g3"] = o.mass_class.map(emp).astype(float)
    o["log_m500c_target_a"] = o.log_m500c_cat
    o["log_m500c_target_lit"] = o.log_m500c_cat - np.log10(o.one_minus_b_literature)
    o["log_m500c_target_empirical_g3"] = o.log_m500c_cat - np.log10(o.one_minus_b_empirical_g3)
    if spec.matching_mass == "catalogue":
        o["log_m500c_target"] = o.log_m500c_target_a
        o["one_minus_b_applied"] = 1.0
        o["one_minus_b_applied_source"] = "none (catalogue mass)"
    elif spec.matching_mass == "literature_one_minus_b":
        o["log_m500c_target"] = o.log_m500c_target_lit
        o["one_minus_b_applied"] = o.one_minus_b_literature
        o["one_minus_b_applied_source"] = o.one_minus_b_literature_source
    else:
        raise ValueError(f"unknown matching_mass {spec.matching_mass!r}")
    o["mass_convention"] = spec.matching_mass
    return o

@cites("tngcluster", "psz2", "act_dr6", "xmm_mission", "hi4pi")
def build_replica_targets(observations: Any, halos: Any, splits: Any, zoom_target_ids: Any | None = None,
                          spec: ReplicaSpec | None = None, folds: Any | None = None,
                          provenance: dict[str, Any] | None = None) -> tuple[Any, Any, dict[str, Any]]:
    """Draw the replica population.

    Returns ``(targets, coverage, report)``: ``targets`` one row per view
    (``targets_v4.csv``), ``coverage`` one row per reference observation
    (status full / partial / unmatched, ``n_replicas``, ``covered``), and a
    report dict (counts, widened windows, epoch fallbacks, split fractions).
    ``folds`` (obsid -> donor_fold) adds the donor-grouped fold column.
    """
    import numpy as np
    import pandas as pd

    spec = spec or ReplicaSpec()
    ref = select_reference(observations, spec)
    units, urep = prepare_halo_units(halos, splits, zoom_target_ids, spec)
    rng = np.random.default_rng(spec.sampler_seed)
    K, ladder = spec.K, tuple(float(w) for w in spec.mass_window_ladder_dex)
    frac = dict(spec.split_fractions)

    u_snap = units.snap.to_numpy(int)
    u_logm = units.log_m500c.to_numpy(float)
    u_hs = units.hs_index.to_numpy(int)
    u_split = units.split.to_numpy(object)
    used = np.zeros(len(units), bool)
    hs_uses = np.zeros(int(u_hs.max()) + 1 if len(u_hs) else 0, int)
    snap_z = units.groupby("snap").z_snap.first()
    snaps_sorted = snap_z.index.to_numpy(int)
    zs = snap_z.to_numpy(float)
    by_snap = {int(s): np.flatnonzero(u_snap == s) for s in snaps_sorted}

    def epoch_snaps(z: float) -> list[int]:
        o = np.argsort(np.abs(zs - z), kind="stable")[: spec.epoch_candidates]
        return [int(snaps_sorted[i]) for i in o if abs(zs[i] - z) <= spec.epoch_max_dz]

    def candidates(snap: int, mt: float, w: float, exclude_hs: set[int]) -> np.ndarray:
        idx = by_snap[snap]
        ok = (~used[idx]) & (np.abs(u_logm[idx] - mt) <= w + 1e-12) & (hs_uses[u_hs[idx]] < spec.halo_reuse_cap)
        c = idx[ok]
        if exclude_hs:
            c = c[~np.isin(u_hs[c], list(exclude_hs))]
        return c

    n0 = []
    for z, mt in zip(ref.z, ref.log_m500c_target):
        es = epoch_snaps(float(z))
        n0.append(len(candidates(es[0], float(mt), ladder[0], set())) if es else 0)
    ref["n_candidates_0p10"] = n0
    if spec.order == "hardest_first":
        order = []
        for _, g in ref.groupby("n_candidates_0p10", sort=True):
            order.extend(rng.permutation(g.index.to_numpy()))
    else:
        order = list(rng.permutation(ref.index.to_numpy()))

    nview = {s: 0 for s in SPLITS}
    draws: list[dict[str, Any]] = []
    status: dict[int, str] = {}
    for i in order:
        z, mt = float(ref.at[i, "z"]), float(ref.at[i, "log_m500c_target"])
        got: list[int] = []
        got_hs: set[int] = set()
        for k_snap, s in enumerate(epoch_snaps(z)):
            c = np.array([], int)
            w_used = ladder[-1]
            for w in ladder:
                c = candidates(s, mt, w, got_hs)
                w_used = w
                if len(np.unique(u_hs[c])) >= K - len(got) or w == ladder[-1]:
                    break
            while len(got) < K and len(c):
                spl = u_split[c]
                tot = max(1, sum(nview.values()))
                avail = [sp for sp in SPLITS if (spl == sp).any()]
                defi = np.array([frac[sp] - nview[sp] / tot for sp in avail])
                best = np.flatnonzero(np.isclose(defi, defi.max()))
                sp_pick = avail[int(best[rng.integers(len(best))]) if len(best) > 1 else int(best[0])]
                cs = c[spl == sp_pick]
                d = np.abs(u_logm[cs] - mt)
                dmin = d.min()
                tie = cs[np.isclose(d, dmin, rtol=0, atol=1e-12)]
                j = int(tie[rng.integers(len(tie))]) if len(tie) > 1 else int(tie[0])
                got.append(j)
                got_hs.add(int(u_hs[j]))
                used[j] = True
                hs_uses[u_hs[j]] += 1
                nview[str(u_split[j])] += 1
                draws.append(dict(ref_index=i, unit=j, match_window_dex=w_used,
                                  epoch_fallback=k_snap > 0, replica_k=len(got)))
                c = c[(c != j) & (u_hs[c] != u_hs[j])]
            if len(got) >= K:
                break
        status[i] = "full" if len(got) == K else ("partial" if got else "unmatched")

    d = pd.DataFrame(draws)
    t = d.merge(units, left_on="unit", right_index=True).merge(
        ref.add_prefix("ref_"), left_on="ref_index", right_index=True)
    t = _view_columns(t, spec)
    if folds is not None:
        fmap = folds.set_index("obsid")["donor_fold"]
        t["donor_fold"] = t.donor_id.map(fmap).astype("Int64")
    cov = ref[["obsid", "cluster_id", "sample", "mass_class", "z", "m500_catalogue_msun",
               "n_candidates_0p10"]].copy()
    cov["status"] = [status.get(i, "unmatched") for i in ref.index]
    cnt = t.groupby("donor_id").size()
    cov["n_replicas"] = cov.obsid.map(cnt).fillna(0).astype(int)
    cov["covered"] = cov.n_replicas > 0

    from tngxmm.core.provenance import split_run_stamps

    prov = dict(provenance or {})
    for k, v in split_run_stamps(prov)[0].items():     # run stamps go to the sidecar
        t[k] = v
    t["replica_spec_hash"] = spec.spec_hash()
    report = {
        "units": urep, "views": int(len(t)), "halo_snaps_used": int(t.groupby(["snap", "halo_id"]).ngroups),
        "zoom_targets_used": int(t.zoom_target_id.nunique()),
        "max_views_per_zoom_target": int(t.groupby("zoom_target_id").size().max()) if len(t) else 0,
        "views_per_halo_snap": {int(k): int(v) for k, v in
                                t.groupby(["snap", "halo_id"]).size().value_counts().sort_index().items()},
        "status": cov.status.value_counts().to_dict(),
        "excluded_obsids": [str(x).zfill(10) for x in spec.exclude_obsids],
        "coverage_frac": float(cov.covered.mean()) if len(cov) else 0.0,
        "split_fractions": {s: float((t.split == s).mean()) for s in SPLITS} if len(t) else {},
        "window_widened_frac": float((t.match_window_dex > ladder[0] + 1e-12).mean()) if len(t) else 0.0,
        "epoch_fallback_views": int(t.epoch_fallback.sum()) if len(t) else 0,
        "dlogm_p10_p50_p90": [float(x) for x in np.percentile(t.dlogm_match_dex, [10, 50, 90])] if len(t) else [],
        "dlogm_by_target_mass": mass_match_bias(t).to_dict("records") if len(t) else [],
        "views_z_obs_lt_0p05": int((t.z_obs < 0.05).sum()) if len(t) else 0,
        "fov_fit_false": int((~t.fov_fit).sum()) if len(t) else 0,
        "seed_scheme": sorted(t.seed_scheme.unique().tolist()) if len(t) else [],
        "replica_spec": spec.to_dict(), "replica_spec_hash": spec.spec_hash(),
    }
    return t.reset_index(drop=True), cov.reset_index(drop=True), report

def mass_match_bias(targets: Any, bin_dex: float = 0.2, tolerance_dex: float = MASS_BIAS_TOLERANCE_DEX) -> Any:
    """The matched-minus-target mass offset per bin of target log M500c.

    The window is two-sided, but where the halo supply above a target is exhausted (few massive
    halo-snapshots, the reuse cap) every view of the bin is filled from below: the matched halos are
    then systematically lighter than the target, which the per-view window check never sees.  One
    row per bin: ``n``, ``median_dlogm_dex``, ``frac_below`` (views lighter than the target),
    ``frac_widened`` (window widened beyond the first rung) and ``one_sided`` (|median| above
    ``tolerance_dex``).  Views without a physics donor (``paired`` False, release 1.2) have no target and are left
    out (:func:`tngxmm.donors.pair_projections.drop_unpaired`)."""
    import numpy as np
    import pandas as pd

    from tngxmm.donors.pair_projections import drop_unpaired

    targets, _ = drop_unpaired(targets)
    lm = targets.log_m500c_target.astype(float)
    lo = np.floor(lm.min() / bin_dex) * bin_dex
    edges = np.arange(lo, lm.max() + bin_dex + 1e-9, bin_dex)
    b = pd.cut(lm, edges, right=False)
    w0 = float(targets.match_window_dex.min())
    g = targets.assign(_b=b).groupby("_b", observed=True)
    out = pd.DataFrame({
        "log_m500c_target_lo": [float(i.left) for i in g.groups],
        "n": g.size().to_numpy(int),
        "median_dlogm_dex": g.dlogm_match_dex.median().to_numpy(float),
        "frac_below": g.dlogm_match_dex.apply(lambda x: float((x < 0).mean())).to_numpy(float),
        "frac_widened": g.match_window_dex.apply(lambda x: float((x > w0 + 1e-12).mean())).to_numpy(float),
    })
    out["one_sided"] = out.median_dlogm_dex.abs() > tolerance_dex
    return out.reset_index(drop=True)




@cites("xmm_mission", "hi4pi")
def _view_columns(t: Any, spec: ReplicaSpec) -> Any:
    """Per-view columns: ids, observed redshift, copied (jittered) conditions, flags, seeds."""
    import numpy as np
    import pandas as pd

    from tngxmm.schemas.ids import make_view_id

    out = pd.DataFrame(index=t.index)
    out["view_id"] = [make_view_id(spec.campaign, s, h, v) for s, h, v in zip(t.snap, t.halo_id, t.view)]
    out["campaign"] = spec.campaign
    out["sim"] = spec.sim
    out["snap"] = t.snap.astype(int)
    out["halo_id"] = t.halo_id.astype(int)
    out["view"] = t.view.astype(int)
    out["rep"] = 0
    out["mock_obsid"] = [make_mock_obsid_v4(s, h, v) for s, h, v in zip(t.snap, t.halo_id, t.view)]
    out["zoom_target_id"] = t.zoom_target_id.astype(int)
    out["split"] = t.split.astype(str)
    out["z_snap"] = t.z_snap.astype(float)
    out["z_obs_source"] = "replica"
    out["m500c_msun"] = t.m500c_msun.astype(float)
    out["r500c_kpc"] = t.r500c_kpc.astype(float)
    for c in ("m200c_msun", "r200c_kpc", "lx_intr_r500_erg_s"):
        if c in t:
            out[c] = t[c].astype(float)
    out["donor_id"] = t.ref_obsid.astype(str)
    out["ref_obsid"] = out["donor_id"]
    out["cluster_id"] = t.ref_cluster_id.astype(str)
    out["ref_name"] = t.ref_cluster_name.astype(str)
    out["ref_sample"] = t.ref_sample.astype(str)
    out["ref_mass_class"] = t.ref_mass_class.astype(str)
    out["ref_mass_source_key"] = t.ref_m500_catalogue_source.astype(str)
    out["ref_z"] = t.ref_z.astype(float)
    out["ref_m500c_cat_msun"] = t.ref_m500_catalogue_msun.astype(float)
    out["ref_in_sz_defined_set"] = t.ref_in_sz_defined_set.astype(bool)
    out["ref_is_primary_obs"] = t.ref_is_primary_obs.astype(bool)
    out["replica_k"] = t.replica_k.astype(int)
    out["mass_convention"] = spec.matching_mass
    out["one_minus_b_applied"] = t.ref_one_minus_b_applied.astype(float)
    out["one_minus_b_applied_source"] = t.ref_one_minus_b_applied_source.astype(str)
    out["one_minus_b_empirical_g3"] = t.ref_one_minus_b_empirical_g3.astype(float)
    out["log_m500c_target"] = t.ref_log_m500c_target.astype(float)
    out["m500c_target_msun"] = 10 ** out.log_m500c_target
    out["log_m500c_target_empirical_g3"] = t.ref_log_m500c_target_empirical_g3.astype(float)
    out["dlogm_match_dex"] = t.log_m500c.astype(float) - out.log_m500c_target
    out["match_window_dex"] = t.match_window_dex.astype(float)
    out["epoch_fallback"] = t.epoch_fallback.astype(bool)
    out["n_candidates_0p10"] = t.ref_n_candidates_0p10.astype(int)

    zobs, jit, s_z, s_d, scheme = [], [], [], [], []
    exp_app = {c: [] for c in CAMERAS}
    nh_app = []
    for row in t.itertuples():
        sz, sch = seed_for_view(spec.sim, int(row.snap), int(row.halo_id), int(row.view), "z_obs", spec.campaign)
        sd, _ = seed_for_view(spec.sim, int(row.snap), int(row.halo_id), int(row.view), "donor", spec.campaign)
        rz = np.random.default_rng(sz)
        e = float(rz.normal(0.0, spec.z_jitter_frac)) if spec.z_jitter_frac > 0 else 0.0
        z = float(row.ref_z) * math.exp(e)
        if spec.z_floor is not None:
            z = max(z, float(spec.z_floor))
        zobs.append(z)
        jit.append(e)
        s_z.append(sz)
        s_d.append(sd)
        scheme.append(sch)
        rd = np.random.default_rng(sd)
        cams = {c: float(getattr(row, f"ref_{_EXP_COL[c]}")) for c in CAMERAS}
        nh = float(row.ref_nh_1e22cm2)
        if spec.exposure_jitter:
            scale = float(np.exp(rd.normal(0.0, spec.jitter_scale_sigma)))
            for c in CAMERAS:
                if np.isfinite(cams[c]):
                    cams[c] = cams[c] * scale * float(np.exp(rd.normal(0.0, spec.jitter_cam_sigma)))
            if np.isfinite(nh):
                nh = nh * float(np.exp(rd.normal(0.0, spec.jitter_nh_sigma)))
        for c in CAMERAS:
            exp_app[c].append(cams[c])
        nh_app.append(nh)
    out["ref_z_obs_jitter_ln"] = jit
    out["z_obs"] = zobs
    out["z_jitter_frac"] = spec.z_jitter_frac
    for c in ("PN", "MOS1", "MOS2"):
        out[_EXP_COL[c]] = exp_app[c]
        out[_EXP_COL[c].replace("_s", "_donor_s")] = t[f"ref_{_EXP_COL[c]}"].astype(float)
    a_tot = sum(_ARF.values())
    out["exposure_eff_s"] = sum(_ARF[c] * out[_EXP_COL[c]].fillna(0.0) for c in CAMERAS) / a_tot
    out["exposure_longest_s"] = out[[_EXP_COL[c] for c in CAMERAS]].max(axis=1)
    out["nh_1e22cm2"] = nh_app
    out["nh_donor_1e22cm2"] = t.ref_nh_1e22cm2.astype(float)
    out["exposure_jittered"] = bool(spec.exposure_jitter)
    for c in ("PN", "MOS1", "MOS2"):
        out[_FILT_COL[c]] = t[f"ref_{_FILT_COL[c]}"].where(t[f"ref_{_FILT_COL[c]}"].notna(), None)
    out["filter"] = out.filter_pn.where(out.filter_pn.notna(), out.filter_mos1)
    out["pn_mode"] = t.ref_pn_mode.where(t.ref_pn_mode.notna(), None)
    # the published fraction of the donor's mode; a donor with a pn exposure in another mode is refused
    from tngxmm.instrument.conditions_donor import pn_oot_fraction

    pn_t = t[f"ref_{_EXP_COL['PN']}"].astype(float)
    out["pn_oot_frac"] = [pn_oot_fraction(m if isinstance(m, str) else "", bool(tp > 0), str(o))
                          for m, tp, o in zip(out.pn_mode, pn_t.fillna(0.0), t.ref_obsid)]
    for c in ("mos1", "mos2"):
        out[f"submode_{c}"] = t[f"ref_submode_{c}"].where(t[f"ref_submode_{c}"].notna(), None)

    out["theta500_arcmin"] = _theta_arcmin(out.r500c_kpc, out.z_obs)
    out["theta_1p5mpc_arcmin"] = _theta_arcmin(np.full(len(out), spec.aperture_max_kpc), out.z_obs)
    out["fov_fit"] = out.theta500_arcmin <= spec.fov_fit_arcmin
    out["fov_fit_r500"] = out["fov_fit"]
    out["fov_fit_1p5mpc"] = out.theta_1p5mpc_arcmin <= spec.fov_fit_arcmin
    if "lx_intr_r500_erg_s" in out:
        flux = _DEPTH_FACTOR * out.lx_intr_r500_erg_s / (4 * np.pi * _lum_dist_cm(out.z_obs) ** 2)
        out["photon_est"] = flux / _E_MEAN_ERG * _AREA_CM2 * out.exposure_longest_s
    else:
        out["photon_est"] = np.nan
    out["photon_est_over_threshold"] = out.photon_est > spec.photon_flag_threshold
    out["seed_z_obs"] = s_z
    out["seed_donor"] = s_d
    out["seed_scheme"] = scheme
    out["sampler_seed"] = int(spec.sampler_seed)
    return out
