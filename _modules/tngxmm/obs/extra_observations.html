Source code for tngxmm.obs.extra_observations

"""Observations added to the reference bank after it was fixed: the rows of ``observations.csv`` with ``in_bank`` False.

The bank of 638 observations is the sample the releases up to 1.1 were made with and stays as it is.  An archive
search of 2026-11-20 matched clusters of MCXC-II, eRASS1, ACT DR6, PSZ2 and the SPT catalogues that the bank does not
hold against the public XMM-Newton observations and graded the pipeline images of 194 of them
(``inputs_extra/candidates_graded.csv``: 170 ``pass``, 13 ``reject``, 11 ``hold``, with the measurements the grade
rests on).  The ``pass`` rows enter the observation table here.  They are fitted like the bank observations and lend
their observing conditions to mock views; a mock is paired with them on the fitted mass of both sides, so the
catalogue mass of a row is a record of what the survey published and is never used as a matching mass:

* ``sample`` is ``extra``; ``m500_catalogue_source`` names the survey of the candidate list (``mcxc2``, ``erass1``,
  ``act_dr6``, ``psz2``, ``spt_sz``, ``spt_ecs``) and ``m500_catalogue_msun`` is its M500c as published, in the
  survey's own cosmology and calibration (``m500_catalogue_derivation`` says so; the bank converts its masses to
  the pipeline cosmology, these are not converted).
* ``mass_class`` is ``extra``: no hydrostatic-bias factor belongs to it, and the reference selections that need one
  (:func:`tngxmm.donors.build_replica_targets.select_reference`) take these rows only when asked for
  (``reference_sample = 'bank638+extra'``), with empty bias columns.
* ``in_bank`` is False, ``is_donor`` True, ``in_sz_defined_set`` False (that set is defined on the bank).

The per-camera columns come from the headers of the pipeline images as for the bank (the live time of a camera is
the sum of the ``EXPOSURE`` keywords of its image exposures), the Galactic columns from the two services of
:mod:`tngxmm.obs.sky_columns` at the catalogue position.  The cluster ids follow the rule of the bank (PSZ2 entry,
else ACT DR6 entry, else the position of the candidate list) on the entries of the two catalogues that lie within
the same-cluster radius of a candidate or carry its name (``inputs_extra/psz2_extract.csv``,
``act_dr6_extract.csv``, cut from the full catalogues; the extracts of the bank hold the bank's clusters only); an
id that a different catalogue entry of the bank already has gets the next free suffix, so no bank row changes.

References
----------
mcxc, erass1, act_dr6, psz2, bleem2015 : the surveys of the candidate list (MCXC-II and the SPT-ECS list are named
    by their ``m500_catalogue_source`` in the table).
xmm_mission, xsa : the observations and their pipeline products.
hi4pi, heasarc_nh_tool : Galactic N_H.
"""

from __future__ import annotations

import re
from pathlib import Path
from typing import Any

from tngxmm.cites import cites, collect_cites

INPUTS_EXTRA_DIR = Path(__file__).resolve().parent / "catalogue" / "inputs_extra"
#: logical name -> file of ``inputs_extra/``
EXTRA_FILES = {"extra_candidates": "candidates_graded.csv", "extra_headers": "camera_headers.csv",
               "extra_sky": "sky_columns.csv", "extra_psz2": "psz2_extract.csv", "extra_act_dr6": "act_dr6_extract.csv"}
SAMPLE = "extra"
MASS_CLASS = "extra"
#: the grade of the candidate list that enters the table
GRADE = "pass"
#: sub-directory of the image tree that holds the extra observations
IMAGE_DIR = "extra"
DERIVATION = "published by the survey of m500_catalogue_source (its cosmology and calibration; not converted)"
REASON = "extra: graded pass in the archive search of 2026-11-20"
_CAMS = {"PN": ("exposure_pn_s", "filter_pn", "pn_s"), "MOS1": ("exposure_mos1_s", "filter_mos1", "mos1_s"),
         "MOS2": ("exposure_mos2_s", "filter_mos2", "mos2_s")}

def cluster_dir(name: str) -> str:
    """Directory name of a cluster in the image tree: its name with every run of other characters than letters,
    digits, ``.``, ``+`` and ``-`` as one underscore."""
    return re.sub(r"[^A-Za-z0-9.+\-]+", "_", str(name)).strip("_")

def load_extra_inputs(inputs_dir: str | Path = INPUTS_EXTRA_DIR) -> dict[str, Any]:
    """The tables of ``inputs_extra/`` (empty dict when the directory holds no candidate list)."""
    import pandas as pd

    d = Path(inputs_dir)
    if not (d / EXTRA_FILES["extra_candidates"]).exists():
        return {}
    rt = ("extra_psz2", "extra_act_dr6")              # FITS values, as the extracts of the bank
    return {k: pd.read_csv(d / f, dtype={"obsid": str}, float_precision="round_trip" if k in rt else None)
            for k, f in EXTRA_FILES.items()}

def camera_exposures(headers: Any) -> Any:
    """``obsid, mos1_s, mos2_s, pn_s, n_files`` from the header rows: per camera the sum of the image exposures."""
    import pandas as pd

    t = headers.groupby(["obsid", "camera"])["exposure_s"].sum().unstack()
    out = pd.DataFrame({col: t[cam] if cam in t else float("nan") for cam, (_, _, col) in _CAMS.items()})
    out["n_files"] = headers.groupby("obsid").size()
    return out.reset_index()[["obsid", "mos1_s", "mos2_s", "pn_s", "n_files"]]

@cites("xmm_mission", "xsa", "hi4pi", "heasarc_nh_tool", "psz2", "act_dr6")
def extra_rows(extra: dict[str, Any], bank: Any, psz2: Any, act_dr6: Any) -> Any:
    """The rows of the extra observations with the columns of ``bank`` (the table of the 638; module docstring)."""
    import numpy as np
    import pandas as pd

    from tngxmm.core.cosmology import r_delta_kpc, theta_arcmin
    from tngxmm.obs.build_observations import assign_cluster_ids
    from tngxmm.obs.read_observation_meta import ARF_BAND_MEAN_CM2, teff_from_obsid
    from tngxmm.schemas.ids import make_cluster_id

    c = extra["extra_candidates"]
    c = c[c.grade.astype(str) == GRADE].assign(obsid=lambda t: t.obsid.str.zfill(10)).sort_values("obsid")
    if c.obsid.duplicated().any() or c.obsid.isin(set(bank.obsid)).any():
        raise ValueError("extra observations: an obsid appears twice or is already in the bank")
    hdr = extra["extra_headers"].assign(obsid=lambda t: t.obsid.str.zfill(10))
    hdr = hdr[hdr.header_ok.astype(bool)]
    sky = extra["extra_sky"].assign(obsid=lambda t: t.obsid.str.zfill(10)).set_index("obsid")
    cams = camera_exposures(hdr)
    for name, have in (("image headers", set(cams.obsid)), ("Galactic columns", set(sky.index))):
        gone = sorted(set(c.obsid) - have)
        if gone:
            raise ValueError(f"extra observations: no {name} for {len(gone)} observations, e.g. {gone[:3]}")

    names = [re.sub(r"\s*\(\s*\)\s*$", "", str(n)).strip() for n in c.name]       # 'MCXC J... ()': no alias given
    o = pd.DataFrame({"obsid": c.obsid.to_numpy(), "cluster_name": names, "sample": SAMPLE,
                      "ra_deg": c.ra.to_numpy(float), "dec_deg": c.dec.to_numpy(float), "centre_source": "catalogue",
                      "z": c.z.to_numpy(float), "m500_catalogue_msun": c.m500_1e14_msun.to_numpy(float) * 1e14,
                      "m500_catalogue_source": c.mass_source.to_numpy(), "mass_class": MASS_CLASS,
                      "mass_source_metadata": c.mass_source.to_numpy()})
    o["nh_1e22cm2"] = o.obsid.map(sky.nh_1e22).astype(float)
    o["nh_source"] = "hi4pi:w3nh"
    ci = cams.set_index("obsid")
    longest = hdr.sort_values(["exposure_s", "exp_id"], ascending=[False, True]).drop_duplicates(["obsid", "camera"])
    lk = longest.set_index(["obsid", "camera"])
    n_exp = hdr.groupby(["obsid", "camera"]).size()
    for cam, (ecol, fcol, tcol) in _CAMS.items():
        o[ecol] = o.obsid.map(ci[tcol])
        o[fcol] = [lk["filter"].get((ob, cam)) for ob in o.obsid]
        o[f"submode_{cam.lower()}"] = [lk["submode"].get((ob, cam)) for ob in o.obsid]
        o[f"n_exp_{cam.lower()}"] = [int(n_exp.get((ob, cam), 0)) for ob in o.obsid]
    o["pn_mode"] = o.pop("submode_pn")
    o["n_files"] = o.obsid.map(ci.n_files).astype(int)
    first = hdr.sort_values(["obsid", "camera", "exp_id"]).drop_duplicates("obsid").set_index("obsid")
    for col in ("ra_pnt_deg", "dec_pnt_deg", "pa_pnt_deg", "date_obs", "object"):
        o[col] = o.obsid.map(first[col])
    o["teff_s"] = [teff_from_obsid(ob, cams, ARF_BAND_MEAN_CM2) for ob in o.obsid]
    o["r500_catalogue_kpc"] = [r_delta_kpc(m, z) for m, z in zip(o.m500_catalogue_msun, o.z)]
    o["theta500_arcmin"] = [theta_arcmin(r, z) for r, z in zip(o.r500_catalogue_kpc, o.z)]
    o["lx_r500_erg_s"] = np.nan
    o["lx_source"] = None

    # cluster ids by the rule of the bank; the suffix of a truncated id continues after the bank's entries
    both = [pd.concat([a, extra[k]], ignore_index=True).drop_duplicates("name").reset_index(drop=True)
            for a, k in ((psz2, "extra_psz2"), (act_dr6, "extra_act_dr6"))]
    o = assign_cluster_ids(o, both[0], both[1])
    taken: dict[str, dict[str, str]] = {}
    for cid, entry in zip(bank.cluster_id, bank.cluster_id_entry):
        taken.setdefault(str(cid)[:len(make_cluster_id(0.0, 0.0))], {})[str(entry)] = str(cid)
    ids = []
    for cid, entry in zip(o.cluster_id, o.cluster_id_entry):
        base = str(cid)[:len(make_cluster_id(0.0, 0.0))]
        known = taken.setdefault(base, {})
        if entry not in known:
            used = set(known.values())
            known[entry] = next(x for x in [base] + [f"{base}-{s}" for s in "bcdefghijklmnopqrstuvwxyz"]
                                if x not in used)
        ids.append(known[entry])
    o["cluster_id"] = ids

    o["m500_catalogue_quantity"] = "M500c"
    o["m500_catalogue_derivation"] = DERIVATION
    o["m500_catalogue_cosmology"] = "survey"
    o["m500_catalogue_ref_id"] = [f"{s}:M500c:{n}" for s, n in zip(o.m500_catalogue_source, o.cluster_name)]
    o["m_ref_value_msun"] = o.m500_catalogue_msun
    o["m_ref_quantity"] = "M500c"
    o["m_ref_radius_kpc"] = np.nan
    o["m_ref_radius_definition"] = "r500 of the mass"
    o["m_ref_source"] = [f"{s}: candidate list of the archive search (inputs_extra/{EXTRA_FILES['extra_candidates']})"
                         for s in o.m500_catalogue_source]
    o["m_ref_method"] = "mass estimate of the survey catalogue"
    o["m_ref_cosmology"] = "survey"
    o["m_ref_bias_adjustment"] = "as published"
    o["in_psz2"] = o.cluster_id_source == "psz2"
    o["has_dr6_match"] = o.cluster_id_source == "act_dr6"
    o["in_act_dr6"] = o.has_dr6_match
    in_bank_cluster = o.cluster_id.isin(set(bank.cluster_id))
    best = o.sort_values(["teff_s", "obsid"], ascending=[False, True]).drop_duplicates("cluster_id")
    o["is_primary_obs"] = o.index.isin(best.index) & ~in_bank_cluster
    for col in ("in_cccp", "in_wtg", "has_wl", "in_g3_obsrc", "is_lowz_zdonor", "in_sz_defined_set", "in_bank",
                "in_reference_sample"):
        o[col] = False
    o["is_donor"] = True
    o["n_ref_masses"] = 0
    o["reference_reason"] = REASON
    o["catalogue_source_keys"] = [";".join(sorted({k, "xsa", "hi4pi"})) for k in o.m500_catalogue_source]
    missing = [col for col in bank.columns if col not in o.columns]
    for col in missing:                       # the reference-mass columns of the bank: none for these rows
        o[col] = np.nan if bank[col].dtype.kind == "f" else None
    return o[list(bank.columns)].reset_index(drop=True)

@cites("xmm_mission", "xsa")
def build_extra_inputs(graded_csv: str | Path, images_root: str | Path, psz2_fits: str | Path, dr6_fits: str | Path,
                       out_dir: str | Path = INPUTS_EXTRA_DIR) -> dict[str, int]:
    """Write ``candidates_graded.csv`` (the graded list without its local paths), ``camera_headers.csv`` (the
    headers of the band-2 images under ``<images_root>/extra/``) and the two catalogue extracts (the entries of the
    PSZ2 union catalogue and of the ACT DR6 catalogue within the same-cluster radius of a ``pass`` candidate, or
    with its name; the columns of the bank's extracts) into ``out_dir``; the counts written."""
    import numpy as np
    import pandas as pd
    from astropy.io import fits

    from tngxmm.obs.build_observations import SAME_CLUSTER_ARCMIN, _sep_arcmin
    from tngxmm.obs.read_observation_meta import read_observation_meta

    out = Path(out_dir)
    out.mkdir(parents=True, exist_ok=True)
    g = pd.read_csv(graded_csv, dtype={"obsid": str})
    g["obsid"] = g.obsid.str.zfill(10)
    g = g.drop(columns=[c for c in ("dir",) if c in g.columns]).sort_values("obsid")
    g.to_csv(out / EXTRA_FILES["extra_candidates"], index=False, float_format="%.10g")
    h = read_observation_meta(images_root, pattern=f"{IMAGE_DIR}/*/P*IMAGE_2000.FTZ")
    h.to_csv(out / EXTRA_FILES["extra_headers"], index=False, float_format="%.15g")
    ok = g[g.grade == GRADE]
    with fits.open(str(psz2_fits)) as f:
        d = f[1].data
        pz = pd.DataFrame({"name": np.char.strip(np.asarray(d["Name"]).astype(str)),
                           "ra_deg": np.asarray(d["RAJ2000"], float), "dec_deg": np.asarray(d["DEJ2000"], float),
                           "z": np.asarray(d["z"], float), "msz_1e14msun": np.asarray(d["MSZ"], float),
                           "index": np.asarray(d["Index"], int)})
    with fits.open(str(dr6_fits)) as f:
        d = f[1].data
        ac = pd.DataFrame({"name": np.char.strip(np.asarray(d["name"]).astype(str)),
                           "ra_deg": np.asarray(d["RADeg"], float), "dec_deg": np.asarray(d["decDeg"], float),
                           "z": np.asarray(d["redshift"], float), "m500c_1e14msun": np.asarray(d["M500c"], float),
                           "snr": np.asarray(d["fixed_SNR"], float), "flags": np.asarray(d["flags"], float)})
    n = {}
    for key, cat in (("extra_psz2", pz), ("extra_act_dr6", ac)):
        keep = cat.name.isin(set(ok.name)).to_numpy()
        for ra, de in zip(ok.ra.astype(float), ok.dec.astype(float)):
            keep |= _sep_arcmin(cat.ra_deg.to_numpy(), cat.dec_deg.to_numpy(), ra, de) < SAME_CLUSTER_ARCMIN
        cat[keep].sort_values("name").to_csv(out / EXTRA_FILES[key], index=False, float_format="%.17g")
        n[key] = int(keep.sum())
    return {"candidates": int(len(g)), "pass": int(len(ok)), "header_rows": int(len(h)),
            "observations_with_headers": int(h.obsid.nunique()), **n}
