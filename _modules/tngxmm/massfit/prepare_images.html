Source code for tngxmm.massfit.prepare_images

"""Crop, rebin and seed the backgrounds of the per-camera images for mbproj2d.

Crop + rebin PPS images around a cluster and build mbproj2d ``Image`` objects,
with each image's background parameter seeded from its own outer annulus.

Why this is the speed fix, not a cosmetic one:

* The raw PPS frame is 648x648 = 4.2e5 pixels at 4"/px, but a cluster's R500
  subtends only ~2-8 arcmin, so the overwhelming majority of pixels are pure
  background.  Fitting the full frame makes the per-image background
  parameters dominate the likelihood and sit in a shallow, degenerate valley,
  so the gradient-free optimiser crawls for >10^4 evaluations.
* Cropping to a few x R500 and rebinning to ~10"/px leaves a cluster-dominated
  image with a thin background annulus: the background is well determined, and
  seeding each background parameter from that annulus starts the optimiser
  essentially at the solution.

Rebinning bookkeeping (mbproj2d predicts counts = SB * pixsize^2 * expmap)::

    counts  -> SUM over the RxR block          (photons add)
    expmap  -> MEAN over the block  [seconds]  (exposure per unit area is intensive)
    pixsize -> R * original

so counts = SB * (R*pix)^2 * mean(expmap) reproduces the summed block counts.
Chip gaps / FOV edges (expmap == 0) become the mask.

Switches:

``optical_axis``
    ``"header"``: the unvignetted exposure map divides the CCF vignetting out
    about ``OPTAXX/OPTAXY`` (0-indexed, mock products) or the PPS EXPMAP keys
    ``XCEN/YCEN`` (1-indexed image pixels) when present, falling back to CRPIX.
    ``"crpix"``: CRPIX, which is the pointing for real PPS files but the
    dithered cluster for mocks.
``rebin_centre``
    ``"block-centre"``: the convention of mbproj2d's own ``Image.binUp``,
    ``(ox + 0.5)/f - 0.5``, which places the origin at the centre of the block
    that holds it; ``"(ox-0.5)/f"`` is the same for f = 2, the 8" fits.
    ``"ox/f"``: the cluster position on the rebinned grid is the native
    position divided by the rebin factor; with f = 2 this puts the model 0.25
    rebinned px (2") off the data along each axis.
``sources``
    point sources (ra, dec, det_ml) masked by zeroing the exposure map
    (``mask_point_sources.source_mask``), identically on mocks and real data.
``psf``
    ``None`` (Sanders et al. 2018): no PSF in the model.
    ``"king-onaxis"``: each image's cluster model is convolved with the on-axis
    King profile of its camera (``fit_psf``: for mocks exactly the instrument
    stage's PSF, the first ``KING_PARAMS`` row of the header's ``PSFFILE``; for
    real data the CCF's on-axis King parameters at the band energy), with a
    centred, pixel-integrated core and a block-averaged tail of unit total;
    the images are zero-padded by the core radius (mbproj2d ``optimal_size``/
    ``pad``) so the FFT convolution does not wrap.  Background models are not
    convolved.
    ``"king-angle"``: the King profile of the CCF at each pixel's off-axis
    angle (``fit_psf.KingAnglePSF``; needs the optical axis of vignetting
    ``'sub-band'``), a mixture over three energies of the band weighted by its
    counts (``fit_psf.band_energy_mixture``, ``psf_kt_keV``); a mock made with
    the on-axis King gets that profile.
``pn_oot``
    ``None``: pn out-of-time events not modelled.
    ``"data-template"``: per pn image a fixed OoT template (``pn_oot``: the
    image's counts smeared along its readout columns, times f_OoT) is added to
    the model, and a mock's pn exposure maps (``OOTCORR = F``) are scaled by
    (1 - f_OoT); real PPS exposure maps already carry that factor.

References
----------
mbproj2d (image model and the ``Image.binUp`` origin convention), epic_pn (OoT),
xmm_ccf_sas (XRT vignetting curves of the ``XAREAEF`` CCF), sanders2018
(no PSF deconvolution, as MBProj2).
"""

def ccf_dir() -> str:
    """CCF directory: ``$TNGXMM_CCF_PATH``, else ``$CCF_PATH``, else the site one."""
    val = os.environ.get("TNGXMM_CCF_PATH") or os.environ.get("CCF_PATH") or DEFAULT_CCF_DIR
    if not val:
        raise site.SiteNotConfigured("no XMM CCF directory: set $CCF_PATH (or $TNGXMM_CCF_PATH, "
                                     "$TNGXMM_SITE_CCF)")
    return val




# ---- unvignetted exposure map (for BackModelVigNoVig) ------------------------
# The PPS EXPMAP is VIGNETTED: expmap_vig = ontime * vig(theta,E) * chip_mask.
# The particle background (QPB) is NOT vignetted (flat in detector coords), so
# BackModelVigNoVig needs the UNvignetted map,
#     expmap_novig = ontime * chip_mask = expmap_vig / vig(theta,E).
# The PPS map is divided by the CCF vignetting curve vig(theta,E) at each
# pixel's off-axis angle and the band mean energy.  vig comes from the
# XRT*_XAREAEF VIGNETTING extension (the CCF used for the ARFs), so real data
# and mocks share one vignetting model.

@cites("xmm_ccf_sas")
def vig_curve(camera: str, ccf: str | None = None):
    """(theta_arcmin[nr], energy_keV[ne], vig[ne, nr]) for a camera, cached."""
    import numpy as np
    from astropy.io import fits

    ccf = ccf or ccf_dir()
    key = (camera, ccf)
    if key in _VIG_CACHE:
        return _VIG_CACHE[key]
    files = glob.glob(os.path.join(ccf, f"{_XRT[camera]}_XAREAEF_*.CCF"))
    path = sorted(files, key=lambda p: int(p.split("_")[-1].split(".")[0]))[-1]
    with fits.open(path) as f:
        v = f["VIGNETTING"].data
        ene = np.asarray(v["ENERGY"], float) / 1e3            # eV -> keV
        vf = np.asarray(v["VIGNETTING_FACTOR"], float)        # (ne, nr)
        dth = f["VIGNETTING"].header["D_THETA"] * 60.0        # deg -> arcmin
    theta = np.arange(vf.shape[1]) * dth                       # arcmin
    _VIG_CACHE[key] = (theta, ene, vf)
    return _VIG_CACHE[key]

@cites("xmm_ccf_sas")
def vig_map(camera, band_mean_keV, ny, nx, cx, cy, pix_arcmin, ccf=None):
    """vig(theta, E_band) on the image grid (optical axis at cx, cy), clipped to [0.05, 1]."""
    import numpy as np

    theta, ene, vf = vig_curve(camera, ccf)
    ie = int(np.clip(np.searchsorted(ene, band_mean_keV) - 1, 0, len(ene) - 1))
    vig_of_theta = vf[ie]
    yy, xx = np.mgrid[0:ny, 0:nx]
    th = np.hypot(xx - cx, yy - cy) * pix_arcmin
    vig = np.interp(th, theta, vig_of_theta, left=vig_of_theta[0], right=vig_of_theta[-1])
    return np.clip(vig, 0.05, 1.0)

def rebin_sum(a, f):
    """Sum over f x f blocks (trailing rows/columns that do not fill a block are dropped)."""
    ny, nx = a.shape
    a = a[:ny - ny % f, :nx - nx % f]
    return a.reshape(a.shape[0] // f, f, a.shape[1] // f, f).sum((1, 3))

def rebin_mean(a, f):
    """Mean over f x f blocks (trailing rows/columns that do not fill a block are dropped)."""
    ny, nx = a.shape
    a = a[:ny - ny % f, :nx - nx % f]
    return a.reshape(a.shape[0] // f, f, a.shape[1] // f, f).mean((1, 3))

def rebin_origin(o: float, f: int, convention: str = "ox/f") -> float:
    """Position ``o`` (0-indexed native px, pixel centres at integers) on the grid rebinned by ``f``."""
    if convention == "ox/f":
        return o / f
    if convention == "block-centre":
        return (o + 0.5) / f - 0.5
    if convention == "(ox-0.5)/f":
        return (o - 0.5) / f
    raise ValueError(f"rebin_centre must be one of {REBIN_CENTRE_MODES}, got {convention!r}")

def cluster_pixel(header, ra, dec):
    """(x, y) 0-indexed pixel of the cluster in the native image."""
    from astropy.wcs import WCS

    wcs = WCS(header)
    x, y = wcs.all_world2pix(ra, dec, 0)
    return float(x), float(y)

def optical_axis_pixel(img_header, expmap_header=None, shape=None, mode: str = "crpix"):
    """0-indexed (x, y) of the optical axis on the native grid and the key it came from.

    ``crpix``: CRPIX - 1 of the image header.  ``header``: OPTAXX/OPTAXY
    (0-indexed px, mocks), else OPTAX_RA/OPTAX_DEC (sky position, converted with
    the image WCS), else XCEN/YCEN of the EXPMAP (PPS, 1-indexed), else CRPIX; the
    returned key ``CRPIX(fallback)`` flags the fallback for the provenance.
    """
    if mode not in OPTICAL_AXIS_MODES:
        raise ValueError(f"optical_axis must be one of {OPTICAL_AXIS_MODES}, got {mode!r}")
    ny, nx = shape if shape is not None else (img_header.get("NAXIS2", 0), img_header.get("NAXIS1", 0))
    if mode == "header":
        for h in (img_header, expmap_header):
            if h is not None and "OPTAXX" in h and "OPTAXY" in h:
                return float(h["OPTAXX"]), float(h["OPTAXY"]), "OPTAXX"
        for h in (img_header, expmap_header):
            if h is not None and "OPTAX_RA" in h and "OPTAX_DEC" in h:
                from astropy.wcs import WCS

                x, y = WCS(img_header).all_world2pix(float(h["OPTAX_RA"]), float(h["OPTAX_DEC"]), 0)
                return float(x), float(y), "OPTAX_RA"
        for h in (expmap_header, img_header):
            if h is not None and "XCEN" in h and "YCEN" in h:
                return float(h["XCEN"]) - 1.0, float(h["YCEN"]) - 1.0, "XCEN"
    oax = float(img_header.get("CRPIX1", nx / 2)) - 1.0
    oay = float(img_header.get("CRPIX2", ny / 2)) - 1.0
    return oax, oay, ("CRPIX" if mode == "crpix" else "CRPIX(fallback)")

@cites("xmm_ccf_sas", "sanders2018")
def preprocess_one(spec, ra, dec, r500_guess_kpc, kpc_per_arcsec,
                   crop_r500=3.0, target_pixsize_as=10.0, min_halfbox=48,
                   crop_cap_arcmin=None,
                   sources=None, mask_rad_as=30.0, protect_core_as=16.487,
                   optical_axis="crpix", rebin_centre="ox/f", ccf=None, pn_oot=None,
                   pn_oot_energy=None, oot_band_specs=None, bg_seed="annulus", img_name=None,
                   mask_radius="fixed", annulus_split=False, qpb_provider=None, vignetting="expmap",
                 psf_kt_keV=None):
    """Return a dict (counts, expmap, expmap_novig, pixsize_as, origin, bg_init,
    bg_src, npix, optical_axis_key, oot_template, oot_info) for one per-camera
    band image.

    ``vignetting = "sub-band"`` (``massfit.vignetting_model``) adds ``expmap_unvig``, the unvignetted exposure the
    cluster model of :func:`tngxmm.massfit.band_response.subband_cluster` is multiplied by: the camera's 0.5-1 keV
    exposure map divided by the CCF vignetting at 0.75 keV about the optical axis (where the vignetting barely
    depends on energy, whatever energy the map was vignetted at), carried to this band with every mask and scale of
    its exposure map; and ``vig_axis`` (x, y), the optical axis on the cropped and rebinned grid.

    If ``sources`` (list of (ra, dec, detml)) is given, point-source pixels are
    masked by zeroing the exposure map there (so they drop out of the image, the
    background and the coverage at once), except within ``protect_core_as`` of
    the cluster centre (protects the BCG / cool core).  ``mask_radius``: ``"fixed"`` masks
    ``mask_rad_as`` (30") around every source; ``"psf"`` the PSF-derived radius of the campaign Lx photometry
    (:func:`~tngxmm.massfit.mask_point_sources.psf_mask_radii_as`: 30" to 150" by the source's counts).

    ``pn_oot="data-template"`` (pn images only): the out-of-time template of
    :func:`tngxmm.massfit.pn_oot.oot_template`, built on the full native frame
    from the unmasked counts, cropped and rebinned like the counts
    (``oot_template``); a mock's exposure maps are scaled by (1 - f_OoT).  With
    ``pn_oot_energy = "ccf-cti"`` and the camera's band specs ``oot_band_specs``, the templates of
    all bands are built together with the CTI energy mis-correction of the OoT events
    (:func:`tngxmm.massfit.pn_oot.oot_templates`; the CTI line of the image's ``SUBMODE`` and
    ``DATE-OBS``, :func:`tngxmm.instrument.pn_oot_energy.cti_line`; cached per camera).

    ``bg_seed`` (the background seed when the r > 2 R500 annulus is off the detector; spec field
    ``massfit.bg_seed``): ``"annulus"`` falls back to the outer 40% of the crop measured
    from the frame corner and then to the constant ln SB = -15; ``"exposed"`` measures that outer
    ring over the exposed pixels only."""
    if bg_seed not in BG_SEED_MODES:
        raise ValueError(f"bg_seed must be one of {BG_SEED_MODES}, got {bg_seed!r}")
    if pn_oot not in PN_OOT_MODES:
        raise ValueError(f"pn_oot must be one of {PN_OOT_MODES}, got {pn_oot!r}")
    import numpy as np
    from astropy.io import fits

    img = fits.getdata(spec["img"]).astype(np.float64)
    exp = fits.getdata(spec["expmap"]).astype(np.float64)
    hdr = spec["header"]
    native_pix_as = round(abs(hdr["CDELT1"]) * 3600.0, 6)      # CDELT [deg] carries float rounding

    cx, cy = cluster_pixel(hdr, ra, dec)
    exp_raw = exp

    if sources:
        from tngxmm.massfit.mask_point_sources import psf_mask_radii_as, source_mask
        if mask_radius not in ("fixed", "psf"):
            raise ValueError(f"mask_radius must be 'fixed' or 'psf', got {mask_radius!r}")
        radii = psf_mask_radii_as(img, exp, hdr, native_pix_as, sources) if mask_radius == "psf" else None
        keep = source_mask(img.shape, hdr, native_pix_as, sources, ra, dec,
                           mask_rad_as=mask_rad_as, protect_core_as=protect_core_as, radii_as=radii)
        exp = exp * keep

    # unvignetted exposure map on the NATIVE grid: divide out the CCF vignetting
    # curve about the optical axis, at the band mean energy.
    camera = spec.get("camera")
    e_mean = 0.5 * (spec.get("emin_keV", 1.0) + spec.get("emax_keV", 2.0))
    axis_key = None
    if camera in _XRT:
        exp_hdr = None
        if optical_axis == "header":
            exp_hdr = spec.get("expmap_header")
            if exp_hdr is None:
                exp_hdr = fits.getheader(spec["expmap"])
        oax, oay, axis_key = optical_axis_pixel(hdr, exp_hdr, img.shape, optical_axis)
        vig = vig_map(camera, e_mean, img.shape[0], img.shape[1], oax, oay,
                      native_pix_as / 60.0, ccf=ccf)
        exp_novig = np.where(exp > 0, exp / vig, 0.0)
    else:
        vig = None
        exp_novig = exp.copy()
    unvig_ratio = None
    if vignetting == "sub-band":
        if camera not in _XRT:
            raise ValueError(f"vignetting 'sub-band' needs the camera of {spec['img']}")
        from tngxmm.instrument.effective_area import vignetting as ccf_vignetting
        from tngxmm.massfit.band_response import UNVIG_BAND, UNVIG_E_KEV

        e2 = exp_raw if spec.get("band") == UNVIG_BAND else fits.getdata(
            str(spec["expmap"]).replace(f"EXPMAP{spec['band']}", f"EXPMAP{UNVIG_BAND}")).astype(np.float64)
        yy0, xx0 = np.mgrid[0:img.shape[0], 0:img.shape[1]]
        v075 = ccf_vignetting(camera, np.hypot(xx0 - oax, yy0 - oay) * native_pix_as / 60.0, UNVIG_E_KEV,
                              ccf or ccf_dir())
        u_raw = np.where(v075 > 0, e2 / np.maximum(v075, 1e-30), 0.0)
        unvig_ratio = np.where(exp_raw > 0, u_raw / np.where(exp_raw > 0, exp_raw, 1.0), 0.0)

    # pn out-of-time events: fixed template from the full native frame, and the
    # in-time exposure of a mock (OOTCORR = F) scaled by (1 - f_OoT)
    oot_t, oot_info = None, None
    if pn_oot == "data-template" and camera == "PN":
        from tngxmm.massfit import pn_oot as po

        exp_hdr_oot = spec.get("expmap_header")
        if exp_hdr_oot is None:
            exp_hdr_oot = fits.getheader(spec["expmap"])
        fr = po.oot_fraction(hdr, exp_hdr_oot)
        ep = os.path.basename(str(spec["expmap"]))
        geom = po.readout_geometry(exp_raw, key=(os.path.dirname(str(spec["expmap"])),
                                                 ep.split("EXPMAP")[0], exp_raw.shape))
        exp_novig_raw = np.where(exp_raw > 0, exp_raw / vig, 0.0) if vig is not None else exp_raw
        k_rec = None
        if pn_oot_energy not in (None, "none", "ccf-cti"):
            raise ValueError(f"pn_oot_energy must be None, 'none' or 'ccf-cti', got {pn_oot_energy!r}")
        if pn_oot_energy == "ccf-cti" and oot_band_specs:
            from tngxmm.instrument.pn_oot_energy import cti_line

            date = str(hdr.get("DATE-OBS", "")).strip()
            if not date:
                raise ValueError(f"{spec['img']}: no DATE-OBS, needed by pn_oot_energy 'ccf-cti'")
            line = cti_line(str(hdr.get("SUBMODE", "")).strip() or None, date)
            tkey = (os.path.dirname(str(spec["expmap"])), ep.split("EXPMAP")[0], pn_oot_energy, line.submode,
                    round(line.mjd, 3), float(fr["f"]), exp_raw.shape)
            if tkey not in _OOT_CACHE:
                cb = {b["band"]: ((b["emin_keV"], b["emax_keV"]), fits.getdata(b["img"]).astype(np.float64))
                      for b in oot_band_specs}
                k_band = [float(v) for v in line.mean_k([(e[0] * e[1]) ** 0.5 for e, _ in cb.values()])]
                _OOT_CACHE.clear()
                _OOT_CACHE[tkey] = (po.oot_templates(cb, exp_novig_raw, fr["f"], geom["phi_deg"],
                                                     geom["s_boundary_px"], native_pix_as, k_band), k_band)
            oot_all, k_rec = _OOT_CACHE[tkey]
            oot_t = oot_all[spec["band"]]
        else:
            oot_t = po.oot_template(img, exp_novig_raw, fr["f"], geom["phi_deg"], geom["s_boundary_px"],
                                    native_pix_as)
        scale = 1.0 if fr["exposure_corrected"] else 1.0 - fr["f"]
        exp, exp_novig = exp * scale, exp_novig * scale
        oot_info = dict(fr, in_time_exposure_scale=scale, **geom, energy=str(pn_oot_energy or "none"), k_per_row=k_rec,
                        template_counts_native=float(oot_t.sum()))

    # particle background and soft protons as XMM-ESAS / X-COP (massfit.qpb_model = 'esas-fwc-corners'): the provider
    # gives the expected QPB counts and the relative soft-proton map on the native frame (massfit.esas_background)
    qpb_t = sp_t = None
    if qpb_provider is not None:
        got = qpb_provider(spec, img.shape)
        qpb_t, sp_t = got[0], got[1]
        ccd_map = got[2] if len(got) > 2 else None
        if spec.get("mask_ccds"):
            # MOS CCDs in the anomalous state left out, as X-COP and XMM-ESAS do (spec 'mask_ccds', set by the fit)
            if ccd_map is None:
                raise ValueError(f"{spec['img']}: CCDs {spec['mask_ccds']} to mask but no CCD map")
            off = np.isin(np.asarray(ccd_map), [int(c) for c in spec["mask_ccds"]])
            exp, exp_novig = np.where(off, 0.0, exp), np.where(off, 0.0, exp_novig)
    exp_unvig = None if unvig_ratio is None else exp * unvig_ratio     # every mask and scale of exp carried over
    # crop half-width in native pixels: crop_r500 * R500, floor for tiny clusters
    r500_as = r500_guess_kpc / kpc_per_arcsec
    half = max(int(round(crop_r500 * r500_as / native_pix_as)), min_halfbox)
    # cap the crop at a fixed ANGULAR radius so the fit region never leaves the
    # field (3 R500 exceeds the ~13.5' usable radius once theta500 > 4.5'); the
    # same cap applies to mocks and observations.
    if crop_cap_arcmin is not None and crop_cap_arcmin > 0:
        half = min(half, int(round(crop_cap_arcmin * 60.0 / native_pix_as)))
    x0, x1 = int(round(cx)) - half, int(round(cx)) + half
    y0, y1 = int(round(cy)) - half, int(round(cy)) + half
    # clip to frame
    x0, y0 = max(x0, 0), max(y0, 0)
    x1, y1 = min(x1, img.shape[1]), min(y1, img.shape[0])
    img = img[y0:y1, x0:x1]
    exp = exp[y0:y1, x0:x1]
    exp_novig = exp_novig[y0:y1, x0:x1]
    if exp_unvig is not None:
        exp_unvig = exp_unvig[y0:y1, x0:x1]
    if oot_t is not None:
        oot_t = oot_t[y0:y1, x0:x1]
    if qpb_t is not None:
        qpb_t, sp_t = qpb_t[y0:y1, x0:x1], sp_t[y0:y1, x0:x1]
    ox, oy = cx - x0, cy - y0                      # cluster in cropped frame
    axis_c = None if unvig_ratio is None else (oax - x0, oay - y0)

    # rebin factor to reach ~target pixel size
    f = max(int(round(target_pixsize_as / native_pix_as)), 1)
    if f > 1:
        img = rebin_sum(img, f)
        exp = rebin_mean(exp, f)
        exp_novig = rebin_mean(exp_novig, f)
        if oot_t is not None:
            oot_t = rebin_sum(oot_t, f)
        if qpb_t is not None:
            qpb_t, sp_t = rebin_sum(qpb_t, f), rebin_mean(sp_t, f)
        if exp_unvig is not None:
            exp_unvig = rebin_mean(exp_unvig, f)
        ox, oy = rebin_origin(ox, f, rebin_centre), rebin_origin(oy, f, rebin_centre)
        if axis_c is not None:
            axis_c = (rebin_origin(axis_c[0], f, rebin_centre), rebin_origin(axis_c[1], f, rebin_centre))
    pix_as = native_pix_as * f

    # mbproj2d requires even dimensions (optimal_size=False path)
    ny, nx = img.shape
    ny -= ny % 2
    nx -= nx % 2
    img, exp, exp_novig = img[:ny, :nx], exp[:ny, :nx], exp_novig[:ny, :nx]
    if exp_unvig is not None:
        exp_unvig = exp_unvig[:ny, :nx]
    if oot_t is not None:
        oot_t = oot_t[:ny, :nx]
        oot_info["template_counts_fit"] = float(oot_t[exp > 0].sum())
    if qpb_t is not None:
        qpb_t, sp_t = qpb_t[:ny, :nx], sp_t[:ny, :nx]

    # seed background from outer annulus (r > 2 R500), else outer frame
    yy, xx = np.mgrid[0:ny, 0:nx]
    r_as = np.hypot(xx - ox, yy - oy) * pix_as
    outer = (exp > 0) & (r_as > 2.0 * r500_as)
    bg_src = "annulus"                       # clean: beyond 2 R500 and on-detector
    bg_vf_init = None
    if outer.sum() < 50:
        # "annulus": r_as.max() is the distance to the frame corner, outside the field of view, so
        # the ring is empty for a crop clipped to the detector; "exposed": the exposed pixels
        rmax = r_as[exp > 0].max() if (bg_seed != "annulus" and (exp > 0).any()) else r_as.max()
        outer = (exp > 0) & (r_as > 0.6 * rmax)
        bg_src = "fallback"                  # outer 40% of the crop: INSIDE the cluster for large theta500
    # counts = exp(param) * pix_as^2 * expmap  ->  param = log(mean SB / pix^2)
    if outer.sum() >= 10:
        # the pn OoT template is a separate model component: it must not also enter the
        # background seed (it would be counted twice under the 0.05 dex seed prior)
        cts = img[outer].sum() - (oot_t[outer].sum() if oot_t is not None else 0.0)
        sb = max(cts, 0.0) / (exp[outer].sum() * pix_as ** 2)
        bg_init = math.log(max(sb, 1e-12))
    else:
        bg_init = -15.0
        bg_src = "none"                      # no usable pixels: arbitrary constant
    if annulus_split and bg_src == "annulus":
        # the annulus beyond 2 R500 decomposed into the model's two components (fitfix annulus_seed = 'split'):
        # counts = a E + b rho F with rho = median(E / F) (mbproj2d rescale_novig), Poisson maximum likelihood
        # (instrument.donor_background.fit_two_poisson); the seed is then in the model's own units,
        # ln SB = ln((a + b) / pix^2) and the vignetted fraction a / (a + b), measured on the data
        from tngxmm.instrument.donor_background import fit_two_poisson

        ok = (exp > 0) & (exp_novig > 0)
        rho = float(np.median(exp[ok] / exp_novig[ok])) if ok.any() else 1.0
        c_out = img[outer].astype(float) - (oot_t[outer] if oot_t is not None else 0.0)
        a_, b_ = fit_two_poisson(np.clip(c_out, 0.0, None), exp[outer], rho * exp_novig[outer])
        if a_ + b_ > 0:
            fv = min(max(a_ / (a_ + b_), 0.01), 0.99)
            bg_init = math.log((a_ + b_) / pix_as ** 2)
            bg_vf_init = math.log(fv / (1.0 - fv))
    return dict(counts=img.astype(np.float32), expmap=exp.astype(np.float32),
                expmap_novig=exp_novig.astype(np.float32),
                pixsize_as=pix_as, origin=(oy, ox), bg_init=bg_init, bg_src=bg_src, bg_vf_init=bg_vf_init,
                npix=ny * nx, optical_axis_key=axis_key,
                oot_template=None if oot_t is None else oot_t.astype(np.float32), oot_info=oot_info,
                qpb_template=None if qpb_t is None else qpb_t.astype(np.float32),
                sp_template=None if sp_t is None else sp_t.astype(np.float32),
                expmap_unvig=None if exp_unvig is None else exp_unvig.astype(np.float32),
                vig_axis=None if axis_c is None else (float(axis_c[0]), float(axis_c[1])))

@cites("mbproj2d", "sanders2018")
def build_images(specs, ra, dec, r500_guess_kpc, kpc_per_arcsec,
                 response_for, crop_r500=3.0, target_pixsize_as=10.0,
                 crop_cap_arcmin=None,
                 sources=None, return_sources=False,
                 optical_axis="crpix", rebin_centre="ox/f", ccf=None, mb=None, info=None,
                 psf=None, pn_oot=None, pn_oot_energy=None, bg_seed="annulus", protect_core_as=16.487,
                 mask_radius="fixed", annulus_split=False, qpb_provider=None, vignetting="expmap",
                 psf_kt_keV=None):
    """mbproj2d Image list + per-image background seeds (log SB) [+ seed sources].

    ``info``: optional list that receives one dict per image (name, optical-axis
    key, rebin factor origin) for the provenance.

    ``sources``: optional list of (ra, dec, detml) point sources to mask.

    ``psf=None``: like Sanders+2018 (MBProj2) the PSF is not modelled.
    Because the identical spatial pipeline is applied to mocks and observations,
    PSF smearing is common-mode and enters the measured mass bias.  ``psf="king-onaxis"`` and ``pn_oot="data-template"``: see the module
    docstring; a pn image's OoT template is attached as ``image.oot_template``
    (counts, the image's padded shape), the fit adds it as a fixed component."""
    import numpy as np

    if psf not in PSF_MODES:
        raise ValueError(f"psf must be one of {PSF_MODES}, got {psf!r}")
    if pn_oot not in PN_OOT_MODES:
        raise ValueError(f"pn_oot must be one of {PN_OOT_MODES}, got {pn_oot!r}")
    if mb is None:
        from tngxmm.massfit.patch_apec import import_mbproj2d
        mb = import_mbproj2d()
    images, bg_inits, bg_srcs = [], [], []
    for s in specs:
        pp = preprocess_one(s, ra, dec, r500_guess_kpc, kpc_per_arcsec,
                            crop_r500=crop_r500, target_pixsize_as=target_pixsize_as,
                            crop_cap_arcmin=crop_cap_arcmin,
                            sources=sources, optical_axis=optical_axis,
                            rebin_centre=rebin_centre, ccf=ccf, pn_oot=pn_oot, pn_oot_energy=pn_oot_energy,
                            protect_core_as=protect_core_as, bg_seed=bg_seed, mask_radius=mask_radius,
                            annulus_split=annulus_split, qpb_provider=qpb_provider, vignetting=vignetting,
                            img_name=f"{s['camera']}_{s['band']}",
                            oot_band_specs=[b for b in specs if b.get("camera") == "PN"])
        if pp is None:
            continue
        rmf, arf = response_for(s["header"])
        mask = pp["expmap"] > 0
        expmaps = {"expmap": pp["expmap"], "expmap_novig": pp["expmap_novig"]}
        if pp.get("expmap_unvig") is not None:      # vignetting 'sub-band': the cluster's exposure
            expmaps["expmap_unvig"] = pp["expmap_unvig"]
            mask = mask & (pp["expmap_unvig"] > 0)
        psf_obj, psf_info, pad = None, None, 0
        if psf == "king-onaxis":
            from tngxmm.massfit import fit_psf

            psf_obj, psf_info = fit_psf.king_psf_for(s["camera"], s["header"], s["emin_keV"],
                                                     s["emax_keV"], ccf=ccf)
            pad = fit_psf.pad_pixels(pp["pixsize_as"])
        elif psf == "king-angle":
            from tngxmm.massfit import fit_psf

            if pp.get("vig_axis") is None:
                raise ValueError("psf 'king-angle' needs the optical axis of the image (vignetting 'sub-band')")
            if psf_kt_keV is None:
                raise ValueError("psf 'king-angle' needs the first-guess temperature (psf_kt_keV)")
            psf_obj, psf_info = fit_psf.king_angle_psf_for(s["camera"], s["header"], s["emin_keV"],
                                                           s["emax_keV"], ccf=ccf, arf=str(arf),
                                                           kt_keV=float(psf_kt_keV))
            pad = fit_psf.pad_pixels(pp["pixsize_as"])
        image = mb.Image(
            f"{s['camera']}_{s['band']}", pp["counts"],
            emin_keV=s["emin_keV"], emax_keV=s["emax_keV"],
            rmf=str(rmf), arf=str(arf),
            pixsize_as=pp["pixsize_as"],
            expmaps=expmaps,
            mask=mask,
            origin=pp["origin"],
            psf=psf_obj,
            optimal_size=psf_obj is not None,     # zero padding against FFT wrap-around
            pad=pad,
        )
        if pp.get("oot_template") is not None:
            t = np.zeros(image.shape, dtype=np.float32)
            t[:pp["oot_template"].shape[0], :pp["oot_template"].shape[1]] = pp["oot_template"]
            image.oot_template = t
        for attr in ("qpb_template", "sp_template"):              # massfit.qpb_model = 'esas-fwc-corners'
            if pp.get(attr) is not None:
                t = np.zeros(image.shape, dtype=np.float32)
                t[:pp[attr].shape[0], :pp[attr].shape[1]] = pp[attr]
                setattr(image, attr, t)
        if pp.get("bg_vf_init") is not None:
            image.bg_vf_init = float(pp["bg_vf_init"])        # start of the vignetted fraction
        if pp.get("vig_axis") is not None:
            image.vig_axis_px = pp["vig_axis"]                # vignetting 'sub-band': optical axis (x, y) on the grid
            if hasattr(image.psf, "set_axis"):
                image.psf.set_axis(pp["vig_axis"])            # psf 'king-angle': the off-axis angle of each pixel
        images.append(image)
        if info is not None:
            info.append(dict(image=f"{s['camera']}_{s['band']}", optical_axis_key=pp.get("optical_axis_key"),
                             origin_yx=[float(v) for v in pp["origin"]], bg_src=pp.get("bg_src"),
                             **({"psf": psf_info} if psf_info else {}),
                             **({"pn_oot": pp["oot_info"]} if pp.get("oot_info") else {})))
        bg_inits.append(pp["bg_init"])
        bg_srcs.append(pp.get("bg_src", "annulus"))
    return (images, bg_inits, bg_srcs) if return_sources else (images, bg_inits)
