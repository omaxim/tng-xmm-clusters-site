Source code for tngxmm.massfit.fit_psf

"""The EPIC King PSF in the fit model: kernel, calibration and convolution (on axis, or at each pixel's off-axis
angle: :class:`KingAnglePSF`, ``massfit.psf = 'king-angle'``).

Model.  The point-spread function is the King profile of the ``XRT*_XPSF``
CCF (``KING_PARAMS``),

    K(r) = (alpha - 1) / (pi r0^2) (1 + r^2/r0^2)^-alpha     [per arcsec^2],

normalised to 1 over the plane; the encircled fraction is
EEF(R) = 1 - (1 + R^2/r0^2)^(1 - alpha).  With alpha = 1.4-1.6 the wings are
heavy: 6-11 % of the flux lies beyond 1', 0.4-1.6 % beyond 10'.

Calibration (:func:`king_params`).
* Mocks (header ``PSFMODEL = KING-ONAXIS``): the first row of ``KING_PARAMS``
  of the header's ``PSFFILE`` (225 eV, on axis), exactly what the instrument
  stage draws its photon offsets from (``instrument.load_responses``:
  ``pars[0]``), for every band.
* Real data: the on-axis rows of the camera's newest ``XPSF`` CCF interpolated
  (linear in log E) to the band's effective energy sqrt(E_min E_max), clipped
  to 0.225-10.25 keV (the 13.25 keV row has alpha < 1 and is not used).  The
  energy dependence is weak (pn r0 5.5 -> 5.1", alpha 1.59 -> 1.52 from 0.7 to
  7 keV); the off-axis broadening is not modelled.

Kernel (:class:`KingPSF`), built per image geometry and cached:
* core, offsets with pixel-centre distance <= ``CORE_RADIUS_ARCSEC``: the
  King profile INTEGRATED over each image pixel (sub-sampled on a symmetric
  sub-pixel grid, finer in the central pixels), centred exactly on pixel
  (0, 0) of the wrapped FFT grid, so it is symmetric under x -> -x, y -> -y
  and x <-> y and introduces no shift;
* tail, everything beyond: convolved on a coarse grid (blocks of b x b image
  pixels, b ~ ``TAIL_PIX_ARCSEC`` / pixel) with the exact block-averaged
  transfer function (a source uniform in its block, the received flux spread
  uniformly over the receiving block), zero-padded to twice the image so the
  tail reaches across the whole image without wrap-around; flux scattered
  beyond the image leaves it, as in the data;
* normalisation: core + tail over the infinite plane = 1 exactly; the core
  sum is set to 1 - (1 + R_eq^2/r0^2)^(1 - alpha), R_eq the radius of the
  circle with the core's area (no renormalisation of a truncated kernel).

Why not mbproj2d's own ``PSF``: it resamples a tabulated kernel onto the image
grid without centring it on a pixel (an 8" pixel and a 1" kernel table left
the kernel 0.055 px off centre with 30 % asymmetric neighbours), and the
image had to be padded by the full kernel radius (600"), which made the
low-z fits 2-3 times slower.  Here the image is padded by the core radius
only (``pad_pixels``).

Off axis (:class:`KingAnglePSF`).  The XPSF calibration tabulates the King
parameters in energy and off-axis angle (KING_PARAMS, 0-14' in steps of 2';
the 'Extended' PSF of Read et al. 2011): off axis the slope falls (MOS1 at 6
keV: alpha 1.37 on axis, 1.25 at 12'), so the wings hold more of the flux.
The instrument stage draws the cluster photons from that table at each
photon's energy and angle; a model with the on-axis King then misplaces the
wings of a cluster that fills the field (STATUS B13: the deep 8e14 Msun
cell at z = 0.06 fitted 0.96 of its mass, 0.995 with this PSF).  The model is
a sum over the tabulated angles of the node's kernel applied to the image
times the pixel's hat weight; each node's kernel is the count-weighted sum of
the King profiles at three energies of the band (:func:`band_energy_mixture`,
:class:`KingSumPSF`), as the stage scatters each photon at its own energy.

References
----------
xmm_ccf_sas (XPSF CCF, KING_PARAMS), xmm_simulator (the instrument stage's
King sampler), read2011 (the King PSF in energy and off-axis angle),
mbproj2d (image model, ``ConvPSFHelper``), pyfftw / fftw.
"""

from __future__ import annotations

import glob
import math
import os
from typing import Any

from tngxmm.cites import cites, collect_cites

_XRT = {"PN": "XRT3", "MOS1": "XRT1", "MOS2": "XRT2"}
#: radius of the directly convolved core [arcsec]; the image is padded by it
CORE_RADIUS_ARCSEC = 120.0
#: target size of the coarse tail grid pixels [arcsec]
TAIL_PIX_ARCSEC = 32.0
#: energy range of the CCF rows used for real data [keV]
CCF_E_RANGE_KEV = (0.225, 10.25)
#: kernel version recorded in the provenance
KERNEL_VERSION = "king-v2: pixel-integrated, centred core 120\" + block-averaged tail, unit total"

_PARAM_CACHE: dict = {}
_KERNEL_CACHE: dict = {}

def ccf_dir() -> str:
    """The directory of the XMM calibration files (``CCF_PATH``; :func:`tngxmm.massfit.prepare_images.ccf_dir`)."""
    from tngxmm.massfit.prepare_images import ccf_dir as _c

    return _c()




def _newest(camera: str, ccf: str) -> str:
    files = glob.glob(os.path.join(ccf, f"{_XRT[camera]}_XPSF_*.CCF"))
    if not files:
        raise FileNotFoundError(f"no {_XRT[camera]}_XPSF_*.CCF in {ccf}")
    return os.path.basename(sorted(files, key=lambda p: int(p.split("_")[-1].split(".")[0]))[-1])


def _onaxis_table(path: str):
    """(E [keV], r0 [arcsec], alpha) of the on-axis rows, and the first row."""
    import numpy as np
    from astropy.io import fits

    with fits.open(path) as f:
        d = f["KING_PARAMS"].data
        e = np.asarray(d["ENERGY"], float) / 1e3
        th = np.asarray(d["THETA"], float)
        p = np.asarray(d["PARAMS"], float)
    on = th == 0
    order = np.argsort(e[on])
    return e[on][order], p[on][order, 0], p[on][order, 1], (float(p[0, 0]), float(p[0, 1]))

@cites("xmm_ccf_sas", "xmm_simulator")
def king_params(camera: str, header=None, emin_keV: float | None = None, emax_keV: float | None = None,
                ccf: str | None = None) -> dict[str, Any]:
    """King (r0 [arcsec], alpha) of one camera image and how they were chosen.

    Mock products (``PSFMODEL = KING-ONAXIS``): first row of the header's
    ``PSFFILE`` (the instrument stage's PSF).  Otherwise: the on-axis rows of
    the newest CCF of the camera's mirror at sqrt(emin emax) (log-E linear)."""
    import numpy as np

    ccf = ccf or ccf_dir()
    mock = header is not None and str(header.get("PSFMODEL", "")).strip().upper().startswith("KING-ONAXIS")
    name = None
    if mock and header.get("PSFFILE"):
        cand = os.path.basename(str(header["PSFFILE"]).strip())
        if os.path.exists(os.path.join(ccf, cand)):
            name = cand
    name = name or _newest(camera, ccf)
    if name not in _PARAM_CACHE:
        _PARAM_CACHE[name] = _onaxis_table(os.path.join(ccf, name))
    e, r0s, als, first = _PARAM_CACHE[name]
    if mock or emin_keV is None or emax_keV is None:
        return dict(r0_as=first[0], alpha=first[1], psf_file=name,
                    calibration="instrument stage (KING_PARAMS row 0: 225 eV, on axis)")
    eeff = float(np.clip(math.sqrt(emin_keV * emax_keV), *CCF_E_RANGE_KEV))
    ok = (e >= CCF_E_RANGE_KEV[0] - 1e-9) & (e <= CCF_E_RANGE_KEV[1] + 1e-9)
    le = np.log(e[ok])
    r0 = float(np.interp(math.log(eeff), le, r0s[ok]))
    alpha = float(np.interp(math.log(eeff), le, als[ok]))
    return dict(r0_as=r0, alpha=alpha, psf_file=name, calibration=f"CCF on-axis at {eeff:.3f} keV")

def king_density(r_as, r0_as: float, alpha: float):
    """King surface density [per arcsec^2], unit integral over the plane."""
    import numpy as np

    return (alpha - 1.0) / (math.pi * r0_as ** 2) * (1.0 + (np.asarray(r_as, float) / r0_as) ** 2) ** (-alpha)

def king_eef(r_as, r0_as: float, alpha: float):
    """Encircled fraction within radius ``r_as``."""
    import numpy as np

    return 1.0 - (1.0 + (np.asarray(r_as, float) / r0_as) ** 2) ** (1.0 - alpha)

def pad_pixels(pix_as: float, core_radius_as: float = CORE_RADIUS_ARCSEC) -> int:
    """Zero padding [image px] that keeps the core convolution free of wrap-around."""
    return int(math.ceil(core_radius_as / pix_as)) + 1




def _pixel_integrated(dx, dy, pix_as: float, r0_as: float, alpha: float, nsub: int):
    """King integrated over the pixels at integer offsets (dx, dy) [px] (midpoint rule
    on a symmetric nsub x nsub sub-grid)."""
    import numpy as np

    sub = ((np.arange(nsub) + 0.5) / nsub - 0.5)
    out = np.zeros(np.broadcast(dx, dy).shape)
    for sy in sub:
        for sx in sub:
            out += king_density(np.hypot((dx + sx) * pix_as, (dy + sy) * pix_as), r0_as, alpha)
    return out * (pix_as ** 2) / nsub ** 2

@cites("xmm_ccf_sas")
def core_kernel(r0_as: float, alpha: float, pix_as: float, core_radius_as: float = CORE_RADIUS_ARCSEC,
                nsub: int = 7, nsub_central: int = 41, central_px: int = 2):
    """(kernel (2h+1)^2 centred at [h, h], h, total) of the core: King integrated over
    each pixel whose centre lies within ``core_radius_as``; its sum is
    1 - (1 + R_eq^2/r0^2)^(1 - alpha) with R_eq = sqrt(N_core / pi) pixels."""
    import numpy as np

    h = int(math.floor(core_radius_as / pix_as))
    ax = np.arange(-h, h + 1, dtype=float)
    dx, dy = np.meshgrid(ax, ax)
    inside = np.hypot(dx, dy) * pix_as <= core_radius_as
    k = _pixel_integrated(dx, dy, pix_as, r0_as, alpha, nsub)
    c = (np.abs(dx) <= central_px) & (np.abs(dy) <= central_px)
    k[c] = _pixel_integrated(dx[c], dy[c], pix_as, r0_as, alpha, nsub_central)
    k = np.where(inside, k, 0.0)
    # exact symmetry (x <-> -x, y <-> -y, x <-> y), then the analytic normalisation
    k = 0.5 * (k + k[::-1, :])
    k = 0.5 * (k + k[:, ::-1])
    k = 0.5 * (k + k.T)
    r_eq = math.sqrt(inside.sum() / math.pi) * pix_as
    total = float(king_eef(r_eq, r0_as, alpha))
    k *= total / k.sum()
    return k, h, total

def tail_transfer(r0_as: float, alpha: float, pix_as: float, b: int, cy: int, cx: int,
                  core_radius_as: float = CORE_RADIUS_ARCSEC):
    """Block-averaged tail transfer T[Dy, Dx] on the wrapped (2 cy, 2 cx) coarse grid:
    expected flux into block J from a unit source spread uniformly over block I,
    D = J - I, counting only fine offsets beyond the core radius."""
    import numpy as np
    from scipy.ndimage import convolve1d

    ny, nx = 2 * cy * b + 2 * b + 1, 2 * cx * b + 2 * b + 1
    oy, ox = ny // 2, nx // 2
    dy = (np.arange(ny) - oy)[:, None]
    dx = (np.arange(nx) - ox)[None, :]
    r = np.hypot(dx, dy) * pix_as
    kt = np.where(r > core_radius_as, king_density(r, r0_as, alpha) * pix_as ** 2, 0.0)
    if b > 1:
        w = (b - np.abs(np.arange(-(b - 1), b))) / b
        kt = convolve1d(convolve1d(kt, w, axis=0, mode="constant"), w, axis=1, mode="constant")
    Dy = np.arange(2 * cy)
    Dy = np.where(Dy >= cy, Dy - 2 * cy, Dy)
    Dx = np.arange(2 * cx)
    Dx = np.where(Dx >= cx, Dx - 2 * cx, Dx)
    return kt[oy + Dy[:, None] * b, ox + Dx[None, :] * b]

    def matchImage(self, imgshape, img_pixsize_as):
        """Build (or fetch from the cache) the wrapped core kernel, its FFT helper and
        the coarse tail transfer for this image shape and pixel size."""
        import numpy as np

        key = self.key(imgshape, img_pixsize_as)
        if key not in _KERNEL_CACHE:
            from mbproj2d import utils as mbu

            ny, nx = imgshape
            k, h, core_total = self._core(img_pixsize_as)
            if 2 * h + 1 > min(ny, nx):
                raise ValueError(f"image {imgshape} smaller than the PSF core ({2 * h + 1} px)")
            wrapped = mbu.zeros_aligned((ny, nx), dtype=np.float32)
            iy = np.arange(-h, h + 1) % ny
            ix = np.arange(-h, h + 1) % nx
            wrapped[np.ix_(iy, ix)] = k.astype(np.float32)
            conv = mbu.ConvPSFHelper(wrapped)
            tail = None
            if self.tail:
                import pyfftw

                b = max(1, int(round(self.tail_pix_as / img_pixsize_as)))
                cy, cx = -(-ny // b), -(-nx // b)
                T = self._tail_transfer(img_pixsize_as, b, cy, cx)
                big = mbu.zeros_aligned((2 * cy, 2 * cx), dtype=np.float32)
                spec = mbu.zeros_aligned((2 * cy, cx + 1), dtype=np.complex64)
                fwd = pyfftw.FFTW(big, spec, axes=(0, 1), direction="FFTW_FORWARD",
                                  flags=["FFTW_MEASURE"])
                bwd = pyfftw.FFTW(spec, big, axes=(0, 1), direction="FFTW_BACKWARD",
                                  flags=["FFTW_MEASURE"])
                big[:, :] = T / (b * b)        # the receiving block's flux spread over b x b pixels
                fwd()
                fT = spec.copy()
                big[:, :] = 0.0
                tail = dict(b=b, cy=cy, cx=cx, fT=fT, total=float(T.sum()), big=big, spec=spec,
                            fwd=fwd, bwd=bwd, blk=np.zeros((cy * b, cx * b), np.float32),
                            fine=np.zeros((cy * b, cx * b), np.float32))
            _KERNEL_CACHE[key] = dict(conv=conv, kernel=wrapped, core_total=core_total, h=h, tail=tail)
        c = _KERNEL_CACHE[key]
        self.shape = tuple(imgshape)
        self.conv, self.resample, self._tail = c["conv"], c["kernel"], c["tail"]
        self.core_total = c["core_total"]

    def tail_image(self, inimg):
        """The tail's contribution (flux scattered beyond the core radius) of ``inimg``
        (a view of an internal buffer, valid until the next call)."""
        self._tail_forward(inimg)
        t = self._tail
        t["spec"] *= t["fT"]
        return self._tail_backward(inimg.shape)



    def _tail_forward(self, inimg):
        """Block sums of ``inimg`` on the tail's coarse grid and their forward transform (in ``_tail['spec']``)."""
        import numpy as np

        t = self._tail
        b, cy, cx = t["b"], t["cy"], t["cx"]
        ny, nx = inimg.shape
        blk, big = t["blk"], t["big"]
        blk[:ny, :nx] = inimg                    # the rest of blk stays zero
        # b x b block sums by strided adds (much faster than a 4-d reshape-sum)
        rows = t.get("rows")
        if rows is None:
            rows = t["rows"] = np.empty((cy, cx * b), np.float32)
        np.copyto(rows, blk[0::b])
        for i in range(1, b):
            rows += blk[i::b]
        # the inverse transform filled all of big: zero what lies outside the block sums, then write them
        big[cy:, :] = 0.0
        big[:cy, cx:] = 0.0
        cs = big[:cy, :cx]
        cs[:, :] = rows[:, 0::b]
        for j in range(1, b):
            cs += rows[:, j::b]
        t["fwd"]()

    def _tail_backward(self, shape):
        """The inverse transform of ``_tail['spec']`` spread from the coarse grid over the image ``shape`` (a view of
        an internal buffer)."""
        t = self._tail
        b, cy, cx = t["b"], t["cy"], t["cx"]
        t["bwd"]()                               # pyfftw normalises the inverse transform
        # each block's value into its b x b pixels (one broadcast copy)
        fine = t["fine"]
        fine.reshape(cy, b, cx, b)[:, :, :, :] = t["big"][:cy, :cx][:, None, :, None]
        return fine[:shape[0], :shape[1]]

class KingSumPSF(KingPSF):
    """A weighted sum of King profiles [(weight, r0 [arcsec], alpha)] as one kernel: the core kernels and the tail
    transfers are summed with the weights (a convolution is linear), so the sum costs one convolution.  The weights
    sum to 1."""

    def __init__(self, components, core_radius_as: float = CORE_RADIUS_ARCSEC, tail_pix_as: float = TAIL_PIX_ARCSEC,
                 tail: bool = True):
        """``components``: [(weight, r0 [arcsec], alpha)]."""
        self.components = [(float(w), float(r), float(a)) for w, r, a in components]
        tot = sum(w for w, _, _ in self.components)
        if not self.components or abs(tot - 1.0) > 1e-6:
            raise ValueError(f"King sum: weights sum to {tot}, not 1")
        w0, r0, a0 = max(self.components)
        super().__init__(r0, a0, core_radius_as, tail_pix_as, tail)

            t = tail_transfer(r0, al, pix_as, b, cy, cx, self.core_radius_as)
            out = w * t if out is None else out + w * t
        return out

def angle_weights(theta_arcmin: Any, node_thetas: Any) -> list:
    """Linear (hat) weights of the nodes at ``node_thetas`` for the angles ``theta_arcmin`` (clipped to the nodes'
    range); they sum to 1 at every pixel."""
    import numpy as np

    th = np.clip(np.asarray(theta_arcmin, float), node_thetas[0], node_thetas[-1])
    out = []
    n = len(node_thetas)
    for j in range(n):
        w = np.zeros(th.shape, np.float32)
        if n == 1:
            w[:] = 1.0
        if j > 0:
            lo, c = node_thetas[j - 1], node_thetas[j]
            m = (th >= lo) & (th <= c)
            w[m] = (th[m] - lo) / (c - lo)
        if j < n - 1:
            c, hi = node_thetas[j], node_thetas[j + 1]
            m = (th >= c) & (th < hi)
            w[m] = (hi - th[m]) / (hi - c)
        out.append(w)
    return out

    def applyTo(self, inimg, minval=1e-10):
        """Convolve ``inimg`` (float32, the image shape) in place."""
        import numpy as np
        from mbproj2d import fast

        if self.weights is None:
            raise RuntimeError("king-angle PSF without an optical axis (set_axis)")
        active = self._active
        if active is None or self._buf[0].shape != inimg.shape:
            # the nodes with weight somewhere on the image (fixed once the axis is set) and two aligned buffers
            from mbproj2d import utils as mbu

            active = self._active = [(p, w) for p, w in zip(self.psfs, self.weights) if w.any()]
            self._buf = (mbu.zeros_aligned(inimg.shape, dtype=np.float32),
                         mbu.zeros_aligned(inimg.shape, dtype=np.float32))
            self._spec = None
            if all(p._tail is not None for p, _ in active):
                c, t = active[0][0].conv, active[0][0]._tail
                self._spec = (mbu.zeros_aligned(c.temp_cmplx.shape, dtype=np.complex64),
                              mbu.zeros_aligned(t["spec"].shape, dtype=np.complex64))
        out, tmp = self._buf
        if self._spec is not None:
            self._fourier_sum(inimg, active, out, tmp)
        else:
            out.fill(0.0)
            for p, w in active:
                np.multiply(inimg, w, out=tmp)      # float32 x float32, as (inimg * w).astype(np.float32)
                p.applyTo(tmp, minval=0.0)
                out += tmp
        inimg[:, :] = out
        fast.clip2DMax(inimg, minval)



    def _fourier_sum(self, inimg, active, out, tmp):
        """sum_j K_j (x) [w_j S] into ``out``, the sum taken over the transforms: per node the forward transforms of
        w_j S (the core on the image grid, the tail on its coarse grid) times the node's kernel transforms are
        added up, then one inverse transform of each.  A convolution is linear, so this is the sum of the node by
        node convolutions up to rounding; the node by node sum clipped each node's image at zero (the FFT's rounding
        noise below zero), which is left to the final clip."""
        import numpy as np

        core, tail = self._spec
        core.fill(0.0)
        tail.fill(0.0)
        for p, w in active:
            np.multiply(inimg, w, out=tmp)
            p._tail_forward(tmp)
            t = p._tail
            np.multiply(t["spec"], t["fT"], out=t["spec"])
            tail += t["spec"]
            c = p.conv
            c.fwd_fftimg.update_arrays(tmp, c.temp_cmplx)
            c.fwd_fftimg()
            np.multiply(c.temp_cmplx, c.temp_cmplxpsf, out=c.temp_cmplx)
            core += c.temp_cmplx
        p0 = active[0][0]
        c, t = p0.conv, p0._tail
        c.bkd_fftout.update_arrays(core, out)
        c.bkd_fftout()                          # pyfftw normalises the inverse transform
        t["spec"][:, :] = tail
        out += p0._tail_backward(inimg.shape)

@cites("xmm_ccf_sas", "sarazin1988")
def band_energy_mixture(arf: str, emin_keV: float, emax_keV: float, kt_keV: float,
                        parts: int = BAND_ENERGY_PARTS, ngrid: int = BAND_ENERGY_GRID) -> list[tuple[float, float]]:
    """[(weight, energy [keV])] of a band: the band cut into ``parts`` equal intervals in ln E, each with its share
    of the band's counts and its count-weighted mean energy.  Counts per energy: the image's ARF (``SPECRESP`` at
    the bin centre) times the photon spectrum of thermal bremsstrahlung at ``kt_keV``, dN/dE ~ exp(-E / kT) / E
    (the continuum of a hot plasma, Sarazin 1988; Gaunt factor 1, no lines, no absorption), integrated over ``ngrid`` bins equal in ln E.  The weights sum to 1."""
    import numpy as np
    from astropy.io import fits

    with fits.open(arf) as f:
        t = f["SPECRESP"].data
        ea = 0.5 * (np.asarray(t["ENERG_LO"], float) + np.asarray(t["ENERG_HI"], float))
        aa = np.asarray(t["SPECRESP"], float)
    edges = np.exp(np.linspace(math.log(emin_keV), math.log(emax_keV), ngrid + 1))
    e = np.sqrt(edges[1:] * edges[:-1])
    # dN = exp(-E/kT) / E dE, and dE = E d(ln E) on bins equal in ln E
    w = np.interp(e, ea, aa) * np.exp(-e / float(kt_keV))
    if not np.isfinite(w).all() or w.sum() <= 0:
        raise ValueError(f"{arf}: no effective area in {emin_keV}-{emax_keV} keV")
    w = w / w.sum()
    out = []
    for idx in np.array_split(np.arange(ngrid), int(parts)):
        ws = float(w[idx].sum())
        out.append((ws, float((w[idx] * e[idx]).sum() / ws)))
    return out

    info = dict(psf_file=p["psf_file"], kt_keV=float(kt_keV), energies=recs, kernel=kernel + "; count-weighted "
                f"mixture of {len(mix)} energies", calibration=f"CCF KING_PARAMS, {len(mix)} energies of the band",
                core_radius_as=CORE_RADIUS_ARCSEC, tail_pix_as=TAIL_PIX_ARCSEC)
    return KingAnglePSF(nodes), info
