Source code for tngxmm.massfit.guess_mass

"""First-guess M500 and c200 of a hydrostatic fit (optimiser start and the fit region; never the fitted quantity).

The first-guess R500 sets the fit's crop (``massfit.crop_r500`` R500), its pixel size, the model's outer radius and the
centre search of a mock, so it is taken from a published OBSERVED relation and the image itself, the same way for mocks
and real observations:

* relation: Bulbul et al. (2019), SPT-selected clusters observed with XMM-Newton, soft-band luminosity in the rest
  frame 0.5-2 keV inside R500, core included, against the SZ-based M500 (their eq. 26 and the values on p. 22)::

      L_X = 4.12e44 erg/s (M500 / 6.35e14 Msun)^1.89 (E(z) / E(0.45))^2 ((1 + z) / 1.45)^-0.20,  sigma_lnL = 0.27

  in the cosmology of that paper (flat LCDM, H0 = 67.74, Omega_M = 0.304), the pipeline's to within 0.005 in Omega_M,
  so no conversion is applied.  The band and aperture are those of the photometry here (measure_lx.seed_lx), so no
  band correction enters;
* iteration: R500 follows from the mass, the luminosity is measured on the image inside that R500 (with the ECF at
  the virial temperature of the trial mass) and inverted through the relation, until the mass changes by less than
  :data:`ITER_RTOL` in ln.  The procedure is the one of the MCXC (Piffaretti et al. 2011, Sect. 3.4), which iterates
  L500 and R500 from an L-M relation; it starts at the relation's pivot mass.  The fixed point is the root of
  g(ln M) = ln M_rel(L(R500(M))) - ln M, which falls with M wherever the measured luminosity grows more slowly than
  M^1.89 (always, for a cluster profile); the steps are those of the plain iteration while they stay inside the
  bracket of the root the signs of g have established so far, and bisect the bracket otherwise (an image whose noise or
  substructure makes the plain iteration oscillate);
* aperture beyond the field: the seed photometry fills the uncovered part of the aperture with the azimuthal mean of
  its covered pixels at the same radius, the ring fill that already replaces masked pixels
  (:func:`tngxmm.photometry.estimate_lx_background.ring_filled_sum`), up to the same limit of half the aperture;
* fallback: a trial mass whose luminosity is not measurable (flags ``nondetection``, ``masked``, ``uncovered``)
  bounds the search: before any trial was measurable the mass steps down a decade (its R500 shrinks into the field),
  after one it bisects back towards the last measurable trial; a root beyond what the image can measure keeps the
  largest measurable mass (``field-edge``); an image with no measurable trial keeps the pivot mass (``pivot``);
* c200 start: the median of Dutton & Maccio (2014) at the first-guess mass (:func:`c200_dutton14`); the fit's prior on
  c200 is flat (Sanders et al. 2018), so the start only places the optimiser.

References
----------
bulbul2019 (relation), piffaretti2011 (iteration), dutton2014 (c200), nfw (M500 / M200), cosmology_planck2015 (E(z)).
"""

from __future__ import annotations

import math
from pathlib import Path
from typing import Any

from tngxmm.cites import cites, collect_cites
from tngxmm.core.cosmology import OL0, OM0

OM, OL = OM0, OL0
#: Bulbul et al. (2019), eq. 26, L_X,cin 0.5-2 keV (p. 22): normalisation [erg/s], pivot mass [Msun], mass slope,
#: pivot redshift, E(z) exponent, (1 + z) exponent, scatter in ln L
BULBUL2019 = dict(norm_erg_s=4.12e44, m_piv_msun=6.35e14, slope=1.89, z_piv=0.45, ez_exp=2.0, opz_exp=-0.20,
                  sigma_lnl=0.27)
#: the tngxmm modules the centre-and-first-guess procedure loads when it runs: :func:`centre_and_seed` and its two
#: drivers, build_obs_seed_lx.measure_one (real observations) and build_fit_targets.measure_seed (mocks), with the
#: settings, ECF, cosmology, source-list and exposure-map code they call, under either source-list rule
#: (``massfit.source_list``: 'sas', the SAS lists; 'default', the PPS lists and the simulated sources above the
#: donor's threshold, as the closure grid's diagnostic arm).  The digest of each module's code
#: (core.hashing.python_code_digest: the syntax tree without docstrings) is part of the seed hash
#: (build_obs_seed_lx.seed_hash).  tests/unit/test_seed_hash.py runs the procedure in a fresh interpreter under both
#: rules and fails when it loads a tngxmm module that is neither here nor in SEED_PROCEDURE_UNHASHED, or no longer
#: loads one listed here
SEED_PROCEDURE_MODULES = (
    "tngxmm.cites", "tngxmm.core", "tngxmm.core.cosmology", "tngxmm.core.hashing", "tngxmm.core.paths",
    "tngxmm.core.site", "tngxmm.instrument", "tngxmm.instrument.compute_ecf", "tngxmm.instrument.point_sources",
    "tngxmm.massfit", "tngxmm.massfit.build_fit_targets", "tngxmm.massfit.build_source_thresholds",
    "tngxmm.massfit.guess_mass", "tngxmm.massfit.mask_point_sources",
    "tngxmm.photometry", "tngxmm.photometry._ecf_adapter", "tngxmm.photometry._settings",
    "tngxmm.photometry.build_obs_seed_lx", "tngxmm.photometry.catalogue_positions",
    "tngxmm.photometry.estimate_lx_background", "tngxmm.photometry.find_centre", "tngxmm.photometry.measure_lx",
    "tngxmm.photometry.measure_obs_lx")
#: modules the procedure loads whose code is not hashed, with the reason
SEED_PROCEDURE_UNHASHED = {
    "tngxmm": "the package version and the stage names; the version is recorded in each table's provenance",
    "tngxmm.config": "the presets: what the procedure reads from its preset is hashed as values, the spec sections "
                     "build_obs_seed_lx.SEED_SECTIONS",
    "tngxmm.instrument.donor_tables": "the provenance checks of the catalogue-position and source-threshold tables "
                                      "(check_table): they refuse a table and change no value of a seed; the tables' "
                                      "sha256 is in the hash (seed_data_files)",
    "tngxmm.core.provenance": "the reader of those sidecars: refuses, computes nothing of a seed",
    "tngxmm.obs": "where the packaged observation table lies, which those checks compare by its sha256",
}


_PROCEDURE_HASH: dict[tuple, str] = {}

def seed_procedure_hash() -> str:
    """sha256 of the code of :data:`SEED_PROCEDURE_MODULES`: each module's name and
    :func:`tngxmm.core.hashing.python_code_digest`, in that order.  Remembered per process until a module file
    changes (modification time and size): every reader of a seed table asks for it."""
    import hashlib
    import importlib.util
    import os

    from tngxmm.core.hashing import python_code_digest

    origins = []
    for name in SEED_PROCEDURE_MODULES:
        found = importlib.util.find_spec(name)
        if found is None or not found.origin:
            raise ModuleNotFoundError(f"{name}, a module of the seed procedure, has no source file")
        origins.append((name, found.origin))
    key = tuple((n, o, os.stat(o).st_mtime_ns, os.stat(o).st_size) for n, o in origins)
    if key not in _PROCEDURE_HASH:
        h = hashlib.sha256()
        for name, origin in origins:
            h.update(f"{name} {python_code_digest(origin)}\n".encode())
        _PROCEDURE_HASH[key] = h.hexdigest()
    return _PROCEDURE_HASH[key]

    """The data files the procedure reads besides each target's own images, exposure maps and source lists: the

@cites("cosmology_planck2015")
def Ez(z):
    """E(z) of the pipeline's flat LCDM (Om = 0.3089)."""
    import numpy as np

    return np.sqrt(OM * (1.0 + np.asarray(z, float)) ** 3 + OL)

@cites("bulbul2019", "cosmology_planck2015")
def lx500_of_m500(m500_msun, z):
    """L_X(0.5-2 keV, < R500, core included) [erg/s] of Bulbul et al. (2019) at M500 [Msun] and ``z``."""
    b = BULBUL2019
    return (b["norm_erg_s"] * (m500_msun / b["m_piv_msun"]) ** b["slope"]
            * (Ez(z) / Ez(b["z_piv"])) ** b["ez_exp"] * ((1.0 + z) / (1.0 + b["z_piv"])) ** b["opz_exp"])

@cites("bulbul2019", "cosmology_planck2015")
def m500_of_lx500(lx_erg_s, z):
    """M500 [Msun] of a luminosity L_X(0.5-2 keV, < R500) [erg/s] through the inverted relation (NaN if L <= 0)."""
    import numpy as np

    b = BULBUL2019
    lx = np.asarray(lx_erg_s, float)
    with np.errstate(invalid="ignore", divide="ignore"):
        x = lx / (b["norm_erg_s"] * (Ez(z) / Ez(b["z_piv"])) ** b["ez_exp"]
                  * ((1.0 + z) / (1.0 + b["z_piv"])) ** b["opz_exp"])
        m = b["m_piv_msun"] * np.power(x, 1.0 / b["slope"])
    return np.where(np.isfinite(m) & (lx > 0), m, np.nan)

@cites("bulbul2019", "piffaretti2011")
def iterate_m500(measure: Any, z: float, rtol: float = ITER_RTOL, max_iter: int = ITER_MAX) -> dict[str, Any]:
    """First-guess M500 of one image: ``measure(m500_msun, r500_kpc) -> L_X`` [erg/s] (0.5-2 keV, inside R500, NaN
    when unmeasurable) inverted through :func:`m500_of_lx500` until the mass is self-consistent (module docstring:
    the plain iteration from the pivot mass, safeguarded by bisection).  A trial whose aperture is unmeasurable bounds
    the search: before any trial was measurable the mass steps down a decade at a time (a smaller R500 that fits the
    field); after, the search bisects back towards the last measurable trial.  Returns ``m500_seed_msun``,
    ``r500_seed_kpc``, ``lx500_seed`` (the luminosity at that R500), ``seed_n_iter`` and ``seed_status``:
    ``converged``; ``field-edge`` (the relation asks for a larger R500 than the image can measure: the largest
    measurable trial is kept); ``pivot`` (no trial measurable: the pivot mass) or ``not-converged`` (the last
    measured mass is kept)."""
    x = math.log(float(BULBUL2019["m_piv_msun"]))
    lo = hi = None                      # ln M with g > 0 (root above) and with g < 0 (root below)
    hi_unmeasurable = False             # the upper bound comes from an unmeasurable trial
    good = None                         # (ln M, L) of the last measurable trial
    for k in range(1, int(max_iter) + 1):
        m = math.exp(x)
        r = r500_kpc(m, z)
        lx = float(measure(m, r))
        if not (math.isfinite(lx) and lx > 0):
            if good is None:
                hi, hi_unmeasurable = x, True
                x -= ITER_MAX_STEP
                continue
            # unmeasurable here: the root lies between this trial and the last measurable one
            if x > good[0]:
                if hi is None or x < hi:
                    hi, hi_unmeasurable = x, True
            else:
                lo = x if lo is None else max(lo, x)
            x = 0.5 * (x + good[0])
            continue
        good = (x, lx)
        g = math.log(float(m500_of_lx500(lx, z))) - x
        if abs(g) < rtol:
            return dict(m500_seed_msun=m, r500_seed_kpc=r, lx500_seed=lx, seed_n_iter=k, seed_status="converged")
        if g > 0:
            lo = x if lo is None else max(lo, x)
        else:
            hi, hi_unmeasurable = (x, False) if hi is None or x < hi else (hi, hi_unmeasurable)
        # the plain iteration's next mass, at most a decade per step while one side of the root is still open (a
        # numerical guard of the search path; the root it finds is the same)
        x_new = x + max(-ITER_MAX_STEP, min(ITER_MAX_STEP, g))
        if lo is not None and hi is not None:
            if hi - lo < rtol:
                status = "field-edge" if hi_unmeasurable and g > 0 else "converged"
                return dict(m500_seed_msun=m, r500_seed_kpc=r, lx500_seed=lx, seed_n_iter=k, seed_status=status)
            if not lo < x_new < hi:
                x_new = 0.5 * (lo + hi)
        x = x_new
    if good is None:
        m = float(BULBUL2019["m_piv_msun"])
        return dict(m500_seed_msun=m, r500_seed_kpc=r500_kpc(m, z), lx500_seed=float("nan"),
                    seed_n_iter=int(max_iter), seed_status="pivot")
    x, lx = good
    m = math.exp(x)
    return dict(m500_seed_msun=m, r500_seed_kpc=r500_kpc(m, z), lx500_seed=lx, seed_n_iter=int(max_iter),
                seed_status="not-converged")

def seed_lx_iterated(merged: Any, z_obs: float, exposure_s: float, nh_1e22: float, expmap: Any = None, *,
                     spec: Any, **kw: Any) -> dict[str, Any]:
    """The seed of a fit target, mock or real, from its own image: :func:`iterate_m500` with the luminosity inside
    each trial R500 measured by :func:`tngxmm.photometry.measure_lx.seed_lx` (the campaign estimator: masked point
    sources, sector-clipped local annulus, ECF at the virial temperature of the trial mass).  ``kw`` goes to
    ``seed_lx`` (sources, centre, kind, obsid, pps_cache).  Returns the seed_lx columns at the final R500
    (``lx_0p35mpc``, ``lx_r500`` = the luminosity inside the first-guess R500, ``lx_1mpc``, flags) and the iteration's
    columns.  No catalogue or true mass enters."""
    from tngxmm.photometry.measure_lx import seed_lx, t_vir_keV

    trials: dict[float, dict[str, Any]] = {}

    def measure(m: float, r: float) -> float:
        t_kev = t_vir_keV(m, r, mu=float(spec.constants.mu_tvir))      # the running preset's mu
        res = seed_lx(merged, r, z_obs, exposure_s, nh_1e22, t_kev, expmap, spec=spec, **kw)
        trials[float(r)] = res
        return float(res["lx_r500"])

    it = iterate_m500(measure, float(z_obs))
    r = float(it["r500_seed_kpc"])
    if r not in trials:                     # the pivot fallback: its photometry at the pivot R500
        measure(float(it["m500_seed_msun"]), r)
    return {**trials[r], **it}

@cites("pacaud2016", "xmm_ccf_sas")
def peak_search_image(merged: Any, image: Any, nominal_xy: tuple[float, float], protect_arcsec: float, *, spec: Any,
                      pix_arcsec: float = 4.0, sources: Any = None, sources_xy: Any = None, kind: str = "mock",
                      obsid: str | None = None, pps_cache: str | None = None, **_: Any) -> Any:
    """``image`` with the listed point sources set to zero over their mask radius (the photometry's rule), for the peak
    search: an AGN or a star is not taken for the cluster.  The list is the one the seed photometry masks with (the
    SAS edetect_chain list of the image under ``massfit.source_list = 'sas'``).  Sources within ``protect_arcsec`` of
    the nominal position (the catalogue's 90 per cent error circle, at least the PSF half-energy width) are kept: there
    the catalogue places the cluster, and its core can be detected as point-like (Pacaud et al. 2016)."""
    import numpy as np
    from astropy.io import fits

    from tngxmm.photometry.estimate_lx_background import LxBackgroundOptions, source_radii_px
    from tngxmm.photometry.measure_lx import sources_to_pixels, view_lx_sources

    opts = LxBackgroundOptions.from_spec(spec)
    if sources is None and sources_xy is None:
        sas = str(getattr(spec.massfit, "source_list", "default")) == "sas"
        sources, _ = view_lx_sources(merged, opts, kind=kind, obsid=obsid, pps_cache=pps_cache, sas=sas)
    hdr = fits.getheader(str(merged))
    xy = sources_to_pixels(sources or [], hdr)
    if sources_xy is not None and len(sources_xy):
        xy = np.vstack([xy, np.asarray(sources_xy, float).reshape(-1, 2)])
    out = np.array(image, dtype=float, copy=True)
    if not len(xy):
        return out
    far = np.hypot(xy[:, 0] - nominal_xy[0], xy[:, 1] - nominal_xy[1]) * pix_arcsec > protect_arcsec
    xy = xy[far]
    if not len(xy):
        return out
    rad = source_radii_px(out, np.ones(out.shape, bool), xy, pix_arcsec, opts)
    yy, xx = np.mgrid[:out.shape[0], :out.shape[1]]
    for (x, y), r in zip(xy, rad):
        out[np.hypot(xx - x, yy - y) <= r] = 0.0
    return out

def centre_and_seed(merged: Any, image: Any, nominal_xy: tuple[float, float], z_obs: float, exposure_s: float,
                    nh_1e22: float, expmap: Any = None, *, spec: Any, pix_arcsec: float = 4.0,
                    r90_arcsec: float = 0.0, **kw: Any) -> dict[str, Any]:
    """The centre and the first guess of a fit target, mock or real, in one procedure: (1) the seed
    (:func:`seed_lx_iterated`) at the nominal position ``nominal_xy`` (a real cluster's catalogue position; a mock's
    catalogue position, its detection on its Compton-y map filtered as the replicated cluster's catalogue filters
    (:mod:`tngxmm.sim.catalogue_centre`), or an
    X-ray position for the three entries of neither SZ catalogue, moved by an error drawn from the published positional
    error of that entry, :func:`tngxmm.photometry.find_centre.mock_nominal`); ``z_obs`` is the catalogue redshift (a
    mock's carries the drawn error of a photometric one, :func:`tngxmm.photometry.catalogue_positions.mock_redshift`);
    (2) the guarded X-ray peak of ``image`` (the
    merged 0.5-2 keV counts, its listed point sources masked outside the catalogue's error circle,
    :func:`peak_search_image`) within max(first-guess theta500, ``r90_arcsec``) of the nominal position, the radius
    within which PSZ2 accepts an X-ray counterpart (:func:`tngxmm.photometry.find_centre.find_peak_centre_guarded`);
    (3) the seed again at that peak, the centre the fit uses.  Every number comes from ``spec`` (its photometry
    settings, estimator options, source-list rule and constants), never from another preset.  Returns the seed columns
    of step 3, ``m500_seed1_msun`` (step 1), ``centre_x``, ``centre_y`` (0-indexed pixels of ``image``),
    ``centre_flag`` and ``search_arcsec``."""
    from tngxmm.core.cosmology import kpc_per_arcsec
    from tngxmm.massfit.mask_point_sources import PSF_HEW_ARCSEC
    from tngxmm.photometry._settings import photometry_settings
    from tngxmm.photometry.find_centre import find_peak_centre_guarded

    s1 = seed_lx_iterated(merged, z_obs, exposure_s, nh_1e22, expmap, spec=spec,
                          center=(float(nominal_xy[0]), float(nominal_xy[1])), **kw)
    theta = float(s1["r500_seed_kpc"]) / kpc_per_arcsec(float(z_obs))
    search = peak_search_image(merged, image, nominal_xy, max(float(r90_arcsec), PSF_HEW_ARCSEC), spec=spec,
                               pix_arcsec=pix_arcsec, **kw)
    px, py, info = find_peak_centre_guarded(search, nominal_xy, max(theta, float(r90_arcsec)), pix_arcsec=pix_arcsec,
                                            settings=photometry_settings(spec))
    s2 = seed_lx_iterated(merged, z_obs, exposure_s, nh_1e22, expmap, spec=spec, center=(px, py), **kw)
    return {**s2, "m500_seed1_msun": s1["m500_seed_msun"], "centre_x": float(px), "centre_y": float(py),
            "centre_flag": info["flag"], "search_arcsec": float(info["search_arcsec"])}




def _nfw_m(x):
    import numpy as np

    return np.log1p(x) - x / (1.0 + x)

def m500_over_m200_nfw(c200):
    """M500c / M200c of an NFW halo of concentration ``c200`` (solves the 500 / 200 density ratio)."""
    from scipy.optimize import brentq

    c = float(c200)
    f = lambda x: _nfw_m(c * x) / (x ** 3 * _nfw_m(c)) - 500.0 / 200.0     # noqa: E731
    x = brentq(f, 1e-3, 1.0)                  # r500 / r200
    return float(_nfw_m(c * x) / _nfw_m(c))

@cites("dutton2014", "nfw")
def c200_dutton14(m500_1e14, z, h=0.6774):
    """Median NFW c200 of Dutton & Maccio (2014) for a (relaxed) halo of M500c = ``m500_1e14`` [1e14 Msun] at ``z``.

    ``log10 c200 = a + b log10(M200 / (1e12 h^-1 Msun))``, ``a = 0.520 + (0.905 - 0.520)
    exp(-0.617 z^1.21)``, ``b = -0.101 + 0.026 z`` (relaxed haloes, Planck cosmology); the scatter
    is :data:`DM14_SCATTER_DEX` = 0.11 dex.  M200 follows from M500 through the NFW halo of the
    relation itself (a short fixed-point iteration)."""
    import numpy as np

    zz = float(z)
    a = 0.520 + (0.905 - 0.520) * np.exp(-0.617 * zz ** 1.21)
    b = -0.101 + 0.026 * zz
    c = 4.0
    for _ in range(6):
        m200 = float(m500_1e14) * 1e14 / m500_over_m200_nfw(c)
        c = 10.0 ** (a + b * np.log10(m200 * h / 1e12))
    return float(c)

def c200_guess(m500_1e14, z=0.0):
    """Start of c200: the Dutton & Maccio (2014) median at M500 = ``m500_1e14`` [1e14 Msun] and ``z``."""
