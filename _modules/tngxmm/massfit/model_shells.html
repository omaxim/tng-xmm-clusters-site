Source code for tngxmm.massfit.model_shells

"""Radial shells of the cluster model finer than the fit pixel (``massfit.model_shells_per_pixel``).

mbproj2d builds the cluster of an image on spherical shells one image pixel wide
(``ClusterBase.__init__``: ``Radii(pixsize_kpc, num)``, ``num = int(maxradius_kpc / pixsize_kpc) + 1``).  Each shell
takes the density, temperature and metallicity of its centre and is treated as being of constant density, as in
MBProj2 (Sanders et al. 2018, Appendix B: the pressures "computed at the centres of the radial bins", the acceleration
"assuming constant density in a shell", the projection "a matrix containing the volume of each shell visible in an
annulus on the sky").  The projected surface brightness of every annulus is then painted onto the pixels by linear
interpolation from the annulus centres (``add_sb_prof`` of ``fast/project_cc.cc``, argument ``rbin``: the annulus
width in pixels).  A shell one pixel wide is a midpoint rule for the emission measure: at z = 0.5 a fit pixel of 8"
is 50 kpc against core radii of 130-170 kpc, and the model then predicts 1-2 per cent more counts within R500 than the
same profile integrated finely (STATUS B12).

With ``k`` shells per pixel the shells are ``pixsize_kpc / k`` wide (the same outer radius, ``k * num`` shells), the
hydrostatic integration and the projection run on them, and the image is painted with ``rbin = 1 / k``.  Nothing
else changes: the images, their pixels, the PSF convolution, the likelihood and the parameters are those of the fit
pixel.  ``k = 1`` is mbproj2d as released.  The cost is the projection matrix (``(k num)^2``) and the profile
evaluation (``k num``), small against the image work, which is unchanged.

Finer shells do not change where the profile is read: ``add_sb_prof`` still evaluates it at the centre of each image
pixel, whereas a detector pixel collects everything that falls on its whole area.  For a profile that is curved on the
scale of a pixel, a cuspy core under a 24" pixel for instance, the value at the centre is larger than the mean over the
pixel, and the model is too peaked.  ``fitfix pixel_average = n`` paints the profile on a grid ``n`` times finer within
:data:`PIXEL_AVERAGE_HALF_WIDTH` pixels of the cluster centre and takes the mean over each image pixel, before the PSF
(:func:`paint_profile`).  Further out the profile is close to linear across a pixel and the two agree.  ``n = 1`` is
mbproj2d's painting.

References
----------
mbproj2d (the shells and the painting this refines), sanders2018 (the MBProj2 shell model).
"""

from __future__ import annotations

import types
from typing import Any

from tngxmm.cites import cites, collect_cites

#: the attribute a cluster class sets when its ``compute`` paints with ``self.shell_rbin`` (band_response's

USES_SHELL_RBIN = "uses_shell_rbin"
#: half width [image pixels] of the square about the cluster centre where ``fitfix pixel_average`` > 1 averages the
#: model over each pixel (a numerical setting: doubling it, or the sub-pixel count, is checked to change nothing)
PIXEL_AVERAGE_HALF_WIDTH = 24

def pixel_average(value: Any) -> int:
    """``fitfix pixel_average`` as an integer >= 1 (None = 1, the profile read at each pixel's centre)."""
    if value is None:
        return 1
    n = int(value)
    if n != float(value) or n < 1:
        raise ValueError(f"pixel_average = {value!r}: an integer >= 1")
    return n

@cites("mbproj2d", "sanders2018")
def refine_shells(cluster, k: Any, mb=None, pixel_avg: Any = 1):
    """Replace the one-pixel shells of an mbproj2d cluster by ``k`` shells per pixel (in place; returns it), and
    paint it averaged over ``pixel_avg`` sub-pixels per side (:func:`paint_profile`).

    ``k = 1`` and ``pixel_avg = 1`` leave the cluster as mbproj2d made it.  The outer radius of every pixel size is
    kept."""
    k = shells_per_pixel(k)
    n = pixel_average(pixel_avg)
    cluster.shell_rbin = 1.0
    cluster.shells_per_pixel = k
    cluster.pixel_average = n
    if k > 1:
        if mb is None:
            from mbproj2d.profile import Radii
        else:
            Radii = mb.Radii
        for pix, radii in list(cluster.pixsize_Radii.items()):
            cluster.pixsize_Radii[pix] = Radii(float(radii.rshell_kpc) / k, int(radii.num) * k)
        cluster.shell_rbin = 1.0 / k
    if (k > 1 or n > 1) and not getattr(type(cluster), USES_SHELL_RBIN, False):
        cluster.compute = types.MethodType(compute_with_rbin, cluster)
    return cluster




class _CircularPaint:
    """mbproj2d's circular painting (``add_sb_prof`` of ``fast/project_cc.cc``, the branch ``addSBToImg_Comb`` takes
    for e = 1, no slosh, no multipole) on one fixed grid, with the radius of every pixel worked out once.

    The C code reads, for each pixel of the box ``centre +- rbin nbins``, r = sqrt((x - xc)^2 + (y - yc)^2) / rbin,
    rc = max(r - 1/2, 0), i = int(rc), w1 = rc - i and adds sb[i] (1 - w1) + sb[i + 1] w1 (sb[nbins] = 0, the index
    capped at nbins), all in single precision and without fused multiply-adds (the shipped binary has none).  Here
    the same operations, in the same order and precision, are done once for the geometry (i and the two weights) and
    the profile is then read with two gathers: the values are the C code's to the bit (the unit tests compare them).
    It pays for the sub-pixel grid of :func:`paint_profile`, the same for every image, node group and likelihood
    call of a fit."""

    def __init__(self, rbin: float, nbins: int, xc: float, yc: float, shape: tuple) -> None:
        import numpy as np

        f32 = np.float32
        yw, xw = int(shape[0]), int(shape[1])
        rb, xcf, ycf = f32(rbin), f32(xc), f32(yc)
        reach = rb * f32(nbins)
        x1, x2 = max(int(xcf - reach), 0), min(int(xcf + reach), xw - 1)
        y1, y2 = max(int(ycf - reach), 0), min(int(ycf + reach), yw - 1)
        self.shape, self.nbins = (yw, xw), int(nbins)
        self.box = (slice(y1, y2 + 1), slice(x1, x2 + 1))
        self.full = (y1, x1, y2, x2) == (0, 0, yw - 1, xw - 1)
        inv = f32(1) / rb
        dx = np.arange(x1, x2 + 1).astype(f32) - xcf
        dy = np.arange(y1, y2 + 1).astype(f32) - ycf
        r = np.sqrt(dx[None, :] * dx[None, :] + dy[:, None] * dy[:, None]) * inv
        rc = np.maximum(r - f32(0.5), f32(0))
        ri = rc.astype(np.int64)
        self.index = np.ascontiguousarray(np.minimum(ri, self.nbins).ravel().astype(np.intp))
        self.w1 = np.ascontiguousarray((rc - ri.astype(f32)).ravel())
        self.w0 = np.ascontiguousarray(f32(1) - self.w1)
        self._a = np.empty(self.index.size, f32)
        self._b = np.empty(self.index.size, f32)

    def values(self, sb):
        """The painted box (float32, a buffer valid until the next call) of the profile ``sb`` (nbins values)."""
        import numpy as np

        sb = np.asarray(sb, np.float32)
        lo = np.empty(self.nbins + 1, np.float32)
        lo[:-1] = sb
        lo[-1] = 0.0
        hi = np.empty_like(lo)
        hi[:-1] = lo[1:]
        hi[-1] = 0.0                            # i + 1 is capped at nbins as well
        a, b = self._a, self._b
        # every index is in 0 ... nbins, so mode 'clip' takes the same values as 'raise', without its check
        np.take(lo, self.index, out=a, mode="clip")
        np.multiply(a, self.w0, out=a)
        np.take(hi, self.index, out=b, mode="clip")
        np.multiply(b, self.w1, out=b)
        np.add(a, b, out=a)
        return a.reshape(self.box[0].stop - self.box[0].start, self.box[1].stop - self.box[1].start)


#: :class:`_CircularPaint` of the sub-pixel grids in use, by geometry (one per image grid and centre)
_PAINT_CACHE: dict = {}
_PAINT_CACHE_MAX = 32
#: whether :func:`_block_mean` reproduces numpy's mean on this machine, by block grid (Y, X): checked on the first
#: window of each grid (False = numpy's mean is used for that grid)
_BLOCK_MEAN_OK: dict = {}


def _circular_paint(rbin, nbins: int, xc, yc, shape: tuple) -> _CircularPaint:
    """The cached :class:`_CircularPaint` of a geometry (keyed by the single-precision values the C code sees)."""
    import numpy as np

    f32 = np.float32
    key = (f32(rbin).item(), int(nbins), f32(xc).item(), f32(yc).item(), tuple(shape))
    geo = _PAINT_CACHE.get(key)
    if geo is None:
        if len(_PAINT_CACHE) >= _PAINT_CACHE_MAX:
            _PAINT_CACHE.clear()
        geo = _PAINT_CACHE[key] = _CircularPaint(rbin, nbins, xc, yc, shape)
    return geo


def _circular(e, slosh, mulm) -> bool:
    """Whether ``addSBToImg_Comb`` paints with its circular branch (e and slosh as the binding converts them, single
    precision; mulm an integer)."""
    import numpy as np

    return int(mulm) <= 0 and np.float32(e) == 1 and np.float32(slosh) == 0


def _pairwise8(a):
    """numpy's pairwise sum of 8 contiguous float32 values (``pairwise_sum`` of ``loops_utils.h``, n = 8) over the
    last axis: ((a0 + a1) + (a2 + a3)) + ((a4 + a5) + (a6 + a7))."""
    return ((a[..., 0] + a[..., 1]) + (a[..., 2] + a[..., 3])) + ((a[..., 4] + a[..., 5]) + (a[..., 6] + a[..., 7]))


def _block_mean(fine, n: int):
    """``fine.reshape(Y, n, X, n).mean(axis=(1, 3))`` for float32 and n = 8, in the order numpy sums it (each row
    of 8 by its pairwise sum, the 8 rows of a block one after the other, then the division by 64), four times
    faster.  Any other case, and a block grid on which the first comparison with numpy's mean fails, takes numpy's
    mean itself."""
    import numpy as np

    Y, X = fine.shape[0] // n, fine.shape[1] // n
    v = fine.reshape(Y, n, X, n)
    if n != 8 or X < 2 or fine.dtype != np.float32:
        # X = 1: numpy merges the two summed axes into one run of 64 and sums it in another order
        return v.mean(axis=(1, 3))
    ok = _BLOCK_MEAN_OK.get((Y, X))
    if ok is None:
        ok = _BLOCK_MEAN_OK[(Y, X)] = bool(np.array_equal(_block_mean_8(v).view(np.int32),
                                                          v.mean(axis=(1, 3)).view(np.int32)))
    return _block_mean_8(v) if ok else v.mean(axis=(1, 3))


def _block_mean_8(v):
    import numpy as np

    p = _pairwise8(v)                           # (Y, 8, X)
    s = p[:, 0] + p[:, 1]
    for j in range(2, 8):
        s += p[:, j]
    s /= np.float32(64)
    return s

def paint_profile(rbin, sb, cx, cy, e, slosh, theta, mulm, mulmag, img, n: int = 1):
    """Add the projected profile ``sb`` (annuli ``rbin`` pixels wide, per image pixel) to ``img``: mbproj2d's
    ``addSBToImg_Comb`` with, for ``n`` > 1, the mean over ``n`` x ``n`` sub-pixels of every pixel within
    :data:`PIXEL_AVERAGE_HALF_WIDTH` of the centre ``(cx, cy)`` [pixels].

    A circular profile is painted on the image and on the sub-pixel grid by :class:`_CircularPaint` (the numbers of
    ``addSBToImg_Comb``, the radii of each grid computed once per geometry) and averaged by :func:`_block_mean`
    (numpy's mean, summed in numpy's order): the result is the one of painting with ``addSBToImg_Comb`` and
    averaging with ``mean``, bit for bit, in under half the time."""
    import numpy as np
    from mbproj2d.fast import addSBToImg_Comb

    if n == 1:
        addSBToImg_Comb(rbin, sb, cx, cy, e, slosh, theta, mulm, mulmag, img)
        return
    w = PIXEL_AVERAGE_HALF_WIDTH
    fast = img.dtype == np.float32 and _circular(e, slosh, mulm)

    def paint(rb, xc, yc, shape):
        # a zero grid of ``shape`` with the profile painted on it: by the cached geometry, else by mbproj2d
        if not fast:
            out = np.zeros(shape, dtype=img.dtype)
            addSBToImg_Comb(rb, sb, xc, yc, e, slosh, theta, mulm, mulmag, out)
            return out
        geo = _circular_paint(rb, len(sb), xc, yc, shape)
        if geo.full:
            return geo.values(sb)
        out = np.zeros(shape, dtype=img.dtype)
        out[geo.box] = geo.values(sb)
        return out

    one = paint(rbin, cx, cy, img.shape)
    ny, nx = img.shape
    x0, x1 = max(int(round(cx)) - w, 0), min(int(round(cx)) + w + 1, nx)
    y0, y1 = max(int(round(cy)) - w, 0), min(int(round(cy)) + w + 1, ny)
    if x1 > x0 and y1 > y0:
        # pixel i of the image covers sub-pixels i n ... i n + n - 1, so its centre is at (i + 1/2) n - 1/2
        fine = paint(rbin * n, (cx - x0 + 0.5) * n - 0.5, (cy - y0 + 0.5) * n - 0.5, ((y1 - y0) * n, (x1 - x0) * n))
        one[y0:y1, x0:x1] = _block_mean(fine, n)
    img += one

@cites("mbproj2d")
def paint_matrix(shape: tuple, rbin: float, nbins: int, cx: float, cy: float, n: int = 1):
    """The painting of :func:`paint_profile` for a circular profile as a sparse matrix (scipy CSR, float64,
    ``shape[0] shape[1]`` x ``nbins``): image (flattened) = matrix @ sb.

    Each pixel holds the interpolation weights of :class:`_CircularPaint` (single precision, as the C code computes
    them); within the averaged square they are the mean of the weights of its n x n sub-pixels.  The painting is
    linear in sb, so the matrix gives the same image up to rounding: the products and sums are done once in double
    precision instead of per sub-pixel in single precision (at most 2e-7 of a pixel on the pilot fits)."""
    import numpy as np
    import scipy.sparse as sp

    ny, nx = int(shape[0]), int(shape[1])
    w = PIXEL_AVERAGE_HALF_WIDTH
    rows, cols, vals = [], [], []

    def add(row, index, w0, w1, scale):
        m0 = index < nbins
        m1 = index + 1 < nbins                  # sb[nbins] = 0: the outer neighbour of the last shell adds nothing
        rows.extend((row[m0], row[m1]))
        cols.extend((index[m0], index[m1] + 1))
        vals.extend((w0[m0].astype(np.float64) * scale, w1[m1].astype(np.float64) * scale))

    g = _CircularPaint(rbin, nbins, cx, cy, (ny, nx))
    yy, xx = np.mgrid[g.box[0], g.box[1]]
    x0, x1 = max(int(round(cx)) - w, 0), min(int(round(cx)) + w + 1, nx)
    y0, y1 = max(int(round(cy)) - w, 0), min(int(round(cy)) + w + 1, ny)
    averaged = n > 1 and x1 > x0 and y1 > y0
    outside = np.ones(yy.size, bool)
    if averaged:
        outside = ~((yy >= y0) & (yy < y1) & (xx >= x0) & (xx < x1)).ravel()
    add((yy * nx + xx).ravel()[outside], g.index[outside], g.w0[outside], g.w1[outside], 1.0)
    if averaged:
        f = _CircularPaint(rbin * n, nbins, (cx - x0 + 0.5) * n - 0.5, (cy - y0 + 0.5) * n - 0.5,
                           ((y1 - y0) * n, (x1 - x0) * n))
        fy, fx = np.mgrid[f.box[0], f.box[1]]
        add(((y0 + fy // n) * nx + (x0 + fx // n)).ravel(), f.index, f.w0, f.w1, 1.0 / (n * n))
    m = sp.coo_matrix((np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
                      shape=(ny * nx, int(nbins))).tocsr()
    m.sum_duplicates()
    return m

@cites("mbproj2d")
def compute_with_rbin(self, pars, imgarrs):
    """mbproj2d's ``ClusterBase.compute`` (commit ff497101) with the annulus width ``self.shell_rbin`` [pixels]
    passed to ``addSBToImg_Comb`` in place of 1, painted through :func:`paint_profile`."""

    def opt(key, default):
        name = f"{self.name}_{key}"
        return default if name not in pars else pars[name].v

    cy_as, cx_as = pars[f"{self.name}_cy"].v, pars[f"{self.name}_cx"].v
    e, slosh, theta = opt("e", 1), opt("slosh", 0), opt("theta", 0)
    mulm, mulmag = int(opt("mulm", 0)), opt("mulmag", 0)
    prof = {}
    for pixsize, radii in self.pixsize_Radii.items():
        ne, T, Z = self.computeProfiles(pars, radii)
        prof[pixsize] = (self.nesqd_to_norm * ne ** 2, T, Z)
    for img, imgarr in zip(self.images, imgarrs):
        norm, T, Z = prof[img.pixsize_as]
        emiss = self.image_RateCalc[img].get(T, Z, norm)
        sb = self.pixsize_Radii[img.pixsize_as].project(emiss)
        # per pixel: the projection is per kpc2 of sky
        sb *= (self.cosmo.as_kpc * img.pixsize_as) ** 2
        pix_cy = cy_as * img.invpixsize + img.origin[0]
        pix_cx = cx_as * img.invpixsize + img.origin[1]
        paint_profile(float(self.shell_rbin), sb, pix_cx, pix_cy, e, slosh, theta, mulm, mulmag, imgarr,
                      getattr(self, "pixel_average", 1))
