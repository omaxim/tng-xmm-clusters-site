"""Data, analytical model surface and residual maps of a finished fit.

Renders ANY finished fit directory written by ``fit_hydrostatic.run_one``,
whatever its switches, by rebuilding exactly what ``run_one`` built:

* the settings come from the fit's own ``result.json`` and ``provenance.json``
  (``fit_options``: centre, N_H, mass and c200 seeds, crop, cap, pixel size,
  background prior, optical axis, rebin centre, metallicity and its prior,
  abundance table, mu convention and composition, PSF; ``source_masking``: the
  masked list and its threshold, re-read with the same
  ``mask_point_sources.sources_for_target``);
* images and model are built by ``fit_hydrostatic.build_fit_setup``, the same
  building blocks ``run_one`` calls in the same order;
* checks before anything is drawn: the image tree digest equals the one in the
  provenance, the image count, pixel size, background-seed sources and
  free-parameter count equal ``result.json``, the free-parameter names equal
  the chain's ``thawed_params``, and the log-likelihood recomputed at the best
  stored sample equals the stored value (reported as ``dlogL``).

The posterior is the stored chain after the ``burn`` of its own header (the
samples ``run_one`` drew M500 from).

Figure ``<stem>_v4.png``:

* row 1, 0.5-2 keV: the data (counts summed over cameras) with the image
  bank's normalisation (``bank.render_bank.normalize_image``, colour map
  ``inferno``); the best-fit ANALYTICAL model surface on the same grid and the
  same normalisation (intrinsic cluster model, PSF as fitted, plus the flat
  background level, no exposure map, no chip gaps, no noise; rate x the peak
  exposure of each camera, i.e. counts at on-axis exposure); the residual
  significance (data - full model incl. exposure map and background) / sigma
  in 0.5-2 keV and in 0.5-12 keV (2-px Gaussian kernel);
* row 2: surface-brightness profiles (0.5-2 and 2-12 keV; data, model total
  with its 68 % band, cluster, background) and the annulus residuals;
* row 3: n_e, kT and M(<r) with 68 % bands; the quoted (real) or true (mock)
  M500 / R500 as markers, and for mocks the intrinsic hydrostatic M500 of the
  view (the preset's headline, else Barnes et al. 2021) and, when the halo's
  truth-profile file is found (``sim.truth_profiles``; :func:`find_truth_profile`),
  the true n_e (rms and volume-weighted) and T_sl / T_mw of the mock's cells, the
  true M_tot(<r) and the HSE-truth curves (else the stored true profiles).

``<stem>_v4_profiles.npz`` holds the profiles.  :func:`render_truth_comparison`
(``--truth-comparison``) draws a mock's M(<r) band against its true M_tot(<r)
and HSE-truth curves with markers at R500 (the mock analogue of a literature
comparison).

astropy, scipy, matplotlib

def align(arr, origin, shape, origin_to):
    """Place `arr` (cluster at `origin` (y,x) px) on a grid of `shape` whose
    cluster pixel is `origin_to`, by integer shift; outside -> 0."""
    import numpy as np

    out = np.zeros(shape, dtype=arr.dtype)
    dy = int(round(origin_to[0] - origin[0]))
    dx = int(round(origin_to[1] - origin[1]))
    y0, x0 = max(dy, 0), max(dx, 0)
    y1, x1 = min(shape[0], arr.shape[0] + dy), min(shape[1], arr.shape[1] + dx)
    if y1 > y0 and x1 > x0:
        out[y0:y1, x0:x1] = arr[y0 - dy:y1 - dy, x0 - dx:x1 - dx]
    return out

def load_fit(fit_dir: str | Path) -> tuple[dict, dict]:
    """(result.json, provenance.json) of a fit directory; provenance is required."""
    fit_dir = Path(fit_dir)
    with open(fit_dir / 'result.json') as f:
        res = json.load(f)
    prov_p = fit_dir / 'provenance.json'
    if not prov_p.exists():
        raise FileNotFoundError(f'{prov_p} missing: a fit without provenance cannot be rebuilt')
    with open(prov_p) as f:
        prov = json.load(f)
    if 'fit_options' not in prov:
        raise ValueError(f'{prov_p}: no fit_options')
    return res, prov

def masking_record(res: dict, prov: dict) -> dict | None:
    """What the fit masked (provenance ``source_masking``, else the mask_* keys of result.json), or None."""
    info = prov.get('source_masking')
    if not info and res.get('source_masking'):
        info = {k: v for k, v in res.items() if k == 'source_masking' or k.startswith('mask_')
                or k == 'n_sources_masked'}
    if not info or not info.get('source_masking'):
        if int(prov['fit_options'].get('n_sources_masked', 0) or 0) > 0:
            raise ValueError('provenance says sources were masked but records no masking list')
        return None
    return info

def fit_sources(res: dict, prov: dict, specs: list[dict]):
    """The point-source list the fit masked, re-read with the fit's own rule, or None."""
    from tngxmm.massfit import mask_point_sources as mps

    info = masking_record(res, prov)
    if info is None:
        return None
    rule = info.get('mask_threshold_rule')
    lst = info.get('mask_source_list')
    if rule == 'caller-list' or not lst:
        raise ValueError(f'masking rule {rule!r}: the masked list was passed by the caller and is not recorded')
    if rule == 'sas-edetect-ext0-det-ml':
        src, _ = mps.sources_for_target('mock' if str(res['name']).startswith('mock_') else 'obs', res['obsid'],
                                        det_ml_min=float(info['mask_det_ml_min']), sas_sources=lst)
    elif rule == 'pps-det-ml':
        src, _ = mps.sources_for_target('obs', res['obsid'], det_ml_min=float(info['mask_det_ml_min']),
                                        pps_cache=str(Path(lst).parent))
    else:
        hdr = specs[0]['header'] if specs else None
        src, _ = mps.sources_for_target('mock', res['obsid'], image_header=hdr,
                                        image_path=specs[0]['img'] if specs else None,
                                        mock_sources=lst, mock_min_flux=info.get('mask_min_flux_erg_s_cm2'))
    n = int(info.get('n_sources_masked', len(src)))
    if len(src) != n:
        raise ValueError(f'masking list re-read gives {len(src)} sources, the fit masked {n}')
    return src

def target_row(targets: str | Path, name: str):
    """The targets-CSV row of ``name`` (pandas Series, obsid zero-filled); raises ``ValueError`` unless exactly one row matches."""
    import pandas as pd

    vt = pd.read_csv(targets, dtype={'obsid': str})
    rows = vt[vt['name'] == name]
    if len(rows) != 1:
        raise ValueError(f'{name}: {len(rows)} matching rows in {targets}')
    r = rows.iloc[0].copy()
    r['obsid'] = str(r['obsid']).zfill(10)
    return r

def bank_norm_params(counts) -> tuple[float, float]:
    """(floor, cutoff) [log10 counts] of the image bank's normalisation of ``counts``
    (``bank.render_bank.normalize_image``: floor = background - 0.25 dex, cutoff =
    99th percentile of the inner 10 % of the frame)."""
    import numpy as np

    from tngxmm.bank.render_bank import JPG_FLOOR_OFFSET_DEX, estimate_background

    counts = np.asarray(counts, dtype=float)
    with np.errstate(divide='ignore'):
        ld = np.where(counts > 0, np.log10(counts), -np.inf)
    h, w = ld.shape
    y, x = np.ogrid[:h, :w]
    dist = np.sqrt((x - w // 2) ** 2 + (y - h // 2) ** 2)
    inner = ld[(dist <= np.sqrt(0.1) * (w / 2)) & np.isfinite(ld)]
    cutoff = float(np.percentile(inner, 99))
    bg = estimate_background(counts)
    floor = math.log10(bg) + JPG_FLOOR_OFFSET_DEX if bg > 0 else cutoff - 4.0
    return floor, cutoff

def bank_norm_like(arr, ref):
    """``arr`` stretched with the bank normalisation of ``ref`` (so data and model share
    one colour scale); ``bank_norm_like(x, x) == normalize_image(x)``."""
    import numpy as np

    floor, cutoff = bank_norm_params(ref)
    arr = np.asarray(arr, dtype=float)
    with np.errstate(divide='ignore'):
        la = np.where(arr > 0, np.log10(arr), -np.inf)
    if cutoff - floor <= 0:
        return np.zeros(arr.shape, dtype=np.float32)
    return np.clip((la - floor) / (cutoff - floor), 0.0, 1.0).astype(np.float32)

def sky_extent(shape, origin, pix_as, east_left=True):
    """imshow extent [arcmin] of an XMM sky image (CDELT1 < 0: RA grows to the LEFT) about
    ``origin`` (y, x): the x axis is the RA offset increasing leftwards (east left, north up),
    so a sky offset (dRA cos dec, dDec) plots at (x, y) = (dRA cos dec, dDec)."""
    oy, ox = origin

            -(oy + 0.5) * pix_as / 60.0, (shape[0] - oy - 0.5) * pix_as / 60.0]

def significance(data, model, exposed, ks=RESID_KERNEL_PX):
    """Significance of the SMOOTHED residual : G*(data - model) /
    sqrt(G^2 * model), G the normalised ``ks``-px Gaussian kernel, so sigma is that of the
    smoothed map under Poisson noise of the model (not the per-pixel sigma)."""
    import numpy as np
    from scipy.ndimage import gaussian_filter
    from scipy.signal import fftconvolve

    num = gaussian_filter(np.where(exposed, data - model, 0.0), ks)
    k = np.zeros((int(8 * ks) + 1,) * 2)
    k[k.shape[0] // 2, k.shape[1] // 2] = 1
    k = gaussian_filter(k, ks)
    var = fftconvolve(np.where(exposed, model, 0.0), k ** 2, mode='same')
    return np.where(exposed & (var > 0), num / np.sqrt(np.clip(var, 1e-30, None)), np.nan)

@cites("mbproj2d")
def rebuild_fit(fit_dir: str | Path, *, cache_dir=None, img_root=None, resp=None, xspec_tmpdir=None,
                verify_images: bool = True) -> dict[str, Any]:
    """Rebuild images, model and posterior of a finished fit exactly as ``run_one``.

    Returns the :func:`fit_hydrostatic.build_fit_setup` dict plus res, prov, post
    (posterior samples), lpost, best, best_logl, dlogl, nburn."""
    import h5py
    import numpy as np

    from tngxmm.massfit import fit_hydrostatic as fh

    fit_dir = Path(fit_dir)
    res, prov = load_fit(fit_dir)
    kw = fh.setup_kwargs_from_fit(res, prov)
    root = Path(img_root) if img_root is not None else Path(prov['image_root'])
    fh.set_rate_cache(cache_dir)
    fh.set_xspec_scratch(xspec_tmpdir, None)
    resp = resp if resp is not None else prov['fit_options'].get('resp')

    specs = fh.find_images(kw['obsid'], root)
    if verify_images:
        names = sorted(Path(s['img']).name for s in specs)
        if names != sorted(prov.get('image_files', names)):
            raise ValueError(f'image files differ from the provenance: {names}')
        if prov.get('image_tree_sha256') and fh.image_tree_digest(specs) != prov['image_tree_sha256']:
            raise ValueError('image tree digest differs from the provenance (images changed since the fit)')
    sources = fit_sources(res, prov, specs)
    st = fh.build_fit_setup(**kw, img_root=root, resp=resp, sources=sources)
    images, pars = st['images'], st['pars']

    if len(images) != res['n_images']:
        raise AssertionError(f'{len(images)} images, the fit had {res["n_images"]}')
    if abs(images[0].pixsize_as - res['pixsize_as']) > 1e-6:
        raise AssertionError(f'pixel size {images[0].pixsize_as} != {res["pixsize_as"]}')
    if ','.join(st['bg_srcs']) != res['bg_seed_sources']:
        raise AssertionError(f'background seeds {st["bg_srcs"]} != {res["bg_seed_sources"]}')
    if pars.numFree() != res['nfree']:
        raise AssertionError(f'{pars.numFree()} free parameters, the fit had {res["nfree"]}')
    with h5py.File(fit_dir / 'chain.hdf5', 'r') as f:
        filefree = [x.decode() for x in f['thawed_params']]
        if filefree != pars.freeKeys():
            raise AssertionError(f'free parameters differ from the chain: {pars.freeKeys()} vs {filefree}')
        nburn = int(f.attrs.get('burn', 150 if res.get('quick') else 500))
        if kw['fitfix']['chain_burn'] == 'stored':
            nburn = 0                    # fitfix chain_burn: the stored chain is already post-burn
        like = np.asarray(f['likelihood'])
        ch = np.asarray(f['chain'])
    # run_one's posterior: loadChainFromFile(burn=nburn) of the stored chain
    post = ch[nburn:].reshape(-1, ch.shape[2]).astype(float)
    lpost = like[nburn:].reshape(-1)
    ibest = int(np.nanargmax(lpost))
    best = post[ibest]
    pars.setFree(best)
    best_logl = float(st['mb'].Likelihood(images, st['totmod'], pars).total)
    st.update(res=res, prov=prov, post=post, lpost=lpost, best=best, nburn=nburn,
              best_logl=best_logl, dlogl=best_logl - float(lpost[ibest]), sources=sources, kw=kw)
    return st

def find_truth_profile(*, view_dir=None, snap: int | None = None, halo: int | None = None,
                       sim: str | None = None, search=None) -> Path | None:
    """The truth-profile file (``sim.truth_profiles``; :mod:`tngxmm.sim.truth_profile`) of a
    mock's halo-snapshot, or None.  ``search``: a file, or a campaign tree root
    (``<root>/<sim>/<snap:03d>/<halo:07d>/truth_profile.h5``) or a halo directory; then the
    halo directory of ``view_dir`` (the parent of ``view<k>/``).  A candidate is accepted
    only when its ``snap`` / ``halo_id`` attributes are those of the mock."""
    import h5py

    from tngxmm.sim.truth_profile import FILENAME

    cands: list[Path] = []
    if search is not None:
        p = Path(search)
        if p.is_file():
            cands.append(p)
        elif snap is not None and halo is not None:
            for s in dict.fromkeys([sim, 'tng-cluster', 'tng300']):
                if s:
                    cands.append(p / s / f'{int(snap):03d}' / f'{int(halo):07d}' / FILENAME)
            cands += [p / f'{int(snap):03d}' / f'{int(halo):07d}' / FILENAME, p / FILENAME]
    if view_dir is not None:
        cands.append(Path(view_dir).parent / FILENAME)
    for c in cands:
        if not c.is_file():
            continue
        if snap is None or halo is None:
            return c
        try:
            with h5py.File(c, 'r') as f:
                if int(f.attrs['snap']) == int(snap) and int(f.attrs['halo_id']) == int(halo):
                    return c
        except (OSError, KeyError):
            continue
    return None

def load_truth_curves(path: str | Path) -> dict[str, Any]:
    """The curves the renderer overlays from one truth-profile file: radius, M_tot(<r),
    M_gas(<r), the HSE-truth masses (the headline estimator on its own grid, and the mock-cell
    T_sl / T_mw and thermal + kinetic ones on the common grid), n_e and T of the mock's cells,
    and the catalogue values."""
    import numpy as np

    from tngxmm.sim.truth_profile import read_truth_profile

    d = read_truth_profile(path)
    a = d['attrs']
    head = str(a.get('hse_headline') or 'barnes21_nosf_fit_mu')
    out = dict(path=str(path), attrs=a, r=np.asarray(d['r_kpc']), R500c_kpc=float(a['R500c_kpc']),
               M500c=float(a['M500c_msun']), z_snap=float(a['z_snap']), snap=int(a['snap']),
               halo=int(a['halo_id']), headline=head,
               M_tot=np.asarray(d['mass/M_tot']), M_gas=np.asarray(d['mass/M_gas']),
               ne_rms=np.asarray(d['gas/mock/ne_rms']), ne_vw=np.asarray(d['gas/mock/ne_vw']),
               T_sl=np.asarray(d['gas/mock/T_sl']), T_mw=np.asarray(d['gas/mock/T_mw']),
               hse={})
    if f'native/{head}/M_msun' in d:
        out['hse'][f'{head} (headline)'] = (np.asarray(d[f'native/{head}/r_kpc']),
                                            np.asarray(d[f'native/{head}/M_msun']))
    for k, lab in (('M_mock_sl', 'mock cells, T$_{sl}$'), ('M_mock_mw', 'mock cells, T$_{mw}$'),
                   ('M_thkin_nonsf', 'thermal + kinetic')):
        if f'hse/{k}' in d:
            out['hse'][lab] = (out['r'], np.asarray(d[f'hse/{k}']))
    return out

def _hse_curves(tr: dict, rmin_r500: float = 0.05):
    """(label, r, M, colour, ls) of the HSE-truth curves, finite and positive, r >= rmin R500c

    import numpy as np

def reference_values(res: dict, row, specs: list[dict], kpc_per_arcsec: float, ra: float, dec: float,

                     truth_profile=None) -> dict[str, Any]:
    """Quoted (real) or true (mock) M500 / R500, the reference centre offset and, for mocks,
    the intrinsic hydrostatic M500 (the preset's headline, else Barnes et al. 2021), the
    truth-profile curves (:func:`find_truth_profile`; ``truth_profile`` = a file or a tree
    root to search first) and the stored true profiles when available.  A mock's truth radii are taken at
    ``z_obs`` (:func:`mock_truth_z`), recorded with the fit's redshift ``z_fit``."""
    import numpy as np

    from tngxmm.massfit import fit_hydrostatic as fh
    from tngxmm.massfit import mask_point_sources as mps

    is_mock = str(res['name']).startswith('mock_')
    ref: dict[str, Any] = dict(is_mock=is_mock, label='true' if is_mock else 'quoted')

    def _get(k):
        if row is None or k not in row:
            return np.nan
        try:
            return float(row[k])
        except (TypeError, ValueError):
            return np.nan

    if is_mock:
        meta = None
        if view_dir is None and specs:
            tab = mps.view_source_table(specs[0]['img'])
            view_dir = tab.parent if tab is not None else None
        if view_dir is not None and (Path(view_dir) / 'obs_meta.json').exists():
            meta = json.loads((Path(view_dir) / 'obs_meta.json').read_text())
            ref['view_dir'] = str(view_dir)
        # the truth is taken at the redshift of the photons (z_obs), not at the fit's (the redshift the view's

        ref.update(z_obs=z_t, z_fit=float(res['z']))

            if zm is not None and zr is not None and np.isfinite(float(zm)) and np.isfinite(float(zr)):
                ref.update(M500=float(zm) / 1e14, R500_kpc=float(zr), truth_quantity='M500c at rho_c(z_obs)')
            else:
                ref.update(M500=meta['M500c'] / 1e14, R500_kpc=meta['R500c_kpc'], truth_quantity='M500c (z_snap)')
            head = meta.get('m500c_hse_msun')
            if head is not None and np.isfinite(float(head)) and meta.get('hse_headline'):
                # the preset's headline intrinsic HSE mass (Barnes+21 on non-SF gas)
                ref['M500_hse'] = float(head) / 1e14
                ref['hse_label'] = str(meta['hse_headline'])
                ref['R500_hse_kpc'] = fh.crop_r500_guess(z_t, ref['M500_hse'])[1]
            elif meta.get('m500_hse_barnes21_msun'):
                ref['M500_hse'] = meta['m500_hse_barnes21_msun'] / 1e14
                ref['R500_hse_kpc'] = fh.crop_r500_guess(z_t, ref['M500_hse'])[1]
            for k in ('m500c_zobs_msun', 'r500c_zobs_kpc'):
                if meta.get(k) is not None and np.isfinite(float(meta[k])):
                    ref[k] = float(meta[k])
            for k in ('com_offset_kpc', 'mbh_bcg_msun', 'bhar_bcg_msun_yr', 'exposure_eff_s', 'donor_id'):
                if k in meta:
                    ref[k] = meta[k]
            snap, halo = int(meta['snap']), int(meta['halo_id'])
        else:
            ref['M500'] = _get('m500_true')
            snap, halo = int(_get('snap')), int(_get('halo_id'))
            if np.isfinite(ref['M500']):
                ref['R500_kpc'] = fh.crop_r500_guess(z_t, ref['M500'])[1]
        tp = Path(true_profiles_dir) / f'true_profiles_{snap:03d}_{halo}.npz'
        if tp.exists():
            ref['prof'] = dict(np.load(tp))
        tpp = find_truth_profile(view_dir=ref.get('view_dir', view_dir), snap=snap, halo=halo,
                                 sim=(meta or {}).get('sim'), search=truth_profile)
        if tpp is not None:
            ref['truth'] = load_truth_curves(tpp)
        if np.isfinite(_get('ra_true')):
            ra_c, dec_c = _get('ra_true'), _get('dec_true')
        else:
            ra_c = dec_c = np.nan
    else:
        m = res.get('published_m500_1e14')
        m = _get('hse') if m is None else m
        if m is not None and np.isfinite(m):
            ref.update(M500=float(m), R500_kpc=fh.crop_r500_guess(res['z'], float(m))[1])
        ra_c, dec_c = _get('ra_cat'), _get('dec_cat')
    if np.isfinite(ra_c) and np.isfinite(dec_c):
        ref['centre_off_amin'] = ((ra_c - ra) * math.cos(math.radians(dec)) * 60.0, (dec_c - dec) * 60.0)
    if 'R500_kpc' in ref:
        # a mock's true radius is seen at z_obs; the fit's kpc per arcsec is at its own (catalogue) redshift
        kpa_t = kpc_per_arcsec
        if is_mock and abs(float(ref['z_obs']) - float(res['z'])) > 1e-9:
            kpa_t = fh.flat_lcdm(fh.H0, fh.OM).kpc_proper_per_arcmin(float(ref['z_obs'])).value / 60.0
        ref['R500_amin'] = ref['R500_kpc'] / kpa_t / 60.0
    return ref

@cites("mbproj2d")
def best_fit_maps(st: dict) -> dict[str, Any]:
    """Best-fit maps of a :func:`rebuild_fit` result on the grid of the first image, summed
    over cameras: data D*, full model M* (exposure map + background), exposure E*, the
    analytical surface S (cluster convolved with the fitted PSF + flat background, no
    exposure map) and the cluster-only surface I, both as rate x each camera's peak exposure;
    * = soft (0.5-2 keV), hard (2-12 keV), all."""
    import numpy as np


    images, pars, totmod = st['images'], st['pars'], st['totmod']
    pars.setFree(st['best'])
    mod_tot = totmod.compute(pars)                                        # cluster + bg, x exposure
    mod_surf = totmod.compute(pars, apply_expmap=False)                   # analytical surface (rate/px)
    mod_int = totmod.compute(pars, apply_back=False, apply_expmap=False)  # cluster only (rate/px)
    shape = images[0].shape
    oy, ox = images[0].origin
    band_of = [im.img_id.split('_')[1] for im in images]
    bands = sorted(set(band_of))
    D, M, E, Surf, Int = ({b: np.zeros(shape) for b in bands} for _ in range(5))
    for im, mt, msf, mi, b in zip(images, mod_tot, mod_surf, mod_int, band_of):
        o = (oy, ox)
        D[b] += align(np.asarray(im.imagearr, float), im.origin, shape, o)
        M[b] += align(np.asarray(mt, float), im.origin, shape, o)
        E[b] += align(np.asarray(im.expmaps['expmap'], float), im.origin, shape, o)
        tpk = float(np.nanmax(im.expmaps['expmap']))                      # on-axis (peak) exposure
        Surf[b] += align(np.asarray(msf, float), im.origin, shape, o) * tpk
        Int[b] += align(np.asarray(mi, float), im.origin, shape, o) * tpk

    def bsum(d, bl):
        return sum(d[b] for b in bl if b in d)
    out = dict(shape=shape, oy=oy, ox=ox, pix_as=images[0].pixsize_as, band_of=band_of)
    for key, bl in (('soft', SOFT), ('hard', HARD)):
        out.update({f'D{key}': bsum(D, bl), f'M{key}': bsum(M, bl), f'E{key}': bsum(E, bl),
                    f'S{key}': bsum(Surf, bl), f'I{key}': bsum(Int, bl)})
    for q in 'DMESI':
        out[f'{q}all'] = out[f'{q}soft'] + out[f'{q}hard']
    out['exposed'] = out['Eall'] > 0
    return out

@cites("cosmology_planck2015", "astropy")
def aperture_in_fit_cosmology(z: float, r_kpc_other: float, h0_other: float, om0_other: float,
                              ode0_other: float | None = None) -> tuple[float, float]:
    """(r [kpc] in the fit cosmology, mass factor) of a physical aperture quoted in another cosmology.

    The aperture is the same ANGLE on the sky: r_fit = r_other x D_A,fit / D_A,other.  A
    hydrostatic mass inside a fixed angle scales as D_A (T and the logarithmic slopes are
    angular), so M_other = M_fit(<r_fit) x factor with factor = D_A,other / D_A,fit."""
    from astropy.cosmology import LambdaCDM

    from tngxmm.core.cosmology import flat_lcdm
    from tngxmm.massfit import fit_hydrostatic as fh

    ours = flat_lcdm(fh.H0, fh.OM)
    # the aperture's own (literature) cosmology: explicit, possibly non-flat
    other = (flat_lcdm(h0_other, om0_other) if ode0_other is None or
             abs(ode0_other + om0_other - 1) < 1e-9 else LambdaCDM(H0=h0_other, Om0=om0_other, Ode0=ode0_other))
    da_f = ours.angular_diameter_distance(z).value
    da_o = other.angular_diameter_distance(z).value
    return r_kpc_other * da_f / da_o, da_o / da_f

@cites("mbproj2d", "nfw")
def mass_within(st: dict, r_kpc, nsamples: int = 800, seed: int = 0):
    """Posterior percentiles (16, 50, 84) [1e14 Msun] of the total hydrostatic mass M(<r) of a
    :func:`rebuild_fit` result at each radius of ``r_kpc`` (fit cosmology): g r^2 / G of
    ``cluster.computeMassProfile`` over ``nsamples`` posterior draws, linearly interpolated
    on a 1-kpc grid."""
    import numpy as np

    mb, pars, cluster = st['mb'], st['pars'], st['cluster']
    r_kpc = np.atleast_1d(np.asarray(r_kpc, float))
    radii = mb.Radii(1.0, int(max(r_kpc.max() * 1.2, 100) / 1.0) + 1)
    post = st['post']
    rng = np.random.default_rng(seed)
    samp = post[rng.choice(len(post), size=min(nsamples, len(post)), replace=False)]
    out = []
    for s in samp:
        pars.setFree(s)
        g, _phi = cluster.computeMassProfile(pars, radii)
        m = g * radii.cent_cm ** 2 / mb.G_cgs / MSUN_G / 1e14
        out.append(np.interp(r_kpc, radii.cent_kpc, m))
    pars.setFree(st['best'])
    return np.percentile(np.asarray(out), [16, 50, 84], axis=0)

@cites("mbproj2d", "nfw", "barnes2021", "tngcluster", "matplotlib")
def render_truth_comparison(fit_dir: str | Path, out_png: str | Path, *, targets: str | Path | None = None,
                            cache_dir=None, img_root=None, resp=None, xspec_tmpdir=None,
                            truth_profile=None, view_dir=None, rmin_r500: float = 0.15,
                            rmax_r500: float = 2.0, nsamples: int = 800, verify_images: bool = True,
                            dlogl_tol: float = 1.0, dpi: int = 200) -> dict[str, Any]:
    """Our hydrostatic M(<r) of a MOCK against its truth (the mock analogue of a literature
    comparison): the posterior M(<r) with its 68 % band (:func:`mass_within`) over
    ``rmin_r500``-``rmax_r500`` x R500c, the true M_tot(<r) of all particles and the HSE-truth
    curves of the halo's truth-profile file (:func:`find_truth_profile`), with markers at R500:
    the fit's M500, the catalogue M500c (rho_c(z_snap)), M500 at rho_c(z_obs) from the
    particles (the fit's convention) and the headline intrinsic HSE M500.  Lower panel: each
    curve over the true M_tot(<r).  Radii are physical kpc; the fit cosmology is TNG's."""
    import matplotlib
    import numpy as np

    from tngxmm.sim.compute_hse_mass import rho_crit_msun_kpc3
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt

    st = rebuild_fit(fit_dir, cache_dir=cache_dir, img_root=img_root, resp=resp,
                     xspec_tmpdir=xspec_tmpdir, verify_images=verify_images)
    if abs(st['dlogl']) > dlogl_tol:
        raise AssertionError(f'replay dlogL {st["dlogl"]:.3f} > {dlogl_tol}: the rebuilt model is not the fitted one')
    res, kw = st['res'], st['kw']
    row = target_row(targets, res['name']) if targets is not None else None
    ref = reference_values(res, row, st['specs'], st['kpc_per_arcsec'], float(kw['ra']), float(kw['dec']),
                           view_dir=view_dir, truth_profile=truth_profile)
    if not ref['is_mock']:
        raise ValueError(f'{res["name"]}: not a mock; the truth comparison needs a TNG halo')
    tr = ref.get('truth')
    if tr is None:
        raise FileNotFoundError(f'{res["name"]}: no truth-profile file found (view dir '
                                f'{ref.get("view_dir")}, search {truth_profile})')
    R5 = tr['R500c_kpc']
    r = np.linspace(rmin_r500 * R5, rmax_r500 * R5, 120)
    band = mass_within(st, r, nsamples=nsamples)
    m, lo, hi = res['m500_1e14'], res['m500_lo'], res['m500_hi']
    mtot = np.exp(np.interp(np.log(r), np.log(tr['r']), np.log(tr['M_tot']))) / 1e14
    hse_lab = HSE_LABELS.get(ref.get('hse_label', 'barnes21'), ref.get('hse_label', ''))
    a = tr['attrs']

    with plt.rc_context({'font.size': 13}):
        fig, (ax, axr) = plt.subplots(2, 1, figsize=(15, 8.6), sharex=True,
                                      gridspec_kw=dict(height_ratios=[3, 1.35], hspace=0.05))
        ax.fill_between(r, band[0], band[2], color='C0', alpha=0.3, lw=0)
        ax.plot(r, band[1], color='C0', lw=2.6, label='this work: hydrostatic M(<r), XMM mock (68 %)')
        ax.plot(tr['r'], tr['M_tot'] / 1e14, color='k', lw=2.0,
                label='TNG truth: M$_{tot}$(<r), all particles (DM + gas + stars + BH)')
        axr.fill_between(r, band[0] / mtot, band[2] / mtot, color='C0', alpha=0.3, lw=0)
        axr.plot(r, band[1] / mtot, color='C0', lw=2.2)
        for lab_h, hr, hm, c, ls in _hse_curves(tr):
            ax.plot(hr, hm / 1e14, color=c, ls=ls, lw=1.6, label=f'HSE truth (3-D gas): {lab_h}')
            ok = np.isfinite(hr)
            if ok.sum() > 1:
                mt_h = np.exp(np.interp(np.log(hr[ok]), np.log(tr['r']), np.log(tr['M_tot'])))
                axr.plot(hr[ok], hm[ok] / mt_h, color=c, ls=ls, lw=1.3)
        ax.errorbar([res['r500_kpc']], [m], yerr=[[m - lo], [hi - m]], fmt='s', color='C0', mec='k', ms=10,
                    capsize=3, zorder=5, label=f'this work: M$_{{500}}$ = {m:.2f}')
        ax.plot(R5, tr['M500c'] / 1e14, 'D', color='lime', mec='k', ms=9, zorder=5,
                label=f'true M$_{{500c}}$ (catalogue, z$_{{snap}}$ = {tr["z_snap"]:.3f}) = {tr["M500c"] / 1e14:.2f}')
        if 'm500c_zobs_msun' in ref and 'r500c_zobs_kpc' in ref:
            ax.plot(ref['r500c_zobs_kpc'], ref['m500c_zobs_msun'] / 1e14, 'D', mfc='none', mec='k', mew=1.5,
                    ms=9, zorder=5, label=f'true M$_{{500}}$ at ρ$_c$(z$_{{obs}}$) = {ref["m500c_zobs_msun"] / 1e14:.2f}')
        if 'M500_hse' in ref:
            # self-consistent R500 of the headline HSE mass at rho_c(z_snap) (its definition)
            r_h = (3 * ref['M500_hse'] * 1e14 / (4 * math.pi * 500 *
                                                 rho_crit_msun_kpc3(tr['z_snap'], float(a['h'])))) ** (1 / 3)
            ax.plot(r_h, ref['M500_hse'], 'v', color='darkorange', mec='k', ms=10, zorder=5,
                    label=f'HSE truth M$_{{500}}$ ({hse_lab}) = {ref["M500_hse"]:.2f}')
        axr.axhline(1.0, color='k', lw=1.0)
        for a_ in (ax, axr):
            a_.axvline(R5, color='lime', lw=1, ls='--', zorder=0)
            a_.axvline(res['r500_kpc'], color='C0', lw=1, ls=':', zorder=0)
            a_.grid(alpha=0.3)
        ax.set_xlim(r[0], r[-1])
        sel = (tr['r'] >= r[0]) & (tr['r'] <= r[-1])
        ymax = max(np.nanmax(band[2]), np.nanmax(tr['M_tot'][sel]) / 1e14)
        ax.set_ylim(0, ymax * 1.12)
        ax.set_ylabel(r'enclosed mass $M(<r)$ [$10^{14}\,M_\odot$]')
        axr.set_ylim(0.4, 1.6)
        axr.set_ylabel(r'$M / M_{\rm tot,true}$')
        axr.set_xlabel('radius r [kpc] (physical)')
        ax.legend(fontsize=9.5, loc='upper left', bbox_to_anchor=(1.01, 1.0), framealpha=0.85)
        name = str(res['name'])
        z_o, z_f = float(ref['z_obs']), float(kw['z'])
        z_lab = f'z$_{{obs}}$ = {z_o:.3f}' + (f', fitted at z = {z_f:.3f}' if abs(z_f - z_o) > 5e-4 else '')
        ax.set_title(f'{name} ({z_lab}; TNG snap {tr["snap"]} halo {tr["halo"]}): '
                     'our hydrostatic mass vs the truth', fontsize=13)
        fig.text(0.5, 0.005, f'Truth: {Path(tr["path"]).name} ({a.get("campaign", "")}, spec {a.get("spec_hash", "")}); '
                             f'{_hse_caption(tr)}; mock cells = the '
                             'cells the mock emits from. Dashed green: R500c, dotted blue: R500 fit.',
                 ha='center', va='bottom', fontsize=8.5, color='0.3', wrap=True)
        out_png = Path(out_png)
        out_png.parent.mkdir(parents=True, exist_ok=True)
        fig.savefig(out_png, dpi=dpi, bbox_inches='tight')
        plt.close(fig)
    m_fit5, m_true5 = float(np.interp(R5, r, band[1])), float(np.interp(R5, r, mtot))
    print(f'wrote {out_png}  (M_fit(<R500c)/M_tot = {m_fit5 / m_true5:.3f}, dlogL = {st["dlogl"]:+.3g})')
    return dict(png=out_png, r_kpc=r, band=band, m_tot=mtot, truth_profile=tr['path'],
                m_fit_r500c=m_fit5, m_true_r500c=m_true5, dlogl=st['dlogl'])

def pars_summary(st: dict) -> str:

    keys = st['pars'].freeKeys()
    return f'{len(keys)} free pars equal to the chain: ' + ', '.join(k for k in keys if not k.startswith('bg_'))

def main(argv: list[str] | None = None) -> int:
    """``tngxmm massfit render-fit-v4 --fit-dir DIR --targets CSV --out DIR [--img-root ...]``."""
    import argparse

    ap = argparse.ArgumentParser(prog='tngxmm massfit render-fit-v4', description=__doc__.split('\n\n')[0])
    ap.add_argument('--fit-dir', required=True, action='append',
                    help='fit directory with result.json, provenance.json, chain.hdf5 (repeatable)')
    ap.add_argument('--targets', default=None, help='targets CSV the fit read (reference mass, centres)')
    ap.add_argument('--img-root', default=None, help='image tree (default: the one in provenance.json)')
    ap.add_argument('--resp', default=None, help='response directory (default: the one in provenance.json)')
    ap.add_argument('--cache-dir', required=True, help='directory CONTAINING mbproj2d_cache/ (a PRIVATE copy)')
    ap.add_argument('--xspec-tmpdir', default=None, help='XSPEC scratch on a rate-cache miss (never /tmp)')
    ap.add_argument('--out', default=None, help='output directory (not needed with --slide)')
    ap.add_argument('--nsamples', type=int, default=100)
    ap.add_argument('--true-profiles', default=str(TRUE_PROF))
    ap.add_argument('--ref-m500', type=float, default=None,
                    help='reference M500 [1e14 Msun] overriding the targets column (e.g. Hoekstra+15 WL)')
    ap.add_argument('--ref-label', default=None, help='label of the reference mass, e.g. "WL (Hoekstra+15)"')
    ap.add_argument('--view-dir', default=None, help='mock view directory (default: from the image links)')
    ap.add_argument('--slide', default=None, help='write only the three-panel slide to this PNG (one --fit-dir)')
    ap.add_argument('--slide-ref-name', default=None,
                    help='reference label on the slide, e.g. "quoted (WtG weak lensing)"')
    ap.add_argument('--slide-ref-short', default=None, help='reference in the mass ratio, e.g. "WL"')
    ap.add_argument('--slide-aperture-kpc', type=float, default=None,
                    help='reference is a mass inside this physical aperture [kpc, H0=70 Om=0.3 OL=0.7]')
    ap.add_argument('--slide-aperture-mass', type=float, nargs=2, default=None, metavar=('M', 'ERR'),
                    help='reference mass inside the aperture and its error [1e14 Msun]')
    ap.add_argument('--slide-half-width', type=float, default=1.8, help='slide half-width in reference theta500')
    ap.add_argument('--no-verify-images', action='store_true',
                    help='skip the image-tree digest check against provenance.json')
    ap.add_argument('--truth-profile', default=None,
                    help='mock truth profile: a truth_profile.h5, or a campaign tree root to search '
                         '(default: the halo directory of the view)')
    ap.add_argument('--truth-comparison', default=None,
                    help='write only the M(<r) vs truth comparison to this PNG (one mock --fit-dir)')
    a = ap.parse_args(argv)

                                verify_images=not a.no_verify_images)

                     resp=a.resp, xspec_tmpdir=a.xspec_tmpdir, half_width_theta500=a.slide_half_width,
                     ref_name=a.slide_ref_name, ref_short=a.slide_ref_short,
                     aperture_kpc=a.slide_aperture_kpc, aperture_mass=a.slide_aperture_mass,
                     verify_images=not a.no_verify_images)

    if not a.out:
        ap.error('--out is required (except with --slide)')
    rows = []
    for d in a.fit_dir:
        r = render_fit_v4(d, a.out, targets=a.targets, cache_dir=a.cache_dir, img_root=a.img_root,
                          resp=a.resp, xspec_tmpdir=a.xspec_tmpdir, nsamples=a.nsamples,
                          true_profiles_dir=a.true_profiles, view_dir=a.view_dir,
                          ref_m500=a.ref_m500, ref_label=a.ref_label,
                          verify_images=not a.no_verify_images, truth_profile=a.truth_profile)
        rows.append(r['summary'])
        s = r['summary']
        print(f'{s["name"]}: M500 = {s["m500"]:.2f} [{s["m500_lo"]:.2f}, {s["m500_hi"]:.2f}]  '
              f'{s["ref_label"]} = {s["m500_ref"]:.2f}  ratio = {s["ratio"]:.3f}  dlogL = {s["dlogl"]:+.3g}')
    return 0
