Source code for tngxmm.massfit.qpb_template

"""Particle-background templates of a real observation as XMM-ESAS and X-COP build them (``massfit.qpb_model =
'esas-fwc-corners'``; ledger ``fit.particle_background_esas``).

For one exposure of one EPIC camera and one PPS band this module makes, on the grid of the exposure's PPS image:

* the expected quiescent particle background (QPB) counts: the CCF filter-wheel-closed (FWC) events of the band
  (camera quality mask, PATTERN limit of the PPS images) of the closed exposures of the observation's epoch in the
  normal CCD state (:mod:`tngxmm.instrument.fwc_state`; the epoch the corner scales refer to, ``fwc_epoch_<cam>``),
  histogrammed per CCD on the ESAS detector grid (780 x 780, 50 DET units = 2.5" per pixel, DETX / DETY from -19500
  to 19500), smoothed within each CCD's area of that epoch by normalised convolution (:data:`SMOOTH_ARCMIN`), divided
  by the CCD's FWC exposure, multiplied by the observation's live time of the CCD and by the observation's corner
  scale (:mod:`tngxmm.donor_tables.qpb_corners`: the camera's scale in the band times the relative scale of the CCD
  (MOS) or quadrant (pn) in the corner band), then carried to the sky;
* the soft-proton template: the ESAS soft-proton detector map of the camera (CCF ``XMM_SPDETMAP``, the map ``proton``
  uses) on the same detector grid times the camera's live time, carried to the sky and normalised to unit mean over the
  exposed pixels (its level is free in the fit).

Detector to sky: an affine map fitted by least squares to the DETX, DETY and X, Y of the exposure's own events,
X, Y then to image pixels through the image's WCS.  The events are those the image is made of: in the field of view,
with a sky position (:func:`defined_sky`) and inside the good time intervals the image carries for their CCD
(:func:`image_good_times`, :func:`in_good_times`).  The PPS list also holds the events outside those intervals, and
an exposure that starts before the telescope has settled on the target has events at another attitude there; the
map is the mean attitude over the time the image integrates, the counterpart of ESAS ``rotdet2sky``.  Each detector
pixel is deposited onto the image grid with 3 x 3 sub-samples, so the counts are conserved.  A map that places the
template worse than one image pixel (:func:`check_det_to_sky`) and a template with nothing on the exposed pixels
(:func:`check_templates`) raise.

Departures from ESAS are those of the corner scales (:mod:`tngxmm.donor_tables.qpb_corners`).
"""

from __future__ import annotations

import math
from pathlib import Path
from typing import Any

from tngxmm.cites import cites, collect_cites

#: the ESAS detector grid: 780 x 780 pixels of 50 DET units (0.05" each) from -19500 to 19500
DET_NPIX = 780
DET_PIX = 50.0
DET_MIN = -19500.0
#: which events the detector-to-sky map of a real template is fitted to (:func:`in_good_times`); part of the name of
#: a cached template, so that one built from the events of the whole exposure is not reused
DET_TO_SKY_EVENTS = "gti"
#: smoothing of the FWC density within a CCD [arcmin] (as instrument.donor_background.FWC_SMOOTH_ARCMIN)
SMOOTH_ARCMIN = 1.0
#: sub-samples per detector pixel and axis when the detector grid is carried to the sky
SUBSAMPLE = 3
#: pn quadrants (CCD numbers)
PN_QUADRANTS = {"q0": (1, 2, 3), "q1": (4, 5, 6), "q2": (7, 8, 9), "q3": (10, 11, 12)}
CAMERA_SUFFIX = {"MOS1": "mos1", "MOS2": "mos2", "PN": "pn"}
SPDETMAP_ROW = {"MOS1": "EMOS1", "MOS2": "EMOS2", "PN": "EPN"}
_FWC_CACHE: dict = {}

def unit_of(cam: str, ccd: int) -> str:
    """The scaling unit of a CCD: the CCD number (MOS) or its pn quadrant."""
    if cam != "PN":
        return str(int(ccd))
    return next(q for q, ccds in PN_QUADRANTS.items() if int(ccd) in ccds)




#: the pn modes XMM-ESAS pnspectra accepts ("SUBMODE must be PrimeFullWindowExtended or PrimeFullWindow"): a Large or
#: Small Window exposure reads part of each CCD and has no usable unexposed corners
ESAS_PN_SUBMODES = ("PrimeFullWindow", "PrimeFullWindowExtended")

def esas_supported(cam: str, submode: str | None) -> bool:
    """Whether the ESAS particle-background model applies to an image of ``cam`` in ``submode`` (MOS: always; pn:
    the full-frame modes of :data:`ESAS_PN_SUBMODES`; a pn image without SUBMODE counts as full frame)."""
    return cam != "PN" or not submode or str(submode).strip() in ESAS_PN_SUBMODES

def ccd_scales(row: Any, cam: str, band: str, ccds: Any) -> dict[int, float]:
    """CCD -> QPB scale of the observation in ``band`` (PPS band id): the camera's corner scale in the band times the
    relative scale of the CCD's unit in the corner band (unit scale over the camera's corner-band scale; a unit without
    corner counts takes 1)."""
    s = CAMERA_SUFFIX[cam]
    cam_band = float(row[f"corner_{band}_{s}"])
    units = {u: float(row.get(f"unit_{u}_{s}", float("nan"))) for u in
             (PN_QUADRANTS if cam == "PN" else [str(k) for k in range(1, 8)])}
    good = [v for v in units.values() if math.isfinite(v) and v > 0]
    mean_unit = sum(good) / len(good) if good else float("nan")
    out = {}
    for c in ccds:
        u = units.get(unit_of(cam, c), float("nan"))
        rel = u / mean_unit if (math.isfinite(u) and u > 0 and math.isfinite(mean_unit)) else 1.0
        out[int(c)] = cam_band * rel
    return out

def det_edges() -> Any:
    import numpy as np

    return DET_MIN + DET_PIX * np.arange(DET_NPIX + 1)

def fwc_band_rates(ccf_dir: str | Path, cam: str, band_ev: tuple[float, float], which: str = "latest",
                   half: int | None = None, epoch_first: int | None = None, ccds: Any = None,
                   rev: int | None = None) -> dict[int, Any]:
    """CCD -> FWC rate per detector pixel [counts / s] of the band on the ESAS detector grid (rows DETY, columns DETX),
    smoothed within the CCD, from the closed exposures of one epoch in the normal CCD state
    (:mod:`tngxmm.instrument.fwc_state`): the epoch that starts at revolution ``epoch_first`` (the corner table's
    ``fwc_epoch_<cam>``), else that of the working CCDs ``ccds`` and revolution ``rev``.  The CCD's area is the one
    with events in the epoch, whatever their state (MOS1 CCD 4 lacks RAWY < 150 after revolution 2382), isolated empty
    pixels filled; cached per (ccf, camera, band, set, half, epoch).  ``which``: the camera's newest FWC set ('latest')
    or its oldest other set ('oldest', the wrong-template test; :func:`tngxmm.instrument.donor_background._oldest_fwc`).
    ``half``: only the events of even (0) or odd (1) index (:data:`tngxmm.instrument.donor_background.FWC_HALVES`),
    each half at twice the rate of its counts per exposure (it is a random half of the same exposure); None all
    events."""
    import numpy as np
    from astropy.io import fits
    from scipy.ndimage import binary_closing, gaussian_filter

    from tngxmm.donor_tables.fwc import fwc_file
    from tngxmm.instrument import donor_corner_qpb as Q
    from tngxmm.instrument import fwc_state as S

    if which == "latest":
        path = fwc_file(ccf_dir, cam)
    else:
        from tngxmm.instrument.donor_background import _oldest_fwc

        path = _oldest_fwc(str(ccf_dir), cam)
    with fits.open(path, memmap=True) as h:
        rows = S.exptime_rows(h["EXPTIME"].data)
        eps = S.epochs(rows)
        ep = S.epoch_by_first(eps, epoch_first) if epoch_first is not None else S.select_epoch(eps, ccds or (), rev)
        missing = ({int(c) for c in ccds or ()} & {c for e in eps for c in e.ccds}) - set(ep.ccds)
        if missing:
            raise ValueError(f"{Path(path).name}: the epoch of revolutions {ep.revs[0]}-{ep.revs[1]} has no CCD "
                             f"{sorted(missing)} of the exposure")
        key = (str(ccf_dir), cam, tuple(band_ev), which, half, ep.revs, S.FWC_REFERENCE_RULE)
        if key in _FWC_CACHE:
            return _FWC_CACHE[key]
        ev = h["EVENTS"].data
        flag = np.asarray(ev["FLAG"]).astype(np.int64)
        pat, pi = np.asarray(ev["PATTERN"]), np.asarray(ev["PI"])
        ccd = np.asarray(ev["CCDNR"]).astype(int)
        x, y = np.asarray(ev["DETX"], float), np.asarray(ev["DETY"], float)
        row_of = S.event_rows(np.asarray(ev["TIME"], float), ccd, rows)
    normal = S.row_selection(rows, ep, normal=True)
    in_epoch = S.selected_events(row_of, S.row_selection(rows, ep, normal=False))
    texp = S.ccd_exposure(rows, normal)
    band_tuple = (band_ev[0], band_ev[1])
    sel = ((pat <= Q.pattern_max(cam, band_tuple)) & (pi >= band_ev[0]) & (pi < band_ev[1])
           & ((flag & Q.BAD_MASK[cam]) == 0) & S.selected_events(row_of, normal))
    if half is not None:
        sel &= (np.arange(len(sel)) % 2) == int(half)
        texp = {c: v / 2.0 for c, v in texp.items()}
    edges = det_edges()
    sig = SMOOTH_ARCMIN * 60.0 / (DET_PIX * 0.05)
    out = {}
    for c in np.unique(ccd[in_epoch]):
        on = (ccd == c) & in_epoch
        occ, _, _ = np.histogram2d(y[on], x[on], bins=[edges, edges])
        occ = binary_closing(occ > 0, structure=np.ones((3, 3))).astype(float)
        if occ.sum() < 100 or texp.get(int(c), 0.0) <= 0:
            continue
        h_, _, _ = np.histogram2d(y[on & sel], x[on & sel], bins=[edges, edges])
        socc = gaussian_filter(occ, sig)
        with np.errstate(invalid="ignore", divide="ignore"):
            sm = np.where(occ > 0, gaussian_filter(h_, sig) / np.maximum(socc, 1e-12), 0.0)
        # keep the band's total counts of the CCD (the normalised convolution redistributes them within the CCD)
        if sm.sum() > 0:
            sm *= h_.sum() / sm.sum()
        out[int(c)] = sm / texp[int(c)]
    _FWC_CACHE[key] = out
    return out

def template_signature(corner_row: Any, cam: str, band: str) -> str:
    """Short digest of what a real template takes from the corner table (the camera's scale in the band, the unit
    scales, the FWC epoch) and of the FWC rule: the name of its cache file, so that a measured-again table never
    reuses a template of the old scales."""
    import hashlib
    import json

    from tngxmm.instrument.fwc_state import FWC_REFERENCE_RULE

    s = CAMERA_SUFFIX[cam]
    keys = [f"corner_{band}_{s}", f"fwc_epoch_{s}"] + [f"unit_{u}_{s}" for u in
                                                        (PN_QUADRANTS if cam == "PN" else [str(k) for k in range(1, 8)])]

    def val(k: str) -> Any:
        try:
            v = float(corner_row[k])
        except (KeyError, IndexError, TypeError, ValueError):
            return None
        return round(v, 9) if v == v else None

def flare_detmap(ccf_dir: str | Path, cam: str, filt: str | None, band: str) -> Any:
    """The soft-proton map of ``cam`` on the ESAS detector grid for PPS band ``band``, as XMM-ESAS ``proton`` builds it
    (SAS 20 and later): the flare maps of the CCF XMM_FLARE for the filter (a filter without maps takes Medium, Kuntz &
    Snowden 2008 Sect. 4.3), each smoothed like the mock stage's (:func:`tngxmm.instrument.donor_background.smooth_detmap`)
    and normalised to mean 1 within
    :data:`tngxmm.instrument.donor_background.FLARE_NORM_RADIUS_ARCMIN` of the detector centre, co-added with the
    share of the soft-proton counts spectrum in the band (:func:`tngxmm.instrument.donor_background.flare_band_shares`,
    the same weights the mock stage uses)."""
    import numpy as np
    from astropy.io import fits

    from tngxmm.instrument import donor_background as DB
    from tngxmm.instrument.donor_corner_qpb import PPS_BANDS_EV
    from tngxmm.instrument.rga_obscuration import ccf_file

    f = filt if filt in DB.FLARE_FILTERS else "Medium"
    maps = {}
    with fits.open(str(ccf_file(ccf_dir, "XMM_FLARE"))) as h:
        for r in h["FLAREMAP"].data:
            b = str(r["BAND"]).strip()
            if str(r["INSTRUME"]).strip() == SPDETMAP_ROW[cam] and str(r["FILTER"]).strip() == f \
                    and b in DB.FLARE_BANDS_KEV:
                maps[DB.FLARE_BANDS_KEV[b]] = DB.smooth_detmap(np.asarray(r["FLAREMAP"], float))
    if len(maps) != len(DB.FLARE_BANDS_KEV):
        raise ValueError(f"XMM_FLARE: {cam} {f} has maps for {sorted(maps)}, not the six bands")
    c = (DET_MIN + (np.arange(DET_NPIX) + 0.5) * DET_PIX) * 0.05 / 60.0          # pixel centres [arcmin]
    inner = np.hypot(*np.meshgrid(c, c)) < DB.FLARE_NORM_RADIUS_ARCMIN
    edges = sorted(maps)
    lo, hi = (v / 1000.0 for v in PPS_BANDS_EV[band])
    share = DB.flare_band_shares(edges, lo, hi)
    out = np.zeros((DET_NPIX, DET_NPIX))
    for k, e in enumerate(edges):
        m = np.clip(maps[e], 0.0, None)
        sel = inner & (m > 0)
        if sel.any():
            out += share[k] * m / m[sel].mean()
    return out

@cites("sas_attcalc")
def defined_sky(header: Any, x: Any, y: Any) -> Any:
    """Whether each event of an event list (``header`` of its EVENTS extension) has a sky position: X and Y are not
    the column's null value and lie within its legal range.

    SAS attcalc computes X, Y from DETX, DETY with the attitude of the event's frame; where the attitude call returns
    invalid pointing data "the X/Y position of the events in question are set to NULL" (attcalc, Sect. 3).  In the
    integer X, Y columns of a PPS event list that NULL is the column's TNULL value (-99999999); TLMIN and TLMAX give
    the legal range (1-51840).  Such events keep their DETX, DETY and FLAG, so the in-field selection alone does not
    remove them.  A column without the keywords falls back to finite values."""
    import numpy as np

    names = {str(header[f"TTYPE{i}"]): i for i in range(1, int(header["TFIELDS"]) + 1)}
    ok = np.ones(len(x), bool)
    for col, v in (("X", x), ("Y", y)):
        v = np.asarray(v, float)
        n = names[col]
        ok &= np.isfinite(v)
        # older PPS lists write TLMIN / TLMAX as strings ('1', '51840')
        if f"TNULL{n}" in header:
            ok &= v != float(header[f"TNULL{n}"])
        if f"TLMIN{n}" in header:
            ok &= v >= float(header[f"TLMIN{n}"])
        if f"TLMAX{n}" in header:
            ok &= v <= float(header[f"TLMAX{n}"])
    return ok

@cites("sas_evselect")
def image_good_times(image_path: str | Path) -> dict[int, Any]:
    """CCD -> good time intervals (N, 2) [s, mission time] of a PPS image.

    SAS evselect writes the time selection of a product into the product: an image has one GTI table per CCD
    (HDUCLAS1 ``GTI``, the CCD in ``CCDID``), whose length is the image's ONTIMEnn (to a few seconds for the pn).  They are
    the intervals the events of the image were taken in, after the pipeline's flare screening.  Raises ``ValueError``
    for an image without such a table: its time selection is not known, and the events of the whole exposure are not
    a substitute (see :func:`in_good_times`)."""
    import numpy as np
    from astropy.io import fits

    good: dict[int, Any] = {}
    with fits.open(image_path) as h:
        for ext in h[1:]:
            if str(ext.header.get("HDUCLAS1", "")).strip() == "GTI" and "CCDID" in ext.header:
                g = np.c_[np.asarray(ext.data["START"], float), np.asarray(ext.data["STOP"], float)]
                good[int(ext.header["CCDID"])] = g[np.argsort(g[:, 0])]
    if not good:
        raise ValueError(f"{Path(image_path).name}: the image has no good-time table (HDUCLAS1 = GTI with CCDID)")
    return good

def in_good_times(time: Any, ccd: Any, good: dict[int, Any]) -> Any:
    """Whether each event (``time`` [s], CCD number ``ccd``) lies inside the good time intervals of its CCD
    (``good``: CCD -> (N, 2), sorted by start, :func:`image_good_times`).  An event of a CCD without intervals is
    outside: the image has nothing from that CCD.

    The detector-to-sky map needs this selection.  DETX, DETY to X, Y is one rotation and shift only while the
    attitude holds; the event list of an exposure that began during the slew or before the pointing had settled has
    events several arcseconds to arcminutes from where the same detector pixel falls later, and the image leaves
    them out with the rest of the bad time."""
    import numpy as np

    t = np.asarray(time, float)
    c = np.asarray(ccd).astype(int)
    inside = np.zeros(len(t), bool)
    for k, g in good.items():
        m = c == int(k)
        if not len(g) or not m.any():
            continue
        i = np.searchsorted(g[:, 0], t[m], side="right") - 1
        inside[m] = (i >= 0) & (t[m] <= g[np.clip(i, 0, None), 1])
    return inside

def fit_det_to_sky(detx: Any, dety: Any, x: Any, y: Any) -> tuple[Any, float]:
    """Affine map (2, 3) with [X, Y] = A @ [DETX, DETY, 1], least squares, and its rms residual [sky units].  The
    events must all have a sky position (:func:`defined_sky`) and belong to one pointing (:func:`in_good_times`);
    fewer than three raise ``ValueError``."""
    import numpy as np

    if len(detx) < 3:
        raise ValueError(f"detector-to-sky map: {len(detx)} events with a sky position, at least 3 needed")
    m = np.c_[np.asarray(detx, float), np.asarray(dety, float), np.ones(len(detx))]
    a = np.linalg.lstsq(m, np.c_[np.asarray(x, float), np.asarray(y, float)], rcond=None)[0].T
    res = np.c_[x, y] - (m @ a.T)
    return a, float(np.sqrt((res ** 2).sum(axis=1).mean()))

DET_HALF_DIAGONAL = -DET_MIN * math.sqrt(2.0)

def check_det_to_sky(affine: Any, rms: float, pixel_sky: float, label: str) -> None:
    """Raise ``ValueError`` (naming ``label``, the observation and image) unless the detector-to-sky map places the
    template to within one image pixel ``pixel_sky`` [sky units].

    The criterion is geometric.  DETX, DETY and X, Y are in the same unit (0.05"), so the linear part L of the map is
    a rotation, possibly with a reflection, and the attitude only turns and shifts the detector.  Two things move a
    template pixel from where its events fall: the scatter of the events about the fitted map (its rms residual,
    the attitude's wander about the mean attitude ESAS rotdet2sky also assumes) and a departure of L from a rotation,
    which at the edge of the grid moves a pixel by up to ||L - R|| times :data:`DET_HALF_DIAGONAL` (R the nearest
    orthogonal matrix, ||.|| the largest singular value).  The template is deposited on the nearest image pixel, so
    either displacement above one image pixel means the template is not where the events are at the resolution of the
    image.  A sky position that is not one (the NULL of attcalc) moves the fit by millions of sky units; events from
    before the pointing settled, by tens to hundreds.  With both left out, what still fails is an attitude that
    moved by more than an image pixel inside the image's own good time, which no single template describes."""
    import numpy as np

    lin = np.asarray(affine, float)[:, :2]
    if not (np.all(np.isfinite(lin)) and math.isfinite(rms)):
        raise ValueError(f"{label}: detector-to-sky map not finite (rms {rms})")
    u, _, vt = np.linalg.svd(lin)
    shear = float(np.linalg.norm(lin - u @ vt, 2)) * DET_HALF_DIAGONAL
    if rms > pixel_sky or shear > pixel_sky:
        raise ValueError(f"{label}: the detector-to-sky map misplaces the template: rms residual {rms:.4g} and "
                         f"departure from a rotation {shear:.4g} sky units at the grid edge, one image pixel is "
                         f"{pixel_sky:.4g}")

def check_templates(qpb: Any, sp: Any, expmap: Any, label: str) -> None:
    """Raise ``ValueError`` (naming ``label``) unless the QPB and soft-proton templates of a real image are finite
    and have a positive sum over the exposed pixels (``expmap`` > 0).  Every exposed pixel of a real image lies on a
    CCD that records particles and on the soft-proton map, so a zero sum means the templates fell off the image."""
    import numpy as np

    e = np.asarray(expmap) > 0
    for name, t in (("QPB", qpb), ("soft-proton", sp)):
        t = np.asarray(t, float)
        if not np.all(np.isfinite(t)):
            raise ValueError(f"{label}: the {name} template has non-finite values")
        s = float(t[e].sum())
        if not s > 0:
            raise ValueError(f"{label}: the {name} template sums to {s:.4g} over the {int(e.sum())} exposed pixels")

def sky_to_pixel(header: Any, wcs: Any, x: Any, y: Any) -> tuple[Any, Any]:
    """Float (column, row) of the image grid of sky coordinates X, Y [0.05"] of the event list ``header``."""
    import numpy as np

    names = {str(header[f"TTYPE{i}"]): i for i in range(1, int(header["TFIELDS"]) + 1)}
    ix_, iy_ = names["X"], names["Y"]
    ra0, dec0 = float(header[f"TCRVL{ix_}"]), float(header[f"TCRVL{iy_}"])
    ra = ra0 + (np.asarray(x, float) - float(header[f"TCRPX{ix_}"])) * float(header[f"TCDLT{ix_}"]) / math.cos(
        math.radians(dec0))
    dec = dec0 + (np.asarray(y, float) - float(header[f"TCRPX{iy_}"])) * float(header[f"TCDLT{iy_}"])
    px, py = wcs.all_world2pix(ra, dec, 0)
    return np.asarray(px), np.asarray(py)

def det_to_image(det_map: Any, affine: Any, header: Any, wcs: Any, shape: tuple[int, int],
                 subsample: int = SUBSAMPLE) -> Any:
    """Deposit a detector-grid map (values per detector pixel) onto the image grid: each detector pixel split into
    ``subsample`` x ``subsample`` points, each carrying its share, nearest image pixel.  Conserves the total within
    the image."""
    import numpy as np

    rows, cols = np.nonzero(det_map)
    if not len(rows):
        return np.zeros(shape)
    v = det_map[rows, cols] / subsample ** 2
    off = (np.arange(subsample) + 0.5) / subsample
    out = np.zeros(shape)
    for oy in off:
        for ox in off:
            dx = DET_MIN + (cols + ox) * DET_PIX
            dy = DET_MIN + (rows + oy) * DET_PIX
            xs = affine[0, 0] * dx + affine[0, 1] * dy + affine[0, 2]
            ys = affine[1, 0] * dx + affine[1, 1] * dy + affine[1, 2]
            px, py = sky_to_pixel(header, wcs, xs, ys)
            ix, iy = np.rint(px).astype(int), np.rint(py).astype(int)
            ok = (ix >= 0) & (iy >= 0) & (ix < shape[1]) & (iy < shape[0])
            np.add.at(out, (iy[ok], ix[ok]), v[ok])
    return out

def observation_templates(event_list: str | Path, image_path: str | Path, cam: str, band: str, corner_row: Any,
                          live: dict[int, float], ccf_dir: str | Path, which: str = "latest",
                          sp_model: str = "flare") -> dict[str, Any]:
    """The QPB counts template and the soft-proton template of one PPS image (``image_path``; band id ``band``) of a
    real exposure, with its own event list ``event_list`` (for the detector-to-sky map, from the events inside the
    image's good time intervals), the observation's corner
    row (:mod:`tngxmm.donor_tables.qpb_corners`) and the live time of each CCD within the image's good time intervals
    ``live``.  Returns ``qpb`` (counts per image pixel), ``sp`` (relative, mean 1 over the exposed pixels), ``ccd``
    (the CCD number of each image pixel from the exposure's in-FOV events, 0 where none fell), ``rms`` (residual of the
    detector-to-sky map [sky units]) and ``total`` (QPB counts on the image).  ``which``: the FWC set
    (:func:`fwc_band_rates`).  ``sp_model``: ``'flare'`` (the CCF flare maps of the image's filter and band,
    :func:`flare_detmap`, as current XMM-ESAS and the mock stage under ``instrument.soft_proton_map = 'flare'``) or
    ``'spdetmap'`` (the single CCF XMM_SPDETMAP of older ESAS versions)."""
    import numpy as np
    from astropy.io import fits
    from astropy.wcs import WCS

    from tngxmm.instrument.donor_corner_qpb import pps_band_parts

    good = image_good_times(image_path)
    with fits.open(event_list, memmap=True) as h:
        hdr = h["EVENTS"].header
        ev = h["EVENTS"].data
        # in-FOV events with a sky position: an event that attcalc could not place has X, Y = NULL
        infov = ((np.asarray(ev["FLAG"]).astype(np.int64) & 0x10000) == 0) & defined_sky(hdr, ev["X"], ev["Y"])
        # the map from the events the image is made of: the list also has those of the bad time, which in an
        # exposure that began before the pointing settled lie at another attitude
        idx = np.flatnonzero(infov & in_good_times(ev["TIME"], ev["CCDNR"], good))
        if len(idx) > 200000:
            idx = idx[:: len(idx) // 200000 + 1]
        if len(idx) < 3:
            raise ValueError(f"{hdr.get('OBS_ID', '')} {Path(image_path).name}: {len(idx)} in-field events inside the "
                             f"image's good time intervals, at least 3 needed for the detector-to-sky map")
        affine, rms = fit_det_to_sky(ev["DETX"][idx], ev["DETY"][idx], ev["X"][idx], ev["Y"][idx])
        # the CCD of each image pixel from all placed in-field events, not only those of the good time: a heavily
        # screened image keeps too few events to reach every exposed pixel, and the label is the majority CCD
        sel = np.flatnonzero(infov)
        if len(sel) > 2000000:
            sel = sel[:: len(sel) // 2000000 + 1]
        ev_x, ev_y = np.asarray(ev["X"])[sel], np.asarray(ev["Y"])[sel]
        ev_ccd = np.asarray(ev["CCDNR"])[sel].astype(int)
    img_hdr = fits.getheader(image_path)
    if not esas_supported(cam, img_hdr.get("SUBMODE")):
        raise ValueError(f"{Path(image_path).name}: pn SUBMODE {img_hdr.get('SUBMODE')} has no unexposed corners "
                         f"(XMM-ESAS pnspectra accepts {ESAS_PN_SUBMODES})")
    shape = (int(img_hdr["NAXIS2"]), int(img_hdr["NAXIS1"]))
    wcs = WCS(img_hdr, naxis=2)
    label = f"{hdr.get('OBS_ID', '')} {Path(image_path).name}"
    check_det_to_sky(affine, rms, image_pixel_sky(hdr, img_hdr), label)
    expmap = fits.getdata(expmap_path(image_path, band))
    # the FWC events in the PI ranges of the PPS image (the pn 4.5-12 keV image leaves out the Cu line at 7.8-8.2 keV),
    # of the epoch and CCD state the corner scales refer to
    epoch = reference_epoch(corner_row, cam, live, which, hdr.get("REVOLUT"))
    rates: dict[int, Any] = {}
    for part in pps_band_parts(cam, band):
        from tngxmm.instrument.donor_background import FWC_TEMPLATE_HALF

        for c, r in fwc_band_rates(ccf_dir, cam, part, which, half=FWC_TEMPLATE_HALF, **epoch).items():
            rates[c] = rates[c] + r if c in rates else r
    scales = ccd_scales(corner_row, cam, band, rates)
    det = np.zeros((DET_NPIX, DET_NPIX))
    for c, r in rates.items():
        det += r * float(live.get(c, 0.0)) * scales[c]
    qpb = np.where(expmap > 0, det_to_image(det, affine, hdr, wcs, shape), 0.0)
    if sp_model == "flare":
        sp_det = flare_detmap(ccf_dir, cam, str(img_hdr.get("FILTER", "")).strip(), band)
    elif sp_model == "spdetmap":
        sp_det = spdetmap(ccf_dir, cam)
    else:
        raise ValueError(f"sp_model must be 'flare' or 'spdetmap', got {sp_model!r}")
    sp = np.where(expmap > 0, det_to_image(sp_det, affine, hdr, wcs, shape), 0.0)
    m = sp[expmap > 0]
    sp = sp / m.mean() if m.size and m.mean() > 0 else sp
    check_templates(qpb, sp, expmap, label)
    return {"qpb": qpb, "sp": sp, "ccd": ccd_label_image(hdr, wcs, shape, ev_x, ev_y, ev_ccd), "rms": rms,
            "total": float(qpb.sum())}

def ccd_label_image(header: Any, wcs: Any, shape: tuple[int, int], x: Any, y: Any, ccd: Any) -> Any:
    """The CCD number of each image pixel: the CCD with the most events (sky X, Y of the event list, header of the
    EVENTS extension) in the pixel; 0 where no event fell."""
    import numpy as np

    px, py = sky_to_pixel(header, wcs, np.asarray(x, float), np.asarray(y, float))
    ix, iy = np.rint(px).astype(int), np.rint(py).astype(int)
    ok = (ix >= 0) & (iy >= 0) & (ix < shape[1]) & (iy < shape[0])
    ix, iy, c = ix[ok], iy[ok], np.asarray(ccd)[ok]
    out = np.zeros(shape, dtype=np.int16)
    best = np.zeros(shape)
    for k in np.unique(c):
        n = np.zeros(shape)
        np.add.at(n, (iy[c == k], ix[c == k]), 1.0)
        take = n > best
        out[take], best[take] = int(k), n[take]
    return out
