Source code for tngxmm.schemas.tables

"""Cross-stage table schemas (INTERFACES section 4).

Stage-internal tables live in ``tngxmm/<stage>/schemas.py``.  Units are in
the column names; ``legacy_renames`` maps the column names of the campaign
results CSV (``results_*_snap_*.csv``), ``result.json`` and the targets CSVs to
the schema names.
"""

from __future__ import annotations

from tngxmm.cites import collect_cites
from tngxmm.schemas._base import Column as C
from tngxmm.schemas._base import TableSchema
from tngxmm.schemas.ids import (
    CLUSTER_ID_PATTERN,
    MOCK_OBSID_PATTERN,
    OBSID_PATTERN,
    VIEW_ID_PATTERN,
)

#: outcome of a campaign aperture Lx (tngxmm.photometry.estimate_lx_background.LX_FLAGS)
LX_FLAGS = ("ok", "nondetection", "masked", "uncovered")
SPLITS = ("train", "val", "test")
CAMERAS = ("PN", "MOS1", "MOS2")
FILTERS = ("Thin1", "Thin2", "Medium", "Thick")
PN_MODES = ("PrimeFullWindow", "PrimeFullWindowExtended", "PrimeLargeWindow",
            "PrimeSmallWindow")

_PROVENANCE = (
    C("spec_hash", "str", "CampaignSpec hash (runtime excluded)"),
    C("package_version", "str", "tngxmm version that wrote the row"),
    C("git_hash", "str", "commit of the producing code", nullable=True),
)

# --------------------------------------------------------------------- views --
VIEW_TABLE = TableSchema(
    name="view",
    owner="sim writes; instrument and photometry fill their columns",
    description="One row per mock view (the campaign results_*_snap_*.csv row).",
    primary_key=("view_id",),
    columns=(
        C("view_id", "str", "canonical view id", regex=VIEW_ID_PATTERN),
        C("campaign", "str", "preset name (v1, v1_tng300, v1_tng100, v1_tng50, v1_flamingo, v1_flamingo_l1m8, v1_simba, v1_eagle)"),
        C("sim", "str", "simulation", allowed=("tng-cluster", "tng300", "tng100", "tng50", "flamingo", "flamingo_l1m8", "simba",
                                               "eagle")),
        C("snap", "int", "snapshot number", min=0, max=151),
        C("halo_id", "int", "halo index in the snapshot's catalogue (Subfind FoF group; SOAP halo for FLAMINGO)",
          min=0),
        C("view", "int", "projection axis 0=z 1=y 2=x", min=0, max=9),
        C("rep", "int", "realisation index (0 unless re-observed)", min=0),
        C("mock_obsid", "str", "PPS-like id of the per-camera products", regex=MOCK_OBSID_PATTERN),
        C("zoom_target_id", "int", "TNG-Cluster zoom region (split unit); the halo's own id in a box", min=0),
        C("split", "str", "frozen split_by_zoom_target.csv (seed 7777)", allowed=SPLITS),
        C("z_snap", "float", "snapshot redshift", min=0.0),
        C("z_obs", "float", "observed redshift", min=0.0),
        C("z_obs_source", "str", "psz2-draw | targets-csv | replica"),
        C("m500c_msun", "float", "catalogue M500c, physical", min=0.0),
        C("r500c_kpc", "float", "catalogue R500c, physical", min=0.0),
        C("m200c_msun", "float", "catalogue M200c, physical", min=0.0),
        C("r200c_kpc", "float", "catalogue R200c, physical", min=0.0),
        C("theta500_arcmin", "float", "R500c angle at z_obs", min=0.0),
        C("donor_id", "str", "obsid of the physics donor observation (mass, redshift, count-matching target)", regex=OBSID_PATTERN),
        C("nh_1e22cm2", "float", "Galactic N_H the view was observed with (the conditions donor's under the rule "
          "nn; the stored photon list is absorbed with physics_nh_1e22)", min=0.0),
        C("exposure_eff_s", "float", "ARF-weighted mean EMSCE live time", min=0.0),
        C("exposure_pn_s", "float", "PN live time of the products (the conditions donor's, jittered, under the rule "
          "nn; NaN if no PN)", nullable=True, min=0.0),
        C("exposure_mos1_s", "float", "MOS1 live time of the products", nullable=True, min=0.0),
        C("exposure_mos2_s", "float", "MOS2 live time of the products", nullable=True, min=0.0),
        C("filter", "str", "EPIC filter of the products (PN, else MOS1)", allowed=FILTERS),
        C("pn_mode", "str", "PN submode of the products", allowed=PN_MODES, nullable=True),
        C("posangle_deg", "float", "roll angle of the products", min=0.0, max=360.0),
        C("conditions_obsid", "str", "obsid of the observation the products were made under (the conditions "
          "donor under the rule nn, else donor_id)", regex=OBSID_PATTERN, required=False),
        C("physics_exposure_pn_s", "float", "PN live time of the physics donor (donor_id): its REAL observation, "
          "not the replica's jittered time", nullable=True, min=0.0, required=False),
        C("physics_exposure_mos1_s", "float", "MOS1 live time of the physics donor (real, not jittered)", nullable=True, min=0.0,
          required=False),
        C("physics_exposure_mos2_s", "float", "MOS2 live time of the physics donor (real, not jittered)", nullable=True, min=0.0,
          required=False),
        C("physics_filter", "str", "EPIC filter of the physics donor", allowed=FILTERS, nullable=True,
          required=False),
        C("physics_pn_mode", "str", "PN submode of the physics donor", allowed=PN_MODES, nullable=True,
          required=False),
        C("physics_nh_1e22cm2", "float", "N_H of the physics donor, the absorption of the stored photon list",
          nullable=True, min=0.0, required=False),
        C("max_dither_arcmin", "float", "dither cap (NaN = uncapped)", nullable=True),
        C("lx_intr_r500_erg_s", "float", "intrinsic rest 0.5-2 keV Lx, cylinder R500c"),
        C("lx_intr_1mpc_erg_s", "float", "intrinsic, cylinder 1 Mpc"),
        C("lx_intr_1p5mpc_erg_s", "float", "intrinsic, cylinder 1.5 Mpc"),
        C("lx_r500_erg_s", "float", "measured aperture Lx inside the TRUE R500c about the true centre, at "
          "the true T_vir, the simulated point sources above the donor's threshold masked (NaN unless "
          "lx_r500_flag == 'ok'); a truth-side quantity, not comparable with a real observation's Lx: the "
          "mock-real pair is the fit-target seed photometry", nullable=True, min=0.0),
        C("lx_1mpc_erg_s", "float", "measured, 1 Mpc (NaN unless "
          "lx_1mpc_flag == 'ok')", nullable=True, min=0.0),
        C("lx_1p5mpc_erg_s", "float", "measured, 1.5 Mpc (NaN "
          "unless lx_1p5mpc_flag == 'ok')", nullable=True, min=0.0),
        *(C(f"{ap}_flag", "str", f"{what}: ok | nondetection (net counts <= 0) | masked "
            "(point-source masks cover > 50 % of the aperture or leave < 200 annulus px) | "
            "uncovered (aperture < 98 % covered, < 200 covered annulus px)",
            allowed=LX_FLAGS, required=False)
          for ap, what in (("lx_r500", "measured Lx, R500c aperture"),
                           ("lx_1mpc", "measured Lx, 1 Mpc"), ("lx_1p5mpc", "measured Lx, 1.5 Mpc"))),
        *(C(f"{ap}_signed_erg_s", "float", f"{what}, signed (net counts not clipped; NaN "
            "only when uncovered)", nullable=True, required=False)
          for ap, what in (("lx_r500", "measured Lx, R500c aperture"),
                           ("lx_1mpc", "measured Lx, 1 Mpc"), ("lx_1p5mpc", "measured Lx, 1.5 Mpc"))),
        *(C(f"{ap}_mask_frac", "float", f"{what}: fraction of the aperture pixels under "
            "point-source masks (> 0.5: flag 'masked')", nullable=True, required=False,
            min=0.0, max=1.0)
          for ap, what in (("lx_r500", "measured Lx, R500c aperture"),
                           ("lx_1mpc", "measured Lx, 1 Mpc"), ("lx_1p5mpc", "measured Lx, 1.5 Mpc"))),
        *(C(f"{ap}_area_corr", "float", f"{what}: masked-area correction factor applied (ring-"
            "filled net counts / net counts of the unmasked pixels; 1 = nothing masked)",
            nullable=True, required=False)
          for ap, what in (("lx_r500", "measured Lx, R500c aperture"),
                           ("lx_1mpc", "measured Lx, 1 Mpc"), ("lx_1p5mpc", "measured Lx, 1.5 Mpc"))),
        C("lx_n_sources_masked", "int", "listed point sources masked about the 1.5 Mpc "
          "aperture (-1: no source list or donor threshold, clipped background only)",
          min=-1, nullable=True, required=False),
        C("mgas_fof_msun", "float", "FoF gas mass", nullable=True),
        C("mstar_bcg_msun", "float", "BCG stellar mass", nullable=True),
        C("mbh_bcg_msun", "float", "BCG black-hole mass", nullable=True),
        C("bhar_bcg_msun_yr", "float", "BCG accretion rate", nullable=True),
        C("sfr_fof_msun_yr", "float", "FoF star-formation rate", nullable=True),
        C("veldisp_km_s", "float", "velocity dispersion", nullable=True),
        C("com_offset_kpc", "float", "centre-of-mass offset", nullable=True),
        C("nsubs", "int", "number of subhaloes", nullable=True),
        C("m500c_hse_msun", "float", "intrinsic hydrostatic M500c; NaN exactly when "
          "hse_crossing is 'none' (the headline M_HSE(<r) never falls through 500 rho_c "
          "beyond 0.4 R500c) or 'fit-failed' (the fitted headline's Vikhlinin fits did not "
          "converge)", nullable=True),
        C("hse_crossing", "str", "downward crossings of 500 rho_c by the headline M_HSE(<r) "
          "beyond 0.4 R500c: none | single | multiple (the headline takes the innermost) | "
          "fit-failed (fitted headline without converged fits; NaN mass)",
          allowed=("none", "single", "multiple", "fit-failed"), required=False),
        C("t_ew_kev", "float", "emission-weighted T within R500c"),
        C("fov_fit", "bool", "largest aperture fits the field (flag only, never a cut)"),
        C("snap_is_mini", "bool", "the snapshot is a TNG mini snapshot (no per-cell GFM_Metals, "
          "GFM_CoolingRate: X_H 0.76, solar ratios, no heated-cell cut)", required=False),
        C("held_out_resolution", "bool", "TNG100-1 / TNG50-1 view: a resolution held out of "
          "training whatever its split", required=False),
        C("seed", "int", "per-view master seed"),
        C("seed_offset", "int", "campaign seed offset"),
        C("status", "str", "ok | ok (cached) | error: ... | excluded: <reason> (a planned view of an excluded donor, not generated)"),
        *_PROVENANCE,
    ),
    legacy_renames=(
        ("M500c", "m500c_msun"), ("R500c_kpc", "r500c_kpc"), ("M200c", "m200c_msun"),

FIGURE_SIDECAR_KEYS: dict[str, str] = {
    "figure_id": "figNN as in figures/paper_figures.yml",
    "script": "figures/figNN_<content>.py",
    "output": "relative path of the rendered file",
    "inputs": "list of {path (relative to data/), sha256}",
    "package_version": "tngxmm version",
    "git_hash": "commit",
    "data_release": "data-package tag, e.g. v0.1-data",
    "created_utc": "ISO-8601 UTC",
    "cites": "citation keys of the plotted quantities",
}




[docs]
def validate_figure_sidecar(d: dict) -> list[str]:
    """Problems in a figure sidecar JSON object (empty list = valid)."""
    errs = [f"missing key {k!r}" for k in FIGURE_SIDECAR_KEYS if k not in d]
    for inp in d.get("inputs", []) or []:
        if not isinstance(inp, dict) or {"path", "sha256"} - set(inp):
            errs.append(f"bad input entry {inp!r} (need path, sha256)")
        elif str(inp["path"]).startswith("/"):
            errs.append(f"input path must be relative: {inp['path']}")
    return errs
