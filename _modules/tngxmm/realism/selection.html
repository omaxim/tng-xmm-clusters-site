Source code for tngxmm.realism.selection

"""The stored views a realism evaluation uses: which donors, which replicas, which are on disk.

The replica targets (``donors.targets``: one row per view with its donor and its replica index k = 1 .. K) say which
views a campaign was asked to make; the view trees say which exist.  :func:`select_views` keeps the views of the
chosen donor set (all donors, or the donors of one role of the frozen split), of the first ``replicas_per_donor``
replicas, that are present on disk, and refuses a view of a mini snapshot.
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

from tngxmm.cites import collect_cites
from tngxmm.realism import split as S

#: columns of the table of selected views
COLUMNS = ("view_id", "donor_id", "role", "sim", "snap", "halo_id", "view", "replica_k", "z_obs", "snap_is_mini",
           "view_dir")

def select_views(targets: Any, split: dict[str, Any], views_root: str | Path, *, donor_set: str = "all",
                 donors: list[str] | None = None, snaps: list[int] | None = None,
                 replicas_per_donor: int = 5, min_replicas: int = 3,
                 exclude_donors: dict[str, str] | None = None) -> Any:
    """The selected views, columns :data:`COLUMNS`, sorted by donor and replica index.

    ``targets``: the replica targets; ``split``: the donor split report (:mod:`tngxmm.realism.split`);
    ``views_root``: the view tree (``<root>/<sim>/<snap>/<halo>/view<v>``); ``donor_set``: ``all`` or a role of the
    split; ``donors``: restrict to these obsids; ``snaps``: restrict to these snapshots; ``replicas_per_donor``: only
    the replicas k <= this; ``min_replicas``: donors with fewer views on disk are dropped.  A view whose
    ``obs_meta.json`` says ``snap_is_mini`` raises ``ValueError``: the evaluation uses full snapshots only.
    ``exclude_donors`` (:func:`tngxmm.core.pool.excluded_donors`): the views of these donors were never generated
    (the donor has no PPS products) and cannot enter the evaluation, whatever the split says.  A view of a
    halo-snapshot without a physics donor (``paired`` False, release 1.2) has no real observation to be compared with
    and is never selected."""
    import pandas as pd

    from tngxmm.core.pool import split_excluded, view_dir
    from tngxmm.donors.pair_projections import drop_unpaired

    S.check_split(split)
    if donor_set != "all" and donor_set not in S.ROLES:
        raise ValueError(f"donor_set must be 'all' or one of {S.ROLES}, got {donor_set!r}")
    role_of = {r["obsid"]: r["role"] for r in split["rows"]}
    targets, _ = drop_unpaired(targets)
    t = targets.assign(donor_id=targets["donor_id"].astype(str).str.zfill(10))
    t, _ = split_excluded(t, exclude_donors)
    t = t[t["donor_id"].isin(role_of)]
    if donor_set != "all":
        t = t[t["donor_id"].map(role_of) == donor_set]
    if donors:
        t = t[t["donor_id"].isin([str(d).zfill(10) for d in donors])]
    if snaps:
        t = t[t["snap"].isin([int(s) for s in snaps])]
    t = t[t["replica_k"] <= int(replicas_per_donor)]
    rows = []
    for r in t.itertuples(index=False):
        vd = view_dir(views_root, r.sim, int(r.snap), int(r.halo_id), int(r.view))
        meta_file = vd / "obs_meta.json"
        if not meta_file.exists() or not (vd / "percam").is_dir():
            continue
        mini = bool(json.loads(meta_file.read_text()).get("snap_is_mini"))
        if mini:
            raise ValueError(f"view {r.view_id} is a mini-snapshot view: the evaluation uses full snapshots only")
        rows.append({"view_id": r.view_id, "donor_id": r.donor_id, "role": role_of[r.donor_id], "sim": r.sim,
                     "snap": int(r.snap), "halo_id": int(r.halo_id), "view": int(r.view),
                     "replica_k": int(r.replica_k), "z_obs": float(r.z_obs), "snap_is_mini": mini,
                     "view_dir": str(vd)})
    out = pd.DataFrame(rows, columns=list(COLUMNS))
    n = out.groupby("donor_id")["view_id"].transform("size") if len(out) else out["view_id"]
    out = out[n >= int(min_replicas)] if len(out) else out
    return out.sort_values(["donor_id", "replica_k"]).reset_index(drop=True)
