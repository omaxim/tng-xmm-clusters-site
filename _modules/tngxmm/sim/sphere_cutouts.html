Source code for tngxmm.sim.sphere_cutouts

"""Periodic sphere cutouts of a box simulation (TNG300-1), and the HaloSource that reads them.

Used by the maximal campaign (not on the production path).  TNG-Cluster's truth columns read the halo's
ORIGINAL ZOOM (``selection='zoom'``: ``loadOriginalZoom``), its *_tngref
columns too; a periodic box has no zoom, and its snapshot is not spatially
indexed (TNG300-1: 600 chunks, 4.2 TB per full snapshot; particles in FoF
order, the outer fuzz in slabs), so a sphere about one halo can only be cut
by scanning every chunk.  One scan per snapshot serves every halo of that
snapshot:

:func:`build_sphere_cutouts` reads ``Coordinates`` of every chunk (gas, DM,
stars, BH), keeps the particles within ``R_cut`` of each target halo's
``GroupPos`` with the PERIODIC distance, reads the other fields only for the
row range kept, and writes one file per halo in the fixture cutout layout
(``tngxmm-fixture-1``, ``tests/fixtures/README.md``) with ``InFoF`` = member
of THIS halo's FoF group (from the group's particle offsets).  Coordinates
are stored as the minimum image about ``GroupPos`` (``x - BoxSize *
round((x - c) / BoxSize)``; unchanged, bit for bit, for every particle that
needs no wrap), so the non-periodic arithmetic downstream (the mock's clip
sphere, the photon projection, the cylinders) is correct for halos that
straddle the box edge.

``R_cut`` (physical) = ``margin * max(2 R200c, 3.15 R500c, hypot(max(R500c,
1.5 Mpc), R200c))``: the mock's FoF clip sphere (2 R200c), the HSE shells
(to 3 R500c, read to 1.05 x that) and the *_tngref cylinders
(:func:`tngxmm.sim.generate_view.reference_clip_kpc`), margin 1.02.  The
particle mass profile of the truth columns beyond ``R_cut`` is therefore
truncated (it is used at r <= 2 R500c only: M500 at z_obs and the literature
estimators).

:class:`CutoutDirectorySource` is the HaloSource of a cutout directory
(``tngxmm_cutouts.json`` marker): the group catalogue and headers come from
the simulation output (exactly as :class:`SnapshotHaloSource`, so halo
selection and catalogue columns are those of the snapshot), the particles of
every selection from the halo's cutout: ``fof`` = ``InFoF == 1`` (inside
``R_cut``), ``zoom`` and ``spatial`` = every particle of the sphere (the box
analogue of "all gas of the original zoom"; the TNG official X-ray maps do not
exist for TNG300, so the *_tngref columns of box views are temet-exact on the
sphere gas, never compared with an official map).

References
----------
illustristng : TNG300-1 snapshots and group catalogues.
illustris_python : group-catalogue reader and particle offsets layout.
h5py : the chunk reads and the cutout files.
"""

def cut_radius_kpc(r500_kpc: float, r200_kpc: float, margin: float = CUT_MARGIN) -> float:
    """Physical cut radius [kpc] (module docstring)."""
    return margin * max(MOCK_CLIP_R200 * r200_kpc, HSE_RMAX_R500 * r500_kpc,
                        math.hypot(max(r500_kpc, 1000.0, 1500.0), r200_kpc))

def minimum_image(x: Any, centre: Any, box: float) -> Any:
    """Coordinates as the periodic image nearest ``centre``; rows needing no wrap are
    returned unchanged bit for bit."""
    import numpy as np

    x = np.asarray(x)
    k = np.round((x - np.asarray(centre)) / box)
    if not k.any():
        return x
    return np.where(k != 0, x - k * box, x).astype(x.dtype, copy=False)

def periodic_select(x: Any, centres: Any, radii: Any, box: float) -> list[Any]:
    """For each sphere (centre, radius) the sorted row indices of ``x`` inside it
    (periodic distance).  Cell-sorted for many spheres, direct for a few."""
    import numpy as np

    x = np.asarray(x, dtype=np.float64)
    centres = np.atleast_2d(np.asarray(centres, dtype=np.float64))
    radii = np.atleast_1d(np.asarray(radii, dtype=np.float64))

    def exact(rows, c, r):
        d = x[rows] - c
        d -= box * np.round(d / box)
        return rows[(d * d).sum(1) <= r * r]

    if len(centres) <= 4 or len(x) < 100_000:
        allrows = np.arange(len(x))
        return [exact(allrows, c, r) for c, r in zip(centres, radii)]
    g = max(1, int(box // max(float(radii.max()), 1e-9)))
    cell = np.floor(x / box * g).astype(np.int64) % g
    key = (cell[:, 0] * g + cell[:, 1]) * g + cell[:, 2]
    order = np.argsort(key, kind="stable")
    ks = key[order]
    out = []
    for c, r in zip(centres, radii):
        c0 = np.floor(c / box * g).astype(np.int64) % g
        span = range(-1, 2) if g >= 3 else range(g)
        cells = {(((c0[0] + i) % g) * g + (c0[1] + j) % g) * g + (c0[2] + k) % g
                 for i in span for j in span for k in span} if g >= 3 else set(range(g ** 3))
        parts = [order[np.searchsorted(ks, q, "left"):np.searchsorted(ks, q, "right")] for q in sorted(cells)]
        rows = np.sort(np.concatenate(parts)) if parts else np.array([], np.int64)
        out.append(exact(rows, c, r))
    return out




def _chunk_path(base: str, snap: int, chunk: int) -> str:
    return os.path.join(base, f"snapdir_{snap:03d}", f"snap_{snap:03d}.{chunk}.hdf5")


def _scan_chunk(task: dict) -> dict:
    """One snapshot chunk: per halo and particle type, the rows inside its sphere
    (global index + fields), written as ``<tmp>/<halo>/pt<t>_c<chunk>.npz``."""
    import h5py
    import numpy as np

    c, tmp = task["chunk"], Path(task["tmp"])
    hal = task["halos"]
    nsel = 0
    with h5py.File(task["path"], "r") as f:
        for pt, fields in FIELDS.items():
            name = f"PartType{pt}"
            if name not in f or "Coordinates" not in f[name] or f[name]["Coordinates"].shape[0] == 0:
                continue
            x = f[name]["Coordinates"][()]
            sel = periodic_select(x, [h["centre"] for h in hal], [h["r_ckpc_h"] for h in hal], task["box"])
            if not any(len(s) for s in sel):
                continue
            lo = min(int(s[0]) for s in sel if len(s))
            hi = max(int(s[-1]) for s in sel if len(s)) + 1
            have = [k for k in fields if k in f[name]]
            block = {k: (x[lo:hi] if k == "Coordinates" else f[name][k][lo:hi]) for k in have}
            for h, s in zip(hal, sel):
                if not len(s):
                    continue
                d = tmp / str(h["halo_id"])
                d.mkdir(parents=True, exist_ok=True)
                arr = {k: v[s - lo] for k, v in block.items()}
                arr["Coordinates"] = minimum_image(arr["Coordinates"], h["centre"], task["box"])
                arr["_gidx"] = task["chunk_start"][pt] + s.astype(np.int64)
                np.savez(d / f"pt{pt}_c{c:04d}.npz", **arr)
                nsel += len(s)
            del x, block
    return {"chunk": c, "n_selected": nsel}


def _write_cutout(path: Path, h: dict, parts_dir: Path, header: dict, grp: dict, sub: dict,
                  sub_id: int, meta: dict) -> dict:
    import glob

    import h5py
    import numpy as np

    counts = {}
    with h5py.File(path, "w") as f:
        f.attrs.update({"format_version": "tngxmm-fixture-1", "simulation": meta["simulation"],
                        "source_base_path": meta["base_path"], "halo_id": int(h["halo_id"]),
                        "snap": int(meta["snap"]), "zoom_target_id": -1,
                        "cutout_radius_ckpc_h": float(h["r_ckpc_h"]),
                        "cutout_radius_kpc": float(h["r_kpc"]),
                        "cutout_centre_ckpc_h": np.asarray(h["centre"], float),
                        "particle_selection": "every particle with periodic distance to GroupPos <= "
                                              "cutout_radius_ckpc_h (all FoF groups and fuzz); "
                                              "Coordinates as the minimum image about GroupPos; "
                                              "InFoF = member of this halo's FoF group",
                        "builder": "tngxmm.sim.sphere_cutouts", "created_utc": meta["created_utc"]})
        hd = f.create_group("Header")
        for k, v in header.items():
            hd.attrs[k] = v
        g = f.create_group("Group")
        for k, v in grp.items():
            g.create_dataset(k, data=v)
        s = f.create_group("Subhalo")
        s.attrs["subhalo_id"] = int(sub_id)
        for k, v in sub.items():
            s.create_dataset(k, data=v)
        npt = np.zeros(6, np.int64)
        for pt, fields in FIELDS.items():
            files = sorted(glob.glob(str(parts_dir / f"pt{pt}_c*.npz")))
            pg = f.create_group(f"PartType{pt}")
            if not files:
                continue
            data: dict[str, list] = {}
            for fn in files:
                with np.load(fn) as z:
                    for k in z.files:
                        data.setdefault(k, []).append(z[k])
            cat = {k: np.concatenate(v) for k, v in data.items()}
            gidx = cat.pop("_gidx")
            off, ln = int(h["offset"][pt]), int(h["length"][pt])
            for k, v in cat.items():
                pg.create_dataset(k, data=v, compression="gzip", compression_opts=1, shuffle=True)
            pg.create_dataset("InFoF", data=((gidx >= off) & (gidx < off + ln)).astype(np.uint8))
            npt[pt] = len(gidx)
            counts[pt] = (int(len(gidx)), int(((gidx >= off) & (gidx < off + ln)).sum()), ln)
        hd.attrs["NumPart_ThisFile"] = npt
        hd.attrs["NumFilesPerSnapshot"] = 1
    return counts

def build_sphere_cutouts(base_path: str, snap: int, halo_ids: Sequence[int], out_dir: str | Path, *,
                         simulation: str = "TNG300-1", workers: int = 1, margin: float = CUT_MARGIN,
                         chunks: Sequence[int] | None = None, keep_parts: bool = False,
                         echo: Any = print) -> dict:
    """Cut the spheres of ``halo_ids`` (snapshot ``snap``) out of ``base_path`` into
    ``<out_dir>/snap_<snap:03d>/<snap:03d>_<halo>.hdf5`` and update the directory
    marker.  ``chunks`` restricts the scan (tests only: an incomplete scan is
    recorded in the marker and the files are flagged ``complete = False``)."""
    import multiprocessing as mp
    import shutil

    import h5py
    import numpy as np

    from tngxmm.sim.read_halo_source import import_illustris_python

    il = import_illustris_python()
    t0 = time.time()
    out = Path(out_dir)
    sdir = out / f"snap_{snap:03d}"
    tmp = sdir / "_parts"
    sdir.mkdir(parents=True, exist_ok=True)
    gc = il.groupcat.loadHeader(base_path, snap)
    hsm, a, box = float(gc["HubbleParam"]), float(gc["Time"]), float(gc["BoxSize"])
    offs_path = os.path.join(os.path.dirname(base_path.rstrip("/")), "postprocessing", "offsets",
                             f"offsets_{snap:03d}.hdf5")
    with h5py.File(offs_path, "r") as f:
        chunk_start = f["FileOffsets/SnapByType"][()]
        grp_off = f["Group/SnapByType"][()]
    with h5py.File(_chunk_path(base_path, snap, 0), "r") as f:
        header = dict(f["Header"].attrs)
    n_chunks = int(header["NumFilesPerSnapshot"])
    header["NumPart_Total_Snapshot"] = header.get("NumPart_Total")
    header["NumPart_Total_HighWord_Snapshot"] = header.get("NumPart_Total_HighWord")
    hal, rows = [], {}
    for hid in halo_ids:
        grp = il.groupcat.loadSingle(base_path, snap, haloID=int(hid))
        r500 = float(grp["Group_R_Crit500"]) * a / hsm
        r200 = float(grp["Group_R_Crit200"]) * a / hsm
        rk = cut_radius_kpc(r500, r200, margin)
        hal.append(dict(halo_id=int(hid), centre=np.asarray(grp["GroupPos"], float), r_kpc=rk,
                        r_ckpc_h=rk * hsm / a, offset=grp_off[int(hid)].astype(np.int64),
                        length=np.asarray(grp["GroupLenType"], np.int64)))
        rows[int(hid)] = grp
    todo = list(range(n_chunks)) if chunks is None else [int(c) for c in chunks]
    tasks = [dict(chunk=c, path=_chunk_path(base_path, snap, c), halos=hal, box=box, tmp=str(tmp),
                  chunk_start=chunk_start[c].astype(np.int64)) for c in todo]
    echo(f"sphere cutouts: snap {snap}, {len(hal)} halos, {len(tasks)} chunks, {workers} workers")
    done = 0
    if workers > 1:
        with mp.get_context("fork").Pool(workers) as pool:
            for _ in pool.imap_unordered(_scan_chunk, tasks):
                done += 1
                if done % 50 == 0:
                    echo(f"  {done}/{len(tasks)} chunks, {time.time() - t0:.0f} s")
    else:
        for t in tasks:
            _scan_chunk(t)
            done += 1
            if done % 50 == 0:
                echo(f"  {done}/{len(tasks)} chunks, {time.time() - t0:.0f} s")
    created = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    meta = dict(simulation=simulation, base_path=base_path, snap=snap, created_utc=created)
    report = {}
    for h in hal:
        grp = rows[h["halo_id"]]
        sid = int(grp["GroupFirstSub"])
        sub = il.groupcat.loadSingle(base_path, snap, subhaloID=sid) if sid >= 0 else {}
        p = sdir / f"{snap:03d}_{h['halo_id']}.hdf5"
        cnt = _write_cutout(p, h, tmp / str(h["halo_id"]), header, grp, sub, sid, meta)
        with h5py.File(p, "a") as f:
            f.attrs["complete"] = chunks is None
        fof_ok = all(c[1] == c[2] for c in cnt.values())
        report[h["halo_id"]] = dict(file=str(p.relative_to(out)), r_cut_kpc=h["r_kpc"],
                                    particles={int(k): v[0] for k, v in cnt.items()},
                                    fof_members_inside={int(k): v[1] for k, v in cnt.items()},
                                    fof_members_total={int(k): v[2] for k, v in cnt.items()},
                                    whole_fof_inside=bool(fof_ok), complete=chunks is None,
                                    bytes=p.stat().st_size)
    if not keep_parts:
        shutil.rmtree(tmp, ignore_errors=True)
    mk = out / MARKER
    m = json.loads(mk.read_text()) if mk.exists() else {"format": FORMAT, "simulation": simulation,
                                                          "base_path": base_path, "files": {}}
    if m.get("base_path") != base_path:
        raise ValueError(f"{mk}: base_path {m.get('base_path')!r} != {base_path!r}")
    m["files"].setdefault(str(snap), {}).update({str(k): v["file"] for k, v in report.items()})
    m.setdefault("builds", []).append({"snap": snap, "halos": len(report), "chunks_scanned": len(tasks),
                                       "complete": chunks is None, "margin": margin,
                                       "seconds": round(time.time() - t0, 1), "created_utc": created})
    mk.write_text(json.dumps(m, indent=1) + "\n")
    echo(f"  done in {time.time() - t0:.0f} s")
    return report

def is_cutout_directory(path: str | Path) -> bool:
    """True when ``path`` is a directory holding the ``tngxmm_cutouts.json`` marker of a set of sphere cutouts."""
    return Path(path).is_dir() and (Path(path) / MARKER).is_file()

class CutoutDirectorySource(SnapshotHaloSource):
    """Group catalogue from the simulation output, particles from the sphere cutouts."""

    kind = "snapshot"

    def __init__(self, cutout_dir: str | Path, snap: int, halo_id: int | None = None) -> None:
        """``cutout_dir``: a directory with the cutout marker; its ``base_path`` entry names the simulation output, from
        which the group catalogue is read; ``snap``, ``halo_id`` as :class:`SnapshotHaloSource`."""
        self.cutout_dir = Path(cutout_dir)
        self.marker = json.loads((self.cutout_dir / MARKER).read_text())
        super().__init__(self.marker["base_path"], snap, halo_id)
        self._cut: dict[int, CutoutHaloSource] = {}

    def cutout(self, halo_id: int | None = None) -> CutoutHaloSource:
        """The :class:`CutoutHaloSource` of the halo's sphere file; ``FileNotFoundError`` when the marker lists none for
        this snapshot and halo."""
        hid = self._hid(halo_id)
        if hid not in self._cut:
            rel = self.marker["files"].get(str(self.snap), {}).get(str(hid))
            if rel is None:
                raise FileNotFoundError(f"{self.cutout_dir}: no sphere cutout of snap {self.snap} "
                                        f"halo {hid} (tngxmm sim sphere-cutouts)")
            self._cut[hid] = CutoutHaloSource(self.cutout_dir / rel)
        return self._cut[hid]

    def particles(self, ptype: int, fields: Sequence[str], selection: str = "fof",
                  radius_ckpc_h: float | None = None, halo_id: int | None = None) -> dict[str, Any]:
        """The particles of the halo's sphere cutout (the group catalogue is the simulation's, the particles the
        cutout's)."""
        return self.cutout(halo_id).particles(ptype, fields, selection, radius_ckpc_h)

def open_cutout_directory(path: str | Path, snap: int | None, halo_id: int | None = None
                          ) -> CutoutDirectorySource:
    """Open a cutout directory as a source; ``snap`` is required."""
    if snap is None:
        raise ValueError("snap is required for a cutout directory")
    return CutoutDirectorySource(path, int(snap), halo_id)
