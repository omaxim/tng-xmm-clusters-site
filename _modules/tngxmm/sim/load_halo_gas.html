Source code for tngxmm.sim.load_halo_gas

"""Load gas cells, apply the X-ray gas selection and compute per-cell emission quantities.

Configuration is injected as keyword arguments whose defaults are the CampaignSpec
values (``sim.t_min_k``, ``sim.sphere_r200c_factor``, ``sim.apec_*``, ``constants.*``).

Gas selection (the standard literature criterion): a cell emits iff
T > 3e5 K, its net cooling rate is negative (``GFM_CoolingRate < 0``) and it
is not star forming (SFR == 0), as Chatzigiannakis et al. (2025).  The
optional ``tvir_frac > 0`` raises the floor to ``tvir_frac * T_vir``
(core-excised variant; off by default).

Which cells are read (``selection``, CampaignSpec ``sim.loader``): ``'fof'`` the
FoF members (``loadHalo``), ``'zoom'`` the halo's original zoom run (``loadOriginalZoom``,
TNG-Cluster only), ``'spatial'`` every cell within ``radius_ckpc_h``, ``'cutout'`` the gas
of the halo's stored spherical truth cutout (:mod:`tngxmm.sim.cutout_gas`).  All but
``'cutout'`` are then clipped to ``sphere_r200c_factor * R200c`` about ``GroupPos``; the
cutout cells are the sphere and are clipped again only when ``clip_radius_kpc`` is given.  The clip
itself is not periodic; with ``sim.periodic_wrap`` the campaign
source has already moved every coordinate to its minimum image about GroupPos
(:class:`tngxmm.sim.read_halo_source.CampaignHaloSource`), so a halo at a box
face keeps the cells across the face.

Reference luminosity (``lx_reference``, off by default): ``'temet-exact'``
also returns, for EVERY loaded cell (no temperature, star-formation or
heating cut), the luminosity exactly as temet computes it for the TNG
official X-ray products (:mod:`tngxmm.sim.temet_xray`: temet's table and
line deposition, X_H = 0.76, star-forming gas at 1000 K, rest-frame band
divided by (1+z)).  The other arrays are unchanged.  The official maps
use all gas of the original zoom: with ``selection='zoom'`` the cell set is
the same (inside the clip sphere); with ``'fof'`` the outer-zoom gas is
missing (1-2 % less in 1-1.5 Mpc cylinders on the z = 0.5 fixture halo).

References
----------
IllustrisTNG (gas fields and units); Chatzigiannakis et al. (2025) gas
selection; APEC table lookup as in temet (see ``compute_apec_spectra``).
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Sequence

from tngxmm.cites import cites, collect_cites
from tngxmm.config import Z_AG89_TABLE_UNIT

_COL_H = 0
_COL_HE = 1
_COL_C = 2
_COL_FE = 8
_COL_OTH = 9

X_H_PRIMORDIAL = 0.76
Z_SUN = 0.0127                   # constants.Z_SUN (TNG GFM solar; descriptive only)
KPC_TO_CM = 3.0857e21
T_MIN_K = 3e5                    # sim.t_min_k
SPHERE_R200C_FACTOR = 2.0        # sim.sphere_r200c_factor
#: ``lx_reference`` values: no reference luminosity, or temet's
LX_REFERENCES = ("none", "temet-exact")
#: ``sim.snapshot_fields`` values: every field of the snapshot, or the field set of a TNG mini snapshot
SNAPSHOT_FIELDS = (None, "mini")

FIELDS_FULL = ('Coordinates', 'Masses', 'Density', 'InternalEnergy',
               'ElectronAbundance', 'StarFormationRate', 'GFM_CoolingRate',
               'GFM_Metals', 'GFM_Metallicity')
FIELDS_NO_MET = ('Coordinates', 'Masses', 'Density', 'InternalEnergy',
                 'ElectronAbundance', 'StarFormationRate', 'GFM_CoolingRate',
                 'GFM_Metallicity')
FIELDS_MINIMAL = ('Coordinates', 'Masses', 'Density', 'InternalEnergy',
                  'ElectronAbundance', 'StarFormationRate',
                  'GFM_Metallicity')

def gas_optional(source: Any, halo_id: Any, fields: Sequence[str],
                 optional: Sequence[str] = ('GFM_Metals', 'GFM_CoolingRate'),
                 **kwargs: Any) -> dict:
    """``source.gas`` that tolerates the mini snapshots: request ``fields``; if a
    field of ``optional`` is absent from the snapshot (TNG mini snapshots carry
    neither ``GFM_Metals`` nor ``GFM_CoolingRate``), drop the optional fields in
    order and retry.  Absent fields are absent keys of the result (the
    :func:`load_halo_gas` fallbacks: X_H primordial, no heated cells)."""
    fields = list(fields)
    for drop in [()] + [tuple(optional[:k]) for k in range(1, len(optional) + 1)]:
        try:
            return source.gas(halo_id, [f for f in fields if f not in drop], **kwargs)
        except Exception:
            if len(drop) == len(optional):
                raise
    raise RuntimeError('unreachable')

@dataclass
class GasParticles:
    """Selected gas cells of one halo in physical units.

    p_kpc (N, 3) float64 positions [physical kpc]; T_K (N,) float32 [K];
    em_cm3 (N,) float64 emission measure n_e n_H V [cm^-3]; Z_solar (N,)
    float32 GFM_Metallicity / 0.0127; Z_elements (N, 7) float32 C, N, O, Ne,
    Mg, Si, Fe in solar units of the abundance table;
    lum_erg_s (N,) float64 rest-frame band luminosity [erg/s]; h_kpc (N,)
    float32 cell effective radius [kpc].
    With ``lx_reference='temet-exact'`` only: p_all_kpc (M, 3) float64 and
    lum_tngref_erg_s (M,) float64 for all M loaded cells, the temet-exact
    luminosity (temet convention: rest-frame band / (1+z)); otherwise None.
    With ``selection='cutout'`` only: ``cutout_info``, the report of the cutout read
    (:meth:`~tngxmm.sim.cutout_gas.CutoutGasSource.cutout_report`: sphere radius, cells
    counted as low-resolution and kept, nearest low-resolution dark matter); otherwise None.
    """

    p_kpc: Any
    T_K: Any
    em_cm3: Any
    Z_solar: Any
    Z_elements: Any
    lum_erg_s: Any
    h_kpc: Any
    p_all_kpc: Any = None
    lum_tngref_erg_s: Any = None
    cutout_info: Any = None
    components: Any = None
    spread: Any = None
    ne_count: Any = None          # (N,) electrons per cell, n_e V; the Compton-y weight with T (None: not given)

def emission_masks(T_K: Any, sfr: Any, cooling_rate: Any, t_cut: float) -> tuple:
    """The mock's emission selection: ``(keep, cold, star_forming, heated)``.

    A cell is EXCLUDED if it is cold (``T_K < t_cut``), star forming
    (``sfr > 0``) or heated (``cooling_rate >= 0``); every other cell is kept.

    Sign convention of TNG's ``GFM_CoolingRate`` (the net cooling rate, du/dt of
    radiative cooling and heating): NEGATIVE when the cell loses energy (net
    cooling), POSITIVE when it gains energy (net heating, its temperature
    increasing).  The mock keeps NET-COOLING cells (``< 0``) and drops heated
    ones (``>= 0``).  Barnes et al. (2021, sec. 2.2.3) drop cells whose "net
    cooling rate is positive (i.e. it is increasing in temperature)": the same
    heating criterion.  ``cooling_rate=None`` (field absent): no cell is heated.
    Used by :func:`load_halo_gas` and by the truth estimators that reuse the
    mock's selection (``compute_hse_mass``)."""
    import numpy as np

    sf = sfr > 0.0
    heat = (cooling_rate >= 0.0) if cooling_rate is not None else np.zeros(len(sf), dtype=bool)
    cold = T_K < t_cut
    return ~(cold | sf | heat), cold, sf, heat

@cites("illustristng")
def t_vir_from_catalog(M500c_msun: float, R500c_kpc: float) -> float:
    """Virial temperature [K]: T_vir = mu m_p G M500c / (2 k_B R500c), mu = 0.59."""
    from astropy.constants import G as G_const
    from astropy.constants import k_B, m_p
    _mu    = 0.59                                    # fully ionised primordial plasma
    M_g    = M500c_msun * 1.989e33                   # M_sun -> g
    R_cm   = R500c_kpc  * 3.0856776e21               # kpc   -> cm
    return (_mu * m_p.cgs.value * G_const.cgs.value * M_g
            / (2.0 * k_B.cgs.value * R_cm))

@cites("illustristng", "chatzigiannakis2025", "temet")
def load_halo_gas(
    source: Any,
    halo_id: int,
    grp: dict | None,
    h: float,
    a: float,
    UM: float,
    UL: float,
    UV: float,
    emin_kev: float,
    emax_kev: float,
    verbose: bool = True,
    obsframe: bool = False,
    tvir_frac: float = 0.0,
    t_vir_K: float = 0.0,
    *,
    selection: str = "fof",
    radius_ckpc_h: float | None = None,
    t_min_k: float = T_MIN_K,
    sphere_r200c_factor: float = SPHERE_R200C_FACTOR,
    z_sun: float = Z_SUN,
    kpc_to_cm: float = KPC_TO_CM,
    z_solar_ag89: float = Z_AG89_TABLE_UNIT,
    apec_table: str | None = None,
    clip_radius_kpc: float | None = None,
    abund_table: str = "lodd",
    apec_version: str = "3.1.3",
    lx_reference: str = "none",
    temet_table: str | None = None,
    abundances: str | None = None,
    untracked: str = "other-solar",
    box_half_r500: float | None = None,
    descriptor: str | None = None,
    photon_spread: str | None = None,
    cell_resample_msun: float | None = None,
    resample_key: tuple = (),
    snapshot_fields: str | None = None,
) -> GasParticles:
    """Load gas cells through a HaloSource and compute the emission arrays.

    ``source.gas(halo_id, fields, selection)`` reads the cells.  Temperature
    cut: ``max(tvir_frac * t_vir_K, t_min_k)`` if ``tvir_frac > 0 and t_vir_K >
    0``, else ``t_min_k``.

    ``clip_radius_kpc`` is the radius of the sphere about GroupPos the cells are
    clipped to, instead of ``sphere_r200c_factor * R200c`` (for
    ``selection='zoom'`` with a line-of-sight depth beyond 2 R200c);
    ``abund_table`` gives the solar mass fractions of ``Z_elements`` (must be
    the table of the photon spectra); ``apec_version`` selects the reference
    table built from that APEC version; ``z_solar_ag89`` is the metal unit of
    the table's metallicity axis (``sim.table_metal_unit``).

    ``cell_resample_msun`` (``sim.cell_resample_msun``, a diagnostic arm; None = off): the cells are degraded to
    that particle mass before the emission (:func:`tngxmm.sim.emission.resample_cells`), with a Generator seeded
    by ``resample_key`` (integers, e.g. snapshot and halo); the record goes to ``cutout_info['resample']``.

    ``snapshot_fields`` (``sim.snapshot_fields``): None = every field the snapshot carries; ``'mini'`` = the cells
    reduced to the fields of a TNG mini snapshot at every snapshot (:func:`tngxmm.ingest.loaders.mini_fields_fn`:
    X_H = 0.76 in the temperature and the emission measure, no net-heating cut), before the resampling and the
    emission.
    """
    import numpy as np

    if lx_reference not in LX_REFERENCES:
        raise ValueError(f"lx_reference must be one of {LX_REFERENCES}, not {lx_reference!r}")

    canonical = selection == "cutout" and hasattr(source, "gas_cells")
    if canonical:
        # a canonical cube (tngxmm.ingest): the cells as stored, no native read
        return _load_canonical(source, halo_id, emin_kev, emax_kev, verbose=verbose, obsframe=obsframe,
                               tvir_frac=tvir_frac, t_vir_K=t_vir_K, t_min_k=t_min_k, z_sun=z_sun,
                               kpc_to_cm=kpc_to_cm, z_solar_ag89=z_solar_ag89, apec_table=apec_table,
                               abund_table=abund_table, apec_version=apec_version,
                               lx_reference=lx_reference, clip_radius_kpc=clip_radius_kpc,
                               abundances=abundances, untracked=untracked,
                               box_half_r500=box_half_r500, photon_spread=photon_spread,
                               cell_resample_msun=cell_resample_msun, resample_key=resample_key,
                               snapshot_fields=snapshot_fields, descriptor=descriptor)
    if snapshot_fields not in SNAPSHOT_FIELDS:
        raise ValueError(f"sim.snapshot_fields must be one of {SNAPSHOT_FIELDS}, not {snapshot_fields!r}")

    def _load(fields):
        # the cutout is the stored sphere: its radius is the clip below, not a read radius
        return source.gas(halo_id, list(fields), selection=selection,
                          radius_ckpc_h=None if selection == "cutout" else radius_ckpc_h)

    try:
        gas = _load(FIELDS_FULL)
        has_metals = True
    except Exception:
        try:
            gas = _load(FIELDS_NO_MET)
            has_metals = False
        except Exception:
            gas = _load(FIELDS_MINIMAL)
            has_metals = False

    n_par   = len(gas['Masses'])

    # Clip to sphere_r200c_factor x R200c about GroupPos (every loader but 'cutout', whose
    # cells are the stored sphere; it is clipped only to clip_radius_kpc, a uniform sphere of
    # sim.gas_radius_r500 R500c, which must lie within the stored sphere).
    cutout_info = None
    if selection == "cutout":
        cutout_info = source.cutout_report(halo_id)
        if cutout_info is None:
            raise ValueError("selection 'cutout' needs a source with cutout_report "
                             "(tngxmm.sim.cutout_gas.CutoutGasSource)")
        cutout_info = dict(cutout_info)
        if clip_radius_kpc is not None:
            r_cut = float(cutout_info["r_cut_kpc"])
            if float(clip_radius_kpc) > r_cut * (1.0 + 1e-9):
                raise ValueError(
                    f"halo {halo_id}: sim.gas_radius_r500 = "
                    f"{float(clip_radius_kpc) / float(cutout_info['r500c_kpc']):.3f} R500c "
                    f"({float(clip_radius_kpc):.0f} kpc) exceeds the stored sphere (r_cut = "
                    f"{r_cut:.0f} kpc = {r_cut / float(cutout_info['r500c_kpc']):.3f} R500c)")
            cutout_info["clip_radius_kpc"] = float(clip_radius_kpc)
    if grp is not None and (selection != "cutout" or clip_radius_kpc is not None):
        _r200c_code = grp.get('Group_R_Crit200', None)
        if _r200c_code is not None or selection == "cutout":
            _centre_kpc = grp['GroupPos'] * (a / h)
            _sphere_fac = float(sphere_r200c_factor)
            _p_raw      = gas['Coordinates'] * (a / h)
            _dist       = np.linalg.norm(_p_raw - _centre_kpc, axis=1)
            if clip_radius_kpc is None:
                _r200c_kpc  = float(_r200c_code) * (a / h)
                _sphere = _dist <= _sphere_fac * _r200c_kpc
                _label = f'{_sphere_fac}xR200c={_sphere_fac*_r200c_kpc:.0f} kpc'
            else:
                _sphere = _dist <= float(clip_radius_kpc)
                _label = f'{float(clip_radius_kpc):.0f} kpc'
            if verbose:
                print(f'    {selection} sphere cut ({_label}): '
                      f'{n_par:,} -> {_sphere.sum():,} particles')
            gas = {k: (v[_sphere] if isinstance(v, np.ndarray) and v.ndim > 0 else v)
                   for k, v in gas.items()}
            n_par = len(gas['Masses'])
            if cutout_info is not None:
                cutout_info["n_clipped"] = int(_sphere.size - _sphere.sum())


    if verbose:
        flag = '' if has_metals else '  [GFM_Metals absent - solar-ratio fallback]'
        print(f'    {n_par:,} gas particles loaded{flag}')

    # the canonical cells (physical units, the simulation's loader in ingest) and their emission (sim.emission): the
    # arithmetic of the legacy loader, operation for operation
    from tngxmm.ingest.loaders import gas_fn
    from tngxmm.sim.emission import emission_from_cells

    from tngxmm.ingest.descriptor import load as load_descriptor

    desc = None if descriptor is None else load_descriptor(str(descriptor))
    cells = gas_fn(desc)(gas, h, a, UM, UL, UV)
    if snapshot_fields == "mini":
        from tngxmm.ingest.loaders import mini_fields_fn

        cells = mini_fields_fn(desc)(cells)
    resample = None
    if cell_resample_msun is not None:
        if lx_reference != "none":
            raise ValueError("sim.cell_resample_msun needs lx_reference 'none' (the reference reads the full gas)")
        cells, resample = _resample(cells, cell_resample_msun, resample_key, verbose)
    z_snap = 1.0 / a - 1.0
    em_out, keep = emission_from_cells(
        cells, z_snap=z_snap, emin_kev=emin_kev, emax_kev=emax_kev, t_min_k=t_min_k,
        tvir_frac=tvir_frac, t_vir_K=t_vir_K, abundances=abundances, untracked=untracked,
        z_sun=z_sun, kpc_to_cm=kpc_to_cm, z_solar_ag89=z_solar_ag89, apec_table=apec_table,
        abund_table=abund_table, apec_version=apec_version, obsframe=obsframe, verbose=verbose,
        photon_spread=photon_spread)
    lum_erg_s = em_out.pop("_lum_all")
    em_out.pop("_n")
    p_kpc = cells.pos_kpc

    if cutout_info is not None and cutout_info.get("n_coarse"):
        # the coarse (low-resolution) cells are kept; their share of the 0.5-2 keV emission
        from tngxmm.sim.cutout_gas import HEAVY_CELL_FACTOR

        # the input masses (code units), or the resampled cells' (Msun): only the ratio to the median is used
        _m = np.asarray(gas['Masses'] if resample is None else cells.mass_msun, np.float64)
        _coarse = _m > HEAVY_CELL_FACTOR * float(np.median(_m))
        _tot = float(lum_erg_s[keep].sum())
        cutout_info["coarse_lum_frac"] = (float(lum_erg_s[keep & _coarse].sum() / _tot)
                                          if _tot > 0 else float('nan'))
        cutout_info["n_coarse_emitting"] = int((keep & _coarse).sum())

    if keep.sum() == 0:
        raise ValueError(
            f'halo {halo_id}: no hot gas particles after cuts '
            f'(T_cut={max(tvir_frac * t_vir_K, t_min_k) if tvir_frac > 0.0 and t_vir_K > 0.0 else t_min_k:.2e} K, '
            f'n_total={n_par})'
        )

    p_all = lum_ref = None
    if lx_reference == "temet-exact":
        from tngxmm.sim import temet_xray

        lum_ref = temet_xray.temet_luminosity(gas, h, z_snap, UL, UM, UV,
                                              f'{emin_kev:.1f}-{emax_kev:.1f}', path=temet_table)
        p_all = p_kpc.astype(np.float64)

    if resample is not None:
        cutout_info = dict(cutout_info or {}, resample=resample)
    return GasParticles(**em_out, p_all_kpc=p_all, lum_tngref_erg_s=lum_ref,
                        cutout_info=cutout_info)




#: first word of the seed of the cell resampling (sim.cell_resample_msun), followed by the resample key
RESAMPLE_SEED_TAG = 20261028


def _resample(cells: Any, mass_msun: float, key: tuple, verbose: bool) -> tuple:
    """:func:`tngxmm.sim.emission.resample_cells` with the Generator of ``(RESAMPLE_SEED_TAG, *key)``."""
    import numpy as np

    from tngxmm.sim.emission import resample_cells

    out, rec = resample_cells(cells, float(mass_msun),
                              np.random.default_rng([RESAMPLE_SEED_TAG, *[int(k) for k in key]]))
    rec["key"] = [int(k) for k in key]
    if verbose:
        print(f"    cell resampling to {float(mass_msun):.3g} Msun: {rec['n_in']:,} -> {rec['n_kept']:,} cells")
    return out, rec


def _load_canonical(source: Any, halo_id: int, emin_kev: float, emax_kev: float, *, verbose: bool,
                    obsframe: bool, tvir_frac: float, t_vir_K: float, t_min_k: float, z_sun: float,
                    kpc_to_cm: float, z_solar_ag89: float, apec_table: str | None, abund_table: str,
                    apec_version: str, lx_reference: str, clip_radius_kpc: float | None,
                    abundances: str | None, untracked: str,
                    box_half_r500: float | None, photon_spread: str | None = None,
                    cell_resample_msun: float | None = None, resample_key: tuple = (),
                    snapshot_fields: str | None = None, descriptor: str | None = None) -> GasParticles:
    """The mock gas of a canonical cube (see :func:`load_halo_gas`).  A cube larger than the
    preset's (``box_half_r500``) is clipped to it (per-axis, physical kpc about the centre); a
    smaller one is refused."""
    import numpy as np

    from tngxmm.sim.cutout_gas import HEAVY_CELL_FACTOR
    from tngxmm.sim.emission import emission_from_cells

    if clip_radius_kpc is not None:
        raise ValueError("a canonical cube is a cube: sim.gas_radius_r500 does not apply")
    if lx_reference != "none":
        raise ValueError("lx_reference on the mock's own cells is not served by a canonical cube; "
                         "the temet reference reads the native FoF gas (sim.lx_reference_gas)")
    if snapshot_fields not in SNAPSHOT_FIELDS:
        raise ValueError(f"sim.snapshot_fields must be one of {SNAPSHOT_FIELDS}, not {snapshot_fields!r}")
    cells = source.gas_cells(halo_id)
    if snapshot_fields == "mini":
        from tngxmm.ingest.descriptor import load as load_descriptor
        from tngxmm.ingest.loaders import mini_fields_fn

        cells = mini_fields_fn(None if descriptor is None else load_descriptor(str(descriptor)))(cells)
    info = dict(source.cutout_report(halo_id) or {})
    at = source.cube(halo_id)["attrs"]
    if box_half_r500 is not None and abs(float(at["half_side_r500c"]) - float(box_half_r500)) > 1e-9:
        if float(at["half_side_r500c"]) < float(box_half_r500):
            raise ValueError(f"halo {halo_id}: the canonical cube has half-side {at['half_side_r500c']} R500c, "
                             f"less than sim.mock_box_r500 = {box_half_r500}")
        half = float(box_half_r500) * float(at["r500c_kpc"])
        d = cells.pos_kpc - np.asarray(at["centre_kpc"], np.float64)
        cells = cells.select(np.all(np.abs(d) <= half, axis=1))
        info.update(half_side_kpc=half, half_side_r500c=float(box_half_r500),
                    n_cube=int(len(cells.mass_msun)), cube_half_side_r500c=float(at["half_side_r500c"]))
    if cell_resample_msun is not None:
        cells, info["resample"] = _resample(cells, cell_resample_msun, resample_key, verbose)
    a = float(source.units()["a"])
    em_out, keep = emission_from_cells(
        cells, z_snap=1.0 / a - 1.0, emin_kev=emin_kev, emax_kev=emax_kev, t_min_k=t_min_k,
        tvir_frac=tvir_frac, t_vir_K=t_vir_K, abundances=abundances, untracked=untracked, z_sun=z_sun,
        kpc_to_cm=kpc_to_cm, z_solar_ag89=z_solar_ag89, apec_table=apec_table, abund_table=abund_table,
        apec_version=apec_version, obsframe=obsframe, verbose=verbose, photon_spread=photon_spread)
    lum_erg_s = em_out.pop("_lum_all")
    em_out.pop("_n")
    if info.get("n_coarse"):
        _m = np.asarray(cells.mass_msun, np.float64)
        _coarse = _m > HEAVY_CELL_FACTOR * float(np.median(_m))
        _tot = float(lum_erg_s[keep].sum())
        info["coarse_lum_frac"] = float(lum_erg_s[keep & _coarse].sum() / _tot) if _tot > 0 else float('nan')
        info["n_coarse_emitting"] = int((keep & _coarse).sum())
    if keep.sum() == 0:
        raise ValueError(f'halo {halo_id}: no hot gas particles after cuts (n_total={len(cells.mass_msun)})')
    return GasParticles(**em_out, cutout_info=info)

@cites("illustristng", "tngcluster", "temet")
def load_reference_cells(
    source: Any,
    halo_id: int,
    grp: dict,
    h: float,
    a: float,
    UM: float,
    UL: float,
    UV: float,
    emin_kev: float,
    emax_kev: float,
    *,
    selection: str = "zoom",
    clip_radius_kpc: float,
    temet_table: str | None = None,
    map_clip_radius_kpc: float | None = None,
    map_bands: Sequence[str] = (),
    verbose: bool = False,
) -> ReferenceCells:
    """Cells and temet-exact luminosities for the ``*_tngref`` columns and, with
    ``map_clip_radius_kpc``, for the per-view reference maps, from ONE snapshot read
    (``sim.lx_reference_gas`` 'fof' or 'zoom', read SEPARATELY from the mock's
    gas, without the mock's temperature, star-formation and cooling cuts).

    ``selection='fof'`` reads every gas cell of the halo's FoF group (``loadHalo``):
    the cell set used in every simulation.

    ``selection='zoom'`` reads every gas cell of the halo's original zoom
    (``loadOriginalZoom``: all its FoF groups and its outer fuzz), the cell set of
    the TNG official X-ray maps (temet ``sphMap_globalZoomOrig``).  Only the six
    fields temet needs plus the coordinates are read.

    ``*_tngref`` cells: within ``clip_radius_kpc`` (physical) of GroupPos (the
    distance is not periodic, as :func:`load_halo_gas`; the campaign source wraps
    the coordinates at load with ``sim.periodic_wrap``), luminosity in ``emin_kev-emax_kev``
    from :func:`tngxmm.sim.temet_xray.temet_luminosity` on exactly those cells, so
    these arrays do not depend on the map options.  Map cells: within
    ``map_clip_radius_kpc`` of GroupPos, one temet-exact luminosity per band of
    ``map_bands`` (e.g. ``'0.5-5.0'``) and the cell radius from Masses / Density.
    temet's zero-luminosity floor (the smallest positive value of the set) is
    taken per cell set, as temet does for the set it is given.

    Cost: one extra snapshot read per halo, of the WHOLE zoom region (TNG-Cluster:
    1-7e7 gas cells, 20-30 s; about 6 GB peak at z = 1 for the snap-50 test halos,
    against about 1e6 FoF cells for the mock).  All luminosities are temet's:
    rest-frame band divided by (1+z).
    """
    import numpy as np

    from tngxmm.sim import temet_xray

    gas = source.gas(halo_id, ["Coordinates", *temet_xray.FIELDS], selection=selection)
    p_kpc = np.asarray(gas["Coordinates"], dtype=np.float64) * (a / h)
    dist = np.linalg.norm(p_kpc - np.asarray(grp["GroupPos"]) * (a / h), axis=1)
    keep = dist <= float(clip_radius_kpc)
    sub = {k: np.asarray(gas[k])[keep] for k in temet_xray.FIELDS}
    msub = mkeep = None
    if map_clip_radius_kpc is not None:
        mkeep = dist <= float(map_clip_radius_kpc)
        msub = {k: np.asarray(gas[k])[mkeep] for k in temet_xray.FIELDS}
    del gas, dist
    if verbose:
        print(f'    tngref gas ({selection}, {float(clip_radius_kpc):.0f} kpc): '
              f'{len(keep):,} -> {int(keep.sum()):,} cells')
    z_snap = 1.0 / a - 1.0
    lum = temet_xray.temet_luminosity(sub, h, z_snap, UL, UM, UV,
                                      f'{emin_kev:.1f}-{emax_kev:.1f}', path=temet_table)
    out = ReferenceCells(p_kpc=p_kpc[keep], lum=lum)
    del sub
    if msub is not None:
        if verbose:
            print(f'    tngref map gas ({float(map_clip_radius_kpc):.0f} kpc): '
                  f'{int(mkeep.sum()):,} cells, bands {list(map_bands)}')
        out.map_p_kpc = p_kpc[mkeep]
        vol = (msub["Masses"] / msub["Density"]).astype(np.float64) * (a / h) ** 3
        out.map_cellrad_kpc = (3.0 * vol / (4.0 * np.pi)) ** (1.0 / 3.0)
        out.map_lum = {str(b): temet_xray.temet_luminosity(msub, h, z_snap, UL, UM, UV, str(b),
                                                           path=temet_table)
                       for b in map_bands}
    return out

def load_reference_luminosity(
    source: Any,
    halo_id: int,
    grp: dict,
    h: float,
    a: float,
    UM: float,
    UL: float,
    UV: float,

    *,

    return rc.p_kpc, rc.lum
