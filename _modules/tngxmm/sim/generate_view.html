"""Generate the mock views of one halo: seeding, gas, photons, absorption, hand-off.

The per-view generation and seeding of the campaign (the process pool, the
gas-load semaphore and the job layer belong to the orchestration stage).  For
each view, in this order:

1. seed the view's streams (:class:`tngxmm.core.rng.ViewStreams`: the global
   numpy stream, then the view Generator); supplement views not listed in
   ``z_obs_fixed`` are skipped BEFORE seeding; cached views make no draw;
2. load the gas once per halo (``load_halo_gas`` through a HaloSource);
3. intrinsic rest-frame 0.5-2 keV luminosities in cylinders (R500c, 1, 1.5 Mpc);
4. z_obs: fixed, from the targets table (``z_obs_fixed``);
5. donor observing conditions (a replica view takes its donor from ``rec['donor_packages']``; otherwise
   ``deps.draw_donor`` when a caller binds one, stream ``donor``; ``default_dependencies`` binds none) and the roll angle (``posangle``);
6. photons for the longest camera time (``generate_photons``), then Galactic
   absorption (``apply_absorption``); only the absorbed list is kept;
7. the pointing-dither cap;
8. the instrument stage and the measured photometry, through ``deps``
   (owned by the instrument and photometry stages; skipped when not given);
9. ``obs_meta.json``.

Cross-stage callables are injected through :class:`ViewDependencies` so this
stage never imports another stage's internals; :func:`default_dependencies`
binds the package implementations.

References
----------
TNG-Cluster (Nelson et al. 2024); the emission and photon models cited by the
called modules.
"""

def campaign_ccf_dir(spec: Any = None) -> str:

def instrument_stage(spec: Any) -> Callable[..., Any]:
    """``run_instrument`` for :class:`ViewDependencies`: the call with ``ccf_dir=``
    mapped onto ``run_xmm_simulator(options=...)``, with the instrument options
    of ``spec``."""
    from tngxmm.instrument.run_xmm_simulator import run_xmm_simulator
    from tngxmm.instrument.simulate_instrument import InstrumentOptions

    core_table: list = []          # donor core-offset table, read once on first use

    def run_instrument(*, ccf_dir: str | Path, view_info: dict | None = None, **kw: Any) -> Any:
        opt = InstrumentOptions.from_spec(spec, ccf_dir)
        if view_info is not None:
            kw.update(instrument_view_kwargs(spec, opt, view_info, core_table))
        return run_xmm_simulator(options=opt, **kw)

    return run_instrument




def _view_id(spec: Any, info: dict) -> str | None:
    """The view's canonical ``view_id`` (header ``VIEW_ID``): the one the replica
    row carries (``info['view_id']`` or the donor package's), else built from
    the preset name and (snap, halo, view, rep)."""
    pkg = info.get("donor_package") or {}
    for vid in (info.get("view_id"), pkg.get("view_id")):
        if isinstance(vid, str) and vid:
            return vid
    if spec is None:
        return None
    from tngxmm.schemas.ids import make_view_id

    rep = info.get("rep")
    rep = int(rep) if isinstance(rep, (int, float)) and rep == rep else 0
    return make_view_id(spec.name, int(info["snap"]), int(info["halo_id"]), int(info["view"]), rep)

def view_soft_protons(spec: Any, opt: Any, obsid: str, table: Any = None) -> dict | None:
    """The residual soft protons of the observation ``obsid`` (the view's conditions donor) under the instrument
    options ``opt``: camera -> ``{'sp_rate', 'source', 'substitute', 'distance'}``
    (:func:`tngxmm.instrument.donor_inout_sp.donor_soft_protons`), None under ``soft_protons = 'off'``.  ``table``:
    the IN/OUT table when the caller has read it (else the one of ``opt``, the data package's by default).

    A camera without a measured rate follows ``opt.sp_rate_unmeasured``: ``'refuse'`` raises; under
    ``'nearest-in-fold'`` the rate of the nearest measured donor of the same donor fold is taken
    (:func:`tngxmm.instrument.donor_inout_sp.nearest_measured_in_fold`), looked up in the condition table of the
    preset's conditions rule (the data package's ``donor_conditions.csv`` unless the rule names another) with the
    rule's donor folds (else the repository's); the condition table is read only when a camera needs a substitute.
    The instrument stage writes the result into the camera headers (SPRATE, SPRFLAG, SPSUBST) and the generation into
    ``obs_meta.json`` and the conditions record (:func:`package_soft_protons`)."""
    if opt.soft_protons == "off":
        return None
    from tngxmm.instrument import donor_inout_sp as SPM
    from tngxmm.instrument import donor_tables as DT

    key = str(obsid).zfill(10)
    if table is None:
        table = DT.read_table("mos_background", opt.mos_background_table)

    def substitute(camera: str) -> dict:
        from tngxmm.instrument.conditions_donor import conditions_of, read_condition_table, read_donor_folds

        cond = conditions_of(spec) if spec is not None else {"rule": "donor"}
        ctab = read_condition_table(str(DT.table_path("conditions", cond.get("table") or None)))
        return SPM.nearest_measured_in_fold(ctab, read_donor_folds(cond.get("folds") or None), table, key, camera)

    return SPM.donor_soft_protons(table, key, opt.sp_rate_unmeasured,
                                  substitute if opt.sp_rate_unmeasured == "nearest-in-fold" else None)

def instrument_view_kwargs(spec: Any, opt: Any, info: dict, core_table: list) -> dict:
    """Per-view keywords of the instrument stage that depend on the donor (the
    same calls as the instrument Kedro node): the mock obsid, the donor's
    filters, PN mode, per-camera submodes (SUBMODE headers, readout windows),
    the header context, the donor core offset and the donor's active CCDs.

    ``info``: ``snap``, ``halo_id``, ``view``, ``donor_package``, ``seed``,
    ``z_obs``, ``nh_1e22`` and optionally ``exposure_scale`` (the conditions donor's exposure jitter,
    default 1: it scales the live times of the donor background parameters)."""
    from tngxmm.instrument.simulate_instrument import (donor_core_offset, donor_filters,
                                                       donor_submodes, view_context)
    from tngxmm.instrument.write_camera_products import mock_obsid

    pkg = info.get("donor_package") or {}
    snap, halo, view = int(info["snap"]), int(info["halo_id"]), int(info["view"])
    out: dict[str, Any] = {}
    out["mockid"] = mock_obsid(snap, halo, view, opt.obsid_style)
    out["filters"] = donor_filters(pkg)
    out["pn_mode"] = pkg.get("pn_submode")
    out["camera_submodes"] = donor_submodes(pkg)
    out["context"] = view_context(spec, snap=snap, halo_id=halo, view=view,
                                  donor_package=pkg, view_id=_view_id(spec, info),
                                  seed=info.get("seed"),
                                  z_obs=info.get("z_obs"), nh_1e22=info.get("nh_1e22"),
                                  mockobs=out.get("mockid"))
    from tngxmm.instrument.sas_headers import package_sas_donor

        from tngxmm.instrument.donor_corner_qpb import donor_unit_factors

        out["qpb_unit_factors"] = donor_unit_factors(DT.read_table("qpb_corners", opt.mos_background_table),

        from tngxmm.instrument.donor_corner_qpb import donor_corner_counts

    if getattr(opt, "soft_proton_map", "flat") == "flare" and opt.soft_protons != "off":
        # the donor's theta500: the flare maps are normalised over the region of its IN/OUT measurement, which left
        # out max(1.5 R500, 3') about its cluster

               conditions_obsid=str(run_pkg.get('obsid') or '').zfill(10),
               physics_nh_1e22=nh, physics_posangle_deg=posangle)
    return out

@dataclass
class SimOptions:
    """The CampaignSpec values this stage reads (the defaults are those of the
    ``v1`` preset).

    ``emax_phot_kev`` is the rest-frame upper photon energy, ``apec_version``
    the AtomDB version of the APEC tables, ``abund_table`` the abundance table
    of the spectra, the solar mass fractions and the TBabs absorption,
    ``table_metal_unit`` the solar metallicity of the gas table,
    ``t_sampling`` (only ``interpolate`` is implemented) the temperature sampling of
    the spectra, ``gas_selection`` (``fof`` / ``zoom`` / ``spatial``) the gas
    particles loaded, with ``gas_radius_r500`` (radius of the loaded and
    clipped gas sphere; ``None``: the ``sphere_r200c_factor`` R200c clip),
    ``truth_columns`` (intrinsic HSE masses per halo and M500 at rho_c(z_obs)
    per view), ``truth_profiles`` ('on': the per-halo ``truth_profile.h5``;
    needs ``truth_columns``).  ``los_depth_r500``: None (default) gives the
    emitting cylinder the half-depth R200c and the radius ``sphere_r200c_factor``
    R200c; a value (loader 'zoom' or 'spatial' with ``gas_radius_r500``) gives it
    the half-depth ``los_depth_r500`` R500c and the radius of the gas sphere, so
    the photons carry the projected outskirts and the gas around the halo.
    ``gas_selection = 'cutout'``: the gas of the halo's stored spherical truth cutout
    (:mod:`tngxmm.sim.cutout_gas`; ``cutout_root`` the tree holding ``truth_cutouts``,
    None = the output tree); the emitting cylinder is bounded by that sphere
    (half-depth and radius ``r_cut``, or the half-depth ``los_depth_r500`` R500c <= r_cut,
    :func:`photon_switches`); with ``gas_radius_r500`` the gas is clipped to the uniform
    sphere of that many R500c (<= r_cut per halo, refused otherwise) and the emitting
    cylinder has that radius and the half-depth ``los_depth_r500`` (default
    ``gas_radius_r500``); the truth columns and profiles of the generation (the halo's FoF
    members, read through the wrapped source) are replaced by the truth-sphere stage as with
    the other loaders.  ``mock_cutout_shape = 'box'`` (loader 'cutout'): ALL the gas of
    the axis-aligned cube of half-side ``mock_box_r500`` R500c (default 4) about GroupPos,
    from ``<mock_cutout_root or cutout_root or output tree>/mock_cutouts``
    (:mod:`tngxmm.sim.mock_cutout`); the cube is the emitting volume, so
    ``gas_radius_r500`` and ``los_depth_r500`` must be unset (no clip, no cylinder
    shorter than the cube).
    """

    seed_offset: int = 1_000_000_000
    rng_campaign: str = "v4"            # campaign name keying the streams

    photon_chunk: int = 4_000_000
    gas_selection: str = "fof"
    gas_radius_r500: float | None = None
    los_depth_r500: float | None = None
    cutout_root: str | None = None
    mock_cutout_shape: str | None = None
    mock_box_r500: float | None = None
    mock_gas_region: str | None = None
    mock_cutout_root: str | None = None
    mock_cylinder_r500: float | None = None
    projection: str | None = None
    apec_version: str = "3.1.3"
    abund_table: str = "lodd"
    table_metal_unit: float = Z_AG89_TABLE_UNIT
    t_sampling: str = "interpolate"
    truth_columns: bool = True
    truth_profiles: str = "on"
    hse_pressure: Any = "thermal-fit-mu"
    hse_gas_phase_cut: Any = "nonsf"
    hse_shell_binning: Any = "linear:50:0:1.5"
    t_weighting: Any = "mass"
    hse_shell_density: str = "filling"
    hse_min_filling: float | None = 0.1
    lx_reference: str = "temet-exact"
    lx_reference_gas: str = "fof"
    periodic_wrap: bool = True
    truth_selection: str = "fof"
    truth_selection_hse: str = "sphere"
    truth_sphere_r200c: float = 3.0
    temet_apec_table: str | None = None

    tngref_map_bands: tuple = ("0.5-5.0", "0.5-2.0")
    tngref_map_npix: int = 2000

    abundances: str | None = None

    def __post_init__(self) -> None:
        from tngxmm.sim.project_tngref_map import TNGREF_MAP_MODES, parse_band

        from tngxmm.sim.generate_photons import T_SAMPLINGS
        from tngxmm.sim.truth_profile import MODES as TRUTH_PROFILE_MODES

        from tngxmm.sim.abundances import MODES as ABUNDANCE_MODES
        from tngxmm.sim.abundances import UNTRACKED_RULES

        if self.abundances not in (None, *ABUNDANCE_MODES):
            raise ValueError(f"sim.emission_abundances must be None or one of {ABUNDANCE_MODES}, "
                             f"not {self.abundances!r}")
        if self.untracked not in UNTRACKED_RULES:
            raise ValueError(f"sim.emission_untracked must be one of {UNTRACKED_RULES}, not {self.untracked!r}")
        from tngxmm.sim.load_halo_gas import SNAPSHOT_FIELDS

        if self.snapshot_fields not in SNAPSHOT_FIELDS:
            raise ValueError(f"sim.snapshot_fields must be one of {SNAPSHOT_FIELDS}, not {self.snapshot_fields!r}")
        if self.snapshot_fields == "mini" and self.abundances != "common":
            raise ValueError("sim.snapshot_fields 'mini' drops the element fractions; it needs "
                             "sim.emission_abundances 'common'")
        if self.cube_store not in ("index", "canonical"):
            raise ValueError(f"runtime.cube_store must be 'index' or 'canonical', not {self.cube_store!r}")
        if self.cube_store == "canonical" and not self.box_mock:
            raise ValueError("runtime.cube_store 'canonical' needs sim.mock_cutout_shape 'box'")
        if self.t_sampling not in T_SAMPLINGS:
            raise ValueError(f"sim.t_sampling must be one of {T_SAMPLINGS} (round_up is not implemented), "
                             f"not {self.t_sampling!r}")
        if not isinstance(self.rng_campaign, str) or not self.rng_campaign:
            raise ValueError(f"sim.rng_campaign must be a non-empty string, not {self.rng_campaign!r}")
        if self.mock_cutout_shape not in (None, "sphere", "box"):
            raise ValueError("sim.mock_cutout_shape must be None, 'sphere' or 'box', "
                             f"not {self.mock_cutout_shape!r}")
        if self.mock_cutout_shape == "box":
            if self.gas_selection != "cutout":
                raise ValueError("sim.mock_cutout_shape 'box' needs loader 'cutout'")
            if self.gas_radius_r500 is not None or self.los_depth_r500 is not None:
                raise ValueError("sim.mock_cutout_shape 'box': the cube is the emitting volume; "
                                 "sim.gas_radius_r500 and sim.los_depth_r500 must be unset (they "
                                 "would clip it)")
            if self.mock_box_r500 is not None and not float(self.mock_box_r500) > 0:
                raise ValueError("sim.mock_box_r500 must be positive")
            if self.mock_gas_region not in (None, *MOCK_GAS_REGIONS):
                raise ValueError(f"sim.mock_gas_region must be one of {MOCK_GAS_REGIONS}, "
                                 f"not {self.mock_gas_region!r}")
        elif (self.mock_box_r500 is not None or self.mock_cutout_root is not None
              or self.mock_gas_region is not None):
            raise ValueError("sim.mock_box_r500, sim.mock_gas_region and sim.mock_cutout_root "
                             "need sim.mock_cutout_shape 'box'")
        if self.mock_cylinder_r500 is not None:
            if self.mock_cutout_shape != "box" or not float(self.mock_cylinder_r500) > 0:
                raise ValueError("sim.mock_cylinder_r500 must be positive and needs sim.mock_cutout_shape 'box'")
            if float(self.mock_cylinder_r500) * math.sqrt(2.0) > self.box_half_r500 * (1.0 + 1e-6):
                raise ValueError(f"sim.mock_cylinder_r500 = {self.mock_cylinder_r500}: the cylinder does not fit in "
                                 f"the cube of half-side {self.box_half_r500} R500c along every line of sight "
                                 "(needs c sqrt(2) <= half-side)")
        if self.projection is not None:
            from tngxmm.sim.project_luminosity import PROJECTIONS

            if self.projection not in PROJECTIONS:
                raise ValueError(f"sim.projection must be one of {PROJECTIONS}, not {self.projection!r}")
            if self.projection != "axes":
                if self.mock_cutout_shape != "box" or self.mock_cylinder_r500 is None:
                    raise ValueError("sim.projection off the simulation axes needs mock_cutout_shape 'box' and "
                                     "sim.mock_cylinder_r500 (a cylinder held by the cube along any axis)")
                if self.tngref_map == "on":
                    raise ValueError("sim.projection off the simulation axes needs tngref_map 'off' (the official "
                                     "maps are axis-aligned)")
        if self.gas_selection == "cutout":
            if self.gas_radius_r500 is not None and not float(self.gas_radius_r500) > 0:
                raise ValueError("sim.gas_radius_r500 must be positive (and, per halo, at most "
                                 "the radius of the stored sphere)")
            if self.los_depth_r500 is not None and not float(self.los_depth_r500) > 0:
                raise ValueError("sim.los_depth_r500 must be positive (and, per halo, at most the "
                                 "radius of the stored sphere)")
            if (self.los_depth_r500 is not None and self.gas_radius_r500 is not None
                    and float(self.los_depth_r500) > float(self.gas_radius_r500)):
                raise ValueError("sim.los_depth_r500 must lie in (0, sim.gas_radius_r500]")
        elif self.cutout_root is not None:
            raise ValueError("sim.cutout_root needs loader 'cutout'")
        elif self.los_depth_r500 is not None:
            # a deeper emitting column needs the gas beyond the FoF / 2 R200c sphere
            if self.gas_selection == "fof" or self.gas_radius_r500 is None:
                raise ValueError("sim.los_depth_r500 needs loader 'zoom' or 'spatial' and "
                                 "sim.gas_radius_r500 (the loaded gas sphere)")
            if not 0 < float(self.los_depth_r500) <= float(self.gas_radius_r500):
                raise ValueError("sim.los_depth_r500 must lie in (0, sim.gas_radius_r500]")
        if self.truth_profiles not in TRUTH_PROFILE_MODES:
            raise ValueError(f"sim.truth_profiles must be one of {TRUTH_PROFILE_MODES}, "
                             f"not {self.truth_profiles!r}")
        if self.truth_profiles == "on" and not self.truth_columns:
            raise ValueError("sim.truth_profiles 'on' needs sim.truth_columns (the profiles "
                             "are binned in the same pass)")

def photon_switches(opts: SimOptions, r500c_kpc: float, r_cut_kpc: float | None = None,
                    box_half_kpc: float | None = None) -> dict:
    """The keyword arguments of ``generate_photons`` set by the options: the
    temperature sampling, the APEC version and the abundance table, and with
    ``los_depth_r500`` the emitting cylinder (half-depth ``los_depth_r500`` R500c,
    radius ``gas_radius_r500`` R500c; ``r500c_kpc`` physical).  Loader ``'cutout'``:
    the cylinder is bounded by the stored sphere of radius ``r_cut_kpc`` [physical kpc]:
    radius ``r_cut_kpc``, half-depth ``r_cut_kpc`` or ``los_depth_r500`` R500c (a column
    deeper than the sphere is refused).  A column shallower than the sphere holds the
    full depth only out to the projected radius sqrt(r_cut^2 - depth^2).  With
    ``gas_radius_r500`` (the gas clipped to that sphere) the cylinder radius is
    ``gas_radius_r500`` R500c and the half-depth ``los_depth_r500`` R500c (default
    ``gas_radius_r500``), both at most ``r_cut_kpc``.

    ``mock_cutout_shape = 'box'``: the stored cube of half-side ``box_half_kpc``
    [physical kpc] about GroupPos holds all the gas there is to select, so the photon
    filter alone bounds the emitting volume, along the view's axis (every view projects
    along a simulation axis).  ``mock_gas_region = 'cylinder'`` (production): the cylinder
    inscribed in the cube, radius and half-depth ``box_half_kpc``, so the projection is a
    disc with the uniform depth +-half-side at every projected radius inside it and the
    cube's corners are dropped; 'cube': the whole cube, i.e. the smallest cylinder that
    encloses it (half-depth ``box_half_kpc``, radius sqrt(2) x ``box_half_kpc``).  The
    depth carries the margin :data:`BOX_CYLINDER_MARGIN` against position round-off (a
    cell of the cube is never lost at its face); the radius of the inscribed cylinder does
    not, it is the stated region."""
    kw = {"t_sampling": opts.t_sampling, "apec_version": opts.apec_version,
          "abund_table": opts.abund_table}
    if opts.box_mock:
        if box_half_kpc is None:
            raise ValueError("mock_cutout_shape 'box': the half-side of the stored cube is needed")
        half = float(box_half_kpc)
        if opts.mock_cylinder_r500 is not None:
            # a cylinder set apart from the cube (sim.mock_cylinder_r500): radius and half-depth c R500c, inside the
            # cube along any line of sight when c sqrt(2) <= the half-side (checked against the stored cube here)
            c = float(opts.mock_cylinder_r500) * float(r500c_kpc)
            if c * math.sqrt(2.0) > half * (1.0 + 1e-6):
                raise ValueError(f"sim.mock_cylinder_r500 = {opts.mock_cylinder_r500} R500c: the cylinder reaches "
                                 f"{c * math.sqrt(2.0):.0f} kpc from the centre, beyond the stored cube's half-side "
                                 f"{half:.0f} kpc")
            kw.update(depth_kpc=c, cyl_radius_kpc=c)
        else:
            depth = half * (1.0 + BOX_CYLINDER_MARGIN)
            rad = half if opts.box_region == "cylinder" else depth * math.sqrt(2.0)
            kw.update(depth_kpc=depth, cyl_radius_kpc=rad)
    elif opts.gas_selection == "cutout":
        if r_cut_kpc is None:
            raise ValueError("loader 'cutout': the radius of the stored sphere is needed")
        rad = (float(r_cut_kpc) if opts.gas_radius_r500 is None
               else float(opts.gas_radius_r500) * float(r500c_kpc))
        if opts.los_depth_r500 is not None:
            depth = float(opts.los_depth_r500) * float(r500c_kpc)
        else:
            depth = rad
        for what, val, nom in (("gas_radius_r500", rad, opts.gas_radius_r500),
                               ("los_depth_r500", depth, opts.los_depth_r500)):
            if val > float(r_cut_kpc) * (1.0 + 1e-9):
                raise ValueError(f"sim.{what} = {nom} R500c ({val:.0f} kpc) exceeds the stored "
                                 f"sphere (r_cut = {float(r_cut_kpc):.0f} kpc)")
        kw.update(depth_kpc=depth, cyl_radius_kpc=rad)
    elif opts.los_depth_r500 is not None:
        kw.update(depth_kpc=float(opts.los_depth_r500) * float(r500c_kpc),
                  cyl_radius_kpc=float(opts.gas_radius_r500) * float(r500c_kpc))
    if opts.projection is not None:
        kw["projection"] = opts.projection
    return kw

def hse_truth_kwargs(opts: SimOptions, grp: dict, a: float, h: float,
                     r200c_kpc: float) -> dict:
    """Keywords of :func:`~tngxmm.sim.compute_hse_mass.compute_hse_truth` for one halo
    under ``opts``: the mock's selection, sphere radius, clip sphere and floor (and
    ``profile`` when ``sim.truth_profiles`` is on)."""
    radius = clip_kpc = None
    if opts.gas_radius_r500 is not None:
        radius = float(opts.gas_radius_r500) * float(grp['Group_R_Crit500'])
        clip_kpc = radius * a / h
    elif opts.box_mock:               # the mock emits from the whole cube: its cells reach the half-side
        clip_kpc = float(opts.box_half_r500) * float(grp['Group_R_Crit500']) * a / h
    kw = dict(mock_selection=opts.gas_selection, mock_radius_ckpc_h=radius,
              mock_clip_radius_kpc=(clip_kpc if clip_kpc is not None
                                    else opts.sphere_r200c_factor * float(r200c_kpc)),
              mock_t_min_k=opts.t_min_k, om0=opts.om0, selection=opts.truth_selection)
    if opts.truth_profiles == 'on':
        kw['profile'] = True
    kw['shell_density'] = opts.hse_shell_density
    kw['min_filling'] = opts.hse_min_filling
    # the fit's abundance table: solar units of the emission-weighted metallicity
    kw['abund_table'] = opts.abund_table
    return kw

def campaign_source(source: Any, opts: SimOptions, cutout_tree: str | Path | None = None,
                    sim: str | None = None) -> Any:
    """The source every read of the generation goes through
    (:class:`~tngxmm.sim.read_halo_source.CampaignHaloSource`): the gas-load slot of
    the campaign pool held for each RAW read only (mock gas, ``*_tngref`` cells,
    truth particles; the per-cell arithmetic and the HSE / profile computation run
    outside it), and with ``sim.periodic_wrap`` the coordinates moved to their
    minimum image about GroupPos.  Idempotent.  Loader ``'cutout'``: the source first
    serves the stored sphere's gas (:class:`~tngxmm.sim.cutout_gas.CutoutGasSource`), from
    ``sim.mock_cutout_root`` (box cubes) or ``sim.cutout_root`` (spheres), else ``cutout_tree`` (the output tree), simulation label ``sim``."""
    from tngxmm.core.pool import gas_load_slot
    from tngxmm.sim.read_halo_source import CampaignHaloSource

    if isinstance(source, CampaignHaloSource):
        return source
    if opts.gas_selection == 'cutout' and getattr(source, 'kind', '') != 'canonical':
        from tngxmm.sim.cutout_gas import CutoutGasSource

        tree = ((opts.mock_cutout_root if opts.box_mock else None) or opts.cutout_root
                or cutout_tree)
        if tree is None or sim is None:
            raise ValueError("loader 'cutout' needs sim.cutout_root or the output tree, "
                             "and the simulation label")
        source = CutoutGasSource(source, tree, sim, shape="box" if opts.box_mock else "sphere",
                                 box_r500=opts.box_half_r500 if opts.box_mock else None)
    return CampaignHaloSource(source, periodic_wrap=opts.periodic_wrap, slot=gas_load_slot)

def load_tngref_cells(source: Any, halo_id: int, grp: dict, sub: dict, meta: Any,
                      opts: SimOptions, *, verbose: bool = False) -> tuple:
    """The ``lx_reference_gas='zoom'`` read of one halo: ``(ref, ref_map)`` with
    ``ref = (p_kpc, lum)`` the ``*_tngref`` cells (clip :func:`reference_clip_kpc`)
    and ``ref_map`` the :class:`~tngxmm.sim.project_tngref_map.MapCells` of the
    per-view maps (``sim.tngref_map='on'``; else None), from ONE snapshot read.
    The read holds the gas-load slot through the campaign source."""
    import numpy as np

    from tngxmm.sim.load_halo_gas import load_reference_cells
    from tngxmm.sim.read_halo_source import cosmo_units

    h, a, _z, UL, UM, UV = cosmo_units(source)
    mclip = centre = None
    if opts.tngref_map == 'on':          # the map cells come from the same read
        from tngxmm.sim.project_tngref_map import map_clip_kpc

        spos = np.asarray(sub['SubhaloPos'], np.float64)
        if opts.periodic_wrap:           # the centre in the same image as the wrapped cells
            from tngxmm.sim.read_halo_source import minimum_image

            spos = minimum_image(spos[None, :], np.asarray(grp['GroupPos'], np.float64),
                                 float(source.header()['BoxSize']))[0]
        centre = spos * a / h
        mclip = map_clip_kpc(meta.R200c_kpc, float(np.linalg.norm(
            centre - np.asarray(grp['GroupPos'], np.float64) * a / h)))
    rc = load_reference_cells(source, halo_id, grp, h, a, UM, UL, UV, opts.emin_lx_kev,
                              opts.emax_lx_kev, selection=opts.lx_reference_gas,
                              clip_radius_kpc=reference_clip_kpc(meta),
                              temet_table=opts.temet_apec_table, map_clip_radius_kpc=mclip,
                              map_bands=opts.tngref_map_bands, verbose=verbose)
    ref_map = None
    if mclip is not None:
        from tngxmm.sim.project_tngref_map import prepare_map_cells

        ref_map = prepare_map_cells(rc.map_p_kpc, rc.map_cellrad_kpc, rc.map_lum, centre,
                                    meta.R200c_kpc, mclip, gas_selection=opts.lx_reference_gas,
                                    periodic_wrap=opts.periodic_wrap)
    return (rc.p_kpc, rc.lum), ref_map

def view_complete(view_dir: str | Path, stored: dict) -> bool:
    """True when the products of a view with the done-marker ``stored`` (its ``obs_meta.json``) are whole: the
    merged image reads to its end and every file of the marker's ``products`` is there with its size.  A view
    made before the marker listed its products is judged on the merged image alone."""
    view_dir = Path(view_dir)

    ``drop_events``, and optionally ``donor_packages`` (replica views), ``seed_offset``, ``no_dither_cap``, ``tvir_frac``.  ``stop_after='posangle'`` stops each view after the roll

    from tngxmm.sim.load_halo_gas import load_halo_gas, t_vir_from_catalog
    from tngxmm.sim.project_luminosity import compute_lx_cylinder
    from tngxmm.sim.read_halo_source import cosmo_units

    opts = opts or SimOptions()
    sim, snap, hid = rec['sim'], rec['snap'], rec['halo_id']
    source = campaign_source(source, opts, out_root, sim)    # slot per raw read; periodic wrap
    snap_info = snapshot_info(int(snap), opts.source)
    seed_offset = int(rec.get('seed_offset', opts.seed_offset))
    no_dither_cap = bool(rec.get('no_dither_cap', not opts.dither_cap))
    _COSMO = get_cosmology()

    rows = []
    gas = None            # loaded lazily, once, only if some view needs it
    halo_ctx: dict = {}
    disk_full = False     # a view of this halo was stopped by a full disk: the later ones are not started

    for view in views:
        zfix = rec.get('z_obs_fixed')
        if zfix is not None and view not in zfix:

        streams = ViewStreams(f"{rng_campaign}-s{snap:03d}-h{hid:07d}-v{view}",
                              campaign=rng_campaign, seed_offset=seed_offset, sim=sim)
        seed = streams.seed('photon_counts')

        out = {k: rec[k] for k in ('halo_id', 'snap', 'sim', 'z_snap', 'M500c',
                                   'R500c_kpc', 'M200c', 'R200c_kpc')}
        out['view'] = view
        out['snap_is_mini'] = snap_info['snap_is_mini']      # the snapshot kind (mini or full)
        out['snap_absent_fields'] = ','.join(snap_info['snap_absent_fields'])
        out.update({k: np.nan for k in MEASURE_COLS + EXTRA_COLS})
        from tngxmm.photometry.measure_lx import LX_FLAG_COLUMNS

        ref_cols = (TNGREF_COLS if opts.lx_reference != 'none' else ()) + LX_FLAG_COLUMNS
        out.update({k: np.nan for k in ref_cols})
        out['status'] = 'pending'

        view_dir = view_dir_of(out_root, sim, snap, hid, view)
        view_dir.mkdir(parents=True, exist_ok=True)
        merged_path = view_dir / 'image_merged.fits'
        meta_path   = view_dir / 'obs_meta.json'
        if merged_path.exists() and meta_path.exists():          # resumable
            try:
                with open(meta_path) as f:
                    stored = json.load(f)
                if zfix is not None and not (
                        stored.get('z_obs_source') in ('targets-csv', 'replica')
                        and int(stored.get('seed_offset', -1)) == seed_offset
                        and abs(float(stored.get('z_obs', -1.0)) - float(zfix[view])) < 1e-6):
                    raise ValueError('cached product does not match the targets request')
                if not cached_position_current(stored, opts):
                    raise ValueError('cached view made under another catalogue-position rule: regenerated')
                if deps.data_inputs is not None and stored.get('data_inputs') != deps.data_inputs():
                    raise ValueError('cached view made from other data-package or AtomDB files: regenerated')
                if not view_complete(view_dir, stored):
                    raise ValueError('cached view with a missing or cut product: regenerated')
                out.update({k: stored.get(k, np.nan)
                            for k in MEASURE_COLS + EXTRA_COLS + ref_cols})
                # the observation and seed columns the fresh path writes (observation_columns, the replica seed
                # columns of INTERFACES 4.1) are in obs_meta.json too; a resumed view keeps them

            pkg['nh_1e22'] = nh
            out['z_obs'], out['nh_1e22'] = z_obs, nh
            if halo_ctx.get('hse') is not None:
                out.update(view_truth_columns(halo_ctx['hse'], z_obs, opts))
            if halo_ctx.get('ref_map') is not None:    # no random draw: streams unchanged
                write_tngref_view_map(view_dir, halo_ctx, view, opts, pkg=pkg, snap=snap,
                                      halo_id=hid)

            # the observing conditions first (the view's own `conditions` and `posangle` streams): the photon
            # list is sized for the LONGEST camera time of the conditions the instrument will use (a conditions
            # donor's, jittered, under 'nn'), so per-camera thinning <= 1 and the camera partition holds
            posangle = float(streams.get('posangle').uniform(0, 360))
            ob = draw_observing_conditions(opts, pkg, nh, streams, rng_campaign, posangle)
            run_exposure, run_pa = ob['exposure_s'], ob['posangle_deg']
            texp_gen = ob['texp_gen_s']
            out.update(observation_columns(pkg, ob['package'], run_exposure, nh, ob['nh_1e22'], run_pa,
                                           posangle))
            from tngxmm.photometry.catalogue_positions import position_rule

            # how the catalogue entry of the physics donor measured its position: the view's own catalogue position
            # (before its measurement error) is made by that rule (sim.catalogue_centre); recorded in obs_meta.json
            catpos = position_rule(str(pkg.get('obsid') or pkg.get('donor_id')),
                                   smoothing_arcsec=opts.peak_smoothing_arcsec)
            evt = generate_photons(gas, center_kpc=group_pos_kpc,
                                   R200c_kpc=meta.R200c_kpc, z_obs=z_obs,
                                   view=view, out_dir=view_dir,
                                   exposure_s=texp_gen, area_cm2=opts.area_cm2,
                                   rng=streams, verbose=verbose,
                                   z_eff_min=opts.z_eff_min,
                                   sphere_r200c_factor=opts.sphere_r200c_factor,
                                   emin_phot_kev=opts.emin_phot_kev,
                                   emax_phot_kev=opts.emax_phot_kev,
                                   phot_nbins=opts.phot_nbins,
                                   emin_lx_kev=opts.emin_lx_kev, emax_lx_kev=opts.emax_lx_kev,
                                   chunk=opts.photon_chunk, h0=opts.h0, om0=opts.om0,
                                   position_rule=catpos, r500c_kpc=meta.R500c_kpc,
                                   **photon_switches(opts, meta.R500c_kpc,
                                                     (halo_ctx['cutout'] or {}).get('r_cut_kpc'),
                                                     (halo_ctx['cutout'] or {}).get('half_side_kpc')))
            catpos_rec = catalogue_position_record(catpos, evt)
            evt_abs = view_dir / f'events_v{view}_abs.h5'
            # absorbed with the column of the conditions the instrument will use (the conditions donor's under 'nn',
            # drawn above), so no photon has to be restored later: a retarget can only remove photons, and a lower
            # conditions column would leave the soft band short
            apply_absorption(evt, evt_abs, float(ob['nh_1e22']), rng=streams, abund_table=opts.abund_table)
            evt.unlink(missing_ok=True)      # keep only the absorbed file

            # pointing-dither cap: keep the largest aperture on-detector
            d_A_kpc  = _COSMO.angular_diameter_distance(z_obs).to('kpc').value
            theta_ap = max(meta.R500c_kpc, 1500.0) * 206265.0 / d_A_kpc / 60.0
            max_dith = (None if no_dither_cap
                        else max(0.0, deps.fov_fit_arcmin - theta_ap))
            out['max_dither_arcmin'] = np.nan if max_dith is None else max_dith
            out['_fingerprint_after_photons'] = streams.fingerprint()

            if stop_after == 'posangle':
                out['status'] = 'ok (photons only)'
                out['_donor_package'] = pkg
                out['_texp_gen_s'] = texp_gen
                out['_seed'] = seed
                rows.append(out)
                continue

            if deps.run_instrument is None:
                raise RuntimeError('no instrument stage given (deps.run_instrument)')
            # one keyed Generator per view for every instrument draw: the
            # instrument stage takes a numpy Generator; per-purpose streams (psf, rmf,
            # background, agn, oot per camera; core.rng.CAMERA_PURPOSES) are not threaded
            # through it yet
            inst_rng = streams.get('partition')
            # the soft-proton rates the instrument will use and where each comes from (a substitute for a camera
            # without a measured rate), recorded with the conditions (conditions.json) and in obs_meta.json
            sp_rec = deps.soft_protons(ob['package']) if deps.soft_protons is not None else None
            if sp_rec is not None and ob['draw'] is not None:
                ob['draw']['record']['soft_protons'] = sp_rec
            ob = stage_observing_conditions(ob, nh, evt_abs, view_dir, opts.abund_table)
            run_pkg, run_nh, run_evt, run_rec = ob['package'], ob['nh_1e22'], ob['evt_file'], ob['record']
            merged = deps.run_instrument(
                evt_file=run_evt, out_dir=view_dir,
                exposure_s=run_exposure,              # per-camera live times
                rng=inst_rng, ccf_dir=deps.ccf_dir, verbose=False,
                posangle_deg=run_pa, nh_background=run_nh,
                pn_oot_frac=run_pkg.get('pn_oot_frac'),   # donor's PN mode
                max_dither_arcmin=max_dith,
                per_camera_dir=(view_dir / 'percam') if opts.write_percam else None,
                mockid=f'{rec["halo_id"]:08d}'[-8:],
                view_info=dict(snap=snap, halo_id=hid, view=view, donor_package=run_pkg,
                               seed=seed, z_obs=z_obs, nh_1e22=run_nh,
                               exposure_scale=(run_rec or {}).get('exposure_scale', 1.0)))

            if rec.get('drop_events'):
                evt_abs.unlink(missing_ok=True)
                if run_evt != evt_abs:
                    Path(run_evt).unlink(missing_ok=True)

            expmap = None
            cov_path = view_dir / 'image_merged_cov.fits'
            if cov_path.exists():
                from astropy.io import fits as _fits

                with _fits.open(str(cov_path)) as hd:
                    expmap = hd[0].data.astype(np.float32)

            if deps.emsce_mean_exposure is not None and deps.campaign_lx is not None:
                t_eff = deps.emsce_mean_exposure(merged)   # ARF-weighted
                out['exposure_eff_s'] = t_eff
                # the truth-side Lx (true R500, true centre, true T_vir), photometry.lx_bkg 'masked-clipped': the
                # simulated point sources above the donor's threshold masked, NaN + lx_<ap>_flag where an aperture
                # is unmeasurable.  Not the mock-real pair, which is the fit targets' seed photometry (SAS lists)
                out.update(deps.campaign_lx(merged, meta.R500c_kpc, z_obs, t_eff, run_nh,
                                            halo_ctx['T_keV'], expmap))
            out['status'] = 'ok'

            fp = out.pop('_fingerprint_after_photons', None)
            prov = recorded_provenance(opts)
            # the done-marker: every file of the view is flushed first (a write error the kernel deferred stops the
            # view here), then the marker is written under a temporary name and renamed
            products = view_products(diskguard.sync_tree(view_dir))
            if not diskguard.fits_complete(merged_path):
                raise OSError(f'{merged_path} is missing or cut')
            diskguard.write_json(meta_path, {**{k: _jsonify(v) for k, v in out.items()},
                           'halo_catalog': halo_ctx['extra'],
                           'donor_package': pkg, 'exposure_s': run_exposure,
                           'catalogue_position': catpos_rec,
                           **({'data_inputs': data_rec} if data_rec is not None else {}),
                           **({'conditions': run_rec} if run_rec is not None else {}),
                           **({'soft_protons': sp_rec} if sp_rec is not None else {}),
                           'texp_gen_s': texp_gen, 'seed': seed,
                           'seed_offset': seed_offset,
                           'z_obs_source': ('replica' if rec.get('donor_packages') is not None
                                            else 'targets-csv'),
                           'dither_cap': not no_dither_cap,
                           'z_donor_obsid': ((rec.get('z_donor') or {}).get(view)),
                           'z_eff_min': (float(opts.z_eff_min) if opts.z_eff_min is not None
                                         else None),
                           'pipeline_version': 'v3.1',
                           'phot_norm_version': 3, 'psf_fix': True,
                           'sampler': 'per-particle', 'camera_partition': True,
                           'exposure_convention': 'arf-weighted-emsce',
                           **prov,
                           **({'gas_cutout': halo_ctx['cutout']}
                              if halo_ctx.get('cutout') is not None else {}),
                           'snap_is_mini': snap_info['snap_is_mini'],
                           'snap_absent_fields': snap_info['snap_absent_fields'],
                           'periodic_wrap': bool(opts.periodic_wrap),
                           'truth_selection': opts.truth_selection,
                           'n_wrapped_particles': {f'{k[0]}:{k[1]}': v for k, v in
                                                   source.wrapped.items()},
                           'products': products,
                           **membudget.meta_fields()})
            out['_fingerprint_after_photons'] = fp
            if rec.get('donor_packages') is not None:
                # replica rows carry the view-schema seed columns (INTERFACES 4.1)
                out.update(seed=int(seed), seed_offset=int(seed_offset), z_obs_source='replica')
        except Exception as e:
            if raise_errors:
                raise
            out['status'] = f'error: {traceback.format_exc().splitlines()[-1][:150]}'
            disk = diskguard.failure_status(e, view_dir)
            if disk is not None:          # not a failure of the view: no marker, made again by the next run
                out['status'] = disk
                disk_full = True
                diskguard.request_stop()
        rows.append(out)

    del gas
    if (opts.truth_profiles == 'on' and not halo_ctx.get('hse') and stop_after is None
            and any(str(r.get('status', '')).startswith('ok') for r in rows)):
        # every view was cached: the truth profile of a tree generated before it existed
        try:
            ensure_truth_profile(source, out_root, sim, opts, verbose=verbose)
        except Exception as e:
            if raise_errors:
                raise
            disk = diskguard.failure_status(e, out_root)
            for r in rows:
                r['status'] = disk or (f'error: truth profile: '
                                       f'{traceback.format_exc().splitlines()[-1][:130]}')
    return rows

@cites("tngcluster")
def generate_view(spec: Any, source: Any, view: int, out_root: str | Path,
                  deps: ViewDependencies | None = None, *, rec: dict | None = None,
                  **kwargs: Any) -> dict:
    """Generate ONE view of the halo held by ``source`` under preset ``spec``.

    ``rec`` defaults to the halo's :func:`select_halos` record (for a cutout,
    its one halo; for a snapshot, ``source.halo_id``); that record carries no ``z_obs_fixed``, so a caller must pass a
    ``rec`` that does (the view otherwise fails with 'no z_obs in the targets').  ``kwargs`` go to
    :func:`generate_halo_views` (``raise_errors``, ``stop_after``, ``verbose``).  Returns the view's row.
    """
    from tngxmm.sim.select_halos import select_halos

    opts = SimOptions.from_spec(spec)
    if rec is None:
        recs = [r for r in select_halos(source, mass_cut=0.0, sim=opts_sim(spec))
                if r['halo_id'] == int(source.halo_id)]
        if not recs:
            raise ValueError(f'halo {source.halo_id} not selectable')
        rec = recs[0]
    rows = generate_halo_views(rec, source, out_root, deps or default_dependencies(spec),
                               opts, views=(int(view),), **kwargs)
    return rows[0]

@cites("tngcluster")
def run_halo_task(task: dict) -> list[dict]:
    """Campaign task function (``tngxmm.core.pool.run_pool``): all views of one halo.

    ``task`` is a :func:`~tngxmm.core.pool.make_halo_tasks` dict: the halo's
    :func:`select_halos` record, the task keys (``z_obs_fixed``,
    ``z_donor``, ``seed_offset``, ``no_dither_cap``, ``tvir_frac``,
    ``drop_events``) and ``preset``, ``out_root``, ``source`` (simulation
    output directory or fixture cutout).  Opens the source, binds
    the dependencies (:func:`default_dependencies`) and options of the preset
    and runs :func:`generate_halo_views`; views go to the campaign layout
    under ``out_root``.  Rows carry no private keys (dicts would not
    fit the results CSV).
    """
    from tngxmm.config import get_preset
    from tngxmm.sim.read_halo_source import read_halo_source

    spec = get_preset(task['preset'])
    opts = SimOptions.from_spec(spec)
    if opts.cube_store == 'canonical':       # the materialised canonical cube (tngxmm.ingest)
        from tngxmm.ingest.canonical_source import open_canonical_source

        # a preset that names a descriptor reads only the cube; the TNG presets keep their native
        # source for the temet check and the generation-time FoF truth
        source = open_canonical_source(opts.cube_root, task['sim'], task['snap'], task['halo_id'],
                                       native=None if opts.source else task.get('source'),
                                       descriptor=opts.source)
    else:
        source = read_halo_source(task['source'], task['snap'], task['halo_id'])
    # node memory budget and per-task peak RSS (core.membudget; off without a budget)
    with membudget.task_memory(lambda: source.group(int(task['halo_id'])),
                               label=f"{task['sim']}/{task['snap']}/{task['halo_id']}",
                               task=task) as tm:
        rows = generate_halo_views(task, source, Path(task['out_root']),
                                   default_dependencies(spec), opts, views=tuple(spec.sim.views))
    if tm is not None and rows:           # after the scope: with the resident at the end
        rows[0]['mem'] = tm.report()
    for r in rows:
        for k in [k for k in r if k.startswith('_')]:
            del r[k]
    return rows

@cites("pyxsim", "apec", "xmm_ccf_sas")
def warm_campaign_caches(spec: Any) -> None:
    """Pre-build the APEC tables of photon generation and the camera responses
    in the PARENT, before the pool forks, so the workers share them
    copy-on-write.  Same cache keys as the per-view calls
    (``generate_photons``); the camera responses depend on the donor (per-donor
    filters, CCF pattern tables) and are built by the workers on first use."""
    from tngxmm.sim.compute_apec_spectra import compute_apec_spectra

    opts = SimOptions.from_spec(spec)
    compute_apec_spectra(opts.emin_phot_kev, opts.emax_phot_kev, opts.phot_nbins,
                         opts.emin_lx_kev, opts.emax_lx_kev, apec_version=opts.apec_version,
                         abund_table=opts.abund_table)
    if opts.abundances is not None:      # the per-element spectra of the abundance modes (same cache keys)
        from tngxmm.sim import abundances as ab
        from tngxmm.sim.compute_apec_spectra import APEC_KT_MAX, APEC_KT_MIN, APEC_N_BINS

        for lo, hi, nb in ((opts.emin_phot_kev, opts.emax_phot_kev, opts.phot_nbins),
                           (opts.emin_lx_kev, opts.emax_lx_kev, APEC_N_BINS)):
            ab.element_spectra(lo, hi, nb, APEC_KT_MIN, APEC_KT_MAX, opts.abund_table, opts.apec_version, None)
    if opts.tngref_map == "on":          # compile the projection kernel once, before the fork
        from tngxmm.sim.project_tngref_map import warm_sph_map

        warm_sph_map()
