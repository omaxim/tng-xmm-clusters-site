Source code for tngxmm.sim.generate_photons

"""Sample an ideal photon list per gas cell, with the (1+z) photon-arrival factor.

A per-particle sampler; configuration is injected as keyword arguments whose
defaults are the CampaignSpec ``sim.*`` values of the ``v1`` preset.

Algorithm.  Each cell's spectrum is a mixture of spectral components
(:func:`_spectral_components`).  Legacy emission (``sim.emission_abundances``
unset): 9 components, the APEC H/He/continuum, the untracked metals scaled by Fe,
and the 7 TNG metals (C, N, O, Ne, Mg, Si, Fe) at their own abundances.  An
abundance mode (:mod:`tngxmm.sim.abundances`): the hydrogen spectrum and the
mode's components at the cell's component abundances.  Expected counts per cell
``lam = lum * (Lambda_ph / Lambda_erg) * A t (1+z) / (4 pi d_L^2)`` (the same
emission model as ``lum_erg_s``); ``N ~ Poisson(lam)``; each photon draws its
component, then its energy from the component's CDF at the cell temperature
(chunks of ``CHUNK`` photons, groups in stable-sorted (component, iT) order),
then a uniform-in-sphere offset within the cell radius.  Output energies are
observer-frame ``E / (1+z_eff)``; ``z_eff = max(z_obs, z_eff_min)``.

Temperature sampling (CampaignSpec ``sim.t_sampling = 'interpolate'``): each
cell has the mixture of the two bracketing APEC grid spectra, weighted
linearly in log T, which is the spectrum at the cell temperature to the order
of the grid spacing; each photon first draws its node (photon-rate weighted),
then its component and energy at that node.

Random draws (RNG contract, in this order): photon_counts (Poisson per cell);
per chunk photon_energy (the node; component, then per group), photon_position
(``standard_normal((n, 3))``, then ``random(n)`` radius).

Output HDF5 (xmm_simulator input format): ``data/{eobs,xsky,ysky}`` float32
[keV, deg, deg]; ``parameters/{area, exp_time, phot_norm_version, z_eff,
sampler}`` and, with a position rule, the view's catalogue position before its
measurement error (``sz_*`` or ``xray_*``, :mod:`tngxmm.sim.catalogue_centre`).
View convention: 0 = LOS z (x-y sky), 1 = LOS y (x-z), 2 = LOS x (y-z).

References
----------
pyXSIM (ZuHone & Hallman 2016; the photon-arrival normalisation it verifies),
SOXS, yt (the pyXSIM stack the sampler replaces and was verified against);
APEC/AtomDB spectra via ``compute_apec_spectra``.
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

from tngxmm.cites import cites, collect_cites
from tngxmm.core.cosmology import H0_KM_S_MPC, flat_lcdm
from tngxmm.core.cosmology import OM0 as _PIPELINE_OM0

LOS_AXIS_IDX = [2, 1, 0]
K_TO_KEV = 8.617333e-8    # Boltzmann in keV/K (photons._K_TO_KEV)
N_ELEMENTS = 7            # C N O Ne Mg Si Fe
FE_IDX = 6                # Fe is index 6 in the 7-element list
CHUNK = 4_000_000         # photons per sampling chunk (RNG contract: part of the draw order)
T_SAMPLINGS = ("interpolate",)

# defaults (CampaignSpec sim.*)
Z_EFF_MIN = None
SPHERE_R200C_FACTOR = 2.0
AREA_CM2 = 4000.0
EXPOSURE_S = 50000.0
EMIN_PHOT_KEV, EMAX_PHOT_KEV, PHOT_NBINS = 0.2, 20.0, 5000
EMIN_LX_KEV, EMAX_LX_KEV = 0.5, 2.0
H0, OM0 = H0_KM_S_MPC, _PIPELINE_OM0


def _peak_rss_gb() -> float:
    import resource

    return resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / 1e6


def _stream(rng: Any, purpose: str):
    """The Generator for ``purpose``: ``rng.get(purpose)`` for a ViewStreams,
    else ``rng`` itself (one shared Generator)."""
    return rng.get(purpose) if hasattr(rng, "get") else rng


def _wendland_c2_q(u: Any) -> Any:
    """Radius over support radius, q in [0, 1], of points distributed as the 3-D Wendland C2 kernel
    W(q) ~ (1 - q)^4 (1 + 4 q) (Dehnen & Aly 2012): the inverse of its radial distribution q^2 W(q), evaluated
    at the uniform variates ``u`` (one variate per photon, as the uniform-sphere rule uses)."""
    import numpy as np

    q = np.linspace(0.0, 1.0, 4097)
    pdf = q ** 2 * (1.0 - q) ** 4 * (1.0 + 4.0 * q)
    cdf = np.concatenate([[0.0], np.cumsum(0.5 * (pdf[1:] + pdf[:-1]) * np.diff(q))])
    cdf /= cdf[-1]
    return np.interp(u, cdf, q)


def _spectral_components(gas: Any, Z_el_p: Any, pmask: Any, emin_phot_kev: float, emax_phot_kev: float,
                         phot_nbins: int, emin_lx_kev: float, emax_lx_kev: float, apec_version: Any,
                         apec_root: Any, abund_table: Any) -> tuple:
    """The spectral components of the photon sampler.

    Legacy emission (``gas.components`` None, ``sim.emission_abundances`` unset): the APEC
    H/He/trace spectrum, the untracked APEC metals at 1 solar unit scaled by the cell's iron,
    and the seven TNG metals at their own abundances (``Z_elements`` columns C N O Ne Mg Si Fe).
    Abundance modes (:mod:`tngxmm.sim.abundances`): the hydrogen spectrum and the mode's
    components, scaled by the cell's component abundances (``Z_elements`` holds them).

    Returns ``(cosmic (nT, nbins), comps (K, nT, nbins), emid, log_kT, Lambda_c_band (nT,),
    Lambda_comps_band (K, nT), log_kT_band, S, cols)``: S the per-cell scale matrix and
    ``cols[k]`` its column for component k."""

    from tngxmm.sim.compute_apec_spectra import (
        APEC_KT_MAX,
        APEC_KT_MIN,
        APEC_N_BINS,
        KEV_TO_ERG,
        get_apec_vapec,
        get_apec_var_spec,
    )

    akw = {k: v for k, v in (("apec_version", apec_version), ("apec_root", apec_root),
                             ("abund_table", abund_table)) if v is not None}
    comps = getattr(gas, "components", None)
    if comps is None:
        log_kT_p, emid_p, cosmic_p, metals_bg_p, var_spec_p = get_apec_var_spec(
            emin_phot_kev, emax_phot_kev, nbins=phot_nbins, **akw)
        log_kT_lx, Lambda_c_lx, Lambda_other_lx, Lambda_metals_lx, *_ = get_apec_vapec(
            emin_lx_kev, emax_lx_kev, zobs=0.0, **akw)
        comps_p = [metals_bg_p] + [var_spec_p[i] for i in range(N_ELEMENTS)]
        Lcomps = [Lambda_other_lx] + [Lambda_metals_lx[i] for i in range(N_ELEMENTS)]
        cols = [FE_IDX] + list(range(N_ELEMENTS))
        return (cosmic_p, comps_p, emid_p, log_kT_p, Lambda_c_lx, Lcomps, log_kT_lx, Z_el_p, cols)
    from tngxmm.sim import abundances as ab

    tab = abund_table or "lodd"
    sp = ab.element_spectra(emin_phot_kev, emax_phot_kev, phot_nbins, APEC_KT_MIN, APEC_KT_MAX,
                            tab, apec_version, apec_root)
    log_kT_p, emid_p, cosmic_p, comps_p = ab.component_spectra(comps, sp)
    bt = ab.element_spectra(emin_lx_kev, emax_lx_kev, APEC_N_BINS, APEC_KT_MIN, APEC_KT_MAX,
                            tab, apec_version, apec_root)
    log_kT_lx, emid_lx, h_lx, c_lx = ab.component_spectra(comps, bt)
    wt = emid_lx * KEV_TO_ERG
    Lc = (h_lx * wt).sum(axis=1)
    Lcomps = [(c * wt).sum(axis=1) for c in c_lx]
    return (cosmic_p, list(comps_p), emid_p, log_kT_p, Lc, Lcomps, log_kT_lx, Z_el_p,
            list(range(len(comps.names))))

    """Write ``<out_dir>/events_v<view>.h5`` for one halo and view; return its path.

    ``rng`` is REQUIRED (no OS-entropy fallback): a ``numpy.random.Generator``
    or a :class:`tngxmm.core.rng.ViewStreams`.

    ``z_eff_min=None``: no clamp, z_eff = z_obs; ``t_sampling='interpolate'``:
    linear-in-log T mixture of the two bracketing grid temperatures (one extra
    ``photon_energy`` draw per photon); the emitting cylinder has the half-depth
    R200c; ``emax_phot_kev`` is the rest-frame upper photon energy;
    ``apec_version`` / ``apec_root`` select the APEC files of the spectra;
    ``abund_table`` the solar abundances of the spectra (must be the table
    ``load_halo_gas`` used for ``Z_elements``).  ``depth_kpc`` / ``cyl_radius_kpc``
    (``sim.los_depth_r500`` / the loaded gas sphere): half-depth and radius of the
    emitting cylinder; None keeps R200c and ``sphere_r200c_factor`` R200c.  ``projection``
    (``sim.projection``, :data:`tngxmm.sim.project_luminosity.PROJECTIONS`): None or 'axes' projects along the
    simulation axis LOS_AXIS_IDX[view]; 'icosahedral6' along the view's icosahedron vertex.
    ``position_rule`` (:func:`tngxmm.photometry.catalogue_positions.position_rule` of the view's physics donor): the
    view's catalogue position before its measurement error is written too
    (:func:`tngxmm.sim.catalogue_centre.catalogue_position`, the detection reached from the halo centre; in the frame
    of ``xsky``/``ysky``): an SZ entry's as ``parameters/sz_centre_deg`` with ``sz_rule`` (the catalogue),
    ``sz_theta500_arcmin``, ``sz_beam_fwhm_arcmin``, ``sz_map_pixel_arcmin`` and, for ACT DR6, ``sz_noise_uk_arcmin`` (the white level of the filter's noise
    model) and ``sz_snr`` (the entry's
    S/N the map is scaled to), from the cells' electron counts x T (an SZ rule without ``gas.ne_count`` raises); an
    X-ray position as ``xray_centre_deg`` with ``xray_smoothing_arcsec`` and ``xray_map_pixel_arcmin``, from the cells'
    band luminosity; both with ``catpos_version``, ``catpos_estimator``, ``catpos_flag``,
    ``catpos_global_offset_arcsec``, ``catpos_edge_weight``, ``catpos_edge_shift_arcsec``,
    ``catpos_edge_tolerance_arcsec`` and ``catpos_saddle_snr``.  The map spans the emitting cylinder; ``r500c_kpc``
    (the halo's R500c, physical) marks a position outside it (flag ``off-halo``).  No random draw is made for it.
    """
    import h5py
    import numpy as np
    if rng is None:
        raise ValueError("rng is required (RNG contract: no OS-entropy fallback)")
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)

    _cosmo   = flat_lcdm(h0, om0)
    if t_sampling not in T_SAMPLINGS:
        raise ValueError(f"t_sampling must be one of {T_SAMPLINGS}")
    z_eff    = float(z_obs) if z_eff_min is None else max(float(z_obs), z_eff_min)
    # -- 1. cylinder filter --------------------------------------------------
    pos    = gas.p_kpc - center_kpc
    R_cyl  = sphere_r200c_factor * R200c_kpc if cyl_radius_kpc is None else float(cyl_radius_kpc)
    depth  = R200c_kpc if depth_kpc is None else float(depth_kpc)
    if (projection or "axes") == "axes":
        los_idx  = LOS_AXIS_IDX[view]
        sky_axes = [i for i in range(3) if i != los_idx]
        r_proj = np.linalg.norm(pos[:, sky_axes], axis=1)
        pmask  = (r_proj <= R_cyl) & (np.abs(pos[:, los_idx]) <= depth)
        sky_a  = np.ascontiguousarray(pos[pmask][:, sky_axes[0]], dtype=np.float64)
        sky_b  = np.ascontiguousarray(pos[pmask][:, sky_axes[1]], dtype=np.float64)
    else:
        # a line of sight off the simulation axes (sim.projection): coordinates on the view's basis
        from tngxmm.sim.project_luminosity import project_offsets

        pa, pb, pl = project_offsets(pos, view, projection)
        r_proj = np.hypot(pa, pb)
        pmask  = (r_proj <= R_cyl) & (np.abs(pl) <= depth)
        sky_a  = np.ascontiguousarray(pa[pmask], dtype=np.float64)
        sky_b  = np.ascontiguousarray(pb[pmask], dtype=np.float64)
    lum_p  = np.asarray(gas.lum_erg_s[pmask], dtype=np.float64)
    T_p    = np.asarray(gas.T_K[pmask],       dtype=np.float64)
    Z_el_p = np.asarray(gas.Z_elements[pmask], dtype=np.float32)   # (N, n): elements or components
    h_p    = np.asarray(gas.h_kpc[pmask],     dtype=np.float64)
    spread = getattr(gas, "spread", None)        # None: uniform in the cell's sphere; 'sph-kernel': Wendland C2

    if verbose:
        print(f'    photons v{view}: {pmask.sum():,} particles  '
              f'lum={lum_p.sum():.3e} erg/s  z={z_eff:.3f}  '
              f'peak_rss={_peak_rss_gb():.1f} GB')

    # -- 2. spectral tables --------------------------------------------------
    # components: index 0 the H(+cosmic) spectrum, then K scaled components; the scale of
    # component k of a cell is column cols[k] of S (clipped at zero)
    cosmic_p, comps_p, emid_p, log_kT_p_grid, Lambda_c_lx, Lcomps_lx, log_kT_lx, S, cols = \
        _spectral_components(gas, Z_el_p, pmask, emin_phot_kev, emax_phot_kev, phot_nbins,
                             emin_lx_kev, emax_lx_kev, apec_version, apec_root, abund_table)
    K = len(cols)
    cosmic_sum = cosmic_p.sum(axis=1)
    comps_sum  = [c.sum(axis=1) for c in comps_p]                  # K x (nT,)

    # -- 3. per-particle expected photon counts ------------------------------
    log_kT = np.log10(np.maximum(T_p * K_TO_KEV, 1e-4))
    it_p   = np.searchsorted(log_kT_p_grid, log_kT).clip(
        0, len(log_kT_p_grid) - 1).astype(np.int32)

    # mixture of the bracketing grid temperatures
    it_hi = it_p
    it_lo = (it_hi - 1).clip(0, len(log_kT_p_grid) - 1).astype(np.int32)
    g_lo, g_hi = log_kT_p_grid[it_lo], log_kT_p_grid[it_hi]
    with np.errstate(divide='ignore', invalid='ignore'):
        w_hi = np.where(it_hi > it_lo, (log_kT - g_lo) / (g_hi - g_lo), 1.0).clip(0.0, 1.0)

    def _scale(k, rows=None):
        col = S[:, cols[k]] if rows is None else S[rows, cols[k]]
        return np.maximum(col.astype(np.float64), 0.0)

    def _lph(it):
        lp = cosmic_sum[it] + _scale(0) * comps_sum[0][it]
        for k in range(1, K):
            lp += _scale(k) * comps_sum[k][it]
        return lp

    lph_lo, lph_hi = _lph(it_lo), _lph(it_hi)
    Lambda_ph = (1.0 - w_hi) * lph_lo + w_hi * lph_hi
    # a photon of the mixture comes from the upper node with probability
    # w L_hi / ((1-w) L_lo + w L_hi) (photon-rate weighted, not w itself)
    with np.errstate(divide='ignore', invalid='ignore'):
        p_hi = np.where(Lambda_ph > 0, w_hi * lph_hi / Lambda_ph, w_hi)
    del lph_lo, lph_hi

    Lambda_erg = np.interp(log_kT, log_kT_lx, Lambda_c_lx)
    Lambda_erg += _scale(0) * np.interp(log_kT, log_kT_lx, Lcomps_lx[0])
    for k in range(1, K):
        Lambda_erg += _scale(k) * np.interp(log_kT, log_kT_lx, Lcomps_lx[k])

    d_L_cm  = _cosmo.luminosity_distance(z_eff).to('cm').value
    d_A_kpc = _cosmo.angular_diameter_distance(z_eff).to('kpc').value
    # the view's catalogue position before its measurement error (no random draw; before the photons, so a rule the
    # gas cannot serve fails before the sampling)
    cp = None
    if position_rule is not None:
        from tngxmm.sim.catalogue_centre import catalogue_position

        ne = getattr(gas, "ne_count", None)
        cp = catalogue_position(position_rule, sky_a, sky_b,
                                y_weight=None if ne is None else np.asarray(ne[pmask], np.float64) * T_p,
                                lum=lum_p, d_a_kpc=d_A_kpc, radius_kpc=R_cyl, r500_kpc=r500c_kpc)
    # includes the (1+z) photon-arrival factor (F_N = Ndot (1+z) / 4 pi dL^2)
    NORM    = area_cm2 * exposure_s * (1.0 + z_eff) / (4.0 * np.pi * d_L_cm ** 2)
    RAD2DEG = 180.0 / np.pi

    with np.errstate(divide='ignore', invalid='ignore'):
        lam = np.where(Lambda_erg > 0, lum_p * (Lambda_ph / Lambda_erg) * NORM, 0.0)
    N_p = _stream(rng, 'photon_counts').poisson(lam)
    total_ph = int(N_p.sum())

    if verbose:
        print(f'    photons v{view}: {total_ph:,} photons  '
              f'peak_rss={_peak_rss_gb():.1f} GB')
    if total_ph == 0:
        raise ValueError(
            f'No photons generated for view {view} '
            f'(z={z_eff:.3f}, R200c={R200c_kpc:.0f} kpc)'
        )

    # -- 4. photon component + energy sampling (chunked, vectorised) ---------
    rng_e = _stream(rng, 'photon_energy')
    rng_x = _stream(rng, 'photon_position')
    keep      = N_p > 0
    par_pool  = np.flatnonzero(keep)
    par_idx   = np.repeat(par_pool, N_p[keep])                 # photon -> particle
    cdf_cache: dict = {}
    all_eobs, all_xsky, all_ysky = [], [], []
    CHUNK = int(chunk)

    for s in range(0, total_ph, CHUNK):
        pj   = par_idx[s:s + CHUNK]
        n    = len(pj)
        # choose the grid node per photon (the linear mixture)
        it_c = np.where(rng_e.random(n) < p_hi[pj], it_hi[pj], it_lo[pj])
        Sp   = S[pj]

        # component weights (photon, 1 + K): the H(+cosmic) spectrum, then the K components
        w = np.empty((n, 1 + K), dtype=np.float64)
        w[:, 0] = cosmic_sum[it_c]
        for k in range(K):
            w[:, 1 + k] = np.maximum(Sp[:, cols[k]].astype(np.float64), 0.0) * comps_sum[k][it_c]
        cum  = np.cumsum(w, axis=1)
        comp = (cum < (rng_e.random(n) * cum[:, -1])[:, None]).sum(axis=1)
        comp = np.clip(comp, 0, K).astype(np.int16)

        # energies: group by (component, iT), one CDF per group
        E = np.empty(n, dtype=np.float64)
        key   = comp.astype(np.int64) * 100_000 + it_c
        order = np.argsort(key, kind='stable')
        ks    = key[order]
        cuts  = np.flatnonzero(np.diff(ks)) + 1
        for grp in np.split(np.arange(n), cuts) if n else []:
            kk = int(ks[grp[0]])
            c, iT = divmod(kk, 100_000)
            cdf = cdf_cache.get(kk)
            if cdf is None:
                row = cosmic_p[iT] if c == 0 else comps_p[c - 1][iT]
                cdf = np.cumsum(row)
                cdf_cache[kk] = cdf
            if cdf[-1] <= 0:
                E[order[grp]] = emid_p[0]
                continue
            idx = np.searchsorted(cdf, rng_e.random(len(grp)) * cdf[-1])
            E[order[grp]] = emid_p[np.clip(idx, 0, len(emid_p) - 1)]

        # positions: particle position + uniform-in-sphere cell scatter
        vec = rng_x.standard_normal((n, 3))
        vec /= np.maximum(np.linalg.norm(vec, axis=1), 1e-30)[:, None]
        u = rng_x.random(n)
        rad = h_p[pj] * (u ** (1.0 / 3.0) if spread is None else _wendland_c2_q(u))
        xs  = (sky_a[pj] + vec[:, 0] * rad) / d_A_kpc * RAD2DEG
        ys  = (sky_b[pj] + vec[:, 1] * rad) / d_A_kpc * RAD2DEG

        all_eobs.append((E / (1.0 + z_eff)).astype(np.float32))
        all_xsky.append(xs.astype(np.float32))
        all_ysky.append(ys.astype(np.float32))

    eobs_arr = np.concatenate(all_eobs)
    xsky_arr = np.concatenate(all_xsky)
    ysky_arr = np.concatenate(all_ysky)

    if verbose:
        print(f'    photons v{view}: E_mean={eobs_arr.mean():.3f} keV (obs)  '
              f'peak_rss={_peak_rss_gb():.1f} GB')

    # -- 5. write HDF5 --------------------------------------------------------
    h5_path = out_dir / f'events_v{view}.h5'
    with h5py.File(str(h5_path), 'w') as f:
        gd = f.create_group('data')
        gd.create_dataset('eobs', data=eobs_arr)
        gd.create_dataset('xsky', data=xsky_arr)
        gd.create_dataset('ysky', data=ysky_arr)
        gp = f.create_group('parameters')
        gp['area']     = float(area_cm2)
        gp['exp_time'] = float(exposure_s)
        gp['phot_norm_version'] = 3   # rest-frame lum x (1+z_eff) arrival factor
        gp['z_eff']    = float(z_eff)
        gp['sampler']  = 'per-particle-v3.1'
        if cp is not None:
            pos = np.array([cp['a_kpc'], cp['b_kpc']]) / d_A_kpc * RAD2DEG
            if cp['kind'] == 'sz':
                gp['sz_centre_deg'] = pos
                gp['sz_rule'] = str(cp['catalogue'])
                gp['sz_theta500_arcmin'] = float(cp['theta500_arcmin'])
                gp['sz_beam_fwhm_arcmin'] = float(cp['beam_fwhm_arcmin'])
                gp['sz_map_pixel_arcmin'] = float(cp['map_pixel_arcmin'])
                if cp.get('snr') is not None:
                    gp['sz_snr'] = float(cp['snr'])
                if cp.get('white_noise_uk_arcmin') is not None:
                    gp['sz_noise_uk_arcmin'] = float(cp['white_noise_uk_arcmin'])
            else:
                gp['xray_centre_deg'] = pos
                gp['xray_smoothing_arcsec'] = float(cp['smoothing_arcsec'])
                gp['xray_map_pixel_arcmin'] = float(cp['map_pixel_arcmin'])
            gp['catpos_version'] = int(cp['version'])
            gp['catpos_estimator'] = str(cp['estimator'])
            gp['catpos_flag'] = str(cp['flag'])
            gp['catpos_global_offset_arcsec'] = float(cp['global_offset_arcsec'])
            gp['catpos_edge_weight'] = float(cp['edge_weight'])
            gp['catpos_edge_shift_arcsec'] = float(cp['edge_shift_arcsec'])
            gp['catpos_edge_tolerance_arcsec'] = float(cp['edge_tolerance_arcsec'])
            gp['catpos_saddle_snr'] = float(cp.get('saddle_snr', float('nan')))
    return h5_path
