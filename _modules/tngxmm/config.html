"""CampaignSpec: every tunable and fixed number of the pipeline, with provenance.

A :class:`CampaignSpec` is the complete, hashable description of one campaign: the
simulation and its gas selection, the donor matching, the instrument, the aperture
photometry, the hydrostatic fit and the run-time settings.  Each field records its
value, unit, the module that reads it, its citation key and a one-line description.

Presets (:data:`PRESETS`):

``v1``         TNG-Cluster: every primary zoom target with M500c >= 1e14 Msun at the full
               snapshots 99-40, three projections each, K = 5 replica views per reference
               observation.  The production campaign (:data:`PRODUCTION_PRESET`).
``v1_tng300``  the same physics on the periodic box TNG300-1 (every FoF central);
``v1_tng100``  TNG100-1 (held-out resolution);
``v1_tng50``   TNG50-1 (held-out resolution).

The field defaults ARE the v1 values: ``CampaignSpec()`` is the v1 physics with the
name ``v1``.  The box presets change only the simulation (``sim.sim_name``,
``sim.base_path``, ``sim.primary_only``) and their random-stream key.

``sim.rng_campaign`` keys the random streams of a view (``core.rng.ViewStreams``; the
sha256 seed message carries it).  The production presets key their streams by the
campaign names ``v4``, ``v4_tng300``, ``v4_tng100`` and ``v4_tng50``, so that every view
pairs with the view of the same (snapshot, halo, projection) of the campaign generated
under that key.  It is a string, not a preset.

The canonical hash (:meth:`CampaignSpec.spec_hash`) is the sha256 of the sorted-key JSON
of every non-runtime field; the ``runtime`` section (workers, paths, concurrency,
timeouts) and the free-text ``description`` never enter it.  The generation hash
(:meth:`CampaignSpec.generation_hash`) leaves out the ``massfit`` section as well: it is
what determines the generated products.

Regenerate ``conf/base/parameters/<preset>.yml`` and the hash table of
``docs/dev/INTERFACES.md`` with ``make params``.
"""

def P(value: Any, unit: str = "", reads: str = "", cite: str | None = None,
      env: str | None = None, doc: str = "", runtime: bool = False,
      omit_if_none: bool = False) -> Any:
    """A spec field with provenance metadata.

    ``reads`` names the module (``tngxmm.`` omitted) whose code applies the value; ``env``
    the environment variable that overrides a run-time value.  ``omit_if_none``: the field
    joins the canonical JSON (the spec hash) only when it is not None, so a field added
    with this flag and the value None leaves the hash of every existing preset unchanged
    and the campaigns produced under them stay valid."""
    md = {"unit": unit, "reads": reads, "cite": cite, "env": env, "doc": doc,
          "runtime": runtime, "omit_if_none": omit_if_none}
    return field(default=value, metadata=md)

def calibration_tables_sha256() -> dict[str, str]:
    """sha256 of every table in the package's ``calibration/`` folders (``*.csv``, path relative to the package),
    entered in the spec hash: :meth:`CampaignSpec.canonical_json`."""
    from tngxmm.core.hashing import sha256_file

    root = Path(__file__).resolve().parent
    return {str(f.relative_to(root)): sha256_file(f) for f in sorted(root.glob("*/calibration/*.csv"))}

def canonical_cell(value: str | None, kind: str) -> str:
    """One cell as it enters :func:`table_columns_digest`.  ``text``: as written, without surrounding blanks.
    ``number``: ``repr(float(cell))``, the shortest text that gives the same double, so ``0.049``, ``0.0490`` and
    ``4.9e-2`` are one value; an empty or NaN cell is the empty string."""
    v = (value or "").strip()
    if kind == "text":
        return v
    return "" if v == "" or v.lower() == "nan" else repr(float(v))

def table_columns_digest(path: str | Path, columns: dict[str, str], key: str,
                         where: tuple[str, str] | None = None) -> str:
    """Hex sha256 of the columns ``columns`` of the CSV table ``path``, whatever else the file holds.  ``where``
    (a column and the text of its cell) keeps only the rows with that cell.

    ``columns`` maps a column name to its kind (:data:`COLUMN_KINDS`).  The digest is taken over the column names in
    sorted order and the rows sorted by the column ``key`` (unique, one of ``columns``), each cell through
    :func:`canonical_cell`.  Other columns, the order of the columns and of the rows, the number of digits a value is
    written with and the line ends do not enter.  (It lives in this module and not in ``core.hashing`` because the
    code of that module is part of the seed hash, ``massfit.guess_mass.SEED_PROCEDURE_MODULES``.)"""
    import csv

    if key not in columns:
        raise KeyError(f"key column {key!r} is not among the hashed columns")
    bad = {k: v for k, v in columns.items() if v not in COLUMN_KINDS}
    if bad:
        raise ValueError(f"column kinds {bad}: one of {COLUMN_KINDS}")
    names = sorted(columns)
    with open(path, newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        missing = [c for c in names if c not in (reader.fieldnames or [])]
        if missing:
            raise KeyError(f"{path}: no column {missing}")
        rows = list(reader)
    if where is not None:
        rows = [r for r in rows if (r[where[0]] or "").strip() == where[1]]
    keys = [canonical_cell(r[key], columns[key]) for r in rows]
    if len(set(keys)) != len(keys):
        raise ValueError(f"{path}: column {key!r} does not identify the rows")
    order = sorted(range(len(rows)), key=keys.__getitem__)
    body = [[canonical_cell(rows[i][c], columns[c]) for c in names] for i in order]
    text = json.dumps({"columns": names, "rows": body}, separators=(",", ":"), ensure_ascii=True)
    return hashlib.sha256(text.encode()).hexdigest()

def _jsonable(v: Any) -> Any:
    if isinstance(v, (tuple, list)):
        return [_jsonable(x) for x in v]
    if isinstance(v, dict):
        return {str(k): _jsonable(x) for k, x in v.items()}
    if isinstance(v, float) and not math.isfinite(v):
        return repr(v)
    return v

    def canonical_json(self) -> str:
        """Sorted-key, compact JSON of every hashed field (runtime and the
        free-text ``description`` excluded)."""
        d = self.to_dict(include_runtime=False)
        d.pop("description", None)
        for sec, _ in SECTIONS:                       # fields that hash only when set
            if sec in d:
                for f in fields(getattr(self, sec)):
                    if f.metadata.get("omit_if_none") and getattr(getattr(self, sec), f.name) is None:
                        d[sec].pop(f.name, None)
        if self.sim.source is not None:               # the descriptor's content is part of the physics
            from tngxmm.ingest.descriptor import descriptor_hash

            d["sim"]["descriptor_sha256"] = descriptor_hash(str(self.sim.source))
        # the package's calibration tables (pn CTI slopes, the CMB spectrum of the SZ filter) are inputs of the
        # generation like the fields: an edited table makes another spec
        d["calibration_tables_sha256"] = calibration_tables_sha256()
        # and so is the observation table: the columns of it that the stages read (OBSERVATION_COLUMNS)
        sample = str(getattr(self.donors, "reference_sample", BANK_SAMPLE))
        d["observation_tables_sha256"] = (observation_tables_sha256(sample) if sample.endswith(EXTRA_SAMPLE_SUFFIX)
                                          else observation_tables_sha256())
        # and the pins of the external data-package files a view is made from (APEC, ECF, reddening tables)
        d["data_manifest_sha256"] = data_manifest_sha256()
        return json.dumps(d, sort_keys=True, separators=(",", ":"), ensure_ascii=True)

    def describe(self) -> list[dict[str, Any]]:
        """One row per field: section, name, value, unit, reads, cite, env, runtime, doc."""
        rows = []
        for f in fields(self):
            if f.metadata:

            for f in fields(obj):

                                 **f.metadata))
        return rows

def _v1_source(name: str, descriptor: str, label: str, cube_store: str = "index") -> CampaignSpec:

    cube index files of the mock-cutout stage and is converted to the canonical cells in memory

    (``'canonical'``, streamed from the data services or read from the SWIFT files); the snapshots nearest the v1 redshifts,

    the cube in the generation.  The generation takes the simulation's own cosmology from the
    descriptor (distances, critical density); ``cosmology`` stays the one analysis cosmology of the
    fits."""
    from tngxmm.ingest.descriptor import load

    d = load(descriptor)
    zs = {int(k): float(v) for k, v in d["snapshots"]["redshift"].items()}
    snaps = []
    for z in V1_SNAPSHOT_Z:
        s = min(zs, key=lambda k: (abs(zs[k] - z), -k))
        if s not in snaps:
            snaps.append(s)
    return CampaignSpec(
        name=name,
        description=f"The v1 physics on {label} (descriptor {descriptor}): the halos of the target list at the "
                    "snapshots nearest the v1 redshifts; the 5.66 R500c gas cube, emitting from the 4 R500c cylinder; "
                    "spherical hydrostatic truth; no temet reference.",
        sim=dataclasses.replace(SimSpec(), sim_name=str(d["name"]), base_path=str(d["base_path"]),
                                primary_only=False, rng_campaign=name.replace("v1", "v4", 1),
                                snaps=tuple(snaps), source=str(descriptor), lx_reference="none",
                                tngref_map="off"),
        donors=dataclasses.replace(DonorSpec(), observing_conditions=dict(PRODUCTION_CONDITIONS)),
        runtime=dataclasses.replace(RuntimeSpec(), cube_store=cube_store),
    )

class ResponsePairingError(ValueError):
    """A preset whose mocks would be made with another response than the fit uses."""




#: (instrument.responses, massfit.responses) of a preset, None read as the default of each side: the mocks are made
#: with the response the fit uses, the shared set of the packaged responses or the conditions donor's own SAS
#: responses.  The closure arms that measure a mismatch (v1_obsresp_shared) are built from a preset, not presets.
RESPONSE_PAIRS = (("packaged", "shared"), ("donor", "per-observation"))

def check_response_pairing(spec: CampaignSpec) -> None:
    """Raise :class:`ResponsePairingError` unless the preset's generator and fit responses pair
    (:data:`RESPONSE_PAIRS`) and ``instrument.ccf_selection`` equals ``massfit.ccf_selection``.  Without it a fit of
    mocks made with the packaged responses would use their donors' SAS 22 responses (or the reverse), a mock-side
    error the size of the real-side one the switches remove."""
    pair = (getattr(spec.instrument, "responses", None) or "packaged",
            getattr(spec.massfit, "responses", None) or "shared")
    if pair not in RESPONSE_PAIRS:
        raise ResponsePairingError(f"preset {spec.name}: instrument.responses {pair[0]!r} with massfit.responses "
                                   f"{pair[1]!r}; a preset sets one of {RESPONSE_PAIRS}")
    ccf = (getattr(spec.instrument, "ccf_selection", None) or "highest-issue",
           getattr(spec.massfit, "ccf_selection", None) or "highest-issue")
    if ccf[0] != ccf[1]:
        raise ResponsePairingError(f"preset {spec.name}: instrument.ccf_selection {ccf[0]!r} differs from "
                                   f"massfit.ccf_selection {ccf[1]!r}: both sides read the same calibration files")

def get_preset(name: str) -> CampaignSpec:
    """A fresh CampaignSpec for preset ``name`` (one of :data:`PRESETS`), checked by :func:`check_response_pairing`."""
    try:
        spec = PRESETS[name]()
    except KeyError:
        raise KeyError(f"unknown preset {name!r}; known: {sorted(PRESETS)}") from None
    check_response_pairing(spec)
    return spec

def write_parameters(out_dir: str, presets: tuple[str, ...] = tuple(PRESETS)) -> list[str]:
    """Write ``<out_dir>/<preset>.yml`` (Kedro parameters) for every preset."""
    import pathlib

    import yaml

    out = []
    d = pathlib.Path(out_dir)
    d.mkdir(parents=True, exist_ok=True)
    for name in presets:
        spec = get_preset(name)
        head = (f"# GENERATED from tngxmm.config preset {name!r}; do not edit by hand.\n"
                f"# Regenerate: python -m tngxmm.config --write-params {out_dir}\n"
                f"# spec_hash (runtime excluded): {spec.spec_hash()}\n")
        body = yaml.safe_dump(spec.to_parameters(), sort_keys=False, default_flow_style=None,
                              width=100)
        p = d / f"{name}.yml"
        p.write_text(head + body, encoding="utf-8")
        out.append(str(p))
    return out

def spec_hash_markdown(presets: tuple[str, ...] = tuple(PRESETS)) -> str:
    """The generated block of INTERFACES.md: one table row per preset (hash, field count)."""
    lines = ["| preset | spec_hash (16) | spec_hash (64) | hashed fields |", "|---|---|---|---|"]
    for name in presets:
        s = get_preset(name)
        n = sum(1 for r in s.describe()
                if not r["runtime"] and not (r["omit_if_none"] and r["value"] is None))
        lines.append(f"| `{name}` | `{s.spec_hash()}` | `{s.spec_hash(64)}` | {n} |")
    return "\n".join(lines)

def write_doc_hashes(path: str) -> bool:
    """Rewrite the block between :data:`DOC_HASH_BEGIN` and :data:`DOC_HASH_END` in
    ``path``; True if the file changed."""
    import pathlib

    p = pathlib.Path(path)
    text = p.read_text(encoding="utf-8")
    head, sep, rest = text.partition(DOC_HASH_BEGIN)
    if not sep or DOC_HASH_END not in rest:
        raise ValueError(f"{path}: no spec-hash block markers")
    new = head + DOC_HASH_BEGIN + "\n" + spec_hash_markdown() + "\n" + DOC_HASH_END + \
        rest.split(DOC_HASH_END, 1)[1]
    if new != text:
        p.write_text(new, encoding="utf-8")
    return new != text




def _main(argv: list[str] | None = None) -> int:
    import argparse

    ap = argparse.ArgumentParser(description="CampaignSpec presets")
    ap.add_argument("--write-params", metavar="DIR")
    ap.add_argument("--show", metavar="PRESET")
    ap.add_argument("--hash", metavar="PRESET")
    ap.add_argument("--write-docs", metavar="MD", help="rewrite the spec-hash block of MD")
    a = ap.parse_args(argv)
    if a.write_params:
        for p in write_parameters(a.write_params):
            print(p)
    if a.write_docs:
        print(f"{a.write_docs}: {'updated' if write_doc_hashes(a.write_docs) else 'unchanged'}")
    if a.show:
        for r in get_preset(a.show).describe():
            print(f"{r['section']:>10s}.{r['name']:<28s} = {r['value']!r}  [{r['unit']}]  "
                  f"read by {r['reads']}  cite={r['cite']}")
    if a.hash:
        print(get_preset(a.hash).spec_hash(64))
    return 0
