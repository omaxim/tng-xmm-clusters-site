Source code for tngxmm.embed.extract_dinov3

"""DINOv3 ViT-H+/16 image features: pinned model, preprocessing and three embedding kinds.

The preprocessing is PIL ``convert("RGB")``, torchvision ``ToTensor`` + ImageNet
``Normalize``, NO resize (the 640 x 640 native crop is fed as is), then the forward
call and the CLS read-out; on the same device, precision, thread count and batch the CLS
vectors are bit-identical to the reference embeddings
(``tests/e2e/test_embed_exact.py``).  INTERFACES section 4.7:

* the Hugging Face revision is PINNED (:data:`DINOV3_REVISION`);
* the numerical precision is an explicit, recorded choice
  (:data:`PRECISIONS`): ``fp32`` (TF32 disabled on CUDA) or
  ``bf16-autocast`` (``torch.autocast`` with bfloat16);
* two more embedding kinds from the same forward passes
  (:data:`EMBEDDING_KINDS`): ``cls_tta8``, the CLS token averaged over the
  eight dihedral transforms of the image (:func:`dihedral_views`), and
  ``patch_mean``, the mean of the 1600 patch tokens of the final layer
  (register tokens excluded);
* no ``huggingface_hub.login(token)`` call is made (it would write the token into
  ``$HF_HOME``): the hub reads ``HF_TOKEN`` / ``HF_TOKEN_PATH`` by itself and no token
  is ever stored by this package.

Every ``torch``/``transformers``/``PIL`` import happens inside the function
that uses it (INTERFACES section 1).

References
----------
dinov3 : Simeoni et al. 2025, arXiv:2508.10104; model
    ``facebook/dinov3-vith16plus-pretrain-lvd1689m`` at revision
    ``c807c9eeea853df70aec4069e6f56b28ddc82acc`` (DINOv3 License; the use of
    DINO Materials must be acknowledged in publications, section 1.b.ii).
transformers : Wolf et al. 2020, EMNLP System Demonstrations, 38.
huggingface_hub : revision-pinned download and cache of the weights.
pytorch : Ansel et al. 2024, ASPLOS 2, 929; Paszke et al. 2019, arXiv:1912.01703.
torchvision : ``ToTensor`` and ``Normalize`` transforms.
pillow : PNG decoding.
"""

from __future__ import annotations

import hashlib
import json
import re
from collections.abc import Iterable, Sequence
from pathlib import Path
from typing import Any

from tngxmm.cites import cites, collect_cites
from tngxmm.core.logging import get_logger


log = get_logger(__name__)

#: Hugging Face model id
DINOV3_MODEL_ID = "facebook/dinov3-vith16plus-pretrain-lvd1689m"
#: pinned model revision: the commit the reference embeddings were made with
#: (docs/CITATIONS_NOTES.md section 4)
DINOV3_REVISION = "c807c9eeea853df70aec4069e6f56b28ddc82acc"
#: ImageNet normalisation (also the model's preprocessor_config.json)
IMAGENET_MEAN = (0.485, 0.456, 0.406)
IMAGENET_STD = (0.229, 0.224, 0.225)

#: embedding kinds, in canonical order; ``cls`` is always computed (it is the
#: identity pass every other kind needs, and INTERFACES 4.7 requires the dataset)
EMBEDDING_KINDS: tuple[str, ...] = ("cls", "cls_tta8", "patch_mean")
KIND_DEFINITIONS: dict[str, str] = {
    "cls": "CLS token of the final layer (after the final LayerNorm), identity view",
    "cls_tta8": ("CLS token of the final layer averaged (float32, fixed order) over the eight "
                 "dihedral transforms of the input: " + ", ".join(
                     ("id", "rot90", "rot180", "rot270", "flipx", "flipx_rot90",
                      "flipx_rot180", "flipx_rot270"))),
    "patch_mean": ("mean of the patch tokens of the final layer (after the final LayerNorm), "
                   "identity view; CLS and register tokens excluded"),
}
#: names of the eight transforms of :func:`dihedral_views`, in evaluation order
DIHEDRAL_NAMES: tuple[str, ...] = ("id", "rot90", "rot180", "rot270",
                                   "flipx", "flipx_rot90", "flipx_rot180", "flipx_rot270")
#: numerical precision of the forward pass
PRECISIONS: tuple[str, ...] = ("fp32", "bf16-autocast")

_SHA1_RE = re.compile(r"^[0-9a-f]{40}$")

def check_revision(revision: str) -> str:
    """Return ``revision`` if it is a full 40-hex commit; branch names and tags are refused,
    because a moving reference makes the embeddings irreproducible."""
    if not isinstance(revision, str) or not _SHA1_RE.match(revision):
        raise ExtractionError(f"model revision must be a full 40-hex commit (pinned), got "
                              f"{revision!r}; the tested one is {DINOV3_REVISION}")
    return revision

def normalise_kinds(kinds: Iterable[str]) -> tuple[str, ...]:
    """Requested kinds plus ``cls``, unique, in the canonical order of :data:`EMBEDDING_KINDS`."""
    req = set(kinds) | {"cls"}
    bad = req - set(EMBEDDING_KINDS)
    if bad:
        raise ExtractionError(f"unknown embedding kind(s) {sorted(bad)}; known: {EMBEDDING_KINDS}")
    return tuple(k for k in EMBEDDING_KINDS if k in req)

    """The preprocessing, as data (written into the embeddings file and hashed)."""
    return {
        "decode": "PIL.Image.open(path).convert('RGB')",
        "resize": None,
        "crop": None,

                      "op": "torchvision.transforms.Normalize"},
    }

@cites("pytorch")
def configure_numerics(device: str, precision: str) -> dict[str, Any]:
    """Set the torch flags the precision needs and return them for provenance.

    ``fp32`` disables TF32 for matmul and cuDNN (TF32 keeps 10 mantissa bits:
    results would then depend on the GPU generation); ``bf16-autocast`` leaves the flags alone.
    """
    import torch

    if precision not in PRECISIONS:
        raise ExtractionError(f"precision must be one of {PRECISIONS}, got {precision!r}")
    if device.startswith("cuda") and precision == "fp32":
        torch.backends.cuda.matmul.allow_tf32 = False
        torch.backends.cudnn.allow_tf32 = False
    return {
        "precision": precision,
        "autocast_dtype": "bfloat16" if precision == "bf16-autocast" else "none",
        "cuda_matmul_allow_tf32": bool(torch.backends.cuda.matmul.allow_tf32),
        "cudnn_allow_tf32": bool(torch.backends.cudnn.allow_tf32),
        "torch_num_threads": int(torch.get_num_threads()),
    }

@cites("dinov3", "transformers", "huggingface_hub", "pytorch")
def load_dinov3(device: str = "cpu", *, model_id: str = DINOV3_MODEL_ID,
                revision: str = DINOV3_REVISION, local_files_only: bool = False) -> Any:
    """DINOv3 backbone in eval mode on ``device``, float32 weights, at a pinned revision.

    With the revision cached (``$HF_HUB_CACHE``) nothing is downloaded; with
    ``local_files_only`` the hub is never contacted.  The gated-model token is
    read by huggingface_hub from ``HF_TOKEN``/``HF_TOKEN_PATH`` and never stored.
    """
    import torch
    from transformers import AutoModel

    check_revision(revision)
    if device.startswith("cuda") and not torch.cuda.is_available():
        raise ExtractionError("device 'cuda' requested but torch.cuda.is_available() is False "
                              "(a missing GPU is never replaced by the CPU silently)")
    log.info("loading %s @ %s (local_files_only=%s) on %s", model_id, revision,
             local_files_only, device)
    model = AutoModel.from_pretrained(model_id, revision=revision,
                                      local_files_only=local_files_only)
    resolved = getattr(model.config, "_commit_hash", None)
    if resolved and resolved != revision:
        raise ExtractionError(f"loaded revision {resolved} != requested {revision}")
    model = model.to(device=torch.device(device), dtype=torch.float32)
    model.eval()
    return model

@cites("pillow", "torchvision")
def load_image_tensor(path: str | Path) -> Any:
    """One bank render as the model input: (3, H, W) float32, ImageNet-normalised."""
    import torchvision.transforms as transforms
    from PIL import Image

    tf = transforms.Compose([
        transforms.ToTensor(),
        transforms.Normalize(mean=list(IMAGENET_MEAN), std=list(IMAGENET_STD)),
    ])
    with Image.open(path) as im:
        return tf(im.convert("RGB"))

def dihedral_views(x: Any) -> list[Any]:
    """The eight transforms of the dihedral group D4 of a square (..., H, W) tensor, in
    the order of :data:`DIHEDRAL_NAMES`: rotations by k x 90 degrees (``torch.rot90``,
    k = 0..3), then the same after a left-right flip.  Pixel permutations, so applying
    them after the per-channel normalisation equals applying them to the image."""
    import torch

    if x.shape[-1] != x.shape[-2]:
        raise ExtractionError(f"dihedral views need a square image, got {tuple(x.shape[-2:])}")
    out = []
    for flip in (False, True):
        base = torch.flip(x, dims=(-1,)) if flip else x
        for k in range(4):
            out.append(torch.rot90(base, k, dims=(-2, -1)).contiguous() if k else base)
    return out

@cites("dinov3", "pytorch")
def embed_batch(model: Any, pixel_values: Any, kinds: Iterable[str] = ("cls",), *,
                precision: str = "fp32") -> dict[str, Any]:
    """Embeddings of one batch: ``{kind: (B, D) float32 numpy}``.

    The identity pass gives ``cls`` (``last_hidden_state[:, 0]``) and
    ``patch_mean``; ``cls_tta8`` adds seven passes, one per non-identity
    transform, and averages the eight CLS vectors in float32.
    """
    import torch

    kinds = normalise_kinds(kinds)
    if precision not in PRECISIONS:
        raise ExtractionError(f"precision must be one of {PRECISIONS}, got {precision!r}")
    dev = pixel_values.device
    use_amp = precision == "bf16-autocast"
    n_reg = int(getattr(model.config, "num_register_tokens", 0))
    patch = int(getattr(model.config, "patch_size", 16))
    n_patch = (pixel_values.shape[-2] // patch) * (pixel_values.shape[-1] // patch)

    def forward(x: Any) -> Any:
        with torch.autocast(device_type=dev.type, dtype=torch.bfloat16, enabled=use_amp):
            out = model(pixel_values=x, output_hidden_states=False)
        return out.last_hidden_state

    res: dict[str, Any] = {}
    with torch.no_grad():
        h = forward(pixel_values)
        if h.shape[1] != 1 + n_reg + n_patch:
            raise ExtractionError(f"unexpected token count {h.shape[1]} "
                                  f"(1 CLS + {n_reg} registers + {n_patch} patches)")
        cls = h[:, 0].float()
        res["cls"] = cls.cpu().numpy()
        if "patch_mean" in kinds:
            res["patch_mean"] = h[:, 1 + n_reg:].float().mean(dim=1).cpu().numpy()
        if "cls_tta8" in kinds:
            acc = cls.clone()
            for view in dihedral_views(pixel_values)[1:]:
                acc += forward(view)[:, 0].float()
            res["cls_tta8"] = (acc / len(DIHEDRAL_NAMES)).cpu().numpy()
    return res




class _ImageDataset:
    """(index, tensor) per image path, in input order (torch Dataset protocol)."""

    def __init__(self, paths: Sequence[str | Path]):
        self.paths = [Path(p) for p in paths]

    def __len__(self) -> int:
        return len(self.paths)

    def __getitem__(self, i: int) -> tuple[int, Any]:
        return i, load_image_tensor(self.paths[i])


def _collate(batch: list[tuple[int, Any]]) -> tuple[list[int], Any]:
    import torch

    return [i for i, _ in batch], torch.stack([x for _, x in batch])

def extract_dinov3(paths: Sequence[str | Path], *, kinds: Iterable[str] = ("cls",),
                   model: Any | None = None, device: str = "cpu", precision: str = "fp32",
                   revision: str = DINOV3_REVISION, local_files_only: bool = False,
                   batch_size: int = 16, num_workers: int = 0) -> dict[str, Any]:
    """DINOv3 embeddings of the images at ``paths``: ``{kind: (N, D) float32 numpy}``, row
    ``i`` <-> ``paths[i]`` (manifest order = embedding order).

    Images are decoded by ``num_workers`` DataLoader workers (0 = in process) and
    batched ``batch_size`` at a time, all of one size.  An
    unreadable image raises instead of leaving a NaN row.  ``model`` is loaded with
    :func:`load_dinov3` when not given.
    """
    import numpy as np
    import torch
    from torch.utils.data import DataLoader

    kinds = normalise_kinds(kinds)
    if not paths:
        raise ExtractionError("no images to embed")
    if batch_size < 1:
        raise ExtractionError(f"batch_size must be >= 1, got {batch_size}")
    if model is None:
        model = load_dinov3(device, revision=revision, local_files_only=local_files_only)
    dim = int(model.config.hidden_size)
    out = {k: np.full((len(paths), dim), np.nan, dtype=np.float32) for k in kinds}
    loader = DataLoader(_ImageDataset(paths), batch_size=batch_size, shuffle=False,
                        num_workers=num_workers, collate_fn=_collate,
                        pin_memory=device.startswith("cuda"),
                        persistent_workers=num_workers > 0)
    n_done = 0
    for idx, x in loader:
        res = embed_batch(model, x.to(torch.device(device)), kinds, precision=precision)
        for k in kinds:
            out[k][idx] = res[k]
        n_done += len(idx)
        log.info("embedded %d/%d images", n_done, len(paths))
    for k, v in out.items():
        if not np.isfinite(v).all():
            raise ExtractionError(f"{k}: non-finite values after extraction")
    return out
