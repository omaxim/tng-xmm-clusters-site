Source code for tngxmm.instrument.donor_tables

"""The per-donor tables of the data package that the instrument stage reads.

Every table has one row (or a few) per real observation, keyed by its ten-digit obsid, and is
BUILT by a node of the ``donor_tables`` Kedro pipeline (:mod:`tngxmm.donor_tables`) into
``$TNGXMM_DATA/tables/``.  The instrument options name the tables only to override the package copy
(a study, a test); an empty name means the package table.  Nothing here knows a machine path:
``$TNGXMM_DATA`` comes from the site file (:func:`tngxmm.core.paths.data_root`).

Six tables depend on where each donor's cluster is (:data:`CENTRE_TABLES`).  The core offsets
(``donor_core_offsets.csv``) are the cluster's place on the detector, computed from the guarded centres; the
anomalous-CCD table, the IN/OUT soft-proton table and the in-field background table (``donor_background.csv``) are
measured outside a circle about those centres; the corner QPB table is measured for the donors of the same catalogue
(the centre and seed tables); the condition table copies the core offsets and the background levels.  The other files
of the data package a view is made from that a builder of this package writes (:data:`BUILT_TABLES`: the total
column, the RASS sky, the active CCDs and the CCD maps, the source thresholds, the catalogue positions; the list of a
preset is :func:`tngxmm.sim.data_inputs.stage_files`) are checked the same way.  The package copy of each is read only
when the provenance sidecar written with it (``<table>.provenance.json``) describes its bytes, says it was built from
a committed, clean tree (``git_dirty`` false), names the current versions of the files it was built from by their
sha256 (:func:`current_inputs`: the centre and seed tables for the tables measured about the cluster, for the core
offsets the centre table and ``observations.csv``, for the two corner tables also the anomalous-CCD table, for the
condition table also the core-offset and background tables, and so on), and, for the tables whose measurement is
versioned, records the current measurement version and FWC reference rule (:func:`measurement_record`;
:func:`check_table`).  The inputs that lie outside the package are compared too: the published catalogue files of the
catalogue positions through their pins, the PPS source lists of the source thresholds with the local PPS cache where
it holds them.  An input of the data package with a sidecar of its own must have been built from a clean tree
too, and the package's centre table must itself be current: copied from the current seed table, which is current for
the production preset (:func:`tngxmm.photometry.build_obs_centres_guarded.check_centres_table`, the check the real
fits make).  A table named explicitly (an override) is the caller's and is read as given.
"""

TABLE_FILES: dict[str, str] = {
    "background": "donor_background.csv",          # in-FOV background levels and live times
    "mos_background": "donor_mos_background.csv",  # MOS corner QPB per band and IN/OUT soft protons
    "qpb_corners": "donor_qpb_corners.csv",        # corner QPB of every camera per band, MOS CCD and pn quadrant (ESAS)
    "nh_total": "donor_nh_total.csv",              # total / atomic Galactic column (Willingale et al. 2013)
    "soft_ccd": "donor_soft_ccd.csv",              # anomalous-state MOS CCDs
    "conditions": "donor_conditions.csv",          # the condition space of the conditions donor
    "sky": "donor_sky_rass.csv",                   # LHB and halo levels of the field from RASS (donor_tables.rass_sky)
    "core_offsets": "donor_core_offsets.csv",      # the cluster's place on the detector (donors.build_core_offsets)
}

    "source_thresholds": "donor_source_thresholds.csv",   # flux threshold of the mock point-source mask per donor
    "catalogue_positions": "catalogue_positions_v4.csv",  # the catalogue entry of every observation
}

def table_path(name: str, override: str | Path | None = None) -> Path:
    """The file of table ``name`` (of :data:`TABLE_FILES` or :data:`STAGE_FILES`): ``override`` when given, else the
    data package's."""
    if override:
        return Path(override)
    files = {**TABLE_FILES, **STAGE_FILES}
    if name not in files:
        raise ValueError(f"unknown donor table {name!r}; known: {sorted(files)}")
    from tngxmm.core.paths import data_asset

    return data_asset("tables") / files[name]

def measurement_record(name: str) -> dict[str, Any]:
    """The provenance values the package copy of table ``name`` must record to hold the current measurement: the
    measurement version of its builder (``measure_version``) and, for the two corner tables, whose scales are
    referred to the filter-wheel-closed data, the rule of that reference (``fwc_rule``,
    :data:`tngxmm.instrument.fwc_state.FWC_REFERENCE_RULE`); for the catalogue positions also the file name and
    sha256 of the three published files outside the package it is built from (PSZ2, ACT DR6, the counterpart table:
    :data:`tngxmm.photometry.catalogue_positions.INPUT_PINS`).  Empty for a table without a versioned measurement."""
    if name == "soft_ccd":
        from tngxmm.donor_tables.soft_ccd import MEASURE_VERSION

        return {"measure_version": MEASURE_VERSION}
    if name == "background":
        from tngxmm.donor_tables.background import MEASURE_VERSION

        return {"measure_version": MEASURE_VERSION}
    if name == "conditions":
        from tngxmm.instrument.conditions_donor import MEASURE_VERSION

        return {"measure_version": MEASURE_VERSION}
    if name == "catalogue_positions":
        # the redshift typing of the table and the position rule its columns serve (theta500 of the SZ mass for the
        # PSZ2 filter scale since rule version 3), the columns themselves, and the published files it is built from
        # (outside the package, so compared through their pins, CP.INPUT_PINS)
        from tngxmm.photometry import catalogue_positions as CP

        return {"redshift_rule": CP.REDSHIFT_RULE_VERSION, "position_rule": CP.POSITION_RULE_VERSION,
                "columns": list(CP.COLUMNS),
                **{k: {"file": f, "sha256": h} for k, (f, h) in CP.INPUT_PINS.items()}}
    if name == "source_thresholds":
        from tngxmm.massfit import build_source_thresholds as ST

        return {"measure_version": ST.MEASURE_VERSION, "columns": list(ST.COLUMNS)}
    if name in ("mos_background", "qpb_corners"):
        from tngxmm.donor_tables import mos_background, qpb_corners
        from tngxmm.instrument.fwc_state import FWC_REFERENCE_RULE

        mod = mos_background if name == "mos_background" else qpb_corners
        return {"measure_version": mod.MEASURE_VERSION, "fwc_rule": FWC_REFERENCE_RULE}
    return {}

def _stat_key(p: Path) -> tuple:
    s = p.stat()
    return (str(p), s.st_mtime_ns, s.st_size)


def _sidecar(f: Path) -> dict[str, Any] | None:
    """The provenance record next to ``f``, None without one."""
    from tngxmm.core.provenance import table_provenance_path

    side = table_provenance_path(f)
    return json.loads(side.read_text()) if side.is_file() else None


def _upstream(files: dict[str, Path]) -> list[str]:
    """What is wrong with the inputs themselves: an input of the data package with a sidecar that does not say it was
    built from a clean tree, and the package's centre table when it is not current
    (:func:`tngxmm.photometry.build_obs_centres_guarded.check_centres_table`, which also checks the seed table it was
    copied from).  A table of the repository (``observations.csv``) is held by its commit and not checked here."""
    from tngxmm.core.provenance import portable_path

    bad = []
    centres = files.get("centres")
    chained = False
    if centres is not None:
        from tngxmm.photometry import build_obs_centres_guarded as gc

        try:
            chained = Path(centres).resolve() == gc.default_table_path().resolve()
        except OSError:
            pass
    if chained:
        try:
            gc.check_centres_table(centres, files.get("seed_lx"))
        except (ValueError, FileNotFoundError) as e:
            bad.append(f"centres: {e}")
    for name, f in files.items():
        if (chained and name in ("centres", "seed_lx")) or portable_path(f)["root"] != "data":
            continue        # checked with the centre table, or not of the data package (a repository table is held
                            # by its commit; a file outside both, root 'none', is the caller's)
        rec = _sidecar(f) if f.is_file() else None
        if rec is not None and rec.get("git_dirty") is not False:
            bad.append(f"{name}: {f.name} was itself built from an uncommitted tree (its sidecar records git_dirty "
                       f"{rec.get('git_dirty')!r} at {str(rec.get('git_hash'))[:12]}); rebuild it first")
    return bad

def check_inputs(path: str | Path, current: dict[str, str | Path], rebuild: str,
                 expect: dict[str, Any] | None = None) -> dict[str, Any]:
    """The provenance record of the table ``path`` when its sidecar (``<table>.provenance.json``, written by
    :func:`tngxmm.core.provenance.write_table_provenance`) describes the table's bytes (``table_sha256``), says the
    table was built from a committed, clean tree (``git_dirty`` false: the recorded commit is the code that built it),
    records every value of ``expect`` (the measurement version and FWC rule of :func:`measurement_record`) and, for
    every name of ``current``, the sha256 of that file as it is now (``inputs.<name>.sha256``), and when those inputs are
    clean and current themselves (:func:`_upstream`); else :class:`StaleTableError` naming what differs and
    ``rebuild``, the command that builds the table again.  A check that passed is remembered until a file or a sidecar
    changes (modification time and size)."""
    from tngxmm.core.hashing import sha256_file
    from tngxmm.core.provenance import table_provenance_path

    p = Path(path)
    side = table_provenance_path(p)
    if not side.is_file():
        raise StaleTableError(f"{p}: no provenance ({side.name}), so nothing says which inputs it was built from; "
                              f"rebuild it ({rebuild})")
    files = {k: Path(v) for k, v in current.items()}
    try:
        watched = [f for f in files.values() if f.is_file()]
        watched += [s for s in (table_provenance_path(f) for f in files.values()) if s.is_file()]
        key = (_stat_key(p), _stat_key(side), *(_stat_key(f) for f in watched),
               json.dumps(expect or {}, sort_keys=True))
    except OSError:
        key = None
    if key is not None and key in _CHECKED:
        return json.loads(side.read_text())
    rec = json.loads(side.read_text())
    bad = []
    if rec.get("table_sha256") != sha256_file(p):
        bad.append(f"the table itself (the sidecar describes other bytes, {str(rec.get('table_sha256'))[:12]})")
    if rec.get("git_dirty") is not False:
        bad.append(f"the code that built it (git_dirty {rec.get('git_dirty')!r} at {str(rec.get('git_hash'))[:12]}: "
                   "an uncommitted tree, whose code no commit holds)")
    for k, v in (expect or {}).items():
        if rec.get(k) != v:
            bad.append(f"{k}: measured under {rec.get(k)!r}, the current code is {v!r}")
    got = rec.get("inputs") or {}
    for name, f in files.items():
        want = sha256_file(f) if f.is_file() else None
        have = (got.get(name) or {}).get("sha256")
        if want is None or have != want:
            bad.append(f"{name}: built from sha256 {str(have)[:12]}, the current {f.name} has "
                       f"{str(want)[:12] if want else 'no file'}")
    bad += _upstream(files)
    if bad:
        raise StaleTableError(f"{p}: not built from the current inputs ({'; '.join(bad)}); rebuild it ({rebuild})")
    if key is not None:
        _CHECKED.add(key)
    return rec

def current_inputs(name: str) -> dict[str, Path]:
    """The files the package copy of ``name`` (one of :data:`CHECKED_TABLES`) must have been built from, as they are
    now: the guarded centre table for the tables measured about the cluster; the observation table for the core offsets
    (the pointings), the total column, the RASS sky, the active CCDs and the CCD maps (the observations they cover);
    the seed table for the measured tables (R500, hence the radius of their regions, and the condition table's
    theta500); for the two corner tables the anomalous-CCD table (the CCDs their scales leave out); for the condition
    table also the core-offset and background tables (it copies their values) and the observation table (the donors
    it holds beside those of the target lists); the reddening table for the total
    column (with the reddening of the observations added to the bank, a package input), the total column for the sky,
    the CCD maps for the active CCDs, the core offsets for the anomalous CCDs
    (their pointings), the active CCDs for the condition table (the observation dates of the solar angle).  The source
    thresholds and the catalogue positions take the observation table (the observations they cover); the rest of
    their inputs lies outside the package: the catalogue positions' published files are compared through their pins
    (:func:`measurement_record`), the source thresholds' PPS source lists with the local PPS cache where it holds
    them (:func:`pps_cache_mismatch`)."""
    from tngxmm.core.paths import data_asset

    if name in ("source_thresholds", "catalogue_positions"):
        return {"observations": observations_csv()}            # the observations they cover
    if name == "nh_total":
        out = {"observations": observations_csv(), "ebv_sfd": data_asset("tables") / "donor_ebv_sfd.csv"}
        from tngxmm.obs.extra_observations import EXTRA_FILES, INPUTS_EXTRA_DIR

        extra = INPUTS_EXTRA_DIR / EXTRA_FILES["extra_sky"]        # E(B-V) of the observations added to the bank
        if extra.is_file():
            out["ebv_sfd_extra"] = extra
        return out
    if name == "sky":
        return {"observations": observations_csv(), "nh_total": table_path("nh_total")}
    if name == "ccd_maps":
        return {"observations": observations_csv()}
    if name == "active_ccds":
        return {"observations": observations_csv(), "ccd_maps": table_path("ccd_maps")}
    from tngxmm.photometry.build_obs_centres_guarded import default_table_path as centres_path

    out = {"centres": centres_path()}
    if name == "core_offsets":
        out["observations"] = observations_csv()
        return out
    from tngxmm.photometry.build_obs_seed_lx import default_table_path as seed_path

    out["seed_lx"] = seed_path()
    if name in ("mos_background", "qpb_corners"):
        out["soft_ccd"] = table_path("soft_ccd")
    if name == "soft_ccd":
        out["core_offsets"] = table_path("core_offsets")
    if name == "conditions":
        out.update(core_offsets=table_path("core_offsets"), background=table_path("background"),
                   active_ccds=table_path("active_ccds"), observations=observations_csv())
    return out

def check_table(name: str, path: str | Path | None = None) -> dict[str, Any]:
    """:func:`check_inputs` of ``name`` (default its package copy) against :func:`current_inputs` and
    :func:`measurement_record`; returns the provenance record.  ``ValueError`` for a name that is not one of
    :data:`CHECKED_TABLES`."""
    if name not in CHECKED_TABLES:
        raise ValueError(f"table {name!r} has no provenance check; checked: {sorted(CHECKED_TABLES)}")
    rec = check_inputs(path or table_path(name), current_inputs(name), CHECKED_TABLES[name],
                       measurement_record(name))
    if name == "source_thresholds":
        bad = pps_cache_mismatch(path or table_path(name))
        if bad:
            raise StaleTableError(f"{path or table_path(name)}: not built from the PPS source lists of the local "
                                  f"cache ({'; '.join(bad[:5])}{'; ...' if len(bad) > 5 else ''}, {len(bad)} in all); "
                                  f"rebuild it ({CHECKED_TABLES[name]})")
    return rec

def pps_cache_mismatch(path: str | Path, cache: str | Path | None = None) -> list[str]:
    """What differs between the source-threshold table ``path`` and the PPS source lists of the local cache (default
    :func:`pps_cache`): a row's list (``pps_file``) present in the cache with another sha256 than the row records
    (``pps_sha256``), and a row flagged ``no_pps_list`` whose observation now has a list there.  A list the cache does
    not hold is not compared (the cache of a machine may be partial, or absent: then nothing is compared).  Remembered
    until the table or the cache directory changes."""
    import pandas as pd

    from tngxmm.core.hashing import sha256_file

    p, c = Path(path), Path(cache) if cache else pps_cache()
    if not c.is_dir():
        return []
    t = pd.read_csv(p, dtype=str, keep_default_na=False)
    rows = [(r.obsid, c / r.pps_file, r.pps_sha256) for r in t.itertuples()
            if r.pps_file and r.pps_sha256] if {"pps_file", "pps_sha256"} <= set(t.columns) else []
    held = [(o, f, h) for o, f, h in rows if f.is_file()]
    try:
        key = ("pps", _stat_key(p), _stat_key(c), *(_stat_key(f) for _, f, _ in held))
    except OSError:
        key = None
    if key is not None and key in _CHECKED:
        return []
    bad = []
    for obsid, f, h in held:
        now = sha256_file(f)
        if now != h:
            bad.append(f"{obsid}: {f.name} has sha256 {now[:12]}, the table records {h[:12]}")
    if "flag" in t.columns:
        from tngxmm.massfit.build_source_thresholds import pps_path_for

        for r in t[t["flag"] == "no_pps_list"].itertuples():
            f = pps_path_for(r.obsid, c)                # the builder's own reading of the cache
            if f is not None:
                bad.append(f"{r.obsid}: flagged no_pps_list, the cache now holds {f.name}")
    if not bad and key is not None:
        _CHECKED.add(key)
    return bad

def required_tables(opt: Any, conditions: dict[str, Any] | None = None) -> dict[str, str]:
    """The donor tables the instrument stage reads for ``opt`` (:class:`InstrumentOptions`):
    ``{logical name: override path or ''}``.  The gates are those of
    :func:`tngxmm.sim.generate_view.instrument_view_kwargs` (tests/unit/test_donor_coverage.py runs both and
    compares); ``conditions`` (:func:`tngxmm.instrument.conditions_donor.conditions_of`) adds the condition
    table under the rule 'nn'."""
    req: dict[str, str] = {}
    if opt.background == "donor" or opt.live_time == "donor-expmap":
        req["background"] = opt.background_table
    if opt.galactic_nh != "hi4pi":
        req["nh_total"] = opt.nh_total_table
    if opt.mos_soft_ccd != "off":
        req["soft_ccd"] = opt.mos_soft_ccd_table
    if opt.soft_protons != "off":
        req["mos_background"] = opt.mos_background_table
    if opt.qpb_level != "in-fov" or getattr(opt, "qpb_units", "none") == "corners":
        req["qpb_corners"] = opt.mos_background_table
    if getattr(opt, "sky_model", "xmm_simulator") == "donor-rass":
        req["sky"] = getattr(opt, "sky_table", "")
    if (conditions or {}).get("rule", "donor") == "nn":
        req["conditions"] = str((conditions or {}).get("table") or "")
    elif getattr(opt, "soft_proton_map", "flat") == "flare" and opt.soft_protons != "off":
        req["conditions"] = ""                     # the donor's theta500 (normalisation region of the flare maps)
    return req




def _unmeasurable_soft_ccd(obsid: str) -> bool:
    """Whether no MOS camera of the donor has a PPS exposure map (the active-CCD table's source), so its absence
    from the soft-CCD table is by design (:func:`tngxmm.instrument.donor_soft_ccd.donor_soft_ccd`)."""
    from tngxmm.instrument.active_ccds import donor_camera_ccds, read_active_ccds

    try:
        cams = donor_camera_ccds(read_active_ccds(), obsid)
    except KeyError:

    return not any(str(v.get("source")) == "pps-expmap" for c, v in cams.items() if c.startswith("MOS"))

def qpb_unit_gaps(donors: Any, table: Any = None) -> dict[str, list[str]]:
    """``{obsid: cameras}`` of the donors that have a camera with an exposure but no corner unit factor
    (:func:`tngxmm.instrument.donor_corner_qpb.donor_unit_factors`: no corner count in any CCD or quadrant, as for
    a pn exposure in a window mode).  Under ``qpb_units = 'corners'`` such a donor cannot lend its conditions: the
    particle background of that camera has no measured level per unit.  ``table``: the corner table (default: the
    data package's)."""
    from tngxmm.instrument.donor_corner_qpb import CAMS, donor_unit_factors

    t = read_table("qpb_corners", "") if table is None else table
    have = set(t.obsid.astype(str).str.zfill(10))
    out: dict[str, list[str]] = {}
    for d in sorted({str(x).zfill(10) for x in donors}):
        if d not in have:
            continue                                  # absent donors are reported by missing_donors
        row = t[t.obsid.astype(str).str.zfill(10) == d].iloc[0]
        factors = donor_unit_factors(t, d)
        lack = [cam for cam, suf in CAMS.items()
                if cam not in factors and float(row.get(f"n_exp_{suf}", 0) or 0) > 0]
        if lack:
            out[d] = lack
    return out

def missing_donors(donors: Any, opt: Any, conditions: dict[str, Any] | None = None,
                   image_root: str | Path | None = None) -> dict[str, list[str]]:
    """``{table: sorted donors it lacks}`` for the donors (obsids) of a campaign: every table of
    :func:`required_tables`, under ``detector_footprint != 'mask'`` the PPS exposure maps
    (``'exposure_maps'``: no map of any camera in ``image_root``) and under ``responses != 'packaged'`` the
    per-observation responses (``'obs_responses'``: :func:`tngxmm.instrument.obs_responses.missing_responses`).
    Under the rule 'nn' the usable conditions donors (rows of the condition table with ``pool``) are held to the
    same tables, since a view is observed with the tables' rows of the donor it draws.  Empty values are left out."""
    keys = sorted({str(d).zfill(10) for d in donors})
    out: dict[str, list[str]] = {}
    cond_table = None
    pool_donors: set[str] = set()
    req = required_tables(opt, conditions)
    if "conditions" in req:
        cond_table = read_table("conditions", req["conditions"] or None)
        pool = cond_table[cond_table["pool"].astype(bool)] if "pool" in cond_table else cond_table
        pool_donors = set(pool.obsid.astype(str).str.zfill(10))
    for name, override in req.items():
        have = set(read_table(name, override or None).obsid.astype(str).str.zfill(10))
        want = set(keys)
        if cond_table is not None and name != "conditions":
            want |= pool_donors
        gone = sorted(want - have)
        if name == "soft_ccd":            # no PPS MOS image: nothing to measure, `donor_soft_ccd` gives Unmeasured
            gone = [d for d in gone if not _unmeasurable_soft_ccd(d)]
        if gone:
            out[name] = gone
    if opt.detector_footprint != "mask":
        from tngxmm.instrument.donor_footprint import CAMERA_CODES, find_expmaps

        if image_root is None:
            from tngxmm.donors.build_core_offsets import DEFAULT_IMAGE_ROOT as image_root
        keys_maps = sorted(set(keys) | pool_donors)     # a conditions donor is observed through its own maps
        gone = [d for d in keys_maps if not any(find_expmaps(d, c, image_root) for c in CAMERA_CODES)]
        if gone:
            out["exposure_maps"] = gone
    if getattr(opt, "responses", "packaged") != "packaged":
        from tngxmm.instrument.obs_responses import check_set, missing_responses

        root = getattr(opt, "response_set", "") or None
        check_set(root)                  # a manifest, and the regions of the current centre table, before any photon
        gone = sorted(missing_responses(set(keys) | pool_donors, root))
        if gone:
            out["obs_responses"] = gone
    return out

def core_offset_gaps(donors: Any) -> list[str]:

def soft_proton_gaps(donors: Any, opt: Any, spec: Any = None) -> list[str]:
    """``'<obsid> <camera>'`` of the donors (obsids) whose soft-proton rate the instrument could not set under ``opt``
    (``soft_protons`` on): a camera without a measured rate under ``sp_rate_unmeasured = 'refuse'``, or without a
    substitute under ``'nearest-in-fold'`` (:func:`tngxmm.sim.generate_view.view_soft_protons`, the lookup of the
    stage).  Empty with the soft protons off."""
    if opt.soft_protons == "off":
        return []
    from tngxmm.instrument.donor_inout_sp import UnmeasuredSoftProtons
    from tngxmm.sim.generate_view import view_soft_protons

    table = read_table("mos_background", opt.mos_background_table or None)
    out = []
    for d in sorted({str(x).zfill(10) for x in donors}):
        try:
            view_soft_protons(spec, opt, d, table)
        except UnmeasuredSoftProtons as e:
            out.append(f"{d} ({str(e)[:120]})")
        except KeyError:
            continue                                  # not in the IN/OUT table: reported by missing_donors
    return out

def check_donor_tables(donors: Any, spec: Any, image_root: str | Path | None = None) -> None:
    """Fail fast, before any view is generated, when a donor of the campaign is absent from a donor
    table the preset reads (:func:`missing_donors`), except the donors the preset excludes from the generation
    (``runtime.exclude_donors``, :func:`tngxmm.core.pool.excluded_donors`): raises :class:`DonorTablesError` with the list per
    table.  Under the conditions rule ``donor`` with the placement ``donor-offset`` every donor needs a usable core
    offset (:func:`core_offset_gaps`); with the soft protons on every donor that can lend its conditions (the pool
    under ``nn``) needs a rate for every camera, measured or substituted (:func:`soft_proton_gaps`).  First, every
    data-package file a view of the preset is made from must be current
    (:func:`tngxmm.sim.data_inputs.check_stage_inputs`).  Called by ``sim.plan_campaign`` (all targets) and by
    ``campaign run`` (the snapshot's)."""
    from tngxmm.instrument.conditions_donor import conditions_of
    from tngxmm.instrument.simulate_instrument import InstrumentOptions

    from tngxmm.core.pool import excluded_donors

    from tngxmm.sim.data_inputs import check_stage_inputs

    check_stage_inputs(spec)                         # every data-package file a view is made from, current
    skip = excluded_donors(spec)                     # the generation never asks these donors for conditions
    donors = [d for d in donors if str(d).zfill(10) not in skip]
    if not donors:
        return
    opt = InstrumentOptions.from_spec(spec, "")
    cond = conditions_of(spec)
    gone = missing_donors(donors, opt, cond, image_root)
    if not gone:
        lenders = {str(d).zfill(10) for d in donors}
        if cond.get("rule", "donor") == "nn":
            ct = read_table("conditions", cond.get("table") or None)
            lenders |= set(ct[ct["pool"].astype(bool)].obsid.astype(str).str.zfill(10))
        elif opt.placement == "donor-offset":
            core = core_offset_gaps(donors)
            if core:
                gone["core_offsets"] = core
        sp = soft_proton_gaps(lenders, opt, spec)
        if sp:
            gone["soft_protons"] = sp
        if getattr(opt, "qpb_units", "none") == "corners" and cond.get("rule", "donor") == "donor":
            qu = qpb_unit_gaps(donors, read_table("qpb_corners", opt.mos_background_table))
            if qu:
                gone["qpb_units (a camera with an exposure and no corner counts)"] = sorted(qu)
    if gone:
        lines = [f"  {t}: {len(v)} donor(s) {v[:12]}{' ...' if len(v) > 12 else ''}" for t, v in gone.items()]
        hint = ("\nA donor without PPS products cannot lend its observing conditions: exclude it from the targets "
                "(runtime.exclude_donors of the preset, with the reason)")
        if "obs_responses" in gone:
            hint += ("\nobs_responses: build the per-observation set first (tngxmm instrument obs-responses plan | run "
                     "| manifest; the node job OBS_RESPONSES) or exclude the donors it cannot cover")
        raise DonorTablesError(
            f"preset {spec.name}: donors of the targets missing from the donor tables it reads:\n"
            + "\n".join(lines) + hint)
