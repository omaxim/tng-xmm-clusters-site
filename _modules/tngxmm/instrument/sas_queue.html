Source code for tngxmm.instrument.sas_queue

"""Single-node work queue of SAS source detections: one single-core ``run_detection`` per core.

:mod:`tngxmm.instrument.sas_detect` runs ONE detection (a mock view or a real observation).
A campaign needs hundreds (637 real observations, thousands of mock views) and a detection
is single-threaded work of minutes, so a node is used as a pool of single-core workers, each
pinned to its own core (``taskset``), instead of a few detections on four cores.  This module
is that pool and the builders of its queue.  It is the one implementation behind
``tngxmm instrument sas-queue build-real | build-mock | run``, the Kedro node
``instrument.run_sas_detection`` and ``slurm/sas_detect.sh``.

A queue is a JSON list of entries:

``{"name", "kind": "real", "files": [PPS file names], "src_dir", "out", "final"}``
    a real observation.  One cluster directory can hold the products of several observations
    and the adapter globs ``P*``, so the entry is staged as a directory of symbolic links to
    ITS OWN files only (``<tmp>/<name>/``, removed after the job).  ``final`` (or null) is
    the directory the kept files are also copied to: ``$TNGXMM_DATA/sas_detect/<obsid>/``,
    where the fit reads them.
``{"name", "kind": "mock", "view", "out", "final": null}``
    a mock view (``<view>/percam`` holds the SAS-ready products).

Any entry may carry ``"detect": {...}``: fields of :class:`DetectSettings` that replace the run's for that entry only
(the sensitivity test of the DET_ML cut: ``{"likemin": 5, "mlmin": 6}`` with ``out`` in another directory).

Output layout (that of ``run_detection``; ``KEEP``): ``<out>/sas_detect/{emllist.fits,
sources.csv, edetect_chain.log, command.txt, provenance.json}``.

Resumable: an entry is *finished* when ``<out>/sas_detect`` holds a complete detection
(:func:`tngxmm.instrument.sas_detect.detection_complete`: the done-marker ``provenance.json``, the
``emllist.fits`` it records by sha256 and a whole ``sources.csv``) and, when ``final`` is set, the same
``sources.csv`` and ``provenance.json`` there.  :func:`run_queue` runs the others; an entry that raises is
tried once more (two attempts), then logged as FAIL and the queue goes on.  Every entry appends
one record to ``progress.jsonl``.  A re-submission therefore only repeats what is missing.

A full disk is not a failed entry (:mod:`tngxmm.core.diskguard`): the entry is recorded as ``DISK``, not
tried again in this run and not counted as failed; no new entry is taken, the running ones end, and the
caller exits with ``diskguard.EXIT_DISK``.  The scratch and the first output directory are probed before
the first entry and every few minutes.

Environment: ``TNGXMM_SASRUN`` (the ``sasrun`` wrapper; site variable ``TNGXMM_SITE_SASRUN``),
``OMP_NUM_THREADS=1`` (set here if absent).  Nothing here talks to SLURM.
"""

def finished(entry: dict[str, Any]) -> bool:
    """True when the entry's outputs are complete (see the module docstring)."""
    s = Path(entry["out"]) / "sas_detect"
    ok = detection_complete(s)
    if ok and entry.get("final"):
        fin = Path(entry["final"])
        ok = ((fin / "sources.csv").is_file() and (fin / "sources.csv").read_bytes() == (s / "sources.csv").read_bytes()
              and diskguard.read_json(fin / "provenance.json") == diskguard.read_json(s / "provenance.json"))
    return ok




def _scan_real(img_root: Path, wanted: set[str]) -> dict[str, dict[str, list[str]]]:
    """obsid -> product directory -> file names ``P<obsid>*``, for the wanted obsids.

    Pure file-name scan of ``<img_root>/<group>/<cluster>/`` (no FITS reads).  An obsid can
    sit in several cluster directories (a cluster in two surveys)."""
    where: dict[str, dict[str, list[str]]] = {}
    for g in os.scandir(img_root):
        if not g.is_dir():
            continue
        for c in os.scandir(g.path):
            if not c.is_dir():
                continue
            for f in os.scandir(c.path):
                n = f.name
                if n.startswith("P") and len(n) > 11 and n[1:11] in wanted:
                    where.setdefault(n[1:11], {}).setdefault(c.path, []).append(n)
    return where

def build_real_queue(observations: Any, img_root: str | Path, out_root: str | Path,
                     final_root: str | Path | None) -> tuple[list[dict[str, Any]], list[dict[str, str]]]:
    """One queue entry per real observation, and a product report.

    ``observations``: a DataFrame (or iterable of dicts) with the column ``obsid`` (any
    width; zero-padded to ten digits) and, if present, ``name`` (the cluster, for the
    report).  ``img_root``: the PPS image tree ``<img_root>/<group>/<cluster>/P<obsid>*``.
    Outputs go to ``<out_root>/<obsid>/sas_detect/`` and, with ``final_root``, also to
    ``<final_root>/<obsid>/`` (``$TNGXMM_DATA/sas_detect``, where the fit reads them).

    An obsid found in several cluster directories uses the one with the most IMAGE/EXPMAP
    files (the others are named in the report).  A camera is usable when one of its exposure
    sets has band 2 and at least four bands with both IMAGE and EXPMAP; an observation with
    no usable camera is reported (``NO_PRODUCTS`` / ``NO_USABLE_CAMERA``) and not queued.
    The report rows are ``obsid, name, status, note`` with status ``ok``, ``ok_note``
    (several directories or exposure sets) or ``CAMERAS_MISSING_<cams>``."""
    rows = observations.to_dict("records") if hasattr(observations, "to_dict") else list(observations)
    want = {str(r["obsid"]).zfill(10): r for r in rows}
    img_root = Path(img_root)
    where = _scan_real(img_root, set(want))
    jobs: list[dict[str, Any]] = []
    report: list[dict[str, str]] = []
    for obs, r in want.items():
        label = str(r.get("name", obs))
        dirs = where.get(obs, {})
        note: list[str] = []
        if not dirs:
            report.append(dict(obsid=obs, name=label, status="NO_PRODUCTS",
                               note="no file P<obsid>* in any cluster directory"))
            continue
        best = max(dirs, key=lambda d: sum(1 for n in dirs[d] if PPS_PATTERN.match(n)))
        if len(dirs) > 1:
            note.append("in %d dirs (%s); used %s" % (
                len(dirs), ", ".join(os.path.relpath(d, img_root) for d in dirs),
                os.path.relpath(best, img_root)))
        sets: dict[tuple[str, str], dict[tuple[str, int], bool]] = {}
        for n in dirs[best]:
            m = PPS_PATTERN.match(n)
            if m:
                sets.setdefault((m.group(2), m.group(3)), {})[(m.group(4), int(m.group(5)))] = True
        ok_cams: list[str] = []
        exps: dict[str, list[tuple[str, list[int]]]] = {}
        for (cam, ex), d in sorted(sets.items()):
            full = [b for b in (1, 2, 3, 4, 5) if ("IMAGE_", b) in d and ("EXPMAP", b) in d]
            exps.setdefault(cam, []).append((ex, full))
        for cam, lst in exps.items():
            if any(len(f) >= 4 and 2 in f for _, f in lst):
                ok_cams.append(cam)
            else:
                note.append("%s: no exposure set with band 2 and >=4 bands (%s)" % (cam, lst))
        if any(len(lst) > 1 for lst in exps.values()):
            note.append("several exposure sets: " + "; ".join(
                "%s %s" % (c, [e for e, _ in lst]) for c, lst in exps.items() if len(lst) > 1))
        missing = [c for c in ("M1", "M2", "PN") if c not in ok_cams]
        status = ("NO_USABLE_CAMERA" if not ok_cams
                  else "CAMERAS_MISSING_" + "+".join(missing) if missing
                  else "ok_note" if note else "ok")
        report.append(dict(obsid=obs, name=label, status=status, note=" | ".join(note)))
        if ok_cams:
            jobs.append(dict(
                obsid=obs, name=obs, kind="real", cluster=label, src_dir=best,
                files=sorted(n for n in dirs[best] if PPS_PATTERN.match(n)),
                out=str(Path(out_root) / obs),
                final=str(Path(final_root) / obs) if final_root else None))
    return jobs, report

def mock_view_name(view_dir: str | Path) -> str:
    """Unique entry name of a view in the tree layout: ``<sim>_<snap>_<halo>_<view>`` (all
    views end in ``view<k>``, so the last directory alone is not unique).  The same name

def target_view_dirs(targets: str | Path, tree: str | Path) -> list[Path]:
    """The view directories of every row of a target list (columns ``sim``, ``snap``, ``halo_id``, ``view``) in the
    campaign tree ``tree`` (:func:`tngxmm.core.pool.view_dir`), in the order of the list.  Nothing is checked on

        rows = list(csv.DictReader(f))
    return [view_dir(tree, r["sim"], int(r["snap"]), int(r["halo_id"]), int(r["view"])) for r in rows]

def build_mock_queue(view_dirs: Iterable[str | Path] | dict[str, str | Path]) -> list[dict[str, Any]]:
    """One queue entry per mock view: the SAS-ready products stay in the view and the source
    list lands in ``<view>/sas_detect/`` (where ``massfit fit-hydrostatic`` reads it).

    ``view_dirs``: directories (named by :func:`mock_view_name`) or a ``{name: directory}``
    mapping.  A directory without ``percam/`` is a mistake and raises, not a silent skip."""
    items = view_dirs.items() if isinstance(view_dirs, dict) \
        else ((mock_view_name(v), v) for v in view_dirs)
    jobs = []
    for name, v in items:
        v = Path(v)
        if not (v / "percam").is_dir():
            raise FileNotFoundError(f"{v}: no percam/ directory (not a SAS-ready view)")
        jobs.append(dict(name=str(name), kind="mock", view=str(v), out=str(v), final=None))
    names = [j["name"] for j in jobs]
    if len(set(names)) != len(names):
        raise ValueError("duplicate queue entry names: "
                         f"{sorted({n for n in names if names.count(n) > 1})}")
    return jobs

def parse_cores(txt: str) -> list[int]:
    """``"0-3,8,10-11"`` -> ``[0, 1, 2, 3, 8, 10, 11]``."""
    out: list[int] = []
    for part in txt.split(","):
        lo, _, hi = part.partition("-")
        out += list(range(int(lo), int(hi or lo) + 1))
    return out

def stage_links(entry: dict[str, Any], tmp: str | Path) -> Path:
    """Stage a real entry as ``<tmp>/<name>/``: symbolic links to its own files only."""
    d = Path(tmp) / entry["name"]
    if d.exists():
        shutil.rmtree(d)
    d.mkdir(parents=True)
    for n in entry["files"]:
        os.symlink(Path(entry["src_dir"]) / n, d / n)
    return d

def select_cores(cores: str | list[int] | None, workers: int | None) -> list[int]:
    """The cores of the pool: ``cores`` (a cpu list) or the cores this process may use, minus
    :data:`RESERVED_CORES` when there are more than four; at most ``workers`` of them."""
    if isinstance(cores, str):
        cores = parse_cores(cores)
    if not cores:
        cores = sorted(os.sched_getaffinity(0))
        if len(cores) > 4:
            cores = cores[:-RESERVED_CORES]
    return list(cores)[:workers] if workers else list(cores)

def stage_only(jobs: list[dict[str, Any]], tmp: str | Path,
               settings: DetectSettings | None = None) -> list[dict[str, Any]]:
    """Stage the inputs of ``jobs`` and return, per entry, what the detection would run:
    ``name, kind, cameras, bands, filters, date_obs, command`` (no SAS; the staged files are
    removed again).  ``command`` is the ``cifbuild`` + ``edetect_chain`` text (the pinned ``cpus`` is a run-time setting of the worker and does not enter the command)."""
    base = settings or DetectSettings()
    out = []
    for j in jobs:
        src = stage_links(j, tmp) if j["kind"] == "real" else Path(j["view"])
        work = Path(tmp) / ("stage_" + j["name"])
        try:
            st = stage_inputs(src, work, j["kind"], base.bands)
            cmd = edetect_chain_command(st, _entry_settings(base, j))
        finally:
            shutil.rmtree(work, ignore_errors=True)
            if j["kind"] == "real":
                shutil.rmtree(src, ignore_errors=True)
        out.append(dict(name=j["name"], kind=j["kind"], cameras=st.cameras, bands=st.bands,
                        filters=st.filters, date_obs=st.date_obs, command=cmd))
    return out




def _with_cpus(settings: DetectSettings, cpus: str) -> DetectSettings:

def _entry_settings(base: DetectSettings, entry: dict[str, Any]) -> DetectSettings:
    """``base`` with the entry's own ``detect`` fields (none: ``base``)."""
    import dataclasses

    return dataclasses.replace(base, **entry["detect"]) if entry.get("detect") else base


def _copy_final(entry: dict[str, Any]) -> None:
    """Copy the kept files of ``<out>/sas_detect`` to ``final``: each atomically, the done-marker
    (``provenance.json``, the last of :data:`KEEP`) after the others."""
    fin = Path(entry["final"])
    fin.mkdir(parents=True, exist_ok=True)
    (fin / "provenance.json").unlink(missing_ok=True)
    for f in KEEP:
        src = Path(entry["out"]) / "sas_detect" / f
        if src.exists():
            diskguard.copy_file(src, fin / f)

def run_queue(jobs: list[dict[str, Any]], tmp: str | Path, *, cores: str | list[int] | None = None,
              workers: int | None = None, limit: int | None = None, only: str | None = None,
              fake: bool = False, progress: str | Path | None = None, stagger: float = 1.0,
              settings: DetectSettings | None = None, attempts: int = 2) -> dict[str, Any]:
    """Run the unfinished entries of ``jobs`` with one worker per core; return the summary.

    ``only``: comma list of entry names; ``limit``: the first K entries; both applied
    before the resume check.  Each worker takes the next entry, stages it (real: links to
    its own files), runs :func:`run_detection` pinned to its core (``force=True``, work
    directory ``<tmp>/work_<name>``, removed afterwards), copies the kept files to ``final``,
    and appends a record to ``progress`` (JSON lines).  An entry that raises is retried up to
    ``attempts`` times in all, then recorded as ``FAIL``; the queue always goes on.
    ``stagger`` seconds between the worker start-ups spread the container start-ups.
    ``fake`` replaces SAS by :func:`_fake_detection` (tests).  SIGTERM/SIGINT stop taking
    new entries (the running ones finish), and so does the first entry stopped by a full disk (module
    docstring).  Returns ``{"entries", "todo", "done", "failed", "disk", "wall_s", "records"}``; ``disk``
    counts the entries a full disk interrupted, 0 in a run that was not stopped for it."""
    os.environ.setdefault("OMP_NUM_THREADS", "1")
    base = settings or DetectSettings()
    tmp = Path(tmp)
    tmp.mkdir(parents=True, exist_ok=True)
    sel = list(jobs)
    if only:
        sel = [j for j in sel if j["name"] in only.split(",")]
    if limit:
        sel = sel[:limit]
    todo = [j for j in sel if not finished(j)]
    print(f"{len(sel)} entries in the queue, {len(sel) - len(todo)} finished, {len(todo)} to run",
          flush=True)
    pool = select_cores(cores, workers)
    print(f"{len(pool)} workers on cores {pool[0]}..{pool[-1]} of {len(os.sched_getaffinity(0))} "
          f"allowed; host {os.uname().nodename}", flush=True)
    q: queue.Queue = queue.Queue()
    for j in todo:
        q.put(j)
    lock, stop, t0 = threading.Lock(), threading.Event(), time.time()
    count = {"done": 0, "fail": 0, "disk": 0}
    diskguard.remove_stale_tmp(tmp, older_than_s=3600.0)        # probe files of a killed run
    guard = diskguard.SpaceGuard(tmp, *([todo[0]["out"]] if todo else []))
    disk_why: list[str] = []
    try:
        guard.check(force=True)                 # before any detection starts
    except diskguard.DiskFull as e:
        disk_why.append(str(e))
        stop.set()
    records: list[dict[str, Any]] = []
    if threading.current_thread() is threading.main_thread():
        signal.signal(signal.SIGTERM, lambda *_: stop.set())
        signal.signal(signal.SIGINT, lambda *_: stop.set())
    log = Path(progress) if progress else tmp / "progress.jsonl"
    log.parent.mkdir(parents=True, exist_ok=True)

    def worker(core: int, delay: float) -> None:
        time.sleep(delay)
        while not stop.is_set():
            with lock:
                try:
                    guard.check()
                except diskguard.DiskFull as e:
                    disk_why.append(str(e))
                    stop.set()
                    return
            try:
                j = q.get_nowait()
            except queue.Empty:
                return
            t1 = time.time()
            rec: dict[str, Any] = dict(name=j["name"], core=core, host=os.uname().nodename)
            for attempt in range(1, attempts + 1):
                src = None
                try:
                    src = stage_links(j, tmp) if j["kind"] == "real" else Path(j["view"])
                    r = run_detection(src, j["out"], kind=j["kind"],
                                      settings=_with_cpus(_entry_settings(base, j), str(core)), keep_work=False,
                                      force=True, run=_fake_detection if fake else None,
                                      work_dir=tmp / ("work_" + j["name"]))
                    if j.get("final"):
                        _copy_final(j)
                    rec.update(status="ok", n_sources=r.get("n_sources"),
                               n_det_ml_gt_10=r.get("n_det_ml_gt_10"), attempt=attempt,
                               cameras=r.get("cameras"), bands=r.get("bands"))
                    break
                except Exception as e:                      # noqa: BLE001  keep the queue going
                    rec.update(status="FAIL", error=(str(e)[-400:] or repr(e)), attempt=attempt)
                    if diskguard.failure_status(e, j["out"]) is not None:
                        # the disk, not the entry: no second attempt, no new entries
                        rec["status"] = "DISK"
                        stop.set()
                        break
                finally:
                    shutil.rmtree(tmp / ("work_" + j["name"]), ignore_errors=True)
                    if j["kind"] == "real" and src is not None:
                        shutil.rmtree(src, ignore_errors=True)
            rec["seconds"] = round(time.time() - t1, 1)
            rec["t_end"] = time.strftime("%Y-%m-%dT%H:%M:%S")
            with lock:
                count[{"ok": "done", "DISK": "disk"}.get(rec["status"], "fail")] += 1
                records.append(rec)
                if rec["status"] == "DISK":
                    disk_why.append(rec.get("error", ""))
                try:
                    with open(log, "a") as fh:
                        fh.write(json.dumps(rec) + "\n")
                except OSError as e:                         # the progress log is not worth a crash
                    print(f"progress log {log}: {e}", flush=True)
                    if diskguard.is_disk_error(e):
                        disk_why.append(str(e))
                        stop.set()
                n = count["done"] + count["fail"] + count["disk"]
                el = time.time() - t0
                print(f"[{n}/{len(todo)}] {j['name']} {rec['status']} {rec['seconds']:.0f}s "
                      f"N(ML>10)={rec.get('n_det_ml_gt_10')} elapsed {el / 60:.0f} min, "
                      f"ETA {el / n * (len(todo) - n) / 60:.0f} min", flush=True)

    threads = [threading.Thread(target=worker, args=(c, i * stagger)) for i, c in enumerate(pool)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    wall = time.time() - t0
    print(f"done {count['done']}, failed {count['fail']}, wall {wall / 60:.1f} min", flush=True)
    n_disk = 0
    if disk_why:
        n_disk = max(count["disk"], 1)
        print(diskguard.disk_message("SAS queue", count["disk"], tmp, disk_why[0]), flush=True)
    return dict(entries=len(sel), todo=len(todo), done=count["done"], failed=count["fail"], disk=n_disk,
                wall_s=round(wall, 1), records=records)

def real_entry_from_dir(directory: str | Path, out: str | Path | None = None) -> dict[str, Any]:
    """A real entry for a directory of PPS products (all its ``P*`` product files), written
    to ``<out or directory>/sas_detect``.  For a directory that holds ONE observation (the
    Kedro node's ``observations`` list); a cluster directory with several observations is
    queued per obsid by :func:`build_real_queue`."""
    d = Path(directory).resolve()
    files = sorted(n for n in os.listdir(d) if PPS_PATTERN.match(n))
    return dict(name=d.name, kind="real", src_dir=str(d), files=files,
                out=str(out or d), final=None)
