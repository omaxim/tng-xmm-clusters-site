Source code for tngxmm.instrument.sas_detect

"""ESA's standard EPIC source detection (``edetect_chain``) on a mock view or a real observation.

One function, one set of settings, for both: the per-camera images and exposure maps of a
mock view (``<view>/percam``, the SAS-ready products of ``instrument.sas_headers = 'on'``)
or of a real observation (the PPS products of one exposure per camera) are staged under the
names ``edetect_chain`` is documented with (``<cam>_image_b<k>.fits``, exposure map
``<cam>_image_b<k>exp.fits``), the calibration index file is built for the observation date
of the product headers (``cifbuild`` with ``withobservationdate``; for a mock the date of its
donor), and one ``edetect_chain`` call detects on the three cameras and five bands jointly.
The source list is written next to the products:

``<out>/sas_detect/emllist.fits``    the ``emldetect`` list (extension ``SRCLIST``)
``<out>/sas_detect/sources.csv``     its summary rows (one per source: position, DET_ML, extent,
                                     EPIC flux, per-band rates), the table the comparisons read
``<out>/sas_detect/edetect_chain.log``, ``provenance.json`` (settings, SAS version, inputs)

``provenance.json`` is the done-marker of a detection: it is removed before SAS starts, written last
(atomically, like the two lists) and holds the sha256 of ``emllist.fits`` and the number of rows of
``sources.csv``, so :func:`detection_complete` can tell a finished detection from the remains of one that
was killed or ran out of disk.  SAS writes its own files in the work directory, not atomically; nothing
is copied out of it before the task has ended without error and its source list reads to the end.  A SAS
failure with a disk-full message in its output, or after which a probe write in the work or output
directory fails, raises :class:`tngxmm.core.diskguard.DiskFull` (the detection is interrupted, not failed).

Settings (``DetectSettings``; the source of every value is METHOD.md of the SAS prototype,
``SASDET``, which follows the ESA SAS threads "EPIC source finding with edetect_chain" and
the 3XMM-DR6 User Guide):

* bands 1-5 = PI 200-500, 500-1000, 1000-2000, 2000-4500, 4500-12000 eV, per camera
  (the PPS / 4XMM bands; a product set without band 1 runs on bands 2-5);
* ``witheexpmap=no``: the exposure maps are those of the products (mocks have no event list;
  a PPS exposure map is the product of ``eexpmap`` on the observation's attitude);
* ``esp_nsplinenodes=16``, ``esen_mlmin=15`` (thread values; task defaults 13 and 10), every
  other parameter at the ``edetect_chain.par`` default (``likemin=10``: the PPS cut
  ``EP_DET_ML > 10``);
* ECFs: 3XMM-DR6 User Guide Table 8, by camera and filter (they scale the FLUX only);
* ``OMP_NUM_THREADS=1`` and, on the host, the cores of ``DetectSettings.cpus`` (login node: 4).

``witheexpmap=no`` makes the event sets and the attitude set arguments that are only parsed
(``INSTRUME``); the first staged image stands in for both.  SAS runs inside the HEASoft
container through the ``sasrun`` wrapper (``$TNGXMM_SASRUN``, site variable
``TNGXMM_SITE_SASRUN``; ``HEADASNOQUERY`` and ``HEADASPROMPT`` set by its environment file,
without them ``ftkeypar`` fails in batch).

References
----------
Gabriel et al. 2004 (SAS); ESA SAS threads "EPIC source finding with edetect_chain" (SAS v22.0);
3XMM-DR6 User Guide, Table 8 (energy conversion factors; Mateos et al. 2009); the ``emldetect``
documentation (Cruddace et al. 1988; Watson et al. 2009; Traulsen et al. 2019, 2020).

    def edetect_arguments(self) -> dict[str, str]:
        """The ``edetect_chain`` parameters that are not defaults of the task."""
        args = {"esp_nsplinenodes": str(self.esp_nsplinenodes), "esen_mlmin": str(self.esen_mlmin),

            args["mlmin"] = f"{self.mlmin:g}"
        return args

def sasrun_path(settings: DetectSettings | None = None) -> str:
    """The ``sasrun`` wrapper: settings, ``$TNGXMM_SASRUN``, then the site variable
    ``TNGXMM_SITE_SASRUN``."""
    from tngxmm.core import site

    path = (settings.sasrun if settings else None) or os.environ.get("TNGXMM_SASRUN") \
        or site.get("TNGXMM_SITE_SASRUN")
    if not path or not Path(path).exists():
        raise RuntimeError("SAS wrapper not found: set $TNGXMM_SASRUN (the 'sasrun' script that "
                           "runs a command line inside the HEASoft container with SAS sourced)")
    return str(path)

def _read_ftz(path: Path, dest: Path) -> None:
    if str(path).lower().endswith((".ftz", ".gz")):
        with gzip.open(path, "rb") as src, open(dest, "wb") as out:
            shutil.copyfileobj(src, out)
    else:
        shutil.copyfile(path, dest)


def _find_products(src_dir: Path, camera: str, kind: str, band: int) -> list[Path]:
    """PPS-named products of one camera/kind/band in ``src_dir`` (mock ``percam/`` or a PPS
    directory); ``kind`` is ``IMAGE_`` or ``EXPMAP``."""
    return sorted(src_dir.glob(f"P*{CAMERAS[camera][1]}{kind}{band}000.FTZ"))


def _exposure(path: Path) -> float:
    from astropy.io import fits

    with fits.open(path) as f:
        return float(f[0].header.get("EXPOSURE", 0.0))

                 bands: tuple = (1, 2, 3, 4, 5)) -> StagedInputs:
    """Stage the IMAGE and EXPMAP products of ``product_dir`` into ``workdir``.

    ``kind='mock'``: a view directory (``percam/``) or a ``percam`` directory itself;
    ``kind='real'``: a directory of PPS products.  Per camera the exposure with the longest
    ``EXPOSURE`` that has every band is used (the PPS main exposure; scheduled and
    unscheduled exposures of one camera are never mixed).  Cameras and bands the products lack
    are left out (all-or-nothing per camera and band); the band list is reduced to the bands
    every staged camera has."""
    product_dir, workdir = Path(product_dir), Path(workdir)
    if kind not in ("mock", "real"):
        raise ValueError("kind must be 'mock' or 'real'")
    src_dir = product_dir / "percam" if (product_dir / "percam").is_dir() else product_dir
    workdir.mkdir(parents=True, exist_ok=True)
    cams: dict[str, dict[int, tuple[Path, Path]]] = {}
    for cam in CAMERA_ORDER:
        # exposure sets: group the band-2 images by exposure id (PPS: S001, U002, ...)
        ref = _find_products(src_dir, cam, "IMAGE_", 2 if 2 in bands else bands[0])
        best: dict[int, tuple[Path, Path]] | None = None
        best_t = -1.0
        for img2 in ref:
            stem = re.sub(r"IMAGE_\d000\.FTZ$", "", img2.name)
            got = {}
            for b in bands:
                im, ex = src_dir / f"{stem}IMAGE_{b}000.FTZ", src_dir / f"{stem}EXPMAP{b}000.FTZ"
                if im.exists() and ex.exists():
                    got[b] = (im, ex)
            if len(got) < len(bands) - (1 if 1 in bands else 0):
                continue
            t = _exposure(img2)
            if t > best_t:
                best, best_t = got, t
        if best:
            cams[cam] = best
    if not cams:
        raise FileNotFoundError(f"no IMAGE/EXPMAP products of bands {bands} in {src_dir}")
    have = tuple(b for b in bands if all(b in v for v in cams.values()))
    cameras = [c for c in CAMERA_ORDER if c in cams]
    sources: dict[str, list[str]] = {}
    filters: dict[str, str] = {}
    date_obs = ""
    from astropy.io import fits

    for cam in cameras:
        pre = CAMERAS[cam][0]
        for b in have:
            im, ex = cams[cam][b]
            _read_ftz(im, workdir / f"{pre}_image_b{b}.fits")
            _read_ftz(ex, workdir / f"{pre}_image_b{b}exp.fits")
            sources[f"{pre}_image_b{b}.fits"] = [str(im)]
            sources[f"{pre}_image_b{b}exp.fits"] = [str(ex)]
        h = fits.getheader(workdir / f"{pre}_image_b{have[0]}.fits")
        filters[cam] = str(h.get("FILTER", "Thin1")).strip()
        date_obs = date_obs or str(h.get("DATE-OBS", "")).strip()
    if not date_obs:
        raise ValueError("no DATE-OBS in the product headers: the calibration (CIF) cannot be "
                         "chosen (mocks: instrument.sas_headers = 'on')")
    return StagedInputs(workdir, cameras, have, filters, date_obs, sources)

def edetect_chain_command(staged: StagedInputs, settings: DetectSettings) -> str:
    """The shell command run inside the SAS container, in ``staged.workdir``: the CIF of the
    observation date and one ``edetect_chain`` over the staged cameras and bands."""
    images = " ".join(staged.images())
    ref = staged.images()[0]
    pimin = " ".join(" ".join(str(PI_MIN[b - 1]) for b in staged.bands) for _ in staged.cameras)
    pimax = " ".join(" ".join(str(PI_MAX[b - 1]) for b in staged.bands) for _ in staged.cameras)
    ecf = " ".join(" ".join(f"{v:g}" for v in ecf_for(c, staged.filters[c], staged.bands))
                   for c in staged.cameras)
    evsets = " ".join(f"{CAMERAS[c][0]}_image_b{staged.bands[0]}.fits" for c in staged.cameras)
    args = settings.edetect_arguments()
    cifdate = "" if settings.analysisdate is None else f" analysisdate={settings.analysisdate}"
    return (
        f"export OMP_NUM_THREADS={settings.omp_threads} && cd {staged.workdir} && "
        f"cifbuild withobservationdate=yes observationdate={staged.date_obs}{cifdate} "
        f"calindexset=ccf.cif > cifbuild.log 2>&1 && export SAS_CCF={staged.workdir}/ccf.cif && "
        f"edetect_chain imagesets=\"{images}\" eventsets=\"{evsets}\" attitudeset={ref} "
        f"witheexpmap={args['witheexpmap']} pimin=\"{pimin}\" pimax=\"{pimax}\" ecf=\"{ecf}\" "
        f"eboxl_list=all_eboxlist_l.fits eboxm_list=all_eboxlist_m.fits "
        f"esp_nsplinenodes={args['esp_nsplinenodes']} eml_list=all_emllist.fits "
        f"esen_mlmin={args['esen_mlmin']}"
        + "".join(f" {k}={args[k]}" for k in ("likemin", "mlmin") if k in args)
        + " > edetect_chain.log 2>&1"
    )

def read_source_list(emllist: str | Path) -> Any:
    """Summary rows (``ID_INST == 0``, ``ID_BAND == 0``) of an ``emldetect`` list as a
    DataFrame with ``SOURCE_COLUMNS``.  FLUX and FLUX_ERR are REPLACED, for every source with a finite per-camera
    all-band flux, by the inverse-variance weighted mean of the per-camera fluxes (the PPS / catalogue EPIC flux
    convention; the ``emldetect`` summary values are kept only for a source with no such camera row).  The per-band ``RATE_<k>`` are
    the summed rates of the bands 1-5 summary rows."""
    import numpy as np
    from astropy.table import Table

    d = Table.read(str(emllist), "SRCLIST")
    for c in d.colnames:
        d[c].unit = None
    summ = d[(d["ID_INST"] == 0) & (d["ID_BAND"] == 0)]
    cam = d[(d["ID_INST"] > 0) & (d["ID_BAND"] == 0)]
    flux = np.array(summ["FLUX"], dtype=float)
    ferr = np.array(summ["FLUX_ERR"], dtype=float)
    for k, sid in enumerate(summ["ML_ID_SRC"]):
        c = cam[(cam["ML_ID_SRC"] == sid) & np.isfinite(cam["FLUX"]) & (cam["FLUX_ERR"] > 0)]
        if len(c):
            w = 1.0 / np.asarray(c["FLUX_ERR"], dtype=float) ** 2
            flux[k] = np.sum(w * np.asarray(c["FLUX"], dtype=float)) / np.sum(w)
            ferr[k] = 1.0 / np.sqrt(np.sum(w))
    import pandas as pd

    out = pd.DataFrame({c: np.asarray(summ[c]) for c in SOURCE_COLUMNS if c in summ.colnames})
    out["FLUX"], out["FLUX_ERR"] = flux, ferr
    for b in range(1, 6):
        sel = d[(d["ID_INST"] == 0) & (d["ID_BAND"] == b)]
        rate = dict(zip(np.asarray(sel["ML_ID_SRC"]), np.asarray(sel["RATE"], dtype=float)))
        out[f"RATE_{b}"] = [rate.get(i, np.nan) for i in out["ML_ID_SRC"]]
    return out

def run_detection(product_dir: str | Path, out_dir: str | Path | None = None,
                  kind: str = "mock", settings: DetectSettings | None = None,
                  work_dir: str | Path | None = None, keep_work: bool = False,
                  force: bool = False, run: Any = None) -> dict[str, Any]:
    """Detect sources on the products of ``product_dir`` and write the source list next to them.

    ``out_dir`` defaults to ``product_dir`` (mock: the view directory; the files go to
    ``<out_dir>/sas_detect/``); the staged images and SAS products live in ``work_dir``
    (default ``<out_dir>/sas_detect/work``) and are removed unless ``keep_work``.  A finished
    detection (:func:`detection_complete`) is kept unless ``force``; unfinished remains are made again.  ``run(cmd, workdir, settings)`` replaces the SAS
    call (tests).  Returns the provenance record, also written as ``provenance.json`` (the done-marker,
    see the module docstring)."""
    settings = settings or DetectSettings()
    product_dir = Path(product_dir)
    out = Path(out_dir or product_dir) / "sas_detect"
    out.mkdir(parents=True, exist_ok=True)
    result_list = out / "emllist.fits"
    if not force and detection_complete(out):
        return json.loads((out / "provenance.json").read_text())
    (out / "provenance.json").unlink(missing_ok=True)                # not finished until it is written again
    diskguard.remove_stale_tmp(out)
    work = Path(work_dir) if work_dir else out / "work"
    if work.exists():
        shutil.rmtree(work)
    try:
        staged = stage_inputs(product_dir, work, kind, tuple(settings.bands))
        cmd = edetect_chain_command(staged, settings)
        diskguard.write_text(out / "command.txt", cmd + "\n")
        (run or _run_sas)(cmd, work, settings)
        emllist = work / "all_emllist.fits"
        if not emllist.exists():
            tail = (work / "edetect_chain.log").read_text(errors="replace")[-600:] \
                if (work / "edetect_chain.log").exists() else ""
            raise RuntimeError(f"edetect_chain wrote no source list in {work}: {tail}")
        if not diskguard.fits_complete(emllist, ("SRCLIST",)):
            raise RuntimeError(f"edetect_chain left a cut source list in {work}")
        diskguard.copy_file(emllist, result_list)
        diskguard.copy_file(work / "edetect_chain.log", out / "edetect_chain.log")
        table = read_source_list(result_list)
        with diskguard.atomic_path(out / "sources.csv") as tmp:
            table.to_csv(tmp, index=False)
    except diskguard.DiskFull:
        raise
    except Exception as e:
        # SAS and cfitsio do not pass ENOSPC on: ask the message, the logs and the disk itself
        logs = "".join((work / n).read_text(errors="replace")[-4000:] for n in ("cifbuild.log", "edetect_chain.log")
                       if (work / n).exists())
        st = (diskguard.interrupted_status(e) if diskguard.text_says_disk(logs) else None) \
            or diskguard.failure_status(e, work) or diskguard.failure_status(e, out)
        if st is not None:
            raise diskguard.DiskFull(f"SAS detection of {product_dir}: {st}") from e
        raise
    log = (out / "edetect_chain.log").read_text(errors="replace")
    ver = re.search(r"\[(\d+\.\d+\.\d+-[0-9a-f]+-\d+)\]", log)
    n_det = int((table["DET_ML"] > DET_ML_MIN).sum())
    rec = {"kind": kind, "product_dir": str(product_dir), "emllist": str(result_list),
           "sources": str(out / "sources.csv"), "n_sources": int(len(table)), "n_det_ml_gt_10": n_det,
           "sas_version": ver.group(1) if ver else None, "cameras": staged.cameras,
           "bands": list(staged.bands), "filters": staged.filters, "date_obs": staged.date_obs,
           "analysisdate": settings.analysisdate or "now", "ccf_manifest_sha256": CCF_MANIFEST_SHA256,
           "settings": dataclasses.asdict(settings), "command": cmd,
           "inputs": {k: {"product": v[0]} for k, v in staged.sources.items()},
           "emllist_sha256": sha256_file(result_list)}
    diskguard.write_text(out / "provenance.json", json.dumps(rec, indent=1, default=str))
    if not keep_work:
        shutil.rmtree(work, ignore_errors=True)
    return rec

def detection_complete(sas_dir: str | Path) -> bool:
    """True when ``sas_dir`` (``<out>/sas_detect``) holds a finished detection: ``provenance.json`` parses,
    ``emllist.fits`` has the sha256 it records and ``sources.csv`` ends with a whole line and has the recorded
    number of rows.  The remains of a detection that was killed, or files cut by a full disk, fail the test."""
    d = Path(sas_dir)
    rec = diskguard.read_json(d / "provenance.json")
    if not isinstance(rec, dict):
        return False
    lst, src = d / "emllist.fits", d / "sources.csv"
    if not lst.is_file() or not src.is_file() or not diskguard.csv_tail_complete(src):
        return False
    if rec.get("emllist_sha256") and sha256_file(lst) != rec["emllist_sha256"]:
        return False
    if rec.get("n_sources") is not None:
        with open(src, "rb") as fh:
            if sum(1 for _ in fh) != int(rec["n_sources"]) + 1:
                return False
    return True




def _run_sas(cmd: str, workdir: Path, settings: DetectSettings) -> None:
    """Run ``cmd`` inside the SAS container with the ``sasrun`` wrapper."""
    env = dict(os.environ, SAS_WORK=str(workdir))
    argv = [sasrun_path(settings), cmd]
    if settings.cpus:
        argv = ["taskset", "-c", settings.cpus, *argv]
    proc = subprocess.run(argv, env=env, capture_output=True, text=True)
    if proc.returncode != 0:
        tail = ""
        log = Path(workdir) / "edetect_chain.log"
        if log.exists():
            tail = log.read_text(errors="replace")[-600:]
        if diskguard.text_says_disk(proc.stderr) or diskguard.text_says_disk(tail):
            raise diskguard.DiskFull(f"edetect_chain failed (exit {proc.returncode}) with a full disk: "
                                     f"{proc.stderr[-300:]} {tail[-300:]}")
        raise RuntimeError(f"edetect_chain failed (exit {proc.returncode}): {proc.stderr[-400:]} {tail}")
