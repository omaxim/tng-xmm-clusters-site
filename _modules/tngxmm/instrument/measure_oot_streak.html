Source code for tngxmm.instrument.measure_oot_streak

"""Measure the EPIC-pn out-of-time (OoT) streak of the brightest source in a
PN image: the same statistics on real PPS products and on mocks.

The streak runs along the CCD readout direction (RAWY) through the source.
Everything is measured in a frame centred on the source: ``s`` along the
readout direction, ``q`` across it.  The readout direction and the boundary
between the two rows of pn CCDs are read from the exposure map itself (the
inter-CCD gaps parallel to the readout direction are long and many; the row
boundary is the one gap across it), so no roll angle or detector geometry
has to be assumed and real and mock images are treated identically.

Model of the background under the streak: the azimuthal surface-brightness
profile of the image around the source, the MEDIAN over 16 angular sectors
per annulus with the two sectors on the streak left out (robust against
point sources and cluster asymmetry), times the exposure map.

Statistics (``StreakStats``):

* ``f_near``/``f_far``: the OoT fraction implied by the excess counts per
  unit length on the two sides of the source within the source's CCD row
  (``near`` = towards the row boundary, ``far`` = towards the outer edge),
  ``f = density x L_col / C`` with ``L_col`` the CCD column length on the sky
  (200 rows x 4.1") and ``C`` the background-subtracted counts in the strip
  ``|q| < width`` over the source's CCD row (in-time plus OoT).  A streak that
  is uniform over the whole column gives f_near = f_far = the mode's OoT
  fraction; a streak confined between the source and its readout node gives
  f_near = 0.
* ``f_beyond``: the same beyond the row boundary (another CCD; 0 if the
  streak stops at the CCD edge).
* ``width_ratio``: FWHM-equivalent (2.355 x rms) width of the streak's cross
  profile over that of the source's own cross profile.
* ``hr_streak``/``hr_src``: hard/soft (1-2 keV over 0.5-1 keV) count ratio
  of the streak excess and of the source.
* ``dispersion``: variance over expected value of the strip counts in
  0.5'-long cells after removing the mean streak (1 = Poisson).

References
----------
XMM-Newton Users' Handbook section 3.3.10 (EPIC out-of-time events);
SAS epchain documentation (out-of-time events); Strueder et al. 2001.
"""

from __future__ import annotations

from dataclasses import asdict, dataclass
from typing import Any

#: inner radius of the streak measurement (source and cluster core excluded)
R_INNER_ARCMIN = 3.0
#: half width of the strip across the streak
STRIP_HALF_WIDTH_ARCMIN = 1.5

def _filled(mask):
    from scipy import ndimage as ndi
    st = ndi.generate_binary_structure(2, 1)
    m = ndi.binary_closing(mask, st, iterations=4)
    return ndi.binary_fill_holes(m)

@cites("epic_pn")
def readout_angle(expmap, step_deg: float = 0.1) -> float:
    """Readout (RAWY) direction in the image frame [deg, mod 180, from +x
    towards +y]: the projection direction along which the exposure map's gaps
    are sharpest (the long inter-CCD gaps are parallel to RAWY)."""
    import numpy as np
    m = expmap > 0.2 * np.nanmax(expmap)
    fov = _filled(m)
    yy, xx = np.nonzero(fov)
    inside = m[yy, xx].astype(float)

    def score(phi):
        a = np.deg2rad(phi)
        q = -xx * np.sin(a) + yy * np.cos(a)
        qb = np.round(q - q.min()).astype(int)
        f = np.bincount(qb)
        g = np.bincount(qb, weights=inside)
        ok = f > 0.3 * f.max()
        return float(np.sum((1 - g[ok] / f[ok]) ** 2))

    coarse = np.arange(0.0, 180.0, 1.0)
    best = coarse[int(np.argmax([score(p) for p in coarse]))]
    fine = np.arange(best - 1.5, best + 1.5, step_deg)
    return float(fine[int(np.argmax([score(p) for p in fine]))] % 180.0)




def _sector_background(rate, good, r, theta, phi_rad, r_edges, nsec=16):
    """Median-over-sectors rate profile (streak sectors excluded)."""
    import numpy as np
    nr = len(r_edges) - 1
    rel = np.mod(theta - phi_rad, np.pi)
    sec = np.floor(np.mod(theta - phi_rad, 2 * np.pi) / (2 * np.pi / nsec)).astype(int) % nsec
    streak = (rel < np.pi / nsec) | (rel > np.pi - np.pi / nsec)
    rb = np.digitize(r, r_edges) - 1
    use = good & ~streak & (rb >= 0) & (rb < nr)
    idx = rb[use] * nsec + sec[use]
    num = np.bincount(idx, minlength=nr * nsec).reshape(nr, nsec).astype(float)
    tot = np.bincount(idx, weights=rate[use], minlength=nr * nsec).reshape(nr, nsec)
    mean = np.where(num >= 5, tot / np.maximum(num, 1), np.nan)
    nval = np.isfinite(mean).sum(axis=1)
    prof = np.full(nr, np.nan)
    ok_r = (nval >= 4) & (num.sum(axis=1) >= 20)
    if ok_r.any():
        prof[ok_r] = np.nanmedian(mean[ok_r], axis=1)
    # fill holes by interpolation
    c = 0.5 * (r_edges[1:] + r_edges[:-1])
    ok = np.isfinite(prof)
    if ok.sum() >= 2:
        prof = np.interp(c, c[ok], prof[ok])
    return prof

@cites("xmm_mission", "epic_pn", "xmm_ccf_sas")
def measure_streak(img_soft, img_hard, expmap, pix_arcsec: float = 4.0,
                   source_xy: tuple | None = None, phi_deg: float | None = None,
                   width_arcmin: float = STRIP_HALF_WIDTH_ARCMIN,
                   r_inner_arcmin: float = R_INNER_ARCMIN,
                   column_arcmin: float = PN_COLUMN_ARCMIN) -> StreakStats:
    """Streak statistics of the brightest source of a PN image pair
    (0.5-1 keV ``img_soft``, 1-2 keV ``img_hard``, counts) with its exposure
    map (any band; only its support and relative shape are used)."""
    import numpy as np
    from scipy import ndimage as ndi

    soft = np.asarray(img_soft, float)
    hard = np.asarray(img_hard, float)
    img = soft + hard
    exp = np.asarray(expmap, float)
    emax = np.nanmax(exp)
    good = exp > 0.3 * emax
    pa = pix_arcsec / 60.0
    if phi_deg is None:
        phi_deg = readout_angle(exp)
    a = np.deg2rad(phi_deg)
    ny, nx = img.shape
    yy, xx = np.mgrid[0:ny, 0:nx].astype(float)

    if source_xy is None:
        core = ndi.binary_erosion(good, iterations=int(round(0.3 / pa)))
        sm = ndi.gaussian_filter(np.where(good, img, 0.0), 2.0)
        sm[~core] = -1
        y0, x0 = np.unravel_index(int(np.argmax(sm)), sm.shape)
        # centroid within 30"
        rr = np.hypot(xx - x0, yy - y0) < 0.5 / pa
        w = img * rr * good
        x0, y0 = float((w * xx).sum() / w.sum()), float((w * yy).sum() / w.sum())
    else:
        x0, y0 = map(float, source_xy)

    dx, dy = xx - x0, yy - y0
    s = (dx * np.cos(a) + dy * np.sin(a)) * pa          # arcmin
    q = (-dx * np.sin(a) + dy * np.cos(a)) * pa
    r = np.hypot(s, q)
    theta = np.arctan2(dy, dx)

    # row boundary: the gap across the readout direction, from the support
    fov = _filled(exp > 0.2 * emax)
    sb_bins = np.arange(np.floor(s[fov].min()), np.ceil(s[fov].max()) + 1e-9, pa)
    f_all = np.histogram(s[fov], sb_bins)[0].astype(float)
    f_in = np.histogram(s[fov & good], sb_bins)[0].astype(float)
    sc = 0.5 * (sb_bins[1:] + sb_bins[:-1])
    ok = f_all > 0.5 * f_all.max()
    frac = np.where(ok, f_in / np.maximum(f_all, 1), 2.0)
    s_b = float(sc[int(np.argmin(frac))])
    near_sign = 1.0 if s_b > 0 else -1.0

    # background (cluster + sky + particles) model: sector-median rate profile
    rate = np.where(good, img / np.maximum(exp, 1e-9), 0.0)
    r_edges = np.concatenate([np.arange(0, 3, 0.25), np.arange(3, 20, 0.5)])
    prof = _sector_background(rate, good, r, theta, a, r_edges)
    rc = 0.5 * (r_edges[1:] + r_edges[:-1])
    model = np.interp(r, rc, prof) * exp
    prof_s = _sector_background(np.where(good, soft / np.maximum(exp, 1e-9), 0.0),
                                good, r, theta, a, r_edges)
    prof_h = _sector_background(np.where(good, hard / np.maximum(exp, 1e-9), 0.0),
                                good, r, theta, a, r_edges)
    model_s = np.interp(r, rc, prof_s) * exp
    model_h = np.interp(r, rc, prof_h) * exp
    # far-field level (sky + particles): outer annulus median
    far = (rc > 9) & (rc < 12.5)
    far_level = float(np.nanmedian(prof[far]))
    far_s, far_h = float(np.nanmedian(prof_s[far])), float(np.nanmedian(prof_h[far]))

    strip = good & (np.abs(q) < width_arcmin)
    src_row = (s * near_sign < abs(s_b))                 # the source's CCD row
    # C: in-time + OoT counts of the strip over the source's row
    C = float((img - far_level * exp)[strip & src_row].sum())
    Csrc = float((img - far_level * exp)[good & (r < r_inner_arcmin)].sum())

    resid = img - model
    out_r = np.abs(s) >= r_inner_arcmin

    # profile along s (0.5' bins) of the strip excess per arcmin, normalised
    # by C (a streak uniform over the column reads f / L_col)
    sbins = np.arange(-16, 16.01, 0.5)
    scen = 0.5 * (sbins[1:] + sbins[:-1])
    ex = np.histogram(s[strip], sbins, weights=resid[strip])[0]
    vv = np.histogram(s[strip], sbins, weights=np.maximum(img[strip], 0))[0]
    npx = np.histogram(s[strip], sbins)[0]
    full = 2 * width_arcmin / pa * (0.5 / pa)               # pixels of a full bin
    cov = npx / full
    corr = np.where(cov > 0.6, 1.0 / np.maximum(cov, 1e-9), np.nan)
    prof = ex * corr / 0.5 / C
    perr = np.sqrt(vv) * corr / 0.5 / C
    prof_s = (scen, prof, perr)
    lo_e, hi_e = sbins[:-1], sbins[1:]
    if near_sign > 0:
        in_row, beyond_b = hi_e < s_b - 0.25, lo_e > s_b + 0.5
    else:
        in_row, beyond_b = lo_e > s_b + 0.25, hi_e < s_b - 0.5
    outside = np.minimum(np.abs(lo_e), np.abs(hi_e)) >= r_inner_arcmin
    fin = np.isfinite(prof)

    def side(sel):
        sel = sel & fin
        n = int(sel.sum())
        if n < 3:
            return np.nan, np.nan, 0.0
        f = float(np.median(prof[sel])) * column_arcmin
        fe = 1.2533 * float(np.sqrt((perr[sel] ** 2).sum())) / n * column_arcmin
        return f, fe, 0.5 * n

    near_b = in_row & outside & (scen * near_sign > 0)
    far_b = in_row & outside & (scen * near_sign < 0)
    f_near, e_near, l_near = side(near_b)
    f_far, e_far, l_far = side(far_b)
    f_bey, e_bey, _ = side(beyond_b & outside)

    # the same with a LOCAL background: sidebands 2.5' < |q| < 5' at the same
    # s (robust for point sources off-axis, where the azimuthal model is not)
    sbm = good & (np.abs(q) > 2.5) & (np.abs(q) < 5.0)
    cb = np.histogram(s[sbm], sbins, weights=img[sbm])[0]
    eb = np.histogram(s[sbm], sbins, weights=exp[sbm])[0]
    cs_ = np.histogram(s[strip], sbins, weights=img[strip])[0]
    es_ = np.histogram(s[strip], sbins, weights=exp[strip])[0]
    rate_b = cb / np.maximum(eb, 1e-9)
    ok_b = eb > 0
    ex_b = np.where(ok_b, (cs_ - es_ * rate_b) * corr / 0.5 / C, np.nan)
    er_b = np.where(ok_b, np.sqrt(cs_ + (es_ / np.maximum(eb, 1e-9)) ** 2 * cb) * corr / 0.5 / C,
                    np.nan)

    def side_sb(sel, r_in=1.5):
        sel = sel & np.isfinite(ex_b) & (np.minimum(np.abs(lo_e), np.abs(hi_e)) >= r_in)
        n = int(sel.sum())
        if n < 2:
            return np.nan, np.nan
        return (float(ex_b[sel].mean()) * column_arcmin,
                float(np.sqrt((er_b[sel] ** 2).sum())) / n * column_arcmin)

    f_near_sb, e_near_sb = side_sb(in_row & (scen * near_sign > 0))
    f_far_sb, e_far_sb = side_sb(in_row & (scen * near_sign < 0))
    f_bey_sb, e_bey_sb = side_sb(beyond_b)


    # cross profile: streak excess over |s| >= r_inner on the source row
    qbins = np.arange(-width_arcmin, width_arcmin + 1e-9, 0.25)
    qc = 0.5 * (qbins[1:] + qbins[:-1])
    sel = good & src_row & out_r & (np.abs(q) < width_arcmin)
    exq = np.histogram(q[sel], qbins, weights=resid[sel])[0]
    sel2 = good & (r < r_inner_arcmin) & (np.abs(q) < width_arcmin)
    srq = np.histogram(q[sel2], qbins, weights=(img - far_level * exp)[sel2])[0]

    def rms(w):
        w = np.clip(w, 0, None)
        if w.sum() <= 0:
            return np.nan
        m = (w * qc).sum() / w.sum()
        return float(np.sqrt((w * (qc - m) ** 2).sum() / w.sum()))

    wr = rms(exq) / rms(srq) if rms(srq) and np.isfinite(rms(srq)) else np.nan

    # hardness (1-2 keV / 0.5-1 keV) of the streak excess per side and of the
    # source; 0.5' bins with an outlier excess (a point source) are dropped
    exs = np.histogram(s[strip], sbins, weights=(soft - model_s)[strip])[0] * corr
    exh = np.histogram(s[strip], sbins, weights=(hard - model_h)[strip])[0] * corr

    def hardness(sel):
        sel = sel & fin
        if sel.sum() < 3:
            return np.nan, np.nan
        tot_ = (exs + exh)[sel]
        err_ = np.sqrt(vv[sel]) * corr[sel]
        keep = tot_ < np.median(tot_) + 4 * err_
        hs_, ss__ = float(exh[sel][keep].sum()), float(exs[sel][keep].sum())
        if ss__ <= 0 or hs_ <= 0:
            return np.nan, np.nan
        var = float((vv[sel][keep] * corr[sel][keep] ** 2).sum())
        # the soft and hard excess share the background noise: rough error
        rel = np.sqrt(var / 2) * np.sqrt(1 / hs_ ** 2 + 1 / ss__ ** 2)
        return hs_ / ss__, hs_ / ss__ * rel

    hr_near, hr_near_err = hardness(near_b)
    hr_far, hr_far_err = hardness(far_b)
    hr_streak, hr_err = hardness(near_b | far_b)
    sr = good & (r < r_inner_arcmin)
    hs = float((hard - far_h * exp)[sr].sum())
    ss_ = float((soft - far_s * exp)[sr].sum())
    hr_src = hs / ss_ if ss_ > 0 else np.nan

    # dispersion of the strip counts in 0.5' cells about the model plus a
    # linear streak trend per side (|s| >= r_inner, source row); 1 = Poisson
    obs_c = np.histogram(s[strip], sbins, weights=img[strip])[0]
    mod_c = np.histogram(s[strip], sbins, weights=model[strip])[0]
    chi: list = []
    for sel in (near_b, far_b):
        sel = sel & fin & (cov > 0.9) & (np.abs(scen) > 5.0)
        if sel.sum() < 4:
            continue
        res = obs_c[sel] - mod_c[sel]
        trend = np.polyval(np.polyfit(scen[sel], res, 1), scen[sel])
        expct = mod_c[sel] + trend
        ok_ = expct > 5
        chi.extend(list(((res - trend) ** 2 / expct)[ok_]))
    disp = float(np.mean(chi)) if len(chi) > 3 else np.nan

    return StreakStats(
        phi_deg=float(phi_deg), x0=x0, y0=y0, s_boundary=s_b,
        counts_strip=C, counts_src=Csrc,
        f_near=f_near, f_near_err=e_near, f_far=f_far, f_far_err=e_far,
        f_beyond=f_bey, f_beyond_err=e_bey, len_near=l_near, len_far=l_far,
        width_ratio=wr, hr_streak=hr_streak, hr_streak_err=hr_err, hr_src=hr_src,
        hr_near=hr_near, hr_near_err=hr_near_err, hr_far=hr_far, hr_far_err=hr_far_err,
        dispersion=disp, f_near_sb=f_near_sb, f_near_sb_err=e_near_sb, f_far_sb=f_far_sb,
        f_far_sb_err=e_far_sb, f_beyond_sb=f_bey_sb, f_beyond_sb_err=e_bey_sb,
        profile_s=prof_s, profile_q=(qc, exq, srq))

@cites("xmm_sas_epevents", "epic_pn")
def expected_far_hardness(e_kev, rawy, ccd, line, rows: tuple, n_grid: int = 48,
                          bands: tuple = ((0.5, 1.0), (1.0, 2.0)), weights=None) -> dict[str, float]:
    """Expected 1-2 keV / 0.5-1 keV hardness of the far-side out-of-time streak of a column.

    Every event of the column that feeds the streak (energy ``e_kev`` [keV], true RAWY ``rawy``, CCD ``ccd``) is
    out of time with the same probability and is assigned a row uniform over the far-side rows ``rows``
    (:func:`far_rows`); its energy becomes ``e_kev`` times ``line.oot_factor`` (a
    :class:`~tngxmm.instrument.pn_oot_energy.PnCtiLine`; None leaves the energies unchanged).  Returns the expected
    soft and hard counts per out-of-time copy and their ratio ``hr``.  Divided by the hardness of the source, this
    is the far / source ratio the streak should show, whatever the spectrum of the column (the reference of the
    streak test and of the comparison with real streaks, whose columns are read from the PPS event lists).
    ``weights`` (per event, e.g. a negative background) default to 1."""
    import numpy as np

    e = np.asarray(e_kev, float)
    yt = np.asarray(rawy, float)
    w = np.ones_like(e) if weights is None else np.asarray(weights, float)
    c = np.broadcast_to(np.asarray(ccd), e.shape)
    out = np.zeros(len(bands))
    grid = rows[0] + (np.arange(n_grid) + 0.5) / n_grid * (rows[1] - rows[0])
    for yr in grid:
        f = 1.0 if line is None else line.oot_factor(e, c, yt, np.full_like(e, yr))
        ee = e * f
        for i, (lo, hi) in enumerate(bands):
            out[i] += (w * ((ee > lo) & (ee <= hi))).sum()
    out /= n_grid
    return dict(soft=float(out[0]), hard=float(out[1]), hr=float(out[1] / out[0]) if out[0] > 0 else float("nan"))
