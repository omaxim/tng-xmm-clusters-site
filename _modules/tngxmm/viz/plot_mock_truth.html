Source code for tngxmm.viz.plot_mock_truth

"""One figure per mock view: the mock beside the simulation it was made from.

Three panels, left to right:

1. the mock EPIC image (``image_merged.fits``, 0.5-2 keV counts of the three cameras), centred on
   the cluster (``CLUSTX``/``CLUSTY``), cropped to a square of half-width ``fov_r500`` true R500c(z_obs), with the
   true R500c drawn;
2. the intrinsic surface brightness of the same gas (``tngref_sb_map.fits``, the temet-exact
   0.5-2 keV map in the simulation frame), rotated by the view's roll angle so that the two
   panels share one orientation, cropped to the same physical square;
3. the cumulative mass profiles: blue, the X-ray fit (median and 68 % band of the per-sample
   profiles the fitter keeps, ``mass_profile_samples.npz``); red, the intrinsic hydrostatic mass
   of the gas (the truth file's headline estimator, Barnes et al. 2021 with each cell's own mean
   molecular weight); black, the true mass from the particles.  The true R500c(z_obs) is marked,
   with the three masses inside it, and each curve's own 500 rho_c(z_obs) crossing is marked on
   the curve.

Both images are stretched with :func:`tngxmm.bank.render_bank.normalize_image`, the one image
normalisation of the project (ceiling at the 99th percentile of the core, floor 0.25 dex below
the background, inferno), each on its own cropped frame: the two panels are in different units
(counts and erg/s/kpc^2), so they cannot share one scale, but they share one stretch rule.

Orientation.  The photon list stores the sky position of a photon as its simulation coordinates
along the two sky axes of the view; the instrument rotates those positions by the roll angle
about the optical axis and histograms them into the merged image (``simulate_instrument``, the
``xr, yr`` lines).  The temet map is the unrotated frame (``AXIS1``, ``AXIS2`` = the same two
sky axes), so ``ndimage.rotate(map, -posangle)`` puts it in the mock's frame; both are shown with
``origin='lower'``, and the mock's WCS (``CDELT1 < 0``) then reads east left.

The estimands under the panels are those of ``analysis.mock_fits``
(:mod:`tngxmm.analysis.estimands`): blue/red, red/black and blue/black inside the true R500c
(the fixed-radius estimand of claim C) and at each curve's own crossing (SO, claims A and B).  A
view made from a mini snapshot says so in its title (``snap_is_mini``), which also carries the core class
of the halo from the central entropy of its truth profile (:func:`core_class`, after Lehle et al. 2024).

Style (figure width, dpi, colours, the field of view in R500c, the band of the temet map) is
``params:viz.mock_truth`` (``conf/base/parameters/pipeline.yml``), so the figures of the paper are
reproducible from the repository with ``kedro run --pipeline viz --nodes viz.plot_mock_truth``.
"""
from __future__ import annotations

import math
from pathlib import Path
from typing import Any

from tngxmm.cites import cites, collect_cites

#: the three curves and the keys of their ratios in ``analysis.mock_fits``
CURVES = (("blue", "X-ray fit of the XMM mock (median, 68 % band)"),
          ("red", "Barnes+21 HSE truth: Vikhlinin fits, 50 shells to 1.5 R500c (exact)"),
          ("black", "true total mass, particles (exact; the truth grid starts at 0.05 R500c)"))
RATIOS = (("blue/red", "ln_br"), ("red/black", "ln_rt"), ("blue/black", "ln_bt"))
#: the Vikhlinin profiles behind the red curve are fitted inside this radius [R500c]; beyond it the analytic
#: HSE mass is extrapolated and drawn greyed out
RED_TRUSTED_R500 = 1.2

DEFAULT_STYLE: dict[str, Any] = {
    "fov_r500": 1.5,                 # half-width of the image panels [true R500c(z_obs)]
    "band": "SB_0.5-2.0KEV",         # extension of the temet map shown
    "width_in": 15.0, "dpi": 200,
    "colours": {"blue": "C0", "red": "C3", "black": "k", "r500": "w"},
    # the core class in the title: the central entropy K0 inside ``aperture_r500`` x R500c of the truth
    # profile's ``phase`` shells, strong cool core at or below the first threshold [keV cm^2], non-cool core
    # above the second, weak cool core between (Lehle et al. 2024, Table 1, after Hudson et al. 2010)
    "core_class": {"aperture_r500": 0.012, "thresholds_kev_cm2": [22.0, 150.0], "phase": "hot"},
    "r_min_r500": 0.05,              # inner limit of the mass panel [R500c]
    "r_max_r500": 2.5,               # outer limit of the mass panel [R500c]
    "formats": ["png"],
}

def mock_image(view_dir: str | Path) -> dict[str, Any]:
    """The merged mock image: ``counts``, the cluster pixel ``centre`` (x, y; 0-based),
    ``pix_arcsec``, ``posangle_deg``, ``z_obs``, ``snap_is_mini`` and the header."""
    import numpy as np
    from astropy.io import fits

    with fits.open(Path(view_dir) / "image_merged.fits") as hd:
        h, counts = hd[0].header, np.asarray(hd[0].data, dtype=float)
    centre = (float(h["CLUSTX"]), float(h["CLUSTY"]))
    return dict(counts=counts, centre=centre, pix_arcsec=abs(float(h["CDELT2"])) * 3600.0,
                posangle_deg=float(h["POSANGLE"]), z_obs=float(h["Z_OBS"]),
                snap_is_mini=bool(h.get("SNAPMINI", False)), header=h)

def tngref_map(view_dir: str | Path, band: str) -> dict[str, Any]:
    """The temet-exact surface-brightness map of ``band`` as a linear image (erg/s/kpc^2; 0
    where the map holds no emission), its centre pixel, ``pix_kpc`` and the header."""
    import numpy as np
    from astropy.io import fits

    with fits.open(Path(view_dir) / "tngref_sb_map.fits") as hd:
        h = hd[band].header
        log_sb = np.asarray(hd[band].data, dtype=float)
    sb = np.where(np.isfinite(log_sb), 10.0 ** np.where(np.isfinite(log_sb), log_sb, 0.0), 0.0)
    n = int(h["NPIX"])
    return dict(sb=sb, centre=((n - 1) / 2.0, (n - 1) / 2.0), pix_kpc=float(h["PIXKPC"]), header=h)

def rotate_to_mock(img: Any, posangle_deg: float) -> Any:
    """``img`` (simulation frame) rotated about its centre into the mock's frame (module
    docstring); the array keeps its shape, the corners that leave the frame are zero."""
    from scipy import ndimage

    if abs(posangle_deg % 360.0) <= 0.1:
        return img
    return ndimage.rotate(img, -posangle_deg, reshape=False, order=1, mode="constant", cval=0.0)

def crop_square(img: Any, centre: tuple[float, float], half_px: float) -> tuple[Any, list[float]]:
    """The square of half-width ``half_px`` about ``centre`` (x, y), clipped to the image, and
    its ``imshow`` extent in pixels relative to the centre."""
    import numpy as np

    cx, cy = centre
    h, w = img.shape
    x0, x1 = max(0, int(round(cx - half_px))), min(w, int(round(cx + half_px)) + 1)
    y0, y1 = max(0, int(round(cy - half_px))), min(h, int(round(cy + half_px)) + 1)
    sub = np.asarray(img[y0:y1, x0:x1])
    return sub, [x0 - 0.5 - cx, x1 - 0.5 - cx, y0 - 0.5 - cy, y1 - 0.5 - cy]

@cites("barnes2021", "mbproj2d")
def mass_curves(fit_dir: str | Path, truth_h5: str | Path, z_obs: float, red_key: str | None = None) -> dict:
    """The three cumulative mass profiles [kpc, Msun] and their 500 rho_c(z_obs) crossings:
    ``blue`` (``r``, ``lo``, ``med``, ``hi`` over the fit's samples), ``red`` and ``black``
    (``r``, ``m``), the true ``r500c_zobs_kpc`` and the SO masses of
    :func:`tngxmm.analysis.estimands.truth_estimands` plus the fit's own M500 and R500."""
    import numpy as np

    from tngxmm.analysis.estimands import truth_estimands
    from tngxmm.massfit.fit_qc import load_mass_profile_samples

    t = truth_estimands(truth_h5, z_obs, red_key)
    s = load_mass_profile_samples(fit_dir)
    lo, med, hi = np.percentile(s["mtot"], [16, 50, 84], axis=0)
    res = s["result"]
    import h5py

    with h5py.File(truth_h5, "r") as f:
        black = (f["mass_grid/r_kpc"][:], f["mass_grid/M_msun"][:])
        red = (f[f"native/{t['red_key']}/r_kpc"][:], f[f"native/{t['red_key']}/M_msun"][:])
    return dict(blue=dict(r=s["r_kpc"], lo=lo, med=med, hi=hi,
                          m500=float(res["m500_1e14"]) * 1e14, r500=float(res["r500_kpc"]),
                          m500_lo=float(res["m500_lo"]) * 1e14, m500_hi=float(res["m500_hi"]) * 1e14),
                red=dict(r=red[0], m=red[1], m500=t["m500_red_zobs_msun"], r500=t["r500_red_zobs_kpc"]),
                black=dict(r=black[0], m=black[1], m500=t["m500c_zobs_msun"], r500=t["r500c_zobs_kpc"]),
                r500c_zobs_kpc=t["r500c_zobs_kpc"], truth=t)




def _loglog(r, m, x):
    import numpy as np

    r, m = np.asarray(r, float), np.asarray(m, float)
    ok = (r > 0) & np.isfinite(m) & (m > 0)
    if ok.sum() < 2 or not (r[ok][0] <= x <= r[ok][-1]):
        return float("nan")
    return float(np.exp(np.interp(math.log(x), np.log(r[ok]), np.log(m[ok]))))

@cites("lehle2024", "hudson2010")
def core_class(truth_h5: str | Path, aperture_r500: float = 0.012, thresholds_kev_cm2=(22.0, 150.0),
               phase: str = "hot") -> tuple[float, str]:
    """(K0 [keV cm^2], class) of the halo after Lehle et al. 2024 (Table 1; the thresholds of Hudson et
    al. 2010): the central entropy inside ``aperture_r500`` x R500c, ``strong cool core`` at or below
    ``thresholds_kev_cm2[0]``, ``non-cool core`` above ``thresholds_kev_cm2[1]``, ``weak cool core`` between.

    K0 is the gas-mass-weighted mean of the truth profile's shell entropies ``K_mw = T_mw n_e,vw^-2/3`` of
    ``phase`` over the shells whose outer edge lies inside the aperture (weights rho_gas x filling x shell
    volume).  Differences from the paper, which uses the non-star-forming, cooling, T > 10^6 K gas bound to
    the cluster with mass-weighted means: the ``hot`` phase is the non-star-forming gas above 5.5 MK with no
    cooling cut, n_e is volume-weighted within a shell, and the shells start at 0.01 R500c (the sphere inside
    is not in the profile).  The centre is the potential minimum in both."""
    import h5py
    import numpy as np

    with h5py.File(truth_h5, "r") as f:
        r_edges = f["r_edges_kpc"][:]
        edges = r_edges / float(f.attrs["R500c_kpc"])
        g = f[f"gas/{phase}"]
        k, rho, fill = g["K_mw"][:], g["rho_gas"][:], g["fill"][:]
    vol = 4.0 / 3.0 * np.pi * (r_edges[1:] ** 3 - r_edges[:-1] ** 3)
    sel = (edges[1:] <= aperture_r500 * (1.0 + 1e-6)) | (np.arange(len(k)) == 0)     # at least the first shell
    w = np.where(sel & np.isfinite(k) & (rho > 0), rho * fill * vol, 0.0)
    k0 = float(np.sum(w * np.where(np.isfinite(k), k, 0.0)) / np.sum(w)) if w.sum() > 0 else float("nan")
    lo, hi = thresholds_kev_cm2
    label = ("strong cool core" if k0 <= lo else "non-cool core" if k0 > hi else "weak cool core")
    return k0, label

def estimand_row(fit_row: Any) -> dict[str, float]:
    """The ratios of one ``analysis.mock_fits`` row as numbers: ``<a/b>@R500`` (inside the true
    R500c(z_obs)) and ``<a/b>@SO`` (each curve's own crossing)."""
    out = {}
    for label, key in RATIOS:
        for suffix, col in (("R500", f"{key}_1"), ("SO", f"{key}_so")):
            v = fit_row.get(col)
            out[f"{label}@{suffix}"] = math.exp(float(v)) if v is not None and v == v else float("nan")
    return out




def _table_text(ratios: dict[str, float]) -> str:
    """The six ratios as an aligned two-row block (monospace)."""
    head = f"{'':<20}" + "".join(f"{label:>12}" for label, _ in RATIOS)
    rows = [f"{name:<20}" + "".join(f"{ratios[f'{label}@{suffix}']:>12.2f}" for label, _ in RATIOS)
            for name, suffix in (("inside true R500c", "R500"), ("own M500 (SO)", "SO"))]
    return "\n".join([head, *rows])

def write_table(rows: list[dict[str, Any]], out_md: str | Path) -> None:
    """One line per view with the six ratios, as Markdown."""
    keys = [f"{label}@{s}" for label, _ in RATIOS for s in ("R500", "SO")]
    lines = ["| view | z_obs | M500c true [1e14] | core (K0 [keV cm2]) | " + " | ".join(keys) + " | mini |",
             "|---|---|---|---|" + "---|" * len(keys) + "---|"]
    for r in rows:
        lines.append(f"| {r['view_id']} | {r['z_obs']:.3f} | {r['m500_true_1e14']:.2f} | "
                     f"{r['core_class']} ({r['k0_kev_cm2']:.0f}) | "
                     + " | ".join(f"{r[k]:.2f}" for k in keys) + f" | {'yes' if r['snap_is_mini'] else 'no'} |")
    Path(out_md).write_text("\n".join(lines) + "\n")

def plot_mock_truth(mock_fits: Any, views: Any, truth_profiles: Any, tngref_maps: Any, options: dict[str, Any]) -> Any:
    """Every ok mock fit of ``mock_fits`` with a view in ``views``: one figure under ``options['out']``
    and one row (view id, files, the ratios); the table is ``<out>/mock_truth_table.md``.  Views
    without a temet map or a truth profile are skipped with a note in the ``status`` column.  ``views`` needs
    ``view_dir``, ``view_id``, ``snap``, ``halo_id``, ``view``; ``truth_profiles`` also ``path``; ``tngref_maps`` the
    same keys.  ``options``: ``out``, ``red_key`` and the figure style (every other key).  Raises ``RuntimeError``
    when no figure is drawn."""
    import pandas as pd

    out = Path(options["out"])
    style = {k: v for k, v in options.items() if k not in ("out", "red_key")}
    vd = {(int(r.snap), int(r.halo_id), int(r.view)): r for r in views.itertuples(index=False)}
    truth = {(int(r.snap), int(r.halo_id)): str(r.path) for r in truth_profiles.itertuples(index=False)}
    maps = {(int(r.snap), int(r.halo_id), int(r.view)) for r in tngref_maps.itertuples(index=False)}
    rows = []
    for _, f in mock_fits.iterrows():
        if str(f.get("status")) != "ok":
            continue
        key = (int(f["snap"]), int(f["halo_id"]), int(f["view"]))
        v = vd.get(key)
        row = dict(view_id=getattr(v, "view_id", None) if v is not None else f"{key[0]}_{key[1]}_v{key[2]}",
                   snap=key[0], halo_id=key[1], view=key[2], fit_dir=str(f["fit_dir"]), tag=f.get("tag"))
        if v is None or key not in maps or key[:2] not in truth:
            rows.append(dict(row, status="no view, map or truth profile"))
            continue
        res = plot_view(v.view_dir, f["fit_dir"], truth[key[:2]], f.to_dict(), out / f"mock_truth_{row['view_id']}",
                        style=dict(style, red_key=options.get("red_key")), title=row["view_id"])
        rows.append(dict(row, status="ok", **res))
    done = [r for r in rows if r["status"] == "ok"]
    if not done:
        raise RuntimeError(f"viz.plot_mock_truth: no figure drawn ({len(mock_fits)} fit rows, "
                           f"{len(rows)} with status ok, {len(vd)} views, {len(maps)} temet maps, "
                           f"{len(truth)} truth profiles)")
    write_table(done, out / "mock_truth_table.md")
    return pd.DataFrame(rows)
