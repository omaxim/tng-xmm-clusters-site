"""The figures of the documentation, made from results that exist on disk.

Every figure is one function ``fig_<name>(inputs, out_dir)``; :data:`FIGURES` lists them with the files they read
(:data:`INPUTS`).  An input is a pair (root, relative path): the root is a site variable or an INTERFACES section 7
root (``TNGXMM_DATA``, ``TNGXMM_FIXTURES``, ``TNGXMM_DEV``, ``TNGXMM_SITE_PROD``, ``TNGXMM_SITE_OBS_IMAGES``) or
``repo`` (this checkout), so the record names no machine path.  A relative path may be a glob.  Nothing here runs a
simulation or a fit: the functions read tables, images and reports and plot them.

``python -m tngxmm.viz.doc_figures [--only NAME ...] [--out docs/figures]`` writes ``<name>.png`` (and ``<name>.pdf`` for
the paper candidates) and ``manifest.json``: per figure, every input file with its root, relative path, size and
sha256, the commit of the checkout, and whether it is a paper candidate.  ``--check`` recomputes the sha256 of the
inputs and compares them with the manifest without plotting.

The captions, the configuration each input belongs to and the caveats are on the page ``docs/figures.md``; the
numbers quoted there are printed by the functions (``--verbose``).
"""

def root_dir(root: str) -> Path:
    """The directory of an input root: ``repo`` is this checkout, any other name a site variable (environment
    first, then the site file, :mod:`tngxmm.core.site`)."""
    from tngxmm.core import site
    from tngxmm.core.paths import repo_root

    if root == "repo":
        return repo_root()
    val = site.get(root)
    if not val:
        raise KeyError(f"input root ${root} is not configured (environment or site file)")
    return Path(val)

def resolve(name: str) -> list[dict[str, Any]]:
    """The input files of figure ``name``: ``[{root, rel, path}]`` (globs expanded, sorted)."""
    out = []
    for root, rel in INPUTS[name]:
        base = root_dir(root)
        hits = sorted(glob.glob(str(base / rel))) if any(c in rel for c in "*?[") else [str(base / rel)]
        if not hits:
            raise FileNotFoundError(f"{name}: no file for ${root}/{rel}")
        for h in hits:
            p = Path(h)
            if not p.is_file():
                raise FileNotFoundError(f"{name}: ${root}/{rel} is not a file")
            out.append({"root": root, "rel": str(p.relative_to(base)), "path": p})
    return out

def sha256(path: Path) -> str:
    """sha256 of a file (:func:`tngxmm.core.hashing.sha256_file`)."""
    from tngxmm.core.hashing import sha256_file

def _paths(files: list[dict[str, Any]], part: str) -> list[Path]:
    return [f["path"] for f in files if part in f["rel"]]

def _save(fig: Any, out_dir: Path, name: str) -> list[Path]:
    import matplotlib.pyplot as plt

    outs = [out_dir / f"{name}.png"]
    if name in PAPER_CANDIDATES:
        outs.append(out_dir / f"{name}.pdf")
    for p in outs:
        # fixed metadata, so that a rerun on the same inputs writes the same PDF bytes
        meta = {"CreationDate": None, "ModDate": None} if p.suffix == ".pdf" else {"Software": None}
        fig.savefig(p, metadata=meta)
    plt.close(fig)
    return outs


def _pct(x: Any, q: tuple[float, ...] = (16, 50, 84)) -> list[float]:
    import numpy as np

    x = np.asarray(x, dtype=float)
    x = x[np.isfinite(x)]
    return [float(np.percentile(x, v)) for v in q] if x.size else [float("nan")] * len(q)


def _results(paths: list[Path]) -> Any:
    import pandas as pd

    return pd.concat([pd.read_csv(p, low_memory=False, dtype={"view_id": str, "donor_id": str})
                      for p in paths], ignore_index=True)

def lx_truth_ratio(measured: Any, intrinsic: Any) -> dict[str, Any]:
    """The truth ratio of a measured L_X: the median and the 16th and 84th percentiles of log10(measured / intrinsic)
    over the views where both are finite and positive, and their number ``n``.  It is the net ratio of the whole
    measurement chain of :func:`tngxmm.photometry.measure_lx.campaign_lx` (background annulus, exposure map, PSF,
    out-of-time events, ECF), not of one step.  Its median is negative for the TNG-Cluster views of the v1.0.1 trees
    in all three apertures and positive in the 1.5 Mpc aperture of the box simulations (``docs/validation.md``, "The
    L_X estimator against the intrinsic luminosity").  The L_X figures quote it beside the measured luminosities."""
    import numpy as np

    m, t = np.asarray(measured, dtype=float), np.asarray(intrinsic, dtype=float)
    ok = np.isfinite(m) & np.isfinite(t) & (m > 0) & (t > 0)
    p16, p50, p84 = _pct(np.log10(m[ok] / t[ok]))
    return {"median": p50, "p16": p16, "p84": p84, "n": int(ok.sum())}

@cites("matplotlib")
def fig_pipeline_overview(files: list[dict[str, Any]], out_dir: Path, log: Callable = print) -> list[Path]:
    """The data flow of the pipeline: stages as boxes, the products between them on the arrows; the mock and the
    real branch meet in the same photometry and the same hydrostatic fit."""
    import matplotlib.pyplot as plt
    from matplotlib.patches import FancyArrowPatch, FancyBboxPatch

    fig, ax = plt.subplots(figsize=(STYLE["width_two"], 3.9))
    ax.set_xlim(0, 100)
    ax.set_ylim(0, 58)
    ax.axis("off")
    boxes = {
        "obs": (1, 40, "obs\n638 XMM observations\nof SZ / lensing clusters"),
        "donors": (1, 21, "donors\nwhich halo is observed\nas which observation"),
        "tng": (1, 2, "IllustrisTNG\nTNG-Cluster, TNG300,\nTNG100, TNG50 (full snapshots)"),
        "sim": (26, 2, "sim\ngas of a 4 R500c cube;\nAPEC photons (pyXSIM);\nGalactic absorption"),
        "inst": (26, 21, "instrument\nEPIC under a real\nobservation's conditions;\nbackgrounds, AGN, PSF"),
        "truth": (51, 2, "truth\nM500c from particles;\nintrinsic M_HSE\n(Barnes et al. 2021)"),
        "phot": (51, 21, "photometry, SAS\nLx apertures,\nedetect_chain"),
        "real": (51, 40, "real PPS products\n(same photometry,\nsame source rules)"),
        "fit": (76, 21, "massfit\nmbproj2d hydrostatic\nfit with XSPEC: M_X"),
        "out": (76, 2, "comparison\nM_X, M_HSE, M500c;\nrealism tests"),
    }
    w, h = 22, 13
    fs = STYLE["font_size"] - 2
    for key, (x, y, text) in boxes.items():
        face = {"tng": "#eeeeee", "obs": "#e3edf9", "real": "#e3edf9"}.get(key, "#fff4dd")
        ax.add_patch(FancyBboxPatch((x, y), w, h, boxstyle="round,pad=0.3", fc=face, ec="0.3", lw=0.8))
        head, _, rest = text.partition("\n")
        ax.text(x + w / 2, y + h - 2.2, head, ha="center", va="center", fontsize=fs, weight="bold")
        ax.text(x + w / 2, y + (h - 3.2) / 2, rest, ha="center", va="center", fontsize=fs - 0.5)

    def arrow(a: str, b: str, side: str = "h") -> None:
        xa, ya, _ = boxes[a]
        xb, yb, _ = boxes[b]
        if side == "h":
            p, q = (xa + w + 0.5, ya + h / 2), (xb - 0.5, yb + h / 2)
        elif side == "d":                                    # down-right diagonal into the left edge of b
            p, q = (xa + w + 0.5, ya + 2), (xb - 0.5, yb + h - 2)
        else:
            up = yb > ya

    cosmo = flat_lcdm(c.h0_km_s_mpc, c.om0)
    res = _results(_paths(files, "results_"))
    obs = pd.read_csv(_paths(files, "obs_seed_lx_v4.csv")[0], dtype={"obsid": str})

def _parse_r3(path: Path) -> dict[str, tuple[float, float]]:
    out = {}
    for line in path.read_text().splitlines():
        parts = line.split()
        if len(parts) == 2 and "/" in parts[1]:
            lo, bo = parts[1].split("/")
            out[parts[0]] = (float(lo), float(bo))
        elif len(parts) == 2 and parts[0].startswith("dino_pca"):
            out[parts[0]] = (float(parts[1]), float("nan"))
    return out


def _parse_univariate(path: Path) -> dict[str, tuple[float, float]]:
    out, on = {}, False
    for line in path.read_text().splitlines():
        if "auc_mock_higher_base" in line:
            on = True
            continue
        if on:
            parts = line.split()
            if len(parts) != 3:
                break
            out[parts[0]] = (float(parts[1]), float(parts[2]))
    return out

def fig_convergence(files: list[dict[str, Any]], out_dir: Path, log: Callable = print) -> list[Path]:
    """Convergence of the projected emission with the size of the gas cube: for 20 TNG-Cluster halos x 3 views,
    the 90th percentile over halos and views of (left) the fraction of the 0.5-2 keV emission of the annuli inside
    R500c that comes from gas outside a cube of half-side X R500c, and (right) that emission relative to the median
    XMM background of the donors, for every annulus inside X R500c.  The production cube has X = 4 (vertical line); the rule stated before the run is 1 per cent on both."""
    import matplotlib.pyplot as plt
    import pandas as pd

    s = pd.read_csv(_paths(files, "summary_by_annulus_x.csv")[0])
    fig, axes = plt.subplots(1, 2, figsize=(STYLE["width_two"], 2.9))
    ann = sorted({(lo, hi) for lo, hi in zip(s.annulus_lo, s.annulus_hi, strict=True)})
    cmap = plt.cm.viridis
    for i, (lo, hi) in enumerate(ann):
        g = s[(s.annulus_lo == lo) & (s.annulus_hi == hi)].sort_values("x_r500")
        c = cmap(i / max(len(ann) - 1, 1))
        g = g[g.f_pct > 0]                                # no emission beyond X: nothing to draw on a log axis
        if bool(s[(s.annulus_lo == lo) & (s.annulus_hi == hi)].criterion_a_applies.iloc[0]) and len(g):
            axes[0].plot(g.x_r500, 100 * g.f_pct, "o-", ms=3, lw=1, color=c, label=f"{lo:g}-{hi:g} R500c")
        g = g[g.annulus_hi <= g.x_r500 + 1e-9]          # an annulus that reaches beyond the cube is not tested
        if len(g):
            axes[1].plot(g.x_r500, 100 * g.ratio_bkg_pct, "o-", ms=3, lw=1, color=c, label=f"{lo:g}-{hi:g} R500c")
    for ax, lab in ((axes[0], "beyond X / annulus emission [%]"),
                    (axes[1], "beyond X / XMM background [%]")):
        ax.axhline(1, color="k", ls="--", lw=0.8)
        ax.axvline(4, color="0.5", lw=0.8)
        ax.set_yscale("log")
        ax.set_xlabel("cube half-side X [R500c]")
        ax.set_ylabel(lab)
    axes[0].legend(frameon=False, fontsize=STYLE["font_size"] - 2)
    axes[1].legend(frameon=False, fontsize=STYLE["font_size"] - 2.5, ncol=2)
    x4 = s[s.x_r500 == 4.0]
    a4, b4 = x4[x4.criterion_a_applies.astype(bool)], x4[x4.annulus_hi <= 4.0 + 1e-9]
    log(f"convergence at X = 4: max P90 fraction inside R500c {100 * a4.f_pct.max():.3f} %, max P90 background "
        f"ratio of the annuli inside 4 R500c {100 * b4.ratio_bkg_pct.max():.2f} % (annulus "
        f"{b4.loc[b4.ratio_bkg_pct.idxmax(), 'annulus_lo']}-{b4.loc[b4.ratio_bkg_pct.idxmax(), 'annulus_hi']})")
    return _save(fig, out_dir, "convergence")

def _commit() -> str:

        return subprocess.run(["git", "rev-parse", "--short=12", "HEAD"], cwd=repo_root(), check=True,

def make(names: list[str], out_dir: Path, log: Callable = print) -> dict[str, Any]:
    """Make the figures ``names`` in ``out_dir``; returns (and writes) the manifest, merged with an existing one."""
    _style()
    out_dir.mkdir(parents=True, exist_ok=True)
    man_p = out_dir / "manifest.json"
    manifest = json.loads(man_p.read_text()) if man_p.exists() else {"figures": {}}
    for name in names:
        files = resolve(name)
        outs = FIGURES[name](files, out_dir, log=log)
        manifest["figures"][name] = {
            "outputs": [p.name for p in outs], "paper_candidate": name in PAPER_CANDIDATES,
            "function": f"tngxmm.viz.doc_figures.{FIGURES[name].__name__}", "commit": _commit(),
            "inputs": [{"root": f["root"], "rel": f["rel"], "bytes": f["path"].stat().st_size,
                        "sha256": sha256(f["path"])} for f in files],
        }
    man_p.write_text(json.dumps(manifest, indent=1, sort_keys=True) + "\n")
    return manifest

def check(out_dir: Path) -> list[str]:
    """Inputs whose sha256 differs from the manifest (or that are missing)."""
    manifest = json.loads((out_dir / "manifest.json").read_text())
    bad = []
    for name, rec in manifest["figures"].items():
        for i in rec["inputs"]:
            p = root_dir(i["root"]) / i["rel"]
            if not p.is_file():
                bad.append(f"{name}: ${i['root']}/{i['rel']} missing")
            elif sha256(p) != i["sha256"]:
                bad.append(f"{name}: ${i['root']}/{i['rel']} changed")
    return bad

def main(argv: list[str] | None = None) -> int:
    """Command line: make the figures or check their inputs."""
    from tngxmm.core.paths import repo_root

    ap = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    ap.add_argument("--only", nargs="*", choices=sorted(FIGURES), help="figures to make (default: all)")
    ap.add_argument("--out", type=Path, default=repo_root() / "docs" / "figures")
    ap.add_argument("--check", action="store_true", help="compare the inputs with the manifest; plot nothing")
    ap.add_argument("--verbose", action="store_true", help="print the numbers quoted in the captions")
    a = ap.parse_args(argv)
    if a.check:
        bad = check(a.out)
        print("\n".join(bad) if bad else "figure inputs: all sha256 as recorded")
        return 1 if bad else 0
    make(a.only or list(FIGURES), a.out, log=print if a.verbose else (lambda *_: None))
    return 0
